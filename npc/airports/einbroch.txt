airport,143,43,5	script	Airport Staff#airport1a::airport1	90,{
	mes "[Equipe do aeroporto]";
	mes "Bem -vindo ao";
	mes "Aeroporto de Einbroch,";
	mes "onde oferecemos sem parar";
	mes "vos para as cidades de";
	mes "Juno, Lighthalzen e Hugel.";
	next;
	if (select("Embarque no dirigvel: Cancelar") == 1) {
		mes "[Equipe do aeroporto]";
		mes "A taxa de embarque dirigvel";
		mes " 1.200 zeny, mas se voc";
		mes "Tenho um ingresso gratuito para aeronave,";
		mes "A taxa ser dispensada. Vai";
		mes "Voc embarca no dirigvel?";
		next;
		if (select("Sim: No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[Equipe do aeroporto]";
			mes "Sinto muito, mas voc no";
			mes "tem um bilhete grtis para";
			mes "Dirigvel e voc no tem";
			mes "Zeny suficiente para embarcar";
			mes "o dirigvel. Lembre -se, o";
			mes "A taxa de embarque  de 1.200 Zeny.";
			close;
		}
	}
	mes "[Equipe do aeroporto]";
	mes "Obrigado e";
	mes "tenha um bom dia.";
	close;
}
