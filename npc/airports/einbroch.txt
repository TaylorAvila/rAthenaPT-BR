airport,143,43,5	script	Airport Staff#airport1a::airport1	90,{
	mes "[Equipe do aeroporto]";
	mes "Bem -vindo ao";
	mes "Aeroporto de Einbroch,";
	mes "onde oferecemos sem parar";
	mes "vôos para as cidades de";
	mes "Juno, Lighthalzen e Hugel.";
	next;
	if (select("Embarque no dirigível: Cancelar") == 1) {
		mes "[Equipe do aeroporto]";
		mes "A taxa de embarque dirigível";
		mes "é 1.200 zeny, mas se você";
		mes "Tenho um ingresso gratuito para aeronave,";
		mes "A taxa será dispensada. Vai";
		mes "Você embarca no dirigível?";
		next;
		if (select("Sim: Não") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[Equipe do aeroporto]";
			mes "Sinto muito, mas você não";
			mes "tem um bilhete grátis para";
			mes "Dirigível e você não tem";
			mes "Zeny suficiente para embarcar";
			mes "o dirigível. Lembre -se, o";
			mes "A taxa de embarque é de 1.200 Zeny.";
			close;
		}
	}
	mes "[Equipe do aeroporto]";
	mes "Obrigado e";
	mes "tenha um bom dia.";
	close;
}
