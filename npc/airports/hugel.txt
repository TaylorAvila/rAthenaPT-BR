hugel,178,142,0	script	toairplane#hugel	45,1,1,{
	end;
OnTouch:
	mes "Para usar o dirigvel, voc deve pagar 1.200 Zeny ou um bilhete gratuito.";
	mes "Voc gostaria de usar o servio?";
	next;
	if (select("Sim: No") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane",244,58;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane",244,58;
			end;
		}
		mes "Sinto muito, mas voc no tem dinheiro suficiente.";
		mes "Lembre -se de que voc deve pagar 1.200 Zeny para usar o servio.";
		close;
	}
	mes "Obrigado, por favor volte.";
	close;
}
