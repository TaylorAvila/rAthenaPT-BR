-	script	::Airship_Staff_izlude	-1,{
	mes "[Funcionário do dirigível]";
	mes "Bem-vindo! Você gostaria";
	mes "Para embarcar no dirigível que";
	mes "Parte no voo que para";
	mes "em Juno e Rachel?";
	next;
	if (select("Embarque no dirigível para Juno/Rachel .: Cancelar.") == 1) {
		mes "[Funcionário do dirigível]";
		mes "A taxa de embarque é";
		mes "1.200 Zeny. No entanto, isso";
		mes "cobrado é dispensado se você usar";
		mes "um ingresso gratuito para dirigir. Agora,";
		mes "Você ainda gostaria de embarcar?";
		next;
		if (select("Sim: Não") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[Funcionário do dirigível]";
			mes "Sinto muito, mas você não";
			mes "Tenha 1.200 Zeny para pagar";
			mes "para a taxa de embarque.";
			close;
		}
	}
	mes "[Funcionário do dirigível]";
	mes "Obrigado e";
	mes "Por favor, venha novamente.";
	mes "Tenha um bom dia ~";
	close;
}
