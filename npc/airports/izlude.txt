-	script	::Airship_Staff_izlude	-1,{
	mes "[Funcionrio do dirigvel]";
	mes "Bem-vindo! Voc gostaria";
	mes "Para embarcar no dirigvel que";
	mes "Parte no voo que para";
	mes "em Juno e Rachel?";
	next;
	if (select("Embarque no dirigvel para Juno/Rachel .: Cancelar.") == 1) {
		mes "[Funcionrio do dirigvel]";
		mes "A taxa de embarque ";
		mes "1.200 Zeny. No entanto, isso";
		mes "cobrado  dispensado se voc usar";
		mes "um ingresso gratuito para dirigir. Agora,";
		mes "Voc ainda gostaria de embarcar?";
		next;
		if (select("Sim: No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[Funcionrio do dirigvel]";
			mes "Sinto muito, mas voc no";
			mes "Tenha 1.200 Zeny para pagar";
			mes "para a taxa de embarque.";
			close;
		}
	}
	mes "[Funcionrio do dirigvel]";
	mes "Obrigado e";
	mes "Por favor, venha novamente.";
	mes "Tenha um bom dia ~";
	close;
}
