lhz_airport,143,43,5	script	Airport Staff#lhz_air1a::lhz_airport1	90,{
	mes "[Equipe do aeroporto]";
	mes "Bem -vindo ao";
	mes "Aeroporto Lighthalzen,";
	mes "onde oferecemos sem parar";
	mes "Voos para Einbroch, Juno e Hugel.";
	next;
	if (select("Embarque no dirigível .: Cancel.") == 1) {
		mes "[Equipe do aeroporto]";
		mes "A taxa de embarque é";
		mes "1.200 Zeny, mas você pode";
		mes "renunciar à taxa se você resgatar";
		mes "um ingresso gratuito para dirigir.";
		next;
		if (select("Sim: Não") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "lhz_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "lhz_airport",148,51;
				end;
			}
			mes "[Funcionário do dirigível]";
			mes "Sinto muito, mas você não";
			mes "Tenha 1.200 Zeny para pagar";
			mes "para a taxa de embarque.";
			close;
		}
	}
	mes "[Equipe do aeroporto]";
	mes "Obrigado e";
	mes "Por favor, venha novamente.";
	mes "Tenha um bom dia ~";
	close;
}

lhz_airport,143,49,3	script	Arrival Staff#lhz_air2a::lhz_airport2	90,{
	mes "[Equipe de chegada]";
	mes "Bem -vindo ao aeroporto Lighthalzen.";
	mes "Por favor, deixe -me guiá -lo para o";
	mes "Terminal principal se você estiver chegando do seu voo. Caso contrário, por favor";
	mes "embarque no dirigível que partiu para chegar ao destino pretendido.";
	next;
	if (select("Saia para o terminal principal .: Cancel.") == 1) {
		mes "[Equipe de chegada]";
		mes "Quando estiver no terminal principal, você precisará pagar a taxa novamente";
		mes "Para embarcar em um dirigível. Você deve";
		mes "Somente saia se o Lighthalzen for o seu destino pretendido. Devemos nós";
		mes "prosseguir para o terminal principal?";
		next;
		if (select("Sim: Não") == 1) {
			warp "lhz_airport",142,40;
			end;
		}
	}
	mes "[Equipe de chegada]";
	mes "Tudo bem, obrigado";
	mes "para o seu patrocínio";
	mes "E eu espero que você tenha";
	mes "um voo agradável ~";
	close;
}
