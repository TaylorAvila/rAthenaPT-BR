y_airport,143,43,5	script	Airport Staff#y_air1a::y_airport1	90,{
	mes "[Equipe do aeroporto]";
	mes "Bem -vindo ao aeroporto de Juno, onde oferecemos voos domsticos para Einbroch, Lighthalzen e Hugel,";
	mes "e vos internacionais para Izlude e Rachel.";
	mes "Como posso ser de servio?";
	next;
	if (select("Embarque no dirigvel .: Cancel.") == 1) {
		mes "[Equipe do aeroporto]";
		mes "A taxa de embarque para todos";
		mes "Os vos so 1.200 Zeny. Se voc";
		mes "Use um bilhete gratuito para aeronave,";
		mes "A taxa de embarque ser dispensada. Ento voc gostaria de partir?";
		next;
		if (select("Sim: No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[Equipe do aeroporto]";
			mes "Sinto muito, mas voc no";
			mes "Tenha 1.200 Zeny para pagar";
			mes "para a taxa de embarque.";
			close;
		}
	}
	mes "[Equipe do aeroporto]";
	mes "Obrigado e";
	mes "tenha um bom dia.";
	close;
}
