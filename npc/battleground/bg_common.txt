	if (.@menu1$ != "") set .@menu$, .@menu1$;
	else for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		set .@menu$, .@menu$+getitemname(.@Items[.@i])+":";
	set .@i, (select(.@menu$)-1)*2;
	mes "[Erundek]";
	mes "Voc escolheu ^3131ff"+getitemname(.@Items[.@i])+"^000000.";
	switch(.@Items[.@i]) {
		case 2720: mes "Este item  apenas para a classe Mestre Swordman e Taekwon."; break;
		case 2721: mes "Este item  apenas para aula de ladres."; break;
		case 2722: mes "Este item  apenas para aula de aclitos."; break;
		case 2723: mes "Este item  apenas para aula de mgica."; break;
		case 2724: mes "Este item  apenas para a classe Archer."; break;
		case 2725: mes "Este item  apenas para aula de comerciante."; break;
		case 2733: mes "Este item  apenas para pistoleiros."; break;
		default: break;
	}
	mes "Voc pode trocar por este item com ^ff0000"+.@Items[.@i+1]+""+getitemname(7828)+"ou"+.@Items[.@i+1]+""+getitemname(7829)+"^000000.";
	mes "Voc gostaria de trocar?";
	next;
	switch(select("No troque: troca")) {
	case 1:
		mes "[Erundek]";
		mes "Voc precisa de mais tempo para verificar os itens?";
		break;
	case 2:
		mes "[Erundek]";
		mes "Qual crach voc deseja trocar?";
		mes "Voc precisa ^3131ff"+.@Items[.@i+1]+"Crachs^000000 para trocar.";
		next;
		if (.@item[0] < 12269 || .@item[0] > 12273 ) {
			mes "[Erundek]";
			mes "Lembre -se de que os itens de recompensa do campo de batalha so ^FF0000Character Bound ^000000. Tem certeza de que deseja este item?";
			next;
		}
		set .@j, select("Crach de bravura: CLIMENTO DE VALOR: Cancelar");
		mes "[Erundek]";
		if (.@j == 3) {
			mes "Voc cancelou a troca.";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "Obrigado por trocar.";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "Voc no tem o suficiente"+getitemname(.@cost)+"s.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Terra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Abenoado Guillaume!";
		} else if (.@team$ == "Croix") {
			mes "[Laffy]";
			mes "Abenoado Croix!";
		}
		mes "Vamos aproveitar nossa gloriosa vitria!";
		mes strcharinfo(0)+",  uma placa que reflete a vitria.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Voc perdeu, mas  dedicado a esta batalha.";
			mes "Esta  uma recompensa por sua grande dedicao de Guillaume Marollo!";
			mes "Basta tomar essa derrota como uma lio e, da prxima vez, voc definitivamente vencer.";
		} else if (.@team$ == "Croix") {
			mes "[Laffy]";
			mes "Oh,"+strcharinfo(0)+"No fique triste.";
			mes "Mesmo que no vencemos, fizemos o nosso melhor.";
			mes "Este  um presente real de Croix e, por favor, no se esquea desta batalha. Vamos ganhar o prximo.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
