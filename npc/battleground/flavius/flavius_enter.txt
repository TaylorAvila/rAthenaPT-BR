bat_room,133,178,5	script	Flavius Officer#01a	418,{
	if (getbattleflag("Fewence.bgQueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- Espere um minuto !! -";
		mes "- Atualmente voc est carregando -";
		mes "- Muitos itens com voc. -";
		mes "- Por favor, tente novamente -";
		mes "- Depois de perder algum peso. -";
		close;
	}
	mes "[Oficial do Exrcito de Guillaume]";
	mes "Vamos mostrar o poder do Exrcito de Guillaume para aqueles croixs fedidos!";
	next;
	switch(select("Eu quero me juntar ao seu exrcito !: Conversao final")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[Oficial do Exrcito de Guillaume]";
			mes "Estou muito satisfeito por voc querer se juntar a ns e lutar contra o exrcito de Croix, mas me desculpe: no posso enviar um novato como voc para morrer no cruel campo de batalha.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[Oficial do Exrcito de Guillaume]";
				mes "Estou muito satisfeito por voc querer se juntar a ns e lutar contra o exrcito de Croix, mas me desculpe: no posso enviar um novato como voc para morrer no cruel campo de batalha.";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[Oficial do Exrcito de Guillaume]";
					mes "Voc parece ter acabado de voltar do campo de batalha.";
					mes " muito cedo para voc voltar. V descansar e deixe os inimigos para ns!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("Bat_b01");
					if (.@mapcount2 > 0) {
						mes "[Oficial do Exrcito de Guillaume]";
						mes "Recebi um relatrio informando que um corpo de elite foi despachado para o campo de batalha que j lutava com os croixs.";
						mes "Por que voc no vai esperar um pouco? Eu sugiro que voc aprimore suas armas e prepare seus suprimentos at ento.";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[Oficial do Exrcito de Guillaume]";
							mes "Um corpo de elite j est de p para ser despachado para o campo de batalha.";
							mes "Por que voc no vai esperar um pouco? Eu sugiro que voc aprimore suas armas e prepare seus suprimentos at ento.";
						}
						else {
							mes "[Oficial do Exrcito de Guillaume]";
							mes "Voc definitivamente parece estar pronto para a batalha!";
							mes "V mostrar aos Croixs o que o medo realmente significa!";
							mes "Hoje, nosso grito de vitria ecoar em todo o campo de batalha!";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[Oficial do Exrcito de Guillaume]";
		mes "Hoje, seremos vitoriosos!";
		break;
	}
	close;
}
