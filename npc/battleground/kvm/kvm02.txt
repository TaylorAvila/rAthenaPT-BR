-	script	KvM02_BG	-1,{
	end;

OnInit:
	if (getbattleflag("Fewence.bgQueue"))
		bg_unbook "BAT_C02";
	mapwarp "bat_c02","bat_room",154,150;
	setwall "BAT_C02",54,122,6,7,0,"BATC02WALL_A";
	setwall "BAT_C02",55,122,5,7,0,"BATC02WALL_B";
	setwall "BAT_C02",140,56,6,7,0,"BATC02WALL_C";
	setwall "BAT_C02",140,57,5,7,0,"BATC02WALL_D";
	disablenpc "KVM Officer#KVM02A";
	disablenpc "KVM Officer#KVM02B";
	end;

OnGuillaumeQuit:
	if (getbattleflag("Fewence.bgQueue"))
		bg_desert;
//	else
//		set BG_Delay_Tick, gettimetick(2) + 1200;
OnGuillaumeDie:
	if( $@KvM02BG == 2 )
	{
		set .Guillaume_Count, .Guillaume_Count - 1;
		bg_updatescore "BAT_C02",.Guillaume_Count,.Croix_Count;
		if( .Guillaume_Count < 1 ) donpcevent "KvM02_BG::OnCroixWin";
		else {
			mapannounce "BAT_C02", "O nmero de guilas "+.Guillaume_Count+".",bc_map,"0x00FF00";
			mapannounce "BAT_C02", "O nmero de croixes "+.Croix_Count+".",bc_map,"0x00FF00";
		}
	}
	end;

OnCroixQuit:
	if (getbattleflag("Fewence.bgQueue"))
		bg_desert;
//	else
//		set BG_Delay_Tick, gettimetick(2) + 1200;
OnCroixDie:
	if( $@KvM02BG == 2 )
	{
		set .Croix_Count, .Croix_Count - 1;
		bg_updatescore "BAT_C02",.Guillaume_Count,.Croix_Count;
		if( .Croix_Count < 1 ) donpcevent "KvM02_BG::OnGuillaumeWin";
		else {
			mapannounce "BAT_C02", "O nmero de guilas "+.Guillaume_Count+".",bc_map,"0x00FF00";
			mapannounce "BAT_C02", "O nmero de croixes "+.Croix_Count+".",bc_map,"0x00FF00";
		}
	}
	end;

OnGuillaumeActive:
	warp "bat_c02",62,119;
	end;

OnCroixActive:
	warp "bat_c02",137,64;
	end;

OnStart:
	disablenpc "KVM Officer#KVM02A";
	disablenpc "KVM Officer#KVM02B";
	set $@KvM02BG_Victory, 0;
	// Warp Teams
	bg_warp $@KvM02BG_id1,"bat_c02",53,128;
	bg_warp $@KvM02BG_id2,"bat_c02",146,55;
	initnpctimer;
	end;

OnTimer1000:
	mapannounce "BAT_C02", "Em 1 minuto, o KVM comear.",bc_map,"0x00FF00";
	end;

OnTimer3000:
	mapannounce "BAT_C02", "O tempo mximo para uma batalha de KVM  de 5 minutos.",bc_map,"0x00FF00";
	end;

OnTimer6000:
	mapannounce "BAT_C02", "Por favor, prepare -se para a batalha do KVM.",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Voc pode lustrar seu povo.",bc_map,"0x00FF00";
	donpcevent "#A_camp_start02::OnEnable";
	donpcevent "#B_camp_start02::OnEnable";
	end;

OnTimer13000:
	donpcevent "#A_camp_start02::OnDisable";
	donpcevent "#B_camp_start02::OnDisable";
	end;

OnTimer30000:
	mapannounce "BAT_C02", "30 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer45000:
	mapannounce "BAT_C02", "15 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	donpcevent "::OnKvM02One";
	end;

OnTimer50000:
	mapannounce "BAT_C02", "10 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	donpcevent "::OnKvM02Two";
	end;

OnTimer55000:
	mapannounce "BAT_C02", "5 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	donpcevent "::OnKvM02One";
	end;

OnTimer59000:
	mapannounce "BAT_C02", "O KVM est agora comeando.",bc_map,"0x00FF00";
	donpcevent "::OnKvM02Two";
	end;

OnTimer61000:
	// Team Members
	set .Guillaume_Count, bg_get_data($@KvM02BG_id1, 0);
	set .Croix_Count, bg_get_data($@KvM02BG_id2, 0);

	if (!getbattleflag("Fewence.bgQueue")) {
		if (.Guillaume_Count < 5 || .Croix_Count < 5) {
			set $@KvM02BG_Victory, 3;
			set $@KvM02BG, 3;
			mapannounce "BAT_C02","No h jogadores suficientes para comear a batalha",1,0x808080;
			stopnpctimer;
			donpcevent "KvM02_BG::OnStop";
			end;
		}
	}
	bg_updatescore "BAT_C02",.Guillaume_Count,.Croix_Count;
	set $@KvM02BG, 2; // Playing
	bg_warp $@KvM02BG_id1,"bat_c02",62,119;
	bg_warp $@KvM02BG_id2,"bat_c02",137,64;
	end;

OnTimer300000:
	mapannounce "BAT_C02", "1 minuto restante para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer330000:
	mapannounce "BAT_C02", "30 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer345000:
	mapannounce "BAT_C02", "15 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer350000:
	mapannounce "BAT_C02", "10 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer355000:
	mapannounce "BAT_C02", "5 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer360000:
	mapannounce "BAT_C02", "A batalha do KVM acabou.",bc_map,"0x00FF00";
	if( .Croix_Count > .Guillaume_Count )
		donpcevent "KvM02_BG::OnCroixWin";
	else if( .Croix_Count < .Guillaume_Count )
		donpcevent "KvM02_BG::OnGuillaumeWin";
	else
	{ // Draw Game
		set $@KvM02BG, 3;
		set $@KvM02BG_Victory, 3;
		mapannounce "BAT_C02", "O nmero de guilas "+.Guillaume_Count+".",bc_map,"0x00FF00";
		mapannounce "BAT_C02", "O nmero de croixes "+.Croix_Count+".",bc_map,"0x00FF00";
		mapannounce "BAT_C02", "Esta batalha terminou empatada.",bc_map,"0x00FF00";
		donpcevent "KvM02_BG::OnStop";
	}
	end;

OnGuillaumeWin:
	set $@KvM02BG, 3;
	set $@KvM02BG_Victory, 1;
	mapannounce "BAT_C02", "Guillaume vence!",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Parabns aos membros de Guillaume.",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Todos sero transferidos para o ponto de partida.",bc_map,"0x00FF00";
	donpcevent "KvM02_BG::OnStop";
	end;

OnCroixWin:
	set $@KvM02BG, 3;
	set $@KvM02BG_Victory, 2;
	mapannounce "BAT_C02", "Croix vence!",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Parabns aos membros do CROIX.",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Todos sero transferidos para o ponto de partida.",bc_map,"0x00FF00";
	donpcevent "KvM02_BG::OnStop";
	end;

OnStop:
	stopnpctimer;
	enablenpc "KVM Officer#KVM02A";
	enablenpc "KVM Officer#KVM02B";
	// Warp Teams
	bg_warp $@KvM02BG_id1,"bat_c02",53,128;
	bg_warp $@KvM02BG_id2,"bat_c02",146,55;
	donpcevent "KvM02_BG_Out::OnBegin";
	if (getbattleflag("Fewence.bgQueue"))
		bg_reserve "BAT_C02", true;
	end;
}

-	script	KvM02_BG_Out	-1,{
	end;

OnBegin:
	initnpctimer;
	end;

OnTimer1000:
	mapannounce "BAT_C02", "Inscreva -se com o oficial para adquirir pontos KVM.",bc_map,"0x00FF00";
	end;

OnTimer3000:
	mapannounce "BAT_C02", "O oficial conceder os pontos por 30 segundos.",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Em 30 segundos, o oficial ser enviado.",bc_map,"0x00FF00";
	end;

OnTimer5000:
	mapannounce "BAT_C02", "A menos que voc fale com o oficial, no pode ganhar os pontos.",bc_map,"0x00FF00";
	mapannounce "BAT_C02", "Por favor, tenha cuidado.",bc_map,"0x00FF00";
	end;

OnTimer55000:
	mapannounce "BAT_C02", "Voc ser enviado de volta.",bc_map,"0x00FF00";
	end;

OnTimer60000:
	stopnpctimer;
	if (getbattleflag("Fewence.bgQueue"))
		bg_reserve "BAT_C02", true;
	mapwarp "bat_c02","bat_room",154,150;
	set getvariableofnpc(.Croix_Count,"KvM02_BG"), 0;
	set getvariableofnpc(.Guillaume_Count,"KvM02_BG"), 0;
	set $@KvM02BG_Victory, 0;
	if( $@KvM02BG_id1 ) { bg_destroy $@KvM02BG_id1; set $@KvM02BG_id1, 0; }
	if( $@KvM02BG_id2 ) { bg_destroy $@KvM02BG_id2; set $@KvM02BG_id2, 0; }
	if (getbattleflag("Fewence.bgQueue"))
		bg_unbook "BAT_C02";
	disablenpc "KVM Officer#KVM02A";
	disablenpc "KVM Officer#KVM02B";
	set $@KvM02BG, 0;
	end;

OnGuillaumeJoin:
OnCroixJoin:
	if( $@KvM02BG )
		end;
	set .@Guillaume, getwaitingroomstate(0,"Kvm02r_guillaume");
	set .@Croix, getwaitingroomstate(0,"KVM02R_CROIX");

	if( .@Guillaume < 5 || .@Croix < 5 )
		end;

	set $@KvM02BG, 1; // Starting
	donpcevent "KvM02R_Croix::OnEnterBG";
	donpcevent "KvM02R_Guillaume::OnEnterBG";
	donpcevent "KvM02_BG::OnStart";
	end;
}
