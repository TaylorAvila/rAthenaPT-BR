-	script	KvM03_BG	-1,{
	end;

OnInit:
	if (getbattleflag("Fewence.bgQueue"))
		bg_unbook "One_C03";
	mapwarp "bat_c03","bat_room",154,150;
	setwall "One_C03",54,122,6,7,0,"BATC03WALL_A";
	setwall "One_C03",55,122,5,7,0,"BATC03WALL_B";
	setwall "One_C03",140,56,6,7,0,"BATC03WALL_C";
	setwall "One_C03",140,57,5,7,0,"BATC03WALL_D";
	disablenpc "KVM Officer#KVM03A";
	disablenpc "KVM Officer#KVM03B";
	end;

OnGuillaumeQuit:
	if (getbattleflag("Fewence.bgQueue"))
		bg_desert;
//	else
//		set BG_Delay_Tick, gettimetick(2) + 1200;
OnGuillaumeDie:
	if( $@KvM03BG == 2 ) {
		set .Guillaume_Count, .Guillaume_Count - 1;
		bg_updatescore "One_C03",.Guillaume_Count,.Croix_Count;
		if( .Guillaume_Count < 1 )
			donpcevent "KvM03_BG::OnCroixWin";
		else {
			mapannounce "One_C03", "O nmero de guilas "+.Guillaume_Count+".",bc_map,"0x00FF00";
			mapannounce "One_C03", "O nmero de croixes "+.Croix_Count+".",bc_map,"0x00FF00";
		}
	}
	end;

OnCroixQuit:
	if (getbattleflag("Fewence.bgQueue"))
		bg_desert;
//	else
//	set BG_Delay_Tick, gettimetick(2) + 1200;
OnCroixDie:
	if( $@KvM03BG == 2 ) {
		set .Croix_Count, .Croix_Count - 1;
		bg_updatescore "One_C03",.Guillaume_Count,.Croix_Count;
		if( .Croix_Count < 1 )
			donpcevent "KvM03_BG::OnGuillaumeWin";
		else {
			mapannounce "One_C03", "O nmero de guilas "+.Guillaume_Count+".",bc_map,"0x00FF00";
			mapannounce "One_C03", "O nmero de croixes "+.Croix_Count+".",bc_map,"0x00FF00";
		}
	}
	end;

OnGuillaumeActive:
	warp "bat_c03",62,119;
	end;

OnCroixActive:
	warp "bat_c03",137,64;
	end;

OnStart:
	disablenpc "KVM Officer#KVM03A";
	disablenpc "KVM Officer#KVM03B";
	set $@KvM03BG_Victory, 0;
	// Warp Teams
	bg_warp $@KvM03BG_id1,"bat_c03",53,128;
	bg_warp $@KvM03BG_id2,"bat_c03",146,55;
	initnpctimer;
	end;

OnTimer1000:
	mapannounce "One_C03", "Em 1 minuto, o KVM comear.",bc_map,"0x00FF00";
	end;

OnTimer3000:
	mapannounce "One_C03", "O tempo mximo para uma batalha de KVM  de 5 minutos.",bc_map,"0x00FF00";
	end;

OnTimer6000:
	mapannounce "One_C03", "Por favor, prepare -se para a batalha do KVM.",bc_map,"0x00FF00";
	mapannounce "One_C03", "Voc pode lustrar seu povo.",bc_map,"0x00FF00";
	donpcevent "#A_camp_start03::OnEnable";
	donpcevent "#B_camp_start03::OnEnable";
	end;

OnTimer13000:
	donpcevent "#A_camp_start03::OnDisable";
	donpcevent "#B_camp_start03::OnDisable";
	end;

OnTimer30000:
	mapannounce "One_C03", "30 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer45000:
	mapannounce "One_C03", "15 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	donpcevent "::OnKvM03One";
	end;

OnTimer50000:
	mapannounce "One_C03", "10 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	donpcevent "::OnKvM03Two";
	end;

OnTimer55000:
	mapannounce "One_C03", "5 segundos restantes para iniciar a batalha do KVM.",bc_map,"0x00FF00";
	donpcevent "::OnKvM03One";
	end;

OnTimer59000:
	mapannounce "One_C03", "O KVM est agora comeando.",bc_map,"0x00FF00";
	donpcevent "::OnKvM03Two";
	end;

OnTimer61000:
	// Team Members
	set .Guillaume_Count, bg_get_data($@KvM03BG_id1, 0);
	set .Croix_Count, bg_get_data($@KvM03BG_id2, 0);

	if (!getbattleflag("Fewence.bgQueue")) {
		if (.Guillaume_Count < 5 || .Croix_Count < 5) {
			set $@KvM03BG_Victory, 3;
			set $@KvM03BG, 3;
			mapannounce "One_C03","No h jogadores suficientes para comear a batalha",1,0xC0C0C0;
			stopnpctimer;
			donpcevent "KvM03_BG::OnStop";
			end;
		}
	}
	bg_updatescore "One_C03",.Guillaume_Count,.Croix_Count;
	set $@KvM03BG, 2; // Playing
	bg_warp $@KvM03BG_id1,"bat_c03",62,119;
	bg_warp $@KvM03BG_id2,"bat_c03",137,64;
	end;

OnTimer300000:
	mapannounce "One_C03", "1 minuto restante para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer330000:
	mapannounce "One_C03", "30 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer345000:
	mapannounce "One_C03", "15 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer350000:
	mapannounce "One_C03", "10 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer355000:
	mapannounce "One_C03", "5 segundos restantes para terminar a batalha do KVM.",bc_map,"0x00FF00";
	end;

OnTimer360000:
	mapannounce "One_C03", "KVM terminou.",bc_map,"0x00FF00";
	if( .Croix_Count > .Guillaume_Count )
		donpcevent "KvM03_BG::OnCroixWin";
	else if( .Croix_Count < .Guillaume_Count )
		donpcevent "KvM03_BG::OnGuillaumeWin";
	else
	{ // Draw Game
		set $@KvM03BG, 3;
		set $@KvM03BG_Victory, 3;
		mapannounce "One_C03", "O nmero de guilas "+.Guillaume_Count+".",bc_map,"0x00FF00";
		mapannounce "One_C03", "O nmero de croixes "+.Croix_Count+".",bc_map,"0x00FF00";
		mapannounce "One_C03", "Esta batalha terminou empatada.",bc_map,"0x00FF00";
		donpcevent "KvM03_BG::OnStop";
	}
	end;

OnGuillaumeWin:
	set $@KvM03BG, 3;
	set $@KvM03BG_Victory, 1;
	mapannounce "One_C03", "Guillaume vence!",bc_map,"0x00FF00";
	mapannounce "One_C03", "Parabns aos membros de Guillaume.",bc_map,"0x00FF00";
	mapannounce "One_C03", "Todos sero transferidos para o ponto de partida.",bc_map,"0x00FF00";
	donpcevent "KvM03_BG::OnStop";
	end;

OnCroixWin:
	set $@KvM03BG, 3;
	set $@KvM03BG_Victory, 2;
	mapannounce "One_C03", "Croix vence!",bc_map,"0x00FF00";
	mapannounce "One_C03", "Parabns aos membros do CROIX.",bc_map,"0x00FF00";
	mapannounce "One_C03", "Todos sero transferidos para o ponto de partida.",bc_map,"0x00FF00";
	donpcevent "KvM03_BG::OnStop";
	end;

OnStop:
	stopnpctimer;
	enablenpc "KVM Officer#KVM03A";
	enablenpc "KVM Officer#KVM03B";
	// Warp Teams
	bg_warp $@KvM03BG_id1,"bat_c03",53,128;
	bg_warp $@KvM03BG_id2,"bat_c03",146,55;
	donpcevent "KvM03_BG_Out::OnBegin";
	if (getbattleflag("Fewence.bgQueue"))
		bg_reserve "One_C03", true;
	end;
}

-	script	KvM03_BG_Out	-1,{
	end;

OnBegin:
	initnpctimer;
	end;

OnTimer1000:
	mapannounce "One_C03", "Inscreva -se com o oficial para adquirir pontos KVM.",bc_map,"0x00FF00";
	end;

OnTimer3000:
	mapannounce "One_C03", "O oficial conceder os pontos por 30 segundos.",bc_map,"0x00FF00";
	mapannounce "One_C03", "Em 30 segundos, o oficial ser enviado.",bc_map,"0x00FF00";
	end;

OnTimer5000:
	mapannounce "One_C03", "A menos que voc fale com o oficial, no pode ganhar os pontos.",bc_map,"0x00FF00";
	mapannounce "One_C03", "Por favor, tenha cuidado.",bc_map,"0x00FF00";
	end;

OnTimer55000:
	mapannounce "One_C03", "Voc ser enviado de volta.",bc_map,"0x00FF00";
	end;

OnTimer60000:
	stopnpctimer;
	if (getbattleflag("Fewence.bgQueue"))
		bg_reserve "One_C03", true;
	mapwarp "bat_c03","bat_room",154,150;
	set getvariableofnpc(.Croix_Count,"KvM03_BG"), 0;
	set getvariableofnpc(.Guillaume_Count,"KvM03_BG"), 0;
	set $@KvM03BG_Victory, 0;
	if( $@KvM03BG_id1 ) { bg_destroy $@KvM03BG_id1; set $@KvM03BG_id1, 0; }
	if( $@KvM03BG_id2 ) { bg_destroy $@KvM03BG_id2; set $@KvM03BG_id2, 0; }
	if (getbattleflag("Fewence.bgQueue"))
		bg_unbook "One_C03";
	disablenpc "KVM Officer#KVM03A";
	disablenpc "KVM Officer#KVM03B";
	set $@KvM03BG, 0;
	end;

OnGuillaumeJoin:
OnCroixJoin:
	if( $@KvM03BG )
		end;
	set .@Guillaume, getwaitingroomstate(0,"Kvm03r_guillaume");
	set .@Croix, getwaitingroomstate(0,"KVM03R_CROIX");

	if( .@Guillaume < 5 || .@Croix < 5 )
		end;

	set $@KvM03BG, 1; // Starting
	donpcevent "KvM03R_Croix::OnEnterBG";
	donpcevent "KvM03R_Guillaume::OnEnterBG";
	donpcevent "KvM03_BG::OnStart";
	end;
}
