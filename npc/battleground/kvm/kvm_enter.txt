// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	KVM Mercenary Officer#1	418,{
	if (getbattleflag("Fewence.bgQueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[Croix Mercenary Officer]";
		mes "Eu sei que voc j assinou com o Croix. Volte e junte -se  equipe. No aceitaremos traidores para lutar por ns.";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[Guillaume Mercenary Officer]";
		mes "Parece que voc participou recentemente de uma batalha de KVM. Voc deve esperar 5 minutos antes de se inscrever novamente.";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[Guillaume Mercenary Officer]";
	mes "Deixe -os saber o verdadeiro poder de Guillaume!";
	next;
	switch(select("Eu vou lutar com voc.: Conversao.")) {
	case 1:
		mes "[Guillaume Mercenary Officer]";
		mes "Mostre a eles como somos fortes.";
		mes "Hoje, todos ouviro o grito de Triumph de Guillaume!";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[Guillaume Mercenary Officer]";
		mes "Vamos ganhar!";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	Guillaume Knight#kvm	417,{
	mes "[Cavaleiro de Guillaume]";
	mes "Ol.";
	mes "O que voc quer saber?";
	next;
	switch(select("Inscreva -se para KVM.: O que  KVM ?: Como fao para participar do KVM ?: Quero conhecer meus pontos Kreiger.")) {
	case 1:
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue"))
			mes "Se voc estiver pronto para a batalha, clique no boto ^0000ffbattle da interface ^000000 para solicitar a batalha de KVM!";
		else {
			mes "Os aplicativos ainda no esto disponveis.";
			mes "Para se inscrever, voc precisa ir a um oficial mercenrio da KVM.";
		}
		close;
	case 2:
		mes "[Cavaleiro de Guillaume]";
		mes "KVM  a abreviao de Kreiger von Midgard.";
		mes "Aventureiro, voc est ciente de que o caminho para o Novo Mundo foi aberto?";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "Ns, a administrao de Guillaume, temos vrios planos para avanar para o Novo Mundo.";
		mes "E KVM  um deles.";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "Isso significa que ns, um dos muitos pases do continente Midgard,";
		mes "decidiram empregar muitos aventureiros para o avano imediato para o Novo Mundo.";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "Portanto, para selecionar os melhores aventureiros, estamos segurando o KVM.";
		mes "Exibimos respostas de muitos aventureiros.";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "E daremos a eles recompensas por sua participao!";
		mes "Quanto s recompensas, entre em contato com um oficial logstico da KVM.";
		close;
	case 3:
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue")) {
			mes "Basicamente, o KVM  um 3 em 3 batalha.";
			mes "Aplique clicando no boto ^0000ffbattle da interface ^000000, como um grupo ou individualmente.";
		} else {
			mes "Basicamente, o KVM  um 5 em 5 batalha.";
			mes "Primeiro, voc se aplica com uma recepcionista KVM, em um grupo ou individualmente.";
		}
		next;
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue"))
			mes "As inscries em grupo so para quando voc pretende entrar no KVM com seus membros do Partido ou Guilda,";
		else
			mes "Os aplicativos de grupo so para quando voc pretende entrar no KVM com os membros do seu partido,";
		mes "E um aplicativo pessoal  quando voc pretende entrar no KVM individualmente.";
		next;
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue"))
			mes "Observe -se, depois de cancelar o seu ^0000FF, no poder se inscrever novamente por 1 minuto ^000000.";
		else {
			mes "Inscreva -se com um oficial da KVM e ele entrar em contato com voc mais tarde, quando voc estiver em Protera.";
			next;
			mes "[Cavaleiro de Guillaume]";
			mes "Ento voc entra e segue as instrues no campo de batalha.";
			next;
			mes "[Cavaleiro de Guillaume]";
			mes "No entanto, esteja ciente de que, a menos que voc esteja no escritrio da KVM, ele no pode entrar em contato com voc.";
		}
		close;
	case 4:
		mes "[Cavaleiro de Guillaume]";
		mes "Seus pontos Kreiger so:";
		mes ""+kvm_point+".";
		close;
	}
}
