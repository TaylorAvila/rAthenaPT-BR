// KvM Officer - Guillaume
//============================================================
bat_room,164,178,5	script	KVM Mercenary Officer#1	418,{
	if (getbattleflag("Fewence.bgQueue"))
		end;
	/*
	set .@kvm,checkquest(6026,PLAYTIME);
	if ((.@kvm == 0) || (.@kvm == 1)) {
		mes "[Croix Mercenary Officer]";
		mes "Eu sei que você já assinou com o Croix. Volte e junte -se à equipe. Não aceitaremos traidores para lutar por nós.";
		close;
	}
	else if (.@kvm == 2) erasequest 6026;
	*/
	set .@permit,checkquest(6025,PLAYTIME);
	if ((.@permit == 0) || (.@permit == 1)) {
		mes "[Guillaume Mercenary Officer]";
		mes "Parece que você participou recentemente de uma batalha de KVM. Você deve esperar 5 minutos antes de se inscrever novamente.";
		close;
	}
	else if (.@permit == 2) erasequest 6025;
	mes "[Guillaume Mercenary Officer]";
	mes "Deixe -os saber o verdadeiro poder de Guillaume!";
	next;
	switch(select("Eu vou lutar com você.: Conversação.")) {
	case 1:
		mes "[Guillaume Mercenary Officer]";
		mes "Mostre a eles como somos fortes.";
		mes "Hoje, todos ouvirão o grito de Triumph de Guillaume!";
		close2;
		//setquest 6025;
		if (BaseLevel > 79)  warp "bat_room",169,223;
		else if (BaseLevel < 60) warp "bat_room",197,223;
		else warp "bat_room",225,223;
		end;
	case 2:
		mes "[Guillaume Mercenary Officer]";
		mes "Vamos ganhar!";
		close;
	}
}

// Guillaume Knight - KvM
//============================================================
bat_room,167,178,5	script	Guillaume Knight#kvm	417,{
	mes "[Cavaleiro de Guillaume]";
	mes "Olá.";
	mes "O que você quer saber?";
	next;
	switch(select("Inscreva -se para KVM.: O que é KVM ?: Como faço para participar do KVM ?: Quero conhecer meus pontos Kreiger.")) {
	case 1:
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue"))
			mes "Se você estiver pronto para a batalha, clique no botão ^0000ffbattle da interface ^000000 para solicitar a batalha de KVM!";
		else {
			mes "Os aplicativos ainda não estão disponíveis.";
			mes "Para se inscrever, você precisa ir a um oficial mercenário da KVM.";
		}
		close;
	case 2:
		mes "[Cavaleiro de Guillaume]";
		mes "KVM é a abreviação de Kreiger von Midgard.";
		mes "Aventureiro, você está ciente de que o caminho para o Novo Mundo foi aberto?";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "Nós, a administração de Guillaume, temos vários planos para avançar para o Novo Mundo.";
		mes "E KVM é um deles.";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "Isso significa que nós, um dos muitos países do continente Midgard,";
		mes "decidiram empregar muitos aventureiros para o avanço imediato para o Novo Mundo.";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "Portanto, para selecionar os melhores aventureiros, estamos segurando o KVM.";
		mes "Exibimos respostas de muitos aventureiros.";
		next;
		mes "[Cavaleiro de Guillaume]";
		mes "E daremos a eles recompensas por sua participação!";
		mes "Quanto às recompensas, entre em contato com um oficial logístico da KVM.";
		close;
	case 3:
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue")) {
			mes "Basicamente, o KVM é um 3 em 3 batalha.";
			mes "Aplique clicando no botão ^0000ffbattle da interface ^000000, como um grupo ou individualmente.";
		} else {
			mes "Basicamente, o KVM é um 5 em 5 batalha.";
			mes "Primeiro, você se aplica com uma recepcionista KVM, em um grupo ou individualmente.";
		}
		next;
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue"))
			mes "As inscrições em grupo são para quando você pretende entrar no KVM com seus membros do Partido ou Guilda,";
		else
			mes "Os aplicativos de grupo são para quando você pretende entrar no KVM com os membros do seu partido,";
		mes "E um aplicativo pessoal é quando você pretende entrar no KVM individualmente.";
		next;
		mes "[Cavaleiro de Guillaume]";
		if (getbattleflag("Fewence.bgQueue"))
			mes "Observe -se, depois de cancelar o seu ^0000FF, não poderá se inscrever novamente por 1 minuto ^000000.";
		else {
			mes "Inscreva -se com um oficial da KVM e ele entrará em contato com você mais tarde, quando você estiver em Protera.";
			next;
			mes "[Cavaleiro de Guillaume]";
			mes "Então você entra e segue as instruções no campo de batalha.";
			next;
			mes "[Cavaleiro de Guillaume]";
			mes "No entanto, esteja ciente de que, a menos que você esteja no escritório da KVM, ele não pode entrar em contato com você.";
		}
		close;
	case 4:
		mes "[Cavaleiro de Guillaume]";
		mes "Seus pontos Kreiger são:";
		mes ""+kvm_point+".";
		close;
	}
}
