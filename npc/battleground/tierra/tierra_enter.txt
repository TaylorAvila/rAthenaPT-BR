bat_room,124,178,5	script	Tierra Gorge Officer#01a	418,{
	if (getbattleflag("Fewence.bgQueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- Espere um minuto !! -";
		mes "- Atualmente você está carregando -";
		mes "- Muitos itens com você. -";
		mes "- Por favor, tente novamente -";
		mes "- Depois de perder algum peso. -";
		close;
	}
	mes "[Oficial do Exército de Guillaume]";
	mes "Vamos mostrar o poder do Exército de Guillaume para aqueles croixs fedidos!";
	next;
	switch(select("Eu quero me juntar ao seu exército !: Conversação final")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[Oficial do Exército de Guillaume]";
			mes "Estou muito satisfeito por você querer se juntar a nós e lutar contra o exército de Croix, mas me desculpe: não posso enviar um novato como você para morrer no cruel campo de batalha.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[Oficial do Exército de Guillaume]";
				mes "Estou muito satisfeito por você querer se juntar a nós e lutar contra o exército de Croix, mas me desculpe: não posso enviar um novato como você para morrer no cruel campo de batalha.";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[Oficial do Exército de Guillaume]";
					mes "Você parece ter acabado de voltar do campo de batalha.";
					mes "É muito cedo para você voltar. Vá descansar e deixe os inimigos para nós!";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("Conversa_ 01");
					if (.@mapcount > 0) {
						mes "[Oficial do Exército de Guillaume]";
						mes "Recebi um relatório informando que um corpo de elite foi despachado para o campo de batalha que já lutava com os croixs.";
						mes "Por que você não vai esperar um pouco? Eu sugiro que você aprimore suas armas e prepare seus suprimentos até então.";
					}
					else {
						if ($@TierraBG1) {
							mes "[Oficial do Exército de Guillaume]";
							mes "Um corpo de elite já está de pé para ser despachado para o campo de batalha.";
							mes "Por que você não vai esperar um pouco? Eu sugiro que você aprimore suas armas e prepare seus suprimentos até então.";
						}
						else {
							mes "[Oficial do Exército de Guillaume]";
							mes "Você definitivamente parece estar pronto para a batalha!";
							mes "Vá mostrar aos Croixs o que o medo realmente significa!";
							mes "Hoje, nosso grito de vitória ecoará em todo o campo de batalha!";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[Oficial do Exército de Guillaume]";
		mes "Hoje, seremos vitoriosos!";
		break;
	}
	close;
}
