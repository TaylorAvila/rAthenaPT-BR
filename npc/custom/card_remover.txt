//===== Description: =========================================
//= Removes cards from equipped items.
//===== Additional Comments: =================================
//= 1.0 First version. [TyrNemesis^]
//= 1.2 Optimized and fixed getequipname menu. [Kisuka]
//= 1.2a Added 'disable_items' command. [Euphy]
//============================================================

prt_in,28,73,4	script	Wise Old Woman#eAcustom	78,{

	set .zenycost,200000;    // base cost of the card remover services (in Zeny)
	set .percardcost,25000;  // cost per card of the card remover services (in Zeny)
	set .faildestroy,1;      // should the card remover have a chance of failure that destroys items? (1=yes, 0=no)

	disable_items;
	mes "[Idosa sábia]";
	mes "Bom dia, jovem. Eu tenho o poder de remover os cartões que você agravou em seu equipamento. Essa ideia é por favor?";
	next;
	switch(select("Sim, sim .: O que você cobra ?: Não, obrigado.")) {
	case 1:
		mes "[Idosa sábia]";
		mes "Muito bem. Qual item devo examinar para você?";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[Idosa sábia]";
			mes "Jovem ... você não está usando nada lá da qual eu possa remover os cartões.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[Idosa sábia]";
			mes "Jovem ... não há cartões compostos neste item. Não posso fazer nada com isso, receio.";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "^3355ffJUST por minuto!";
			mes "Eu não posso oferecer nenhum dos meus";
			mes "serviços para você porque";
			mes "Você está carregando muito";
			mes "coisa. Coloque seus itens extras em";
			mes "Armazenamento kafra e venha de novo ~";
			close;
		}
		mes "[Idosa sábia]";
		mes "Este item tem" + .@cardcount + "cartões compostos nele. Para realizar minha magia, vou precisar" + (.zenycost+(.@cardcount * .percardcost)) + "Zeny, a ^0000ffstar Crumb ^000000 e a ^0000ffyellow Gemstone ^000000.";
		next;
		if(select("Muito bem. Faça isso.") == 2) {
			mes "[Idosa sábia]";
			mes "Muito bem. Retorne imediatamente se você procurar meus serviços.";
			close;
		}
		if((zeny < (.zenycost+(.@cardcount * .percardcost))) || (countitem(1000) < 1) || (countitem(715) < 1)) {
			mes "[Idosa sábia]";
			mes "Você não tem todos os itens que eu preciso para trabalhar minha mágica, criança. Venha de novo quando você o fizer.";
			close;
		}
		mes "[Idosa sábia]";
		mes "Antes de começar, devo avisá-lo-posso falhar. Se o fizer, posso destruir os cartões, o item ou ambos. Eu não dou reembolsos. Dito isto, o que é mais importante para você: os cartões ou o item?";
		next;
		switch(select("Eu mudei de idéia sobre isso .: o item .: Os cartões.")) {
		case 1:
			mes "[Idosa sábia]";
			mes "Muito bem. Retorne imediatamente se você procurar meus serviços.";
			close;
		case 2:
			set .@failtype,1;
			break;
		case 3:
			set .@failtype,2;
		}
		mes "[Idosa sábia]";
		mes "Muito bem. Eu começarei.";
		set Zeny, Zeny - (.zenycost+(.@cardcount * .percardcost));
		delitem 1000,1; //Star_Crumb
		delitem 715,1; //Yellow_Gemstone
		
		// Replace the constants in the next 3 lines with failure chance values defined in refine_db.txt
		// First value = Total failure chance (item and cards destroyed)
		// Second value = Partial failure chance (one or the other is destroyed, player decides which one is safe)
		// Third value = Harmless failure chance (all that's lost is your investment)

		set .@failchance,rand(100);
		if (.faildestroy==1) {
			if(.@failchance < 2) {
				next;
				failedremovecards .@part,0;
				mes "[Idosa sábia]";
				mes "O processo foi uma falha total. Receio que o item e os cartões tenham sido destruídos.";
				close;
			}

			if(.@failchance < 8) {
				if (.@failtype == 1) {
					next;
					failedremovecards .@part,1;
					mes "[Idosa sábia]";
					mes "Embora eu tenha conseguido remover os cartões do item, eles foram destruídos no processo. O item, no entanto, está bem.";
					close;
				}

				if (.@failtype == 2) {
					next;
					failedremovecards .@part,2;
					mes "[Idosa sábia]";
					mes "Mais lamentável. Consegui remover os cartões, mas o próprio item foi destruído no processo.";
					close;
				}
			}
		}

		if(.@failchance < 10) {
			next;
			failedremovecards .@part,3;
			mes "[Idosa sábia]";
			mes "Eu não consegui remover os cartões. Felizmente, no entanto, o item e os cartões ainda estão bem.";
			close;
		}
		next;
		successremovecards .@part;
		mes "[Idosa sábia]";
		mes "O processo foi um sucesso. Aqui estão seus cartões e seu item. Até a próxima.";
		close;
	case 2:
		mes "[Idosa sábia]";
		mes "I charge a flat fee of "+callfunc("F_InsertComma",.zenycost)+" zeny, plus "+callfunc("F_InsertComma",.percardcost)+" zeny for each card I remove from the item. In addition, I need a star crumb and a yellow gemstone to work my magic.";
		close;
	case 3:
		mes "[Idosa sábia]";
		mes "Muito bem. Retorne imediatamente se você procurar meus serviços.";
		close;
	}
}
