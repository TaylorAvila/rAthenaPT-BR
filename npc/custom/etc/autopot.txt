//===== rAthena Script =======================================
//= Auto-Potion
//===== Description: =========================================
//= Provides an @autopot command to automatically use potions.
//===== Changelogs: ==========================================
//= 1.0 First version. [Euphy]
//= 2.0 rewrite and adding SP option. [sader1992]
//============================================================
-	script	#autopot	-1,{
OnInit:
	.HP_Option = true;	//Allow HP option
	.SP_Option = true;	//Allow SP option
	.MinDelay = 100;	//Minimum Delay Allowed in milliseconds (Default and Recommended: 100) [lower values will increase server strain]
	.MaxDelay = 1000;	//Maximum Delay Allowed in milliseconds (Default and Recommended: 1000)
	setarray .blackList[0],0;	//Array of black listed item ID

	if(!.HP_Option && !.SP_Option){
		debugmes "Aviso: @Autopot HP e SP estão desativados";
		debugmes "Aviso: o script @Autopot foi desativado!";
		end;
	}
	bindatcmd("autopot",strnpcinfo(0)+"::OnCommand",0,99);
end;

L_Help:
	dispbottom "Comandos disponíveis:";
	if(.HP_Option && .SP_Option){
		dispbottom "@AUPOPOT <hp | sp> <item id> {<min hp | sp % [1..100]> {<atraso" + .MinDelay + ".."+.MaxDelay+"]>}}";
		dispbottom "@AUTOPOT <hp | sp> lista";
		dispbottom "@Autopot <hp | sp> <on | off>";
	}else{
		dispbottom "@Autopot <" + (.HP_Option?"HP":"sp") + "> <item id> {<min" + (.HP_Option?"HP":"sp") + "% [1..100]> {<Atraso [" + .MinDelay + ".."+.MaxDelay+"]>}}";
		dispbottom "@Autopot <" + (.HP_Option?"HP":"sp") + "> lista";
		dispbottom "@Autopot <" + (.HP_Option?"HP":"sp") + "> <On | Off>";
	}
	dispbottom "@Autopot Info";
	dispbottom "@autopot ajuda";
	dispbottom "@Autopot Blacklist";
	return;

L_Info:
	dispbottom "---------- Informações sobre Potion Auto -----------";
	if(.HP_Option){
		dispbottom "Poção HP:" + getitemname(@autopot_hp_ID) + "(" + @autopot_hp_ID + ")";
		dispbottom "Min HP:" + @autopot_hp_Min + "%";
	}
	if(.SP_Option){
		dispbottom "SP POTION:" + getitemname(@autopot_sp_ID) + "(" + @autopot_sp_ID + ")";
		dispbottom "MIN SP:" + @autopot_sp_Min + "%";
	}
	dispbottom "ATRASO:" + @autopot_delay + "EM";
	dispbottom "---------------------------------------------";
	return;

L_Start:
	deltimer strnpcinfo(3) + ":: OnStart";
	.@type$  = getarg(0,0);
	.@potion = getarg(1,0);
	setd "@Autopot_" + .@type$ + "_Min",getarg(2,0);
	@autopot_delay = getarg(3,.MinDelay);

	if (getd("@Autopot_" + .@type$ + "_Min")   < 1  || getd("@Autopot_" + .@type$ + "_Min") > 100)  setd "@Autopot_" + .@type$ + "_Min",90;
	if (@autopot_delay < .MinDelay) @autopot_delay = .MinDelay;
	if (@autopot_delay > .MaxDelay) @autopot_delay = .MaxDelay;
	if (getiteminfo(.@potion, ITEMINFO_TYPE) != IT_HEALING) {
		dispbottom getitemname(.@potion) + "não é um item de cura.";
		end;
	}
	if(inarray(.blackList[0],.@potion) != -1){
		message strcharinfo(0), "'" + getitemname(.@potion) + "'está listado preto e não pode ser usado.";
		end;
	}
	if (BaseLevel < getiteminfo(.@potion, ITEMINFO_EQUIPLEVELMIN)) {
		message strcharinfo(0), "Seu nível base é muito baixo para usar '" + getitemname(.@potion) + "'.";
		end;
	}
	setd "@Autopot_" + .@type$ + "_EU IA",.@potion;
	setd "@Autopot_" + .@type$ + "_Active",true;
	addtimer @autopot_delay,strnpcinfo(3) + ":: OnStart";
	message strcharinfo(0), "Potion automático" + strtoupper(.@type$) + "habilitado.";
	callsub L_Info;
	return;

OnCommand:
	if (!getarraysize(.@atcmd_parameters$)) {
		message strcharinfo(0), "Sintaxe inválida.";
		callsub L_Help;
		end;
	}
	.@command$[0] = strtolower(.@atcmd_parameters$[0]);
	.@command$[1] = strtolower(.@atcmd_parameters$[1]);
	
	if(.@command$[0] == "HP" || .@command$[0] == "sp"){
		if((!.HP_Option && .@command$[0] == "HP") || (!.SP_Option && .@command$[0] == "sp")){
			message strcharinfo(0), "Potion automático" + strtoupper(.@command$[0]) + "está desativado do servidor.";
			end;
		}
		if(.@command$[1] == "sobre"){
			if(getd("@Autopot_" + .@command$[0] + "_Active")){
				message strcharinfo(0), "Potion automático" + strtoupper(.@command$[0]) + "já está ligado.";
				end;
			}
			if(!getd("@Autopot_" + .@command$[0] + "_EU IA")){
				dispbottom "Você precisa definir um" + strtoupper(.@command$[0]) + "Poção a ser usada.";
				callsub L_Help;
				end;
			}
			callsub L_Start,.@command$[0],getd("@Autopot_" + .@command$[0] + "_EU IA");
			end;
		}
		if(.@command$[1] == "desligado"){
			message strcharinfo(0), "Potion automático" + strtoupper(.@command$[0]) + "desabilitado.";
			setd "@Autopot_" + .@command$[0] + "_Active",false;
			if(!@autopot_hp_Active || !@autopot_sp_Active)
				deltimer strnpcinfo(3) + ":: OnStart";
			end;
		}
		if(.@command$[1] == "lista"){
			getinventorylist;
			for(.@i = 0;.@i < @inventorylist_count;.@i++){
				if (getiteminfo(@inventorylist_id[.@i], ITEMINFO_TYPE) == IT_HEALING) {
					.@items[.@count] = @inventorylist_id[.@i];
					.@menu$ = .@menu$ + sprintf("~ ^0055ff%s ^000000 (%dx)" + (inarray(.blackList[0],@inventorylist_id[.@i]) != -1? "^364022black listado^000000":"") + (BaseLevel < getiteminfo(@inventorylist_id[.@i], ITEMINFO_EQUIPLEVELMIN)? "^E82C0CNEED Nível mais alto^000000":"") + ":", getitemname(@inventorylist_id[.@i]), countitem(@inventorylist_id[.@i]));
					.@count++;
				}
			}
			if(.@count){
				mes "[Potion automático]";
				mes "Selecione a" + strtoupper(.@command$[0]) + "item.";
				.@select = select(.@menu$ + "^Hahakknak^000") - 1;
				if (.@select != .@count)
					callsub L_Start,.@command$[0],.@items[.@select];
				close2;
			}else{
				message strcharinfo(0), "Não há itens de cura em seu inventário.";
			}
			end;
		}
		.@potion = atoi(.@command$[1]);
		.@percent = atoi(.@atcmd_parameters$[2]);
		.@delay = atoi(.@atcmd_parameters$[3]);
		if(getitemname(.@potion) != "nulo"){
			callsub L_Start,.@command$[0],.@potion,.@percent,.@delay;
			end;
		}
		message strcharinfo(0), "Auto-Potion: Algo deu errado!";
		callsub L_Help;
		end;
	}
	if(.@command$[0] == "Lista negra"){
		dispbottom "-------------- Itens listados em preto ----------------";
		for(.@i=0;.@i<getarraysize(.blackList);.@i++){
			dispbottom "" + getitemname(.blackList[.@i]) + "(" + .blackList[.@i] + ")";
		}
		dispbottom "---------------------------------------------";
		end;
	}
	if(.@command$[0] == "informações"){
		if(@autopot_hp_Active || @autopot_sp_Active){
			message strcharinfo(0), "As informações de pós automáticas são exibidas abaixo.";
			callsub L_Info;
		}else{
			message strcharinfo(0), "O Potion Auto não está ativado.";
		}
		end;
	}
	if(.@command$[0] == "ajuda"){
		message strcharinfo(0), "A lista de comandos é exibida abaixo.";
		callsub L_Help;
		end;
	}
	
	message strcharinfo(0), "Sintaxe inválida.";
	callsub L_Help;
end;

OnStart:
	if (!getstatus(SC_BERSERK) && !getstatus(SC_SATURDAYNIGHTFEVER) && !getstatus(SC_GRAVITATION) &&
	    !getstatus(SC_TRICKDEAD) && !getstatus(SC_HIDING) && !getstatus(SC__SHADOWFORM) && !getstatus(SC__INVISIBILITY) &&
	    !getstatus(SC__MANHOLE) && !getstatus(SC_KAGEHUMI) && !getstatus(SC_HEAT_BARREL_AFTER) &&
		!getstatus(SC_STONE) && !getstatus(SC_FREEZE) && !getstatus(SC_STUN) && !getstatus(SC_SLEEP))
		{
		if(Hp){
			if(@autopot_hp_Active && Hp * 100 / MaxHp < @autopot_hp_Min) {
				if (countitem(@autopot_hp_ID)) {
					delitem @autopot_hp_ID,1;
					consumeitem @autopot_hp_ID;
					specialeffect2 207;
				}
			}
			if(@autopot_sp_Active && Sp * 100 / MaxSp < @autopot_sp_Min) {
				if (countitem(@autopot_sp_ID)) {
					delitem @autopot_sp_ID,1;
					consumeitem @autopot_sp_ID;
					specialeffect2 208;
				}
			}
		}
	}
	if(@autopot_hp_Active || @autopot_sp_Active)
		addtimer @autopot_delay,strnpcinfo(3) + ":: OnStart";
end;
}
