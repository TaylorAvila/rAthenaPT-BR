//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Duplo para baixo".
//= Currently does not allow for "seguro", or "dividindo"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	Black Jack Dealer	57,{
	mes "[Distribuidor]";
	mes "Olá! Você gostaria de jogar um jogo de Black Jack?";
M_Menu:
	next;
	menu "Quais são as regras?",M_0, "Eu quero jogar.",L_Play, "Não, obrigado.",M_End;

M_0:
	mes "[Distribuidor]";
	mes "O que você gostaria de saber?";
sM_Menu0:
	next;
	menu "O básico.",sM_0a, "Ganhar e perder.",sM_0b, "'Double Down'.",sM_0c,
	"Valor da ás.",sM_0d, "Nada.",M_Menu;

sM_0a:
	mes "[Distribuidor]";
	mes "Black Jack é um jogo de cartas no qual o objetivo é obter 21 ou como";
	mes "Perto de 21 pontos possível, sem passar mais de 21 anos.";
	next;
	mes "[Distribuidor]";
	mes "Antes do início do jogo, o jogador deve fazer uma aposta.  Uma vez a aposta";
	mes "foi feito, tanto o revendedor quanto o jogador recebem 2 cartas";
	mes "cada.  Dependendo de quais cartões você tem, você pode optar por ^5533ff'Stay ' ^000000,";
	mes "ou você pode optar por ^5533ff'pull ' ^000000.";
	next;
	mes "[Distribuidor]";
	mes "Quando você escolhe 'ficar', você está dizendo ao revendedor que não";
	mes "Precisa de cartas mais.  Isso permite que o revendedor puxe se ele/ela/ela";
	mes "quer.  Para ficar, você deve ter ^ff3355T pelo menos 13 pontos ^000000. O";
	mes "O revendedor só pode ficar quando ele tem^ff3355 17 pontos ou mais^000000.";
	next;
	mes "[Distribuidor]";
	mes "Quando você escolhe 'puxar', você está dizendo ao revendedor que deseja";
	mes "Outra carta.  Ao puxar mais cartas, você pode aumentar seu ponto";
	mes "total.  Desde que você tenha ^ff3355 sem 21 ^000000 pontos, você pode puxar um";
	mes "Cartão do baralho.";
	goto sM_Menu0;
sM_0b:
	mes "[Distribuidor]";
	mes "Existem três maneiras de vencer e perder no Blackjack.";
	next;
	mes "[Distribuidor]";
	mes "1.) No final de uma rodada de Black Jack, se você tiver mais pontos";
	mes "do que o revendedor você ganhará a rodada.  Por outro lado, se você tiver";
	mes "Menos pontos do que o revendedor que você perderá.";
	next;
	mes "[Distribuidor]";
	mes "2.) Se você puxar um cartão que torne seu ponto total, vá mais de 21 você";
	mes "vai perder automaticamente a rodada.  Isso é chamado de ^5533ff'bust ' ^000000.  Se";
	mes "O revendedor quebra então você ganhará a rodada.";
	next;
	mes "[Distribuidor]";
	mes "3.) Se você tiver um total de 21 com os 2 primeiros cartões, você";
	mes "Ganhará automaticamente a rodada.  Isso é chamado de ^5533ff'black jack ' ^000000";
	mes "e acontece quando você recebe um cartão 'Ace' e um '10 valorizado '. Se o";
	mes "O revendedor recebe um jack preto que ele/ela ganhará automaticamente a rodada.";
	next;
	mes "[Distribuidor]";
	mes "4.) Além de ganhar e perder, você pode amarrar com o revendedor. Se";
	mes "Tanto você quanto o revendedor têm o mesmo ponto total no final de um";
	mes "Rodada, isso resultará em um empate sem vencedor e sem perda ou ganho";
	mes "em dinheiro.";
	next;
	mes "[Distribuidor]";
	mes "Isso é chamado de ^5533ff'push ' ^000000 com o revendedor.  Isso também";
	mes "aplica -se a você e ao revendedor";
	mes "tempo.";
	goto sM_Menu0;
sM_0c:
	mes "[Distribuidor]";
	mes "A opção 'Double Down' permite dobrar sua aposta atual,";
	mes "Mas com a desvantagem de que você só poderá puxar um";
	mes "cartão adicional.  Esta opção está disponível apenas no início de";
	mes "cada rodada.";
	next;
	mes "[Distribuidor]";
	mes "Um exemplo de quando dobrar é útil é quando";
	mes "Seus dois primeiros cartões lhe dão um total de 11. Você tem um bom";
	mes "chance de obter 21 ou 20 com o próximo cartão que você desenha.  Esse";
	mes "Seria uma boa mão para dobrar.";
	goto sM_Menu0;
sM_0d:
	mes "[Distribuidor]";
	mes "A carta 'Ace' é uma carta única no jogo de Black Jack porque";
	mes "Pode ter 2 valores.  Um ás pode ser contado como 11 pontos,";
	mes "ou apenas 1 ponto.  Por exemplo, se você tivesse um ás e um 4, isso seria";
	mes "Dê a você 15 ou 5 pontos.";
	next;
	mes "[Distribuidor]";
	mes "Se você decidisse ficar, o ás seria automaticamente contado como";
	mes "11 pontos para dar a você 15 pontos no total.";
	next;
	mes "[Distribuidor]";
	mes "Se você tivesse decidido puxar e receber um 9, o ás";
	mes "ser considerado automaticamente como 1 ponto para lhe dar um total de 14";
	mes "pontos.  Se o ás fosse contado como 11 pontos, você teria um";
	mes "Point Total acima de 21 e teria perdido.";
	next;
	mes "[Distribuidor]";
	mes "É por causa da flexibilidade que você tem com o 'ás' que faz";
	mes "É a carta mais poderosa do jogo.";
	goto sM_Menu0;
M_End:
	mes "[Distribuidor]";
	mes "Sinta -se à vontade para voltar a qualquer hora";
	close;

//================
L_Play:
	mes "[Distribuidor]";
	mes "Por favor, faça suas apostas ...";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "Muito rico para o meu sangue ....",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[Distribuidor]";
	mes "Sinto muito, mas você não tem Zeny suficiente para fazer essa aposta.";
	close;

L_Cont0:
	mes "(os cartões estão sendo tratados)";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- Aqui estão os cartões ^000000 do ^ff55333Dealer:";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "O revendedor tem: ^ff5533"+@dealTot+"^000000";
	mes "";
	mes "- Aqui estão ^5533ffyour ^000000 cartões:";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "Você tem: ^5533ff" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "Você tem: ^5533ff" +@playTot+ "^000000, ou ^5533ff" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Bata em mim (puxe)",M_Hit, "Ficar",M_Stay, "Duplo para baixo",M_Double;
	if(@dealerTurn == 0) menu "Bata em mim (puxe)",M_Hit, "Ficar",M_Stay;

M_Stay:
	mes "[Distribuidor]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "O revendedor fica.";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "O revendedor vai puxar";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "Sinto muito, mas você não tem um total suficientemente alto para ficar.  Você deve puxar.";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[Distribuidor]";
	mes "O jogador optou por dobrar.  Sua aposta atual será";
	mes "dobrou, e você só poderá puxar 1 cartão extra.";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[Distribuidor]";
	mes "Parabéns, você ganhou!";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[Distribuidor]";
	mes "Sinto muito, mas você perdeu.";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[Distribuidor]";
	mes "É um empurrão.  Você amarrou com o revendedor.";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), "" + getarg(1) + "";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), "UM";
	if(@rnd == 11) set getarg(2), "J";
	if(@rnd == 12) set getarg(2), "Q";
	if(@rnd == 13) set getarg(2), "K";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes ".-----.   .-----.";
	mes "|"+@dealCard$[0]+"|   |"+@dealCard$[1]+"|";
	mes "'-----'   '-----'";
	return;
sF_P2cards:
	mes ".-----.   .-----.";
	mes "|"+@playCard$[0]+"|   |"+@playCard$[1]+"|";
	mes "'-----'   '-----'";
	return;
sF_D3cards:
	mes ".-----.   .-----.   .-----.";
	mes "|"+@dealCard$[0]+"|   |"+@dealCard$[1]+"|   |"+@dealCard$[2]+"|";
	mes "'-----'   '-----'    '-----'";
	return;
sF_P3cards:
	mes ".-----.   .-----.   .-----.";
	mes "|"+@playCard$[0]+"|   |"+@playCard$[1]+"|   |"+@playCard$[2]+"|";
	mes "'-----'   '-----'    '-----'";
	return;
sF_D4cards:
	mes ".-----.  .-----.  .-----.";
	mes "|"+@dealCard$[0]+"|  |"+@dealCard$[1]+"|  |"+@dealCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes ".-----.";
	mes "|"+@dealCard$[3]+"|";
	mes "'-----'";
	return;
sF_P4cards:
	mes ".-----.  .-----.  .-----.";
	mes "|"+@playCard$[0]+"|  |"+@playCard$[1]+"|  |"+@playCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes ".-----.";
	mes "|"+@playCard$[3]+"|";
	mes "'-----'";
	return;
sF_D5cards:
	mes ".-----.  .-----.  .-----.";
	mes "|"+@dealCard$[0]+"|  |"+@dealCard$[1]+"|  |"+@dealCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes ".-----.   .-----.";
	mes "|"+@dealCard$[3]+"|   |"+@dealCard$[4]+"|";
	mes "'-----'   '-----'";
	return;
sF_P5cards:
	mes ".-----.  .-----.  .-----.";
	mes "|"+@playCard$[0]+"|  |"+@playCard$[1]+"|  |"+@playCard$[2]+"|";
	mes "'-----'  '-----'   '-----'";
	mes ".-----.   .-----.";
	mes "|"+@playCard$[3]+"|   |"+@playCard$[4]+"|";
	mes "'-----'   '-----'";
	return;
}
