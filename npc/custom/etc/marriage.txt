function SF_Groom {
	if ($wed_bride_progress == 0) {
		mes "["+@name$+"]";
		mes "Parece que sua noiva ainda não chegou e se registrou.";
		close;
	}
	if (SF_equip_check() == 0)
		close;

	switch($wed_groom_progress) {
	case 1:
		SF_AcceptBride();
		break;
	case 2:
		mes "["+@name$+"]";
		mes "Estou esperando que seu parceiro aceite que você inicie a cerimônia.";
		close;
	case 3:
		SF_RetrieveRingM();
		break;
	case 4:
		mes "["+@name$+"]";
		mes "O anel de casamento do seu parceiro ainda não foi recuperado. A cerimônia começará assim que vocês dois reivindicaram seus anéis.";
		close;
	case 5:
		mes "["+@name$+"]";
		SF_StartCeremony();
		break;
	default:
		mes "["+@name$+"]";
		mes "Por favor, não me interrompa agora.";
		close;
	}
}

function SF_AcceptGroom {
	mes "["+@name$+"]";

	next;

	case 1:
		mes "["+@name$+"]";
		mes "Você o quê!?";
		mes "err .. * tosse * * tosse * muito bem ... volte depois de se decidir.";
		emotion ET_ANGER;
		close;
	case 2:
		mes "["+@name$+"]";
		mes "!!";
		mes "Ah ... err ... ehm ... ok. Vocês dois parecem ter algumas diferenças para se estabelecer primeiro.";
		close2;
		emotion ET_HUK;
		npctalk @name$ +": Senhoras e senhores,"+$wed_bride$+" has rejected to marry "+$wed_groom$+"!";
		SF_wed_end();
		break;
	case 3:
		set $wed_bride_progress,2;
		if ($wed_groom_progress == 2) {
			SF_RingsAccepted();
			break;
		}
		emotion ET_OK;
		mes "["+@name$+"]";
		mes "Depois que seu noivo aprovar, você receberá seus anéis, a cerimônia começará e você será oficialmente casado.";
		close;
	}
}

function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "O que aconteceu com o seu"+getitemname($@wed_ring)+"? Você não perdeu ... Não é? Precisamos continuar com a cerimônia!";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;	//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "Você não parece ter espaço suficiente para carregar o anel ... Vá libertar um pouco de espaço e volte para recuperar o anel do seu parceiro.";
		close;
	}
	mes "Aqui está o anel de casamento para sua noiva.";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "Quando sua noiva recuperar o anel, a cerimônia começará.";
		close;
	}
}

function SF_StartCeremony {
	mes "Agora vou começar a cerimônia de casamento e você será declarado marido e mulher.";
	set $wed_bride_progress,5;
	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;
	if (marriage($@msg$) == 0) {
		next;
		mes "["+@name$+"]";
		mes "Onde está"+$@msg$+"? Não posso me casar com vocês dois se estiver faltando ...";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "Criança, o que você fez com o seu"+getitemname(7170)+"?";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "Criança, você deveria usar um"+getitemname(2338)+"em todos os momentos durante a cerimônia ...";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Criança, você não pode tirar o seu"+getitemname(2206)+"ainda....";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

	function SF_WedProgress;
	function SF_Principles;
	function SF_Procedure;
	function SF_Register;
	function SF_TryRegister;

	if ($wed_progress) {
		SF_WedProgress();
		end;
	}

		mes "["+@name$+"]";
		mes "Casamento ... é uma coisa tão linda.";
		mes "Você gostaria de se casar com alguém?";
		next;
		set @menu, select(
			"Eu estarei solteiro para sempre!",
			"Explique os princípios do casamento.",
			"Explique o procedimento de casamento.",
			"Eu quero me casar com alguém."
		);
		switch (@menu) {
		case 1: //Quit
			mes "["+@name$+"]";
			mes "Nesse caso, aproveite a vida de seu bacharel.";
			close;
		case 2: //Principles
			SF_Principles();
			break;
		case 3: //Procedure
			SF_Procedure();
			break;
		case 4: //Register
			SF_Register();
			break;
		}
	} while (@menu > 1);
	end;

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "Muito bem, de quem você gostaria de se registrar?";
		next;
		set @submenu, select("Noivo","Noiva","Cancelar");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "Muito bem, você se registrará como o noivo?";
		next;
		if (select("Sim","Eu mudei de idéia.")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "Muito bem, você se registrará como a noiva?";
		next;
		if (select("Sim","Eu mudei de idéia.")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Agora você está registrado como o noivo.";
		mes "Diga à sua noiva para se registrar o mais rápido possível.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "Agora você está registrado como a noiva.";
		mes "Diga ao seu noivo para se registrar o mais rápido possível.";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "Volte quando estiver pronto.";
		close;
	}
}

//Subfunction SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "noivo";
	if (@bride)
		set @type$, "noiva";
		
	mes "["+@name$+"]";
	mes "Antes de se registrar como"+@type$+", deixe -me verificar se você atende a todos os requisitos ...";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "Oh querido, você é jovem demais para pensar em casamento!";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "Você deveria estar usando um"+getitemname(@item)+"Se você quiser se casar.";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "Onde está o seu"+getitemname(2206)+"? É um complemento necessário para o seu vestido.";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "Onde está o anel? Você precisa de um"+getitemname($@wed_ring)+"Para a troca do anel, querida.";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else 
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "Sinto muito, mas você não tem o suficiente para pagar pela taxa de registro.";
		mes "Volte depois de coletar"+@cost+"Com."; 
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}
