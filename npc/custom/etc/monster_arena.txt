	set #monpoints,#monpoints-.@sellexp;

	set Zeny, Zeny+.@sellearn;
	mes "[Monster Trainer]";
	mes "Você ganhou ^0000ff" + .@sellearn + "^000000Z.";
	close;

	menu "Continuar",-,"Cancelar",L_Exit;
	mes "[Monster Trainer]";
