//Counts down and starts race
pvp_y_1-5,145,269,5	script	Starter	733,{
	if ($@race != 0) goto L_Started;
	if ($@counting != 0) goto L_Started;
	if ($@racecount == 1) goto L_Started;
L_Menu:
	mes "[Starter de corrida]";
	mes "Por favor, fique no meu lado leste.";
	menu "Start Race",L_Count,"Cancelar",-;
	close;
L_Count:
	set $@counting,1;
	mes "Contando o Down ...";
	addtimer 1000, "Starter :: oncount1000";
	addtimer 2000, "Starter :: oncount2000";
	addtimer 3000, "Starter :: oncount3000";
	addtimer 4000, "Starter :: oncount4000";
	announce strcharinfo(0) + "Iniciou uma contagem regressiva",1;
	announce "Prepare -se para correr!",1;
	close;

OnCount1000:
	announce "[3]",1;
	end;
OnCount2000:
	announce "[2]",1;
	end;
OnCount3000:
	announce "[1]",1;
	end;
OnCount4000:
	emotion ET_GO;
	specialeffect EF_CHIMTO;
	announce "[IR!]",1;
	set $@race,1;
	set $@position,0;
	set $@counting,0;
	set $@raceid,rand(100000,999999);
	end;

L_Started:
	if ((getmapusers("pvp_y_1-5") < 3) && ($@position > 0)) goto L_Menu;
	if ($@position > 2) goto L_Menu;
	mes "[Iniciante]";
	mes "Raça em andamento";
	close;

OnInit:
	set $@race,0;
	set $@position,0;
	set $@racecount,0;
	end;
}

//Checkpoint 1
pvp_y_1-5,144,262,5	script	Check Point 1	111,0,5,{
	end;
OnTouch:
	if (@raceid != $@raceid) goto L_Started;
	if (@race == 6) goto L_Finished;
	if ($@race == 1) goto L_Started;
	mes "A corrida não começou, por favor volte.";
	close;
L_Started:
	set @race,1;
	set @raceid,$@raceid;
	end;
L_Finished:
	mes "Você já completou a corrida.";
	close;
}

pvp_y_1-5,174,244,5	script	Finish Line	111,6,6,{
	end;
OnTouch:
	if (@raceid != $@raceid) goto L_WrongRace;
	if (@race != 5) goto L_Miss;
	set @race,6;
	set $@position,$@position+1;
	announce "[" + strcharinfo(0) +"] chegou à linha de chegada! [Posição:" + $@position + "]",1;
	end;
L_Miss:
	mes "Você perdeu um ponto de verificação. Por favor, volte.";
	close;
L_WrongRace:
	mes "Você não está nesta corrida.";
	close;
}
