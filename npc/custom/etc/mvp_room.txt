prontera,148,174,5	script	Private MVP Room	100,{
	mes "[Sala MVP privada]";
	mes "Selecione uma sala privada de MVP.";
	if ( getvariableofnpc( .rentcost, "MVP Summoner" ) )
		mes "The cost to rent a room for "+ getvariableofnpc( .timeout, "MVP Summoner" ) +" minutes is "+ callfunc("F_InsertComma", getvariableofnpc( .rentcost, "MVP Summoner" ) ) +" zeny.";
	else
		mes "You can only use the room for only "+ getvariableofnpc( .timeout, "MVP Summoner" ) +" minutes.";
	mes "";
	for ( .@i = 1; .@i <= 8; .@i++ )
		if ( getvariableofnpc( .renttime[.@i], "MVP Summoner" ) )
			mes "Sala #"+ .@i +"="+ .color$[ .type[.@i] ] + .whoinuse$[.@i] +"^000000";
	next;
	.@room = select(
		"MVP Room 1 ["+ getmapusers("06Guild_01") +"]",
		"MVP Room 2 ["+ getmapusers("06Guild_02") +"]",
		"MVP Room 3 ["+ getmapusers("06Guild_03") +"]",
		"MVP Room 4 ["+ getmapusers("06Guild_04") +"]",
		"MVP Room 5 ["+ getmapusers("06Guild_05") +"]",
		"MVP Room 6 ["+ getmapusers("06Guild_06") +"]",
		"Sala de MVP 7 ["+ getmapusers("06Guild_07") +"]",
		"MVP Room 8 ["+ getmapusers("06Guild_08") +"]");
	if ( getvariableofnpc( .renttime[.@room], "MVP Summoner" ) ) {
		if ( .inuseid[.@room] == getcharid( .type[.@room] ) ) {
			warp "06guild_0"+ .@room, 0,0;
			close;
		} else {
			mes "[Sala MVP privada]";
			mes "Esta sala é reservada para";
			mes .color$[ .type[.@room] ] + .whoinuse$[.@room] +"^000000.";
			mes "Selecione outro.";
			close;
		}
	}
	mes "[Sala MVP privada]";
	mes "Reserve este quarto para ...";
	next;
	set .@type, select( "Para os membros do meu partido", "Para meus membros da guilda", "Para uso da conta pessoal" );
	if ( !getcharid(.@type) ) {
		mes "[Sala MVP privada]";
		mes "Você não possui um"+( ( .@type == 1 )? "Festa" : "Guilda" )+".";
		close;
	}
	else if ( Zeny < getvariableofnpc( .rentcost, "MVP Summoner" ) ) {
		mes "Você não tem Zeny suficiente para alugar um quarto.";
		close;
	}
	else if ( getvariableofnpc( .renttime[.@room], "MVP Summoner" ) ) {
		mes "[Sala MVP privada]";
		mes "Sinto muito, alguém já registrou esta sala mais rápido que você.";
		close;
	}
	for ( .@i = 1; .@i <= 8; .@i++ ) {
		if ( ( getvariableofnpc( .renttime[.@i], "MVP Summoner" ) ) && .@type == .type[.@i] && getcharid(.@type) == .inuseid[.@i] ) {
			mes "[Sala MVP privada]";
			mes "Você já alugou o quarto#"+ .@i +". Use essa sala.";
			close;
		}
	}
	set .type[.@room], .@type;
	set .inuseid[.@room], getcharid(.@type);
	set .whoinuse$[.@room], strcharinfo( ( .@type == 3 )? 0 : .@type );
	Zeny -= getvariableofnpc( .rentcost, "MVP Summoner" );
	warp "06guild_0"+ .@room, 0,0;
	killmonsterall "06guild_0"+ .@room;
	donpcevent "MVP Summoner#"+ .@room +"::OnEnterMap";
	close;
OnInit:
	.color$[1] =  "^EE8800"; // party color
	.color$[2] =  "^70CC11"; // guild color
	.color$[3] =  "^0000FF"; // account color
	end;
}

			mes "[MVP Invocador]";
			mes "Não posso convocar outro mini-chefe quando há monstros por perto.";
			close;
		} else if ( .bosscost ) {
			mes "[MVP Invocador]";
			mes "The cost to summon a mini-boss is "+ callfunc( "F_InsertComma", .bosscost ) +" zeny.";
			next;
			set .@menu, select(.bossid_menu$) -1;
			mes "[MVP Invocador]";
			if ( Zeny < .bosscost ) {
				mes "Você não tem Zeny suficiente para convocar um mini-chefe.";
				close;
			}
		} else {
			set .@menu, select(.bossid_menu$) -1;
			mes "[MVP Invocador]";
		}
		mes "Por favor, prepare -se.";
		close2;
		if ( Zeny < .bosscost ) end;
		Zeny -= .bosscost;
		monster "esse", 0, 0, "--- e--", .bossid[.@menu], 1, strnpcinfo(3)+":: OnMobdead";
		end;
	case 4:
		if ( mobcount( "esse", strnpcinfo(3)+":: OnMobdead" ) > 0 ) {
			mes "[MVP Invocador]";
			mes "Não posso oferecer esse serviço quando há monstros por perto.";
			close;
		}
		close2;
		callshop "MVP Sala#Branch", 1;
		end;
	case 5:
		mes "[MVP Invocador]";
		mes "Tem certeza de que deseja sair desta sala?";
		next;
		if ( select( "Sim: Não") == 1 )
			warp .respawnmap$, .respawnx, .respawny;
		close;
	case 6:
		mes "[MVP Invocador]";
		if ( getmapusers( strcharinfo(3) ) > 1 ) {
			mes "Ainda existem alguns jogadores nesta sala. Verifique se você é o último membro nesta sala a usar esta opção.";
			close;
		}
		mes "Tem certeza de que deseja desistir deste quarto?";
		if ( .rentcost )
			mes "Você precisará pagar novamente para entrar nesta sala.";
		next;
		if ( select( "Sim", "No" ) == 2 ) close;
		awake strnpcinfo(0);
		end;
	}
	close;

OnMobDead:
	end;

	.@id = atoi( strnpcinfo(2) );
	.renttime[.@id] = gettimetick(2);
	sleep .timeout * 60000;
	mapwarp strnpcinfo(4), .respawnmap$, .respawnx, .respawny;
	.renttime[.@id] = 0;
	killmonsterall strnpcinfo(4);
	end;
