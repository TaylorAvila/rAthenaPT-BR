sec_pri,36,58,1	script	Chief Warder	105,{ 
	mes "[Saddeus]";
	emotion ET_QUESTION;
	mes (Sex==SEX_MALE?"Senhor.":"EM.")+"Prisioneiro #"+BaseLevel+JobLevel+",";
	mes "Qual é o barulho ali?";
	next;
	menu "Deixe -me livre, eu vou pagar!",-, "Deixe -me trabalhar com minha liberdade!",M_Q1, "Nada",M_NO_THANKS;

M_PAY:
	mes "[Saddeus]";
	set .@MUSTPAY,(PRISON+1)*1000000;
	if (.@MUSTPAY<1000000) set .@MUSTPAY,1000000;
	if (.@MUSTPAY>100000000) set .@MUSTPAY,100000000;
	if (PRISON==0) mes "Não me lembro de você. É a sua primeira vez?";
	if (PRISON>0) mes "Você de novo? Alguém não ficou mais sábio desde a primeira visita ...";
	mes "Ok, poderíamos libertá -lo para ^ff0000"+.@MUSTPAY+"Z^000000.";
	next;
	menu "Eu pago em dinheiro!",M_PAYCASH,"Você aceita cheques?",M_PAYBANK,"Eu mudei de idéia.",-;

	mes "[Saddeus]";
	mes "Você tem algum tempo para pensar ...";
	close;

M_PAYCASH:
	if (.@MUSTPAY>Zeny) goto L_NOCASH;
	set Zeny,Zeny-.@MUSTPAY;
	mes "[Saddeus]";
	mes "Ok, cante aqui e ali.";
	goto L_RELEASE;

L_NOCASH:
	mes "[Saddeus]";
	mes "O que é isso? Não é suficiente!";
	close;

M_PAYBANK:
	if (.@MUSTPAY>#kafrabank) goto L_NOBANK;
	set #kafrabank,#kafrabank-.@MUSTPAY;
	mes "[Saddeus]";
	mes "OK, assine seu cheque. E abaixe seu nome no meu livro.";
	goto L_RELEASE;

L_NOBANK:
	mes "[Saddeus]";
	if (#kafrabank==0) mes "Seu caso não diz nada sobre suas contas bancárias.";
	if (#kafrabank!=0) mes "Infelizmente, aí está as informações da sua conta bancária ... ^FF0000"+#kafrabank+"Z^000000 ... No caso. Não é suficiente.";
	mes "Pare seus jogos bobos agora!";
	close;

L_RELEASE:
	set PRISON,PRISON+1;
	next;
	mes "[Saddeus]";
	mes "Você está livre agora!";
	next;
	savepoint "chocado",105,112;
	warp "izlude",105,112;
	close;

M_Q1:
	mes "[Saddeus]";
	mes "Nossas ações estão com falta de saques.  Lembra -se da habilidade da sua manhã?  Que tal algum trabalho em nossa fazenda de cogumelos patrocinados?";
	next;
	menu "Ok, vou trabalhar!",-, "Eu amo essa prisão!",M_NO_THANKS;

	mes "[Saddeus]";
	if(checkcart() || checkfalcon() || checkriding()){
		emotion ET_SCRATCH;
		mes "Sinto muito, você deveria liberar seu carrinho, PECEPECO ou FALCON!";
		close;
	}
	mes "Fale com o nosso superintendente, Oliver.";
	next;

	nude;

	set .@rw,rand(1,4);
	if (.@rw==2) goto L_W2;
	if (.@rw==3) goto L_W3;
	if (.@rw==4) goto L_W4;

L_W1:
	savepoint "sec_in02",179,76;
	warp "sec_in02",179,76;
	close;

L_W2:
	savepoint "sec_in02",139,32;
	warp "sec_in02",139,32;
	close;

L_W3:
	savepoint "sec_in02",100,28;
	warp "sec_in02",100,28;
	close;

L_W4:
	savepoint "sec_in02",107,75;
	warp "sec_in02",107,75;
	close;

M_NO_THANKS:
	mes "[Saddeus]";  
	if (rand(2)) mes "Hoje é o X-Mas Time, hein?";
	mes "Agora cale a boca e o retorno!";
	if (rand(2)) emotion ET_HUK;
	close;  
}

sec_in02,137,57,1	script	Overseer	708,{ 
	mes "[Oliver]";

	delitem 4002,countitem(4002);//Items: Fabre_Card,
	delitem 4009,countitem(4009);//Items: Chonchon_Card,
	delitem 4022,countitem(4022);//Items: Spore_Card,
	delitem 4048,countitem(4048);//Items: Poison_Spore_Card,
	
	if (PRISON_Q <= 0 ) goto L_GET_Q;

	mes "Eu quero que você se reúna ^ff0000"+PRISON_Q+"Cogumelo da rede laranja^000000.";
	if ( countitem(1069)<PRISON_Q ) close;//Items: Orange_Net_Mushroom,
	mes "Ok ... me entregue todos os cogumelos ...";
	mes "Deixe-me ver..."+countitem(1069)+"de quase comestíveis ...";//Items: Orange_Net_Mushroom,
	mes "E"+countitem(1070)+"inúteis mortais.";//Items: Orange_Gooey_Mushroom_,
	next;
	mes "[Oliver]";
	mes "Obrigado. Você é grátis!";
	set PRISON_Q,0;
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,
	next;
	savepoint "chocado",105,112;
	warp "izlude",105,112;
	close;

L_GET_Q:
	set PRISON,PRISON+1;
	if (PRISON > 1) mes "Hmm ... você já esteve aqui ... bem bem ...";

	if ( countitem(1069) || countitem(1070) ) mes "Solte esses cogumelos !! Eles são venenosos!";//Items: Orange_Net_Mushroom, Orange_Gooey_Mushroom_,
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,

	set PRISON_Q, PRISON*3 + BaseLevel/3;
	set PRISON_Q, PRISON_Q * (readparam(bAspd)/55 + 1);
	set PRISON_Q, PRISON_Q+rand(50,60);
	mes "Ok, você deve me trazer ^ff0000"+PRISON_Q+"^000000 comestível ^ff0000RANECE COMUMO DE LENTA ^000000 TOADSTOO.";
	if(readparam(bAspd)>=100) mes "Você é um cara rápido! Você terá que obter mais cogumelos, então.";
	if (rand(10)<4) mes "E você pode comer os mortais no café da manhã.";
	close;
}
