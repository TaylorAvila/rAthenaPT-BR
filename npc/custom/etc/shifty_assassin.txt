//= Players buy ninjas to assassinate other players
//===== Additional Comments: =================================

//= 1.1.2 Updated WoE Check. [Paradox924X]

morocc,148,86,5	script	Shifty Assassin	725,{
	set .@ninja_price,250000;

	// STARTS THE MENU //
M_Start:
	mes "[Assassino sombrio]";
	mes "O que você quer?";
	next;
	menu "Compre ninjas",M_Buy,"Assassinato alguém",M_Kill,"Verifique seus ninjas",M_Check, ( getgmlevel() > 90 ? "Adicione ninjas" : "" ),-,"Cancelar",M_Exit;

	// GM MENU TO ADD NINJAS //
	mes "[Assassino sombrio]";
	mes "Quantos ninjas você deseja disponibilizar?";
	next;
	input .@add;
	set $ninja_avail, $ninja_avail+.@add;
	mes .@add +"Ninjas adicionados.";
	close;

	// BUY NINJAS //
M_Buy:
	mes "[Assassino sombrio]";
	mes "Quantos ninjas você quer comprar?";
	mes "Existem ^0000ff" + $ninja_avail + "^000000 Ninjas disponíveis.";
	mes "Eles custam ^0000ff" + .@ninja_price + "Zeny ^000000ECH.";

	input .@buy;
	next;
	if ($ninja_avail < 1) goto NoNinjas;
	if ($ninja_avail < .@buy) goto NotEnoughNinjas;
	set .@price, .@buy*.@ninja_price;
	if (Zeny < .@price) goto NoZeny;

	mes "[Assassino sombrio]";
	mes "Isso vai custar -lhe ^0000ff"+ .@price +"Zeny^000000.";
	next;
	menu "Continuar",-,"Cancelar",M_Exit;

	set Zeny, Zeny - .@price;
	set #ninjas, #ninjas + .@buy;
	set $ninja_avail, $ninja_avail - .@buy;

	mes "[Assassino sombrio]";
	mes "Obrigado.";
	close;

	// ASSASSINATE SOMEBODY //
M_Kill:
	if (agitcheck()) goto M_Busy;
	mes "[Assassino sombrio]";
	mes "Digite o nome do alvo.";
	mes "^Ff0000TYPE exatamente o nome, caso contrário, não poderei encontrar a vítima.^000000";
	next;
	menu "Continuar",-,"Cancelar",M_Exit;
	input .@name$;
	if ( !getcharid( 3,.@name$ ) ) {
		mes "[Assassino sombrio]";
		mes .@name$ +"não está online.";
		close;
	}
	next;
	mes "[Assassino sombrio]";
	mes "Ninjas ativos:"+#ninjas;
	mes "Ninjas de descanso:"+#ninjasr;
	mes "Quantos você quer enviar?";
	input .@number;
	if (.@number < 1) goto NoNinjasSent;
	if (.@number > #ninjas) goto NotEnoughNinjas1;
	if (.@number > 10) goto TooManyNinjas;
	set .@chance, rand(1,12);
	set #ninjas,#ninjas-.@number;
	set #ninjas,#ninjas+#ninjasr;
	set #ninjasr,0;
	if (.@number < .@chance) goto M_Failure;

	// SUCCESSFUL ATTACK //
	mes "Enviando ninjas agora.";
	next;
	mes "[Assassino sombrio]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr,.@number-.@ninjasurvived;
	mes "Seu ataque foi bem -sucedido, mas apenas ^ff0000" + #ninjasr + "^000000 Ninjas sobreviveu.";
	set .@acc_id, getcharid( 3,.@name$ );
	if ( .@acc_id )
		unitkill .@acc_id;
	announce .@name$ +"foi assassinado por" + strcharinfo(0) +"'S Ninjas.",bc_npc;
	close;

	// FAILED ATTACK //
M_Failure:
	mes "Enviando ninjas agora.";
	next;
	mes "[Assassino sombrio]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr, .@number-.@ninjasurvived;
	mes "Seu ataque falhou e apenas ^ff0000" + #ninjasr + "^000000 Ninjas sobreviveu.";

	announce .@name$+"sobreviveu" + strcharinfo(0) +"é um ataque ninja.",8;
	close;

	// NINJAS BUSY FOR WOE //
M_Busy:
	mes "[Assassino sombrio]";
	mes "Desculpe, todos os meus ninjas estão ocupados fazendo guerra de Empeio.";
	close;

	// CHECK YOUR NINJAS //
M_Check:
	mes "[Assassino sombrio]";
	mes "Você tem:";
	mes "^FF0000" + #ninjas + "^000000 Ninjas ativos.";
	mes "^0000FF" + #ninjasr + "^000000 Ninjas de repouso.";
	next;
	goto M_Start;


	// LIMIT //
NoNinjasSent:
	mes "[Assassino sombrio]";
	mes "Você não pode matar ninguém sem ninjas.";
	next;
	goto M_Start;

TooManyNinjas:
	mes "[Assassino sombrio]";
	mes "Você só pode enviar 10 ninjas máx.";
	next;
	goto M_Start;

NoZeny:
	mes "[Assassino sombrio]";
	mes "Você não tem Zeny suficiente.";
	close;

NotEnoughNinjas:
	mes "[Assassino sombrio]";
	mes "Não há muitos ninjas para comprar.";
	next;
	goto M_Start;

NoNinjas:
	mes "[Assassino sombrio]";
	mes "Não há ninjas para comprar.";
	close;

NotEnoughNinjas1:
	mes "[Assassino sombrio]";
	mes "Você não tem tantos ninjas.";
	next;
	goto M_Start;

M_Exit:
	mes "[Assassino sombrio]";
	mes "Adeus.";
	close;

// TIMER DELAY NINJA ADDER //
OnClock0600:
OnClock1200:
OnClock1500:
OnClock1900:
OnClock2000:
OnClock0000:
	set $ninja_avail,$ninja_avail+2;
	end;
OnClock1800:
	set $ninja_avail,$ninja_avail+3;
	end;
OnInit:
	set $ninja_avail,$ninja_avail+1;
	end;
}
