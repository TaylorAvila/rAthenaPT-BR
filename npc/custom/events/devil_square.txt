prontera,156,185,5	script	Devil Square Guardian#1	406,{
	set .@gm_level, 99; // GM level required to start/stop the event

	mes "[Devil Square Guardian]";
	mes "Esta é a entrada para ^ff0000DEVil Square ^000000.";
	mes "Se você falhar ou morrer, a porta será fechada e você deve esperar até que o evento comece novamente a entrar novamente.";
	next;
	switch(select("Enter.:Information.:" + ((getgmlevel() >= .@gm_level) ? "Menu GM." : "") + ":Cancelar.")) {
	case 1:
		if (.start != 1) { // registration period
			mes "[Devil Square Guardian]";
			mes "Devil Square está agora fechado.";
			mes "Ele será aberto nesses momentos:";
			mes "00:00 , 02:00 , 04:00 , 06:00"; // display your times here
			mes "08:00 , 10:00 , 12:00 , 14:00";
			mes "16:00 , 18:00 , 20:00 , 22:00";
			close;
		}
		if (BaseLevel < 70 || Zeny < 20000) { // requirements
			mes "[Devil Square Guardian]";
			mes "Você não atende aos requisitos para entrar no Devil Square.";
			mes "";
			mes "Requisitos:";
			mes "- Nível de base 70";
			mes "- Taxa de inscrição 20.000Z";
			close;
		}
		Zeny -= 20000;
		announce strcharinfo(0)+"entrou no Devil Square.", bc_npc;
		percentheal 100,100;
		warp "ordeal_1-1",183,182;
		close;
	case 2:
		mes "[Devil Square Guardian]";
		mes "No interior, você encontrará 3 ondas de monstros. Os monstros ficarão mais fortes a cada onda.";
		next;
		mes "[Devil Square Guardian]";
		mes "Se você derrotar todos os inimigos, terá a oportunidade de abrir 10 baús de tesouro com itens, equipamentos e cartões raros dentro!";
		close;
	case 3:
		mes "[Devil Square Guardian]";
		switch (.start) { // event states
			case 0: mes "Devil Square ainda não começou."; break;
			case 1: mes "Devil Square está atualmente aceitando participantes."; break;
			case 2: mes "Devil Square Event está atualmente em rodada"+ .round +"com"+ .mob +"monstros restantes."; break;
		}
		next;
		switch(select("Start Event.:Stop Event.:Cancel.")) {
		case 1:
			mes "[Devil Square Guardian]";
			if (.start)
				mes "Devil Square já começou.";
			else {
				mes "Começando Praça do Diabo ...";
				donpcevent strnpcinfo(0)+"::OnStartEvent";
			}
			close;
		case 2:
			mes "[Devil Square Guardian]";
			if (!.start)
				mes "Devil Square ainda não começou.";
			else {
				mes "Parando a praça do diabo ...";
				donpcevent strnpcinfo(0)+"::OnStopEvent";
			}
			close;
		case 3:
			break;
		}
		break;
	case 4:
		break;
	}
	mes "[Devil Square Guardian]";
	mes "Até a próxima.";
	close;

OnStartEvent:
OnClock0000:
OnClock0200:
OnClock0400:
OnClock0600:
OnClock0800:
OnClock1000:
OnClock1200:
OnClock1400:
OnClock1600:
OnClock1800:
OnClock2000:
OnClock2200:
	.start = 1;
	disablenpc "ord11-12"; //from npc/warps/pvp.txt
	disablenpc "Devil Square Guardian#2";
	mapannounce "ordeal_1-1","Devil Square está agora fechado.", bc_map;
	getmapxy .@map$, .@x, .@y, BL_NPC;
	mapwarp "ordeal_1-1", .@map$, .@x, .@y;
	killmonsterall "ordeal_1-1";
	announce "Devil Square está aberto. O evento começará em 5 minutos.", bc_all;
	sleep 60000;
	announce "Devil Square começará em 4 minutos.", bc_all;
	sleep 60000;
	announce "Devil Square começará em 3 minutos.", bc_all;
	sleep 60000;
	announce "Devil Square começará em 2 minutos.", bc_all;
	sleep 60000;
	announce "Devil Square começará em 1 minuto.", bc_all;
	sleep 60000;
	announce "Devil Square começou!", bc_all;
	if ( !getmapusers("ordeal_1-1") )
		goto OnStopEvent;
	.start = 2;
	.round = 1;
	.mob = 75;
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Esqueleto do soldado",1028,10,strnpcinfo(0)+":: OnDevilDead";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Orc Archer",1189,15,strnpcinfo(0)+":: OnDevilDead";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Sandman",1165,10,strnpcinfo(0)+":: OnDevilDead";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] esqueleto de arqueiro",1016,15,strnpcinfo(0)+":: OnDevilDead";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Minoud",1149,15,strnpcinfo(0)+":: OnDevilDead";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Rest",1164,10,strnpcinfo(0)+":: OnDevilDead";
	end;

OnDevilDead:
	.mob--;
	if ( .mob == 25 || .mob == 5 )
		announce "Diabo Square:"+ .mob +"Monstros restantes na rodada 1.", bc_map;
	else if ( .mob <= 0 ) {
		announce "Devil Square está entrando na 2ª rodada ...", bc_map;
		goto OnDevil2;
	}
	end;

OnDevil2:
	.round = 2;
	.mob = 70;
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Gargoyle",1253,15,strnpcinfo(0)+":: OnDevilDead2";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Alarme",1193,10,strnpcinfo(0)+":: OnDevilDead2";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] alto orc",1213,10,strnpcinfo(0)+":: OnDevilDead2";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] injustiça",1257,10,strnpcinfo(0)+":: OnDevilDead2";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] imitando",1191,15,strnpcinfo(0)+":: OnDevilDead2";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Wraith",1192,10,strnpcinfo(0)+":: OnDevilDead2";
	end;

OnDevilDead2:
	.mob--;
	if ( .mob == 25 || .mob == 5 )
		announce "Diabo Square:"+ .mob +"Monstros restantes na segunda rodada.", bc_map;
	else if ( .mob <= 0 ) {
		announce "Devil Square está entrando na 3ª rodada ...", bc_map;
		goto OnDevil3;
	}
	end;

OnDevil3:
	.round = 3;
	.mob = 58;
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Anolian",1206,15,strnpcinfo(0)+":: OnDevilDead3";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Wanderer",1208,15,strnpcinfo(0)+":: OnDevilDead3";
	areamonster "ordeal_1-1",183,182,246,244,"Idiota",1109,10,strnpcinfo(0)+":: OnDevilDead3";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Joker",1131,15,strnpcinfo(0)+":: OnDevilDead3";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Luar",1150,2,strnpcinfo(0)+":: OnDevilDead3";
	areamonster "ordeal_1-1",183,182,246,244,"[DS] Dark Illusion",1302,1,strnpcinfo(0)+":: OnDevilDead3";
	end;

OnDevilDead3:
	.mob--;
	if ( .mob == 25 || .mob == 5 )
		announce "Diabo Square:"+ .mob +"Monstros restantes na rodada final.", bc_map;
	else if ( .mob <= 0 ) {
		announce "Devil Square: Parabéns. Esses 10 baús de tesouro são seus.", bc_map;
		goto OnDevil4;
	}
	end;

OnDevil4:
	.round = 4;
	.mob = 10;
	monster "ordeal_1-1",231,250,"Baú do tesouro",1324,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",234,247,"Baú do tesouro",1328,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",237,244,"Baú do tesouro",1332,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",240,241,"Baú do tesouro",1336,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",243,238,"Baú do tesouro",1340,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",246,235,"Baú do tesouro",1344,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",249,232,"Baú do tesouro",1348,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",252,229,"Baú do tesouro",1352,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",249,241,"Baú do tesouro",1356,1,strnpcinfo(0)+":: onTreSleSleaDead";
	monster "ordeal_1-1",240,249,"Baú do tesouro",1360,1,strnpcinfo(0)+":: onTreSleSleaDead";
	end;

OnTreasureDead:
	.mob--;
	announce "Diabo Square:"+ strcharinfo(0) +"abriu o baú do tesouro na Devil Square.", bc_all;
	if ( .mob <= 0 )
		goto OnStopEvent;
	end;

OnStopEvent:
	killmonsterall "ordeal_1-1";
	enablenpc "ord11-12"; //from npc/warps/pvp.txt
	enablenpc "Devil Square Guardian#2";
	.start = .round = .mob = 0;
	end;
}

	mes "Até a próxima.";
	close2;
	warp "SavePoint",0,0;
	end;
}
