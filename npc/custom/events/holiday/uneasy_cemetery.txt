//= A periodical quest of the Uneasy Cemetery (Kill undead / Prevent their appearance)
//= Every day, at the midnight Prontera receive a wave of Undeads.
//= They come from Uneasy Cemetery of Prontera. To protect the players
//= from the undeads terror you may either kill the enemy. Or supply Mother Mathana
//= with needed amount of Holy Water. Every citizen can take his part in the 
//= saving of Prontera city. After some days of quiet life... the Cemetery strikes back.
//===== Additional Comments: ================================= 
//= 1.1 More advanced ver. Added some bonus for the one who'd kill the last walking undead
//= 1.2 Added coords to the script to make label OmMobDead working
//= 1.2a Changed item names to item IDs. [Samuray22]

prontera,3,3,3	script	Uneasy_Check	-1,{
	end;

OnHour00:
	set $UNEASY_DL,$UNEASY_DL-1;
	set $UNEASY_BL,$UNEASY_BL+30; //add need of HW for 30 bottles per day
	if ($UNEASY_BL>666) set $UNEASY_BL,666; //keep needed bottles not <=666
	if ($UNEASY_DL < 0) goto L_Start_Undead;
//The Cemetery is OK yet.
	disablenpc "Mother Mathana";
	end;
OnInit:
	if ($UNEASY_DL >= 0) disablenpc "Mother Mathana";
	end;

OnHour06:
	killmonsterall "prontera";	//The Sun kills undead in the morning
	end;

OnHour01:
	if ($@UNEASY_MOB > 0) mapannounce "Prontera","[Mãe Mathana]: Em nome de Odin, termine essas sobras de mortos -vivos!",0;
	end;

OnZombieDead:
	set $@UNEASY_MOB,$@UNEASY_MOB-1;
	if ($@UNEASY_MOB>0) end;
	set $UNEASY_DL,0;
	set $UNEASY_H$,strcharinfo(0);
	if (Sex==SEX_MALE) mapannounce "Prontera","[Mãe Mathana]: Brave"+$UNEASY_H$+"Acabou de matar os últimos mortos -vivos em Protera!",0;
	else mapannounce "Prontera","[Mãe Mathana]: senhora"+$UNEASY_H$+"Acabou de matar os últimos mortos -vivos em Protera!",0;
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	end;

L_Start_Undead:
	killmonsterall "prontera";	//kills any left monsters
	enablenpc "Mother Mathana";
//call some monsters in the city
	set $@UNEASY_MOB, 65;
	areamonster "Prontera",0,0,0,0,"Zumbi",1015,30,"UNEASY_CHECK :: OnZombiedead";
//in the Cemetery
	monster "Prontera",268,349,"Zumbi",1015,30,"UNEASY_CHECK :: OnZombiedead";
	monster "Prontera",269,350,"Ghoul",1036,5,"UNEASY_CHECK :: OnZombiedead";
//announce
	mapannounce "Prontera","[Mãe Mathana]: O cemitério ficou inquieto!  Em nome de Odin, corra para o santuário! Salve a cidade de Protera!",0;
}

prontera,257,313,5	script	Mother Mathana	79,{
	mes "[Mãe Madana]";
	if ($UNEASY_DL <= 0) goto L_Undead_Walk;
	mes "Receio que haja algo errado com nosso antigo cemitério ...";
	if ($UNEASY_H$==strcharinfo(0)) mes "Mas graças a você,"+$UNEASY_H$+", poderemos dormir" + $UNEASY_DL + "Noites!";
	if ($UNEASY_H$!=strcharinfo(0)) mes "Mas graças a"+$UNEASY_H$+"O apoio, nós temos" + $UNEASY_DL + "Noites fáceis!";
	emotion ET_SURPRISE;
	close;

L_Undead_Walk:
	if ($UNEASY_DL == 0) mes "Eles poderiam voltar a noite de amanhã!";
	if ($UNEASY_DL == 0 && $UNEASY_H$==strcharinfo(0)) mes "Obrigado,"+$UNEASY_H$+"! Agora vamos descansar até a próxima meia -noite!";
	if ($UNEASY_DL == 0 && $UNEASY_H$!=strcharinfo(0)) mes "Mas devido a"+$UNEASY_H$+"A ajuda vamos descansar até a próxima meia -noite!";
	mes "Para acalmar o cemitério inquieto, devemos derramar todas essas sepulturas com a água benta. Mas nossas irmãs e irmãos ficaram sem isso.";
	mes "Você poderia nos fornecer água benta?";
	next;
	menu "Sim, tenha toda a minha água benta!",-, "Não, eu preciso disso.",M_NO, "Eu não tenho nenhum.",M_DONT_HAVE;

	if ( countitem(523)<1 ) goto M_DONT_HAVE;
	set $UNEASY_BL,$UNEASY_BL-countitem(523);
	delitem 523,countitem(523);

	if ( $UNEASY_BL > 0 ) goto L_NEED_MORE;
//set quiet days!!! no more undead for this period!
	set $UNEASY_DL,5+((0-$UNEASY_BL)/30);
	set $UNEASY_H$,strcharinfo(0);
	mes "[Mãe Madana]";
	mes "Obrigado,"+$UNEASY_H$+"! Agora temos água benta suficiente!";
	next;
	mes "[Mãe Madana]";
	mes "Ao derramar o cemitério com aquela água, teremos" + $UNEASY_DL + "Noites seguras!";
	next;
	killmonsterall "prontera";	//kills any left monsters
	mes "[Mãe Madana]";
	mes "Ver,"+ $UNEASY_H$ +"? Todos eles se foram agora!";
	next;
	mes "[Mãe Madana]";
	mes "Nossa igreja vai agradecer pessoalmente ...";
	next;
	if (Sex==SEX_MALE) mapannounce "Prontera","[Mãe Mathana]: Em nome de Odin, declaramos bonito"+$UNEASY_H$+"as a Prontera savior!",0;
	else mapannounce "Prontera","[Mãe Mathana]: Em nome de Odin, declaramos lindos"+$UNEASY_H$+"as a Prontera savior!",0;
	mes "[Mãe Madana]";
	mes "Em nome de Odin, abençoamos você e, decentemente, apresentamos um presente modesto apenas do Cellar Wine-Cellar da Mareusis.";
	getitem 505,1; //Blue_Potion
	set JobExp,JobExp+100;
	set BaseExp,BaseExp+50;
	close;

L_NEED_MORE:
	mes "[Mãe Madana]";
	mes "Obrigado, bom"+strcharinfo(0)+", mas ainda precisamos" + $UNEASY_BL + "mais garrafas de água benta.";
	close;

M_NO:
	mes "[Mãe Madana]";
	mes "Receio que o antigo cemitério esteja saindo de controle em breve ... por favor, pegue -nos toda a água benta que você pode obter.";
	close;

M_DONT_HAVE:
	mes "[Mãe Madana]";
	mes "Infelizmente! Nós ainda precisamos" + $UNEASY_BL + "Mais garrafas de água benta ... Por que você não vai pedir a outras pessoas uma água benta extra?";
	mes "O antigo cemitério está saindo de controle em breve ...";
	mes "Por favor, em nome de Odin, ajude a cidade de Pronntera.";
	close;
}
