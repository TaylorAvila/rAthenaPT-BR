	mes "Você pode voltar novamente.";
	mes "Então, quantos você quer?";

	set @needmon,0;
	input @flag_num;
	if (@flag_num <= 0) goto L_NONE;

	mes "Você pode dar isso a alguém como um presente ou se divertir!";
	mes "Mmm .... chocolate doce ...";
	mes "Visite -me a qualquer hora ...!";
	close;

	mes "Sinto muito, mas não posso te dar tantos.";
L_NONE:
	close;

L_NOTENO:
	mes "[Stephen]";
	mes "Sinto muito, mas parece que você não pode comprar isso de mim.";
	close;
}
