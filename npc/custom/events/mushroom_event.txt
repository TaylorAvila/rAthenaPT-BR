prontera,142,228,6	script	Find the Mushroom	1084,{
	mes "[Encontre o cogumelo]";
	if (.status == 1) {
		.@count = mobcount( .event_map$, strnpcinfo(0) + ":: onMobkiled" );
		if (.@count > 0) {
			mes "Há" + .@count + "Cogumelos deixados para dentro" + .event_map$ + "!";
			mes "Encontre e mate os cogumelos para ganhar" + getitemname(.prize) + "!";
			close;
		}
		.event_map$ = ]";
		.status = 0;
		mes "Todos os cogumelos são mortos?";
	}
	mes "Não há evento no momento!";
	if (.status || getgmlevel() < .GM) close;
	mes "Começar o evento?";
	next;
	if (select("- Não:- Sim") == 1) close;
	donpcevent strnpcinfo(0) + "::OnMinute10";
	mes "[Encontre o cogumelo]";
	mes "Evento começou!";
	close;

OnInit:
	.prize = 512;	// Reward item ID
	.amount = 10;	// Reward item amount
	.GM = 60;		// GM level required to access NPC
	setarray .maps_list$[0],"chocado","Se encaixa","Marrocos","Prontera"; // Possible maps
	end;

OnMinute10:	// Start time (every hour)
	if (.status) end;
	.status = 1;
	.@spawn = rand(1,10);	// How many Mushrooms should spawn?
	.event_map$ = .maps_list$[ rand(getarraysize(.maps_list$)) ];
	.@label$ = strnpcinfo(0) + ":: onMobkiled";
	killmonster .event_map$, .@label$;
	monster .event_map$,0,0,"Por favor, não me mate!",1084,.@spawn, .@label$;
	announce "Encontre o cogumelo: total de" + .@spawn + "Cogumelos foram gerados em" + .event_map$ + "!",0;
	sleep 2500;
	announce "Encontre o cogumelo: todo cogumelo que você matará vai te dar" + getitemname(.prize) + "!",0;
	end;

OnMobKilled:
	if (playerattached() == 0)
		end;
	getitem .prize, .amount;
	.@spawn = mobcount( .event_map$, strnpcinfo(0) + ":: onMobkiled" );
	if (.@spawn > 0)
		announce "[" + strcharinfo(0) + "] matou um cogumelo. Existem agora" + .@spawn + "Cogumelos à esquerda.",bc_map;
	else {
		announce "O evento Find the Mushroom terminou. Todos os cogumelos foram mortos.",0;
		.status = 0;
		.event_map$ = ]";
	}
	end;
}
