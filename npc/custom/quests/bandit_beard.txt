umbala,126,129,4	script	Bearded Man#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "[Homem barbudo]";
	mes "Barbas! Pegue suas barbas maravilhosas!";
	mes "Você gostaria de uma barba?";
	next;
	menu "Sim, eu gostaria de um pouco de cabelo facial!",-,"Não, obrigado, isso é nojento!",M_NO;

	mes "[Homem barbudo]";
	mes "Hoho ~ Então você quer alguns bigodes.";
	emotion ET_SMILE;
	mes "Bem, eu te daria minhas barbas, mas não há barato.";
	mes "De fato, só tenho um ...";
	mes "Mas não estou vendendo.";
	mes "Mas não fique bravo! Eu posso fazer de você um.";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "[Homem barbudo]";
	mes "Eu precisaria de alguns ^8080FFTHREAD ^000000 para a barba.";
	mes "Converse com meu amigo em Izlude por alguns ^8080FFTHREAD ^000000.";
	next;

	mes "[Homem barbudo]";
	mes "Além disso, eu precisaria que você me pegue algumas coisas para a barba.";
	mes "Eles são:";
L_LIST:
	mes "^8080 Bandafelástica^000000";
	mes "^8080FF5 Skins de animais^000000";
	mes "^8080ffflack corante material^000000";
	mes "^8080ffgranpa barba^000000";
	mes "^8080FF100 Muco pegajoso^000000";
	mes "^8080ffCounterAgent^000000";
	mes "e um ^8080FF2 Diamante de quilato ^000000";
	close;

M_NO:
	mes "[Homem barbudo]";
	mes "Tudo bem, seja assim!";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "[Homem barbudo]";
	mes "Desculpe, só posso fazer um por pessoa.";
	mes "Não queremos muitas barbas agora. Certo?";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "[Homem barbudo]";
	mes "Vamos fazer de você sua barba!";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080ff ~ você o vê cortando e costurando a barba juntos ~^000000";
	next;
	mes "^8080ff ~ ele entrega a barba acabada ~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "[Homem barbudo]";
	mes "Tenha um bom dia!";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "[Homem barbudo]";
	mes "Que diabos! Não posso fazer uma barba sem os itens!";
	mes "Aqui está a lista novamente:";
	emotion ET_HUK;
	goto L_LIST;
}
