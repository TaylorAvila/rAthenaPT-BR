prontera,165,178,8	script	Old Woman	103,{

	setarray .@Items[0],714,25,1061,100,1095,200,618,10,1551,2,1614,1,727,20;
	set .@n$,"[Mulher velha]";

	mes .@n$;
	if(BerzQuest == 1) {
		mes "Ei, eu me lembro de você! Eu já lhe disse que você pode concluir apenas essa missão uma vez.";
		close; }
	mes "Saudando, rapaz. Encontrei um item realmente bacana enquanto estava escolhendo cogumelos perto de Glast Heim.";
	next;
	if(select("Keep Talking:Leave")==2) goto M_Leave;

	mes .@n$;
	mes "Eu nunca vi um item como este antes. Disseram -me que ele pode ser colocado dentro de qualquer acessório com um slot disponível.";
	next;
	mes .@n$;
	mes "Se você estiver interessado neste item, vou lhe dizer o que você precisa para que eu faça de você um dos seus. Só posso dar um desses a cada jogador; portanto, depois de terminar essa missão, você pode não fazê -lo novamente.";
	next;
	if(select("Tell me more about it:Nah, I don't care about it")==2) goto M_Leave;

	mes .@n$;
	mes "Eu acredito que o item é chamado de cartão ^ff0000berzebub ^000000. Isso faz você lançar feitiços muito rápido!";
	next;
	mes .@n$;
	mes "Eu preciso de todos os seguintes itens:";
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		mes "~"+.@Items[.@i+1]+"x"+getitemname(.@Items[.@i]);
		if (countitem(.@Items[.@i]) < .@Items[.@i+1]) set .@nr,1; }
	next;
	if(select("I have all of that!:I'll get those ASAP")==2) goto M_Leave;

	mes .@n$;
	if (.@nr) {
		mes "Você mente! Saia daqui e pegue esses itens, desculpe desculpa para uma estrela do rock.";
		close; }
	for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		delitem .@Items[.@i], .@Items[.@i+1];
	getitem 4145,1;
	set BerzQuest, 1;
	mes "Parabéns! Você tem todos os itens. Aqui está o seu cartão ^ff0000berzebub ^000000, conforme prometido.";
	emotion ET_CONGRATULATION;
	close;

M_Leave:
	mes .@n$;
	mes "Um ótimo item que tenho aqui ...";
	close;
}
