// Simple item trade-in quest, a person can get a Dead Branch
// by simply providing the items: Log x3 (7201), Wooden Heart x1 (7189), 
// Trunk x5 (1019), and Wooden Gnarl x1 (7222).
//===== Additional Comments: ================================= 
//= 1.1 Optimized, added Bloody Branch with 0.01% chance [Lupus]
//============================================================

niflheim,204,179,3	script	Mister Mobry	121,{

	mes "Yumister Mobrisch";
	mes "As árvores possuem espíritos que você conhece ...";
	next;

	mes "Yumister Mobrisch";
	mes "Na verdade, eu conheço um segredo sobre esses espíritos que podem de alguma forma ... talvez te ajudem ...";
	next;

	mes "Yumister Mobrisch";
	mes "Eu posso criar um ramo espirituoso ... Ou seja, se você pode me fornecer os materiais corretos ...";
	next;
	goto L_MENU;

L_MENU:
	menu "O que eu preciso?",-,"Onde posso encontrar essa porcaria?",M_FIND,"Faça -me um ramo morto!",M_CREATE,"Esqueça...",M_END;

	mes "Yumister Mobrisch";
	mes "Mwehehe ... vejo que você quer controlar as espíritos das árvores, estou certo? Ok ... eu preciso:";
	mes "^1354453 toras^000000";
	mes "^1354451 Coração de madeira^000000";
	mes "^1354455 Trunk^000000";
	mes "^1354451 Gnarl de madeira^000000";
	next;
	goto L_MENU;

M_FIND:
	mes "Yumister Mobrisch";
	mes "Você pode encontrar os seguintes itens de certos monstros heheh ...";
	mes "^135445logs podem ser encontrados em golems de árvores.^000000";
	mes "^135445 Wooden Hearts A essência dos golems de árvores.^000000";
	mes "^135445trunks? Sério, pense em madeira ...^000000";
	mes "^135445wooden gnarl ... Gibbet ...^000000";
	next;
	goto L_MENU;

L_NOLOG:
	mes "Yumister Mobrisch";
	mes "^135445Logs^000000 são o corpo ... eu preciso de mais deles.";
	mes "Pegue -me três toras e eu farei de você o espírito ...";
	close;

L_NOHEART:
	mes "Yumister Mobrisch";
	mes "O coração de 135445wooden ^000000 é a essência do Espírito.";
	mes "Pegue -me um coração de madeira e eu farei de você o espírito ...";
	close;

L_NOTRUNK:
	mes "Yumister Mobrisch";
	mes "O que? Você não conseguiu nem encontrar ^1354455 troncos ^000000?";
	mes "Mate Willows ou algo assim ... Nossa ... sem troncos heh ... patético.";
	close;

L_NOGNARL:
	mes "Yumister Mobrisch";
	mes "Como posso fazer um sem ^135445wooden gnarl ^000000?";
	mes "Eu disse Gibbet ... eles estão por todo esse lugar ...";
	close;

M_CREATE:
	mes "Yumister Mobrisch";
	mes "Bem ... vamos ver o que você me trouxe ...";
	next;
	if(countitem(7201)<3) goto L_NOLOG;
	if(countitem(7189)<1) goto L_NOHEART;
	if(countitem(1019)<5) goto L_NOTRUNK;
	if(countitem(7222)<1) goto L_NOGNARL;
	delitem 7201,3;
	delitem 7189,1;
	delitem 1019,5;
	delitem 7222,1;
	mes "Yumister Mobrisch";
	mes "Heh ... espero que você se divirta com os espíritos da árvore ... Cuidado agora ...";
	mes "Ei ... Cuidado agora? O que eu me importo se você faz ..... não importa.";
	if(rand(1000)==0) goto L_GIVE2;
	getitem 604,1; //Dead Branch
	close;
L_GIVE2:
	getitem 12103,1; //Bloody Branch
	close;

M_END:
	mes "Yumister Mobrisch";
	mes "Sem espírito para você ... então vá!";
	close;
}
