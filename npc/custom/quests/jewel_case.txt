lighthalzen,108,229,2	script	La Fenn	766,{
	mes "[La Fenn]";
	if(BaseLevel<99) {
		if(Q_99LVL) {
			mes "Você ... você tem mais jovem !!! Mas você é definitivamente você!";
			mes "Eu não conseguia pensar que você era capaz de encontrar minha condição ...";
			emotion ET_SCRATCH;
			next;
			mes "[La Fenn]";
			mes "Aqui está o seu prêmio. É seu para sempre.";
			set Q_99LVL,0;
			getitem 12106,1; //Jewel Case
			emotion ET_CONGRATULATION;
		} else {
			mes callfunc("F_Hi");
			mes "...";
			emotion ET_SLEEPY;
			next;
			mes "[La Fenn]";
			mes "Eu tenho um enigma para um homem forte da 2ª classe e 99 nível básico.";
			mes "* Bocejo* Se você encontrar alguns, conte a eles sobre meu enigma.";
		}
		close;
	}
	if(BaseLevel==99 && Upper==0) {
		if(Q_99LVL) {
			mes "Agora vá em frente e fique mais jovem!";
			mes "Se você resolver meu enigma, eu lhe darei uma caixa de jóia mágica com um acessório raro.";
		} else {
			mes callfunc("F_Hi")+" The Hero!";
			if(BaseExp<1000000) {
				mes "Agora você é forte.";
				mes "Mas ... não é muito forte para o meu enigma.";
				if(rand(2)) {
					mes "Matar"+((1000000-BaseExp)/3765+1)+"mais focas e retorno.";
				} else {
					mes "Matar"+((1000000-BaseExp)/81+1)+"Mais PoPorings e volte.";
				}
			} else {
				mes "Agora você é muito forte.";
				mes "E se você ficar mais jovem do que agora eu te darei meu tesouro.";
				set Q_99LVL,1;
				emotion ET_BEST;
			}
		}
		close;
	}
	emotion ET_SCRATCH;
	mes "...";
	close;
}
