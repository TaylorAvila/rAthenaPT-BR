morocc,208,90,6	script	Turban Thief	58,{
	mes "[Ladrão turbante]";
	if(event_umbala < 2) {
		mes "O que você quer comigo? Se foi!";
		close; }
	if(sphmask_q) {
		mes "Você não tem mais negócios comigo, vá embora!";
		close; }
	mes "Olá meu amigo, eu interessaria a essa máscara rara? Seu valor, garanto que você é verdadeiro Mah Frien. É manífico!";
	next;
	callsub L_Menu,1,"1,000,000",1000000;
	callsub L_Menu,2,"750,000",750000;
	callsub L_Menu,3,"500,000",500000;
	next;
	mes "[Ladrão turbante]";
	mes "ACK! Forgez IT! Eu posso fazer Bettaz en elsez onde!";
	set sphmask_q,1;
	close;

L_Menu:
	mes "[Ladrão turbante]";
	switch(getarg(0)) {
		case 1: mes "Wah? Ó ho ho ho, então você sabe que eu roubo desses selvagens não? Bem, eu não vou devolver. Mas eu vou por um pricez ... o que você diz?"; break;
		case 2: mes "Ahh um homem de negócios, você não é?  Tudo bem, que tal ..."; break;
		case 3: mes "Hmmm ... você dirige uma pechincha, ok ... minha oferta final ..."; break; }
	next;
	if(select("Pagar"+getarg(1)+"Z: Sem negócio")==2) return;
	mes "[Ladrão turbante]";
	if (Zeny < getarg(2)) {
		mes "Você está brincando comigo? Você não tem dinheiro EZ!";
		close; }
	set Zeny, Zeny-getarg(2);
	mes "Ó ho ho, é um acordo, então!";
	getitem 7114,1;
	set sphmask_q,1;
	close;
}
