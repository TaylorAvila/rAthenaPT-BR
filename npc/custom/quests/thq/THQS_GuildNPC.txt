yuno_in01,124,164,6	script	Guild Leader	108,{
	mes "[Keegan]";
	mes "O que te traz aqui? Tem algo a dizer?";
	next;
	menu "Eu quero ser um caçador de tesouros",-,"Faça uma nova missão.",N_NQuest,"Leve -me para a loja oficial.",N_OShop,"Deixa para lá.",N_NVM;
	if(Event_THQS==1) goto N_AlreadyReg;
	mes "[Keegan]";
	mes "Bem"+strcharinfo(0)+"Se você quer ser um caçador de tesouros primeiro, deve.";
	mes "";
	mes "^Ff00001.^000000 Você é bem alterado^FF000025.000Z^000000 para treinamento básico e sua prova de ser membro.";
	mes "";
	mes "Umm ... bem, isso é tudo o que você precisa fazer. Hahaha.";
	next;
	menu "PAY ^FF000025.000Z ^000000.",-,"Isso é muito alto !!",N_HighPrice;
	if (Zeny < 25000) goto N_NoZeny;
	set Zeny,Zeny-25000;
	//getitem 7950, 1;
	//getitem 7951, 1;
	set Event_THQS,1;
	mes "[Keegan]";
	mes "Parabéns!";
	emotion ET_CONGRATULATION;
	next;
	mes "[Keegan]";
	mes "Bem -vindo ao Guild of Treasure Hunters.";
	mes "Agora você pode fazer uma missão de caça ao tesouro como achar melhor.";
	close;

	mes "[Keegan]";
	mes "Hmmm você parece não ter ^ff000025.000z ^000000"+strcharinfo(0)+". Por favor, volte quando o fizer.";
	close;

	mes "[Keegan]";
	mes "Bem, se você não pode pagar um pequeno ^FF000025.000Z ^000000, ainda não deve ser um caçador de tesouros.";
	close;
N_AlreadyReg:
	mes "[Keegan]";
	mes "Ha ha ha ... você já é um membro.";
	close;
N_NQuest:
	if (Event_THQS!=1) goto A_NeedReg;
	mes "[Keegan]";
	mes "Bem"+strcharinfo(0)+"Se você deseja fazer uma missão, vá falar com o colega caçador de tesouros na outra sala lá.";
	close;
A_NeedReg:
	mes "[Keegan]";
	mes "Desculpe, você deve ser um membro do Treasure Hunter Guild, se desejar fazer uma missão.";

	mes "Também agora que você é um membro, sinta -se à vontade para usar nossas lojas sempre que desejar. Eles estão todos localizados neste andar.";

	mes "Também temos outra loja apenas para membros, nos quais usamos pequenos diamantes de metal chamados Treasure Hunter Tokens para trocar bens raros.";
	mes "";
	mes "Você até terá acesso à nossa masmorra pessoal exclusiva.";
	close;
N_OShop:
	if (Event_THQS!=1) goto N_NeedMem;
	mes "[Keegan]";
	mes "Sim"+strcharinfo(0)+"Vou levá -lo à nossa loja imediatamente!";
	next;
	warp "prt_in",166,171;
	close;
	end;
N_NeedMem:
	mes "[Keegan]";
	mes "Sinto muito, apenas os membros podem visitar nossa loja particular.";
	close;
N_NVM:
	mes "[Keegan]";
	mes "Tudo bem, volte quando você tiver tempo livre para sobrar.";

}
