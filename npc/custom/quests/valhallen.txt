prontera,147,171,5	script	Lenneth	811,{
	mes "[Lenneth]";
	mes "Eu venho de Valhalla para ajudá -lo.";
	mes "Eu posso transmutar para você muitos dos itens de Valhallen,";
	mes "Embora exijam muitos elementos raros para fazer.";
	next;
	switch(select("Mjolnir: Sleipnir: BRISHAMEN: GLEIPNIR: MEGINGJARD: Asprika: Brynhild: Hervor: Jormungand: Muspellium: Essence of Rune: Fechar")) {
		case 1:
			mes "[Lenneth]";
			mes "^3355ffmjolnir^000000 é o poderoso martelo de Thor.";
			setarray .@Items[0],1530,1522,2,984,20,985,5,969,40,7074,2,7075,4,7078,5,7091,5,7089,5;
			break;
		case 2:
			mes "[Lenneth]";
			mes "^3355ffsleipnir^000000 são botas feitas após o cavalo de guerra de Odin.";
			setarray .@Items[0],2410,2406,2,969,20,985,10,7076,3,7079,5,7083,3,7086,4,7087,3;
			break;
		case 3:
			mes "[Lenneth]";
			mes "^3355fbrisingammen^000000 é o colar mágico de Freyja, deusa da beleza.";
			setarray .@Items[0],2630,2603,1,726,2,722,3,727,10,724,5,969,20,7073,4,7077,4,7088,3,7090,3,7092,3;
			break;
		case 4:
			mes "[Lenneth]";
			mes "O ^3355FGGLEIPNIR ^000000 é uma corda leve, mas forte, necessária para fazer ^3355ffMegingJard ^000000";
			setarray .@Items[0],7058,7080,4,7081,5,7082,4,7084,3,7085,3;
			break;
		case 5:
			mes "[Lenneth]";
			mes "O ^3355FFMEGINGJARD ^000000 é o poderoso cinto de Thor.";
			setarray .@Items[0],2629,7058,1,2627,1,969,20,726,10,984,10;
			break;
		case 6:
			mes "[Lenneth]";
			mes "O ^3355ffasPrika ^000000 é a luz dos deuses.";
			setarray .@Items[0],2541,7835,1,7836,1,7837,1,7838,1,2513,1,7063,100,7291,10,7293,10,985,20;
			break;
		case 7:
			mes "[Lenneth]";
			mes "O ^3355ffbrynhild ^000000 é a armadura do guerreiro.";
			setarray .@Items[0],2383,7830,1,7831,1,7832,1,7833,1,7834,1,7510,100,2357,1,969,10,985,20;
			break;
		case 8:
			mes "[Lenneth]";
			mes "O ^3355FFHERVOR ^000000 recebeu o nome da Valquíria, que garante a vitória por seus guerreiros.";
			setarray .@Items[0],2181,6595,2,6596,1,6594,4,6597,3,6602,1,6604,1,6605,1,2115,1;
			break;
		case 9:
			mes "[Lenneth]";
			mes "O ^3355ffjormungand ^000000 é famoso por sua capacidade de envenenar o inimigo.";
			setarray .@Items[0],2020,6603,4,6599,1,6598,1,6601,4,6600,1,6604,1,6605,1,1473,1;
			break;
		case 10:
			mes "[Lenneth]";
			mes "O ^3355FFMUSPELLIUM ^000000 é necessário para fazer ^3355ffjormungand ^000000 e ^3355ffHerVor ^000000.";
			setarray .@Items[0],6605,984,50,985,50;
			break;
		case 11:
			mes "[Lenneth]";
			mes "A ^3355FESSENCE OF RUNE ^000000 é necessária para fazer ^3355ffjormungand ^000000 e ^3355ffHerVor ^000000.";
			setarray .@Items[0],6604,12735,200,12736,100;
			break;
		default:
			close;
	}
	
	mes "^Ff0000Os itens que eu preciso são os seguintes:^000000";
	set .@itemmissing,0;
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		mes "~"+.@amount+"x"+((countitem(.@id)<.@amount)?("^Cego"):("^"))+getitemname(.@id)+((getitemslots(.@id)>0)?"["+getitemslots(.@id)+"]":"")+"^000000";
		if (countitem(.@id) < .@amount) 
			set .@itemmissing,1;
	}
	if (.@itemmissing) {
		next;
		mes "[Lenneth]";
		mes "Continue e encontre todos esses itens.";
		mes "Vou esperar aqui para ajudá -lo.";
		close;
	}
	next;
	mes "[Lenneth]";
	mes "Você deseja ^3355ff"+getitemname(.@Items[0])+"^000000?";
	next;
	if(select("No","Sim")==1) {
		mes "[Lenneth]";
		mes "Por favor, volte se você mudar de idéia.";
		close;
	}
	mes "[Lenneth]";
	mes "Aqui está o seu ^3355ff"+getitemname(.@Items[0])+"^000000.";
	mes "Que isso o sirva bem.";
	//second check if the player tries to bypass the first
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		if (countitem(.@id) < .@amount) 
			close;
	}
	// Delete the items _after_ the second check
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		delitem .@id, .@amount;
	}
	getitem .@Items[0],1;
	close;
}
