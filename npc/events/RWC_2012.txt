prontera,147,61,3	script	Driller#pron	87,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Você está carregando muitos itens em seu inventário. Visite o armazenamento da Kafra e tente novamente.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Você não pode prosseguir porque está acima do peso.";
		close;
	}
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[DRILLER]";
		mes "Meu trabalho é perfurar um slot de cartão nos acessórios do RWC Memorial.";
		next;
		mes "[DRILLER]";
		mes "Sinto muito, mas você não tem nenhum item equipado na sua posição de acessório correto.";
		close;
	}
	mes "[DRILLER]";
	mes "Meu trabalho é perfurar um slot de cartão nos acessórios do RWC Memorial. Além disso, eu apenas trato os itens ff0000Pure ^000000, ou aqueles que não foram encantados.";
	next;
	set .@equip_id, getequipid(.@part);
	if (.@equip_id != 2966 && .@equip_id != 2968) {
		mes "[DRILLER]";
		mes "No entanto, posso ver que o acessório que você está usando no lado direito não pode ser tratado. Equipe um acessório memorial da RWC.";
		close;
	}
	mes "[DRILLER]";
	mes "Você também deve saber que tornar um cartão de cartão extremamente perigoso. ^FF0000Chances para ter sucesso são cerca de 50%.^000000 Você deseja prosseguir?";
	next;
	if(select("Cancelar: Vamos!") == 1) {
		mes "[DRILLER]";
		mes "Vejo você então.";
		close;
	}
	if (.@equip_id == 2966) {
		set .@slotted, 2967; //RWC_2012_Ring_
		set .@name$,"RWC 2012 Ring Memorial";
		set .@str$,"anel";
	} else if (.@equip_id == 2968) {
		set .@slotted, 2969; //RWC_2012_Pendant_
		set .@name$,"RWC 2012 Pingente Memorial";
		set .@str$,"durante";
	} else {
		mes "[DRILLER]";
		mes "Não consigo identificar o item de acessório que você está usando na mão direita. Eu não posso trabalhar nisso.";
		close;
	}
	if (getequipcardid(.@part,3) > 0) {
		mes "[DRILLER]";
		mes "Este item já foi encantado. Não posso trabalhar nisso, pois é contra as regras.";
		close;
	}

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id))
		close;

	delequip .@part;
	if (rand(1,10) > 5) {
		getitem .@slotted,1;
		specialeffect2 EF_REPAIRWEAPON;
		mes "[DRILLER]";
		mes "Yay! Sucesso! Seu"+.@name$+"agora tem um slot de cartão. Confira!";
		close;
	} else {
		specialeffect2 EF_LORD;
		mes "[DRILLER]";
		mes "Awww ... muito fraco"+.@str$+"... quebrou durante o procedimento. Desculpe.";
		close;
	}
}

prontera,147,59,3	script	Goldberg#pron	878,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Você está carregando muitos itens. Volte depois de organizar seu inventário.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Você não pode prosseguir porque está acima do peso.";
		close;
	}
	mes "[Goldberg]";
	mes "Olá! Estou encarregado de encantar acessórios memoriais da RWC com alguns poderes místicos.";
	next;
	set .@part, EQI_ACC_L;
	if (!getequipisequiped(.@part)) {
		mes "[Goldberg]";
		mes "Sinto muito, mas você não tem nenhum item equipado na sua posição de acessório correto.";
		close;
	}
	set .@equip_id, getequipid(.@part);
	if (.@equip_id < 2966 || .@equip_id > 2969) {
		mes "[Goldberg]";
		mes "No entanto, posso ver que o acessório que você está usando não é algo em que eu possa trabalhar. Equipe um acessório memorial da RWC.";
		close;
	}
	set .@select, select("Desculpe, não está interessado .: Por favor, capacite meu acessório.")-1;
	if (.@select == 0) {
		mes "[Goldberg]";
		mes "Tudo bem, então, vejo você na próxima vez ...";
		close;
	}
	setarray .@equip_card[0], getequipcardid(.@part,0),getequipcardid(.@part,1),getequipcardid(.@part,2),getequipcardid(.@part,3);
	if (.@select == 1) {
		switch(getequipid(.@part)) {
		case 2966:
			setarray .@option[0],2,2,4,4;
			break;
		case 2967:
			setarray .@option[0],0,2,4,4;
			break;
		case 2968:
			setarray .@option[0],1,1,3,3;
			break;
		case 2969:
			setarray .@option[0],0,1,3,3;
			break;
		default:
			mes "[Goldberg]";
			mes "Sinto muito, mas não posso trabalhar no acessório que você está usando atualmente.";
			mes "Se você equipou seu acessório memorial RWC no lado esquerdo, tente trocá -lo para o lado direito.";
			close;
		}
		for(set .@i,3; .@i>=0; set .@i,.@i-1) {
			if (.@equip_card[.@i] == 0) {
				set .@slot, .@i;
				set .@op_type, .@option[.@i];
				break;
			}
		}
		switch(.@op_type) {
		case 4:
			mes "[Goldberg]";
			mes "Qual encantamento você gostaria de infundir?";
			next;
			setarray .@enchant_select[0],1,2,3,4;
			set .@i, select("Cancelar: Lutando Espírito: ATK (%): Max HP: HP")-2;
			break;
		case 3:
			setarray .@enchant_select[0],5,6,7;
			set .@i, select("Cancelar: Feitiço: MATK (%): sp")-2;
			break;
		case 2:
			mes "[Goldberg]";
			mes "^FF0000SE Cuidado! Há cerca de 25% de chance de o encantamento falhar. Se isso acontecer, o item será destruído.^000000 Qual encantamento você gostaria de infundir?";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,14;
			set .@i, select("Cancel:STR:AGI:VIT:INT:DEX:LUK:SP")-2;
			break;
		case 1:
			mes "[Goldberg]";
			mes "^FF0000 Há cerca de 25% de chance de o encantamento falhar. Se isso acontecer, o item será destruído.^000000 Qual encantamento você gostaria de infundir?";
			next;
			setarray .@enchant_select[0],8,9,10,11,12,13,15,16;
			set .@i, select("Cancelar: STR: AGI: VIT: INT: DEX: LUK: MHP: HP")-2;
			break;
		case 0:
			mes "[Goldberg]";
			mes "Seu acessório recebeu tantos encantamentos que dificilmente posso trabalhar nisso.";
			close;
		}
		if (.@i == -1) {
			mes "[Goldberg]";
			mes "Tudo bem, então, vejo você na próxima vez.";
			close;
		}
		mes "[Goldberg]";
		mes "O poder do encantamento será escolhido aleatoriamente. ^ff0000Ence Infundido, o encantamento não pode ser removido.^000000 devemos continuar?";
		next;
		if(select("Não, por favor, pare.") == 1) {
			mes "[Goldberg]";
			mes "Tudo bem, então, vejo você na próxima vez ...";
			close;
		}
		set .@enchant_type, .@enchant_select[.@i];
		if (!getequipisequiped(.@part)) {
			mes "[Goldberg]";
			mes "Não tire seu equipamento enquanto estou trabalhando, ok?";
			close;
		}
		switch(.@enchant_type) {
		case 1:
			setarray .@enc[0],4811,4810,4809; //Fighting_Spirit1,Fighting_Spirit2,Fighting_Spirit3
			break;
		case 2:
			setarray .@enc[0],4819,4766,4767; //Atk1,Atk2,Atk3
			break;
		case 3:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 4:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		case 5:
			setarray .@enc[0],4760,4761,4806; //Matk1,Matk2,Matk3
			break;
		case 6:
			setarray .@enc[0],4815,4814,4813; //Spell1,Spell2,Spell3
			break;
		case 7:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 8:
			setarray .@enc[0],4700,4701,4702; //Strength1,Strength2,Strength3
			break;
		case 9:
			setarray .@enc[0],4730,4731,4732; //Agility1,Agility2,Agility3
			break;
		case 10:
			setarray .@enc[0],4740,4741,4742; //Vitality1,Vitality2,Vitality3
			break;
		case 11:
			setarray .@enc[0],4710,4711,4712; //Inteligence1,Inteligence2,Inteligence3
			break;
		case 12:
			setarray .@enc[0],4720,4721,4722; //Dexterity1,Dexterity2,Dexterity3
			break;
		case 13:
			setarray .@enc[0],4750,4751,4752; //Luck1,Luck2,Luck3
			break;
		case 14:
			setarray .@enc[0],4870,4800,4871; //SP25,SP50,SP75
			break;
		case 15:
			setarray .@enc[0],4861,4862,4867; //MHP1,MHP2,MHP3
			break;
		case 16:
			setarray .@enc[0],4795,4796,4797; //HP100,HP200,HP300
			break;
		default:
			mes "[Goldberg]";
			mes "Temos um problema, deixe -me verificar.";
			close;
		}

		if (.@enchant_type < 8)
			set .@i, rand(1,300); // 0% break chance.
		else
			set .@i, rand(1,400); // 25% break chance.

		     if (.@i < 151) set .@enchant, .@enc[0];
		else if (.@i < 251) set .@enchant, .@enc[1];
		else if (.@i < 301) set .@enchant, .@enc[2];
		else set .@enchant,9;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		    callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]))
			close;

		set .@equip_card[.@slot], .@enchant;
		if (.@slot == 2 && .@enchant == 0) {
			set .@equip_card[3],0;
		} else if (.@slot == 1 && .@enchant == 0) {
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		} else if (.@slot == 0 && .@enchant == 0) {
			set .@equip_card[1],0;
			set .@equip_card[2],0;
			set .@equip_card[3],0;
		}

		set .@equip_refine, getequiprefinerycnt(.@part);
		delequip .@part;
		if (.@enchant == 9) {
			mes "[Goldberg]";
			mes "Oh Deus!";
 			mes "O item não era forte o suficiente para suportar o encantamento e, portanto, foi destruído. Sinto muito.";
			specialeffect2 EF_LORD;
			close;
		}
		if (.@enchant == 0) { // Should never happen.
			mes "[Goldberg]";
			mes "Oh ... parece que houve algum tipo de instabilidade entre todos os poderes infundidos. Isso fez com que todos os encantamentos desaparecessem. É uma pena, mas tente novamente!";
		} else {
			mes "[Goldberg]";
			mes "Ótimo!";
			mes "O encantamento é um sucesso! Será aplicado no soquete nº^990000"+(.@slot+1)+"^000000.";
			specialeffect2 EF_REPAIRWEAPON;
		}

//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];

		close;
	} else if (.@select == 2) {
		mes "[Goldberg]";
		mes "Vou inicializar a opção encantada sem fazer nada com o cartão com fenda. Você quer continuar?";
		next;
		if(select("Eu vou parar.: Sim, com certeza, continue.") == 1) {
			mes "[Goldberg]";
			mes "Volte se você mudar de idéia.";
			close;
		}
		if (countitem(6665) == 0) {
			mes "[Goldberg]";
			mes "Desculpe. Mas você não tem o cupom de inicialização do RWC. Você pode verificar seu inventário?";
			close;
		}
		if (getiteminfo(.@equip_card[3], ITEMINFO_SUBTYPE) != CARD_ENCHANT) {
			mes "[Goldberg]";
			mes "HM ... este equipamento está limpo. Não posso inicializá -lo se não houver nada! Verifique novamente.";
			close;
		}
		specialeffect2 EF_REPAIRWEAPON;
		mes "[Goldberg]";
		mes "A opção encantadora no seu item será inicializada.";
		delitem 6665,1; //RWC_Inicializer
		delequip .@part;
	
//		GetNonSlotItemSock2 .@equip_refine .@equip_id .@equip_card[0] .@equip_card[1] .@equip_card[2] .@equip_card[3]
		for ( .@i = getiteminfo(.@equip_id, ITEMINFO_SLOT); .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		getitem2 .@equip_id,1,1,.@equip_refine,0,.@equip_card[0],.@equip_card[1],.@equip_card[2],.@equip_card[3];
		
		close;
	}
}
