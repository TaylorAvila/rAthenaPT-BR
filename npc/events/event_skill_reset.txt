//= Skill Reset with payment, minimum BaseLv.60 to reset skill.
//= Each 1 BaseLv need 20.000 Zeny, so be careful to use skill.
//============================================================
yuno,138,187,4	script	Hypnotist Teacher	124,{
	set @npcname$,"[^D5a500hypnotist^000000]";
	mes @npcname$;

	if(MISC_QUEST & 1024){
		mes "Eu já lhe disse que você pode concluir apenas este evento uma vez.";
		close;
	}
	if(BaseLevel < 60) {
		mes "Por favor, volte quando chegar à Baselv 60 ou superior.";
		close;
	}
	if(SkillPoint != 0) {
		mes "Você precisará usar todos os seus pontos de habilidade se quiser que eu continue.";
		mes "Por favor, venha de novo em breve!";
		close;
	}
	if(Weight || checkfalcon() || checkcart() || checkriding()) {
		mes "Você gostaria de redefinir habilidades?";
		mes "Sinto muito, mas ...";
		next;
		mes @npcname$;
		if(Weight) {
			mes "Você não pode redefinir habilidades";
			mes "Quando você mantém";
			mes "quaisquer itens.";
		} else if(checkcart())
			mes "Por favor, solte seu carrinho e continuaremos.";
		else if(checkfalcon())
			mes "Por favor, liberte seu Falcon e continuaremos.";
		else if(checkriding())
			mes "Por favor, liberte seu PECOPECO e continuaremos.";
		next;
		mes @npcname$;
		mes "Volte em breve!";
		close;
	}
	mes "Olá, eu sou o reinicializador de habilidades.";
	mes "Seu nome é ^0080ff"+strcharinfo(0)+"^000000.";
	mes "Como posso ajudá-lo?";
	next;
	menu "^009500 Informação sobre as habilidades de redefinição.^000000",L_Info,"^00b6ffreset Habilidades.^000000",L_Reset,"^000088NeverMind^000000",-;

	mes @npcname$;
	mes "Você sabe onde me encontrar,";
	mes "Se você quer uma redefinição !!"; 
	close;

L_Info:
	mes @npcname$;
	mes "Esta redefinição de habilidade não é gratuita !!";
	mes "A despesa para a redefinição da habilidade é ^D5A50020000 Zeny x Baselv ^000000.";
	mes "Sim ... Cada Baselv custa 20000 Zeny para redefinir a habilidade.";
	next;
	mes @npcname$;
	mes "Ah, sim, mais uma coisa!";
	mes "Quaisquer carrinhos, falcões ou pecos que você equipou";
	mes "será removido se você redefinir suas habilidades.";
	next;
	mes @npcname$;
	mes "Apenas uma vez faz novamente para sacudir o ponto de habilidade";
	mes "Cuidado com suas habilidades daqui em diante.";
	close;
	
L_Reset:
	mes @npcname$;
	mes "Antes da redefinição da habilidade para começar.";
	mes "Você deve primeiro me dizer seu nível base."; 
	next;
	mes "^D5a500 [" + strcharinfo(0) + "]^000000";
	mes "Meu nível base é ^aa00Aalevel" + BaseLevel + "Basileia.^000000"; 
	next;
	set @zeny,BaseLevel*20000;
	mes @npcname$;
	mes "Zeny total para a redefinição da quantidade de habilidade ^529dff" + @zeny + "Zeny^000000 para o serviço de redefinição de habilidades.";
	next;
	if(Zeny < @zeny) {
		mes @npcname$;
		mes "Parece que você não tem dinheiro suficiente.";
		mes "Além disso, esperamos a oportunidade.";
		emotion ET_SCRATCH;
		close;
	}
	set Zeny, Zeny-@zeny;
	set MISC_QUEST,MISC_QUEST | 1024;
	resetskill;
	mes @npcname$;
	mes "Obrigado.";
	emotion ET_THANKS;
//	logmes "Evento de redefinição de habilidades";
	close;
}
