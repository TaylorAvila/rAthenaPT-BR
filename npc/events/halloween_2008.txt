payon,162,176,4	script	Halloween Magician#iRO08	704,{
	if (Hallow08 < 1) {
		mes "[Mágico de Halloween]";
		mes "Kkkkkkkkk!";
		mes "Eu tenho um evento especial neste Halloween que testa sua sorte e agilidade.";
		mes "Você está interessado?";
		next;
		mes "[Mágico de Halloween]";
		mes "Vamos! Não seja um wuss!";
		mes "Se você coletar ingressos suficientes, poderá obter bons prêmios!";
		mes "Então, o que você diz?";
		next;
		goto MainMenu;
	}
	if (Hallow08Kill == 1) {
		mes "[Mágico de Halloween]";
		mes "Você é um gato 'Fraidy ou algo assim?!";
		mes "Você sabe que quer tentar de novo ...";
		mes "Você conhece as regras?";
		next;
		set Hallow08Kill,0;
		if(select("Sim, eu sei .: Não, eu não sei.") == 2) {
			goto Rules;
		}
		goto Participate;
	}
	if (Hallow08Kill == 2) {
		mes "[Mágico de Halloween]";
		mes "Oh, muito bem! Você está bem!";
		mes "Não é divertido com zumbis ??";
		mes "Você sabe, os zumbis também eram pessoas!";
		mes "Ha!";
		mes "Kkkkkkk.";
		next;
		getitem 7941,1;
		set Hallow08Kill,0;
		mes "[Mágico de Halloween]";
		mes "Como eu prometi";
		mes "Você pode obter ingressos de Halloween para itens legais.";
		close;
	}
	if (Hallow08 > 0) {
		mes "[Mágico de Halloween]";
		mes "Bem, você quer ouvir as regras novamente ou, apenas volte a isso ...";
		next;
		goto MainMenu2;
	}

		Rules:
			mes "[Mágico de Halloween]";
			mes "Esta vila é como um payon virtual.";
			mes "Existem zumbis e ghouls andando por aí e três saídas do sul, mas apenas um funciona.";
			mes "Isso depende de você para descobrir.";
			next;
			mes "[Mágico de Halloween]";
			mes "Você não pode usar nenhuma habilidade para matar os ghouls ou zumbis.";
			mes "E mais uma coisa ...";
			mes "Você não deveria esquecer ...";
			next;
			mes "[Mágico de Halloween]";
			mes "Todos os participantes não devem usar nada.";
			mes "Coloque todos os pertences em seu armazenamento e volte aqui quando seu peso for '0'.";
			next;
			mes "[Mágico de Halloween]";
			mes "Ah, e mais uma coisa!";
			mes "Você não pode estar montando um PECOPECO ou tem um carrinho.";
			mes "Se você estiver, então eu os removerei antes de entrar.";
			mes "Entendi?";
			next;
			mes "[Mágico de Halloween]";
			mes "Lembre -se de que existem três saídas, mas apenas um funciona aleatoriamente, os zumbis e os ghouls vagando por lá não podem ser mortos e você não pode estar usando nada.";
			next;
			if (Hallow08 > 0) {
				mes "[Mágico de Halloween]";
				mes "Ei...";
				mes "Volte quando estiver pronto.";
				close;
			} else {
				goto MainMenu;
			}

		Participate:
			mes "[Mágico de Halloween]";
			mes "Ok, você está pronto.";
			mes "Deixe -me verificar seu peso.";
			next;
			if (Weight > 0) {
				mes "[Mágico de Halloween]";
				mes "Poxa!";
				mes "Sempre há uma ovelha negra em qualquer lugar.";
				next;
				mes "[Mágico de Halloween]";
				mes "Você acha que eu não notaria que seu peso está acima de '0'?";
				mes "Você está acima do peso ...";
				close;
			} else {
				mes "[Mágico de Halloween]";
				mes "Você parece bom ir, e seu peso está certo.";
				next;
				mes "[Mágico de Halloween]";
				mes "Eu estava bastante inundado com meu trabalho, então estou exausto.";
				next;
				mes "[Mágico de Halloween]";
				mes "Às vezes esqueço de te mandar para lá ...";
				next;
				mes "[Mágico de Halloween]";
				mes "Espero que você volte bem.";
				close2;
				if (Hallow08 < 1) {
					set Hallow08,1;
				}
				set Hallow08Kill,1;
				set @Hallow08Warp, rand(1,3);
				percentheal -98,0;
				setriding 0;
				setcart 0;
				warp "evt_zombie",155,246;
				end;
			}

		TicketExchange:
			mes "[Mágico de Halloween]";
			mes "Você quer trocar ingressos por prêmios?";
			mes "Good job! Kkkkkk!";
			next;
			mes "[Mágico de Halloween]";
			mes "Deixe -me dizer quais itens você pode trocar.";
			next;
			mes "[Mágico de Halloween]";
			mes "5 ingressos para torta de abóbora.";
			mes "20 ingressos para cabeça de abóbora.";
			mes "50 ingressos para a velha caixa azul.";
			mes "70 ingressos para a velha caixa roxa.";
			mes "200 ingressos para antigo álbum de cartão.";
			next;
			mes "[Mágico de Halloween]";
			mes "O que você gostaria de trocar?";
			next;
			switch(select("Torta de abóbora: Cabeça de abóbora: Blue Box: Old Purple Box: Old Card Album")) {
			case 1:
				if (countitem(7941) < 5) {
					goto NotEnough;
				} else {
					delitem 7941,5;
					getitem 12192,1;
					goto Enough;
				}
				break;
			case 2:
				if (countitem(7941) < 20) {
					goto NotEnough;
				} else {
					delitem 7941,20;
					getitem 5134,1;
					goto Enough;
				}
				break;
			case 3:
				if (countitem(7941) < 50) {
					goto NotEnough;
				} else {
					delitem 7941,50;
					getitem 603,1;
					goto Enough;
				}
				break;
			case 4:
				if (countitem(7941) < 70) {
					goto NotEnough;
				} else {
					delitem 7941,70;
					getitem 617,1;
					goto Enough;
				}
				break;
			case 5:
				if (countitem(7941) < 200) {
					goto NotEnough;
				} else {
					delitem 7941,200;
					getitem 616,1;
					goto Enough;
				}
			}

		NextTime:
			mes "[Mágico de Halloween]";
			mes "Ok, vejo você então.";
			mes "Kkkkkkkk.";
			close;

	MainMenu:
		menu "Explique para mim.",Rules,"Eu quero participar.",Participate,"Eu quero trocar ingressos por prêmios.",TicketExchange,"Voltarei na próxima vez.",NextTime;

	MainMenu2:
		menu "Leve -me de volta lá agora!",Participate,"Por favor, me diga as regras",Rules,"Eu quero trocar ingressos por prêmios.",TicketExchange,"Voltarei na próxima vez.",NextTime;

	NotEnough:
		mes "[Mágico de Halloween]";
		mes "Você não tem ingressos suficientes!";
		mes "Você não pode nem contar?";
		mes "Por favor, venha aqui com o número certo de ingressos.";
		close;

	Enough:
		mes "[Mágico de Halloween]";
		mes "Aqui está.";
		mes "Você precisa ...";
		mes "algo mais?";
		next;
		if (Hallow08 == 1) {
			goto MainMenu2;
		} else {
			goto MainMenu;
		}
}
