-	script	Halloween Wizard#iRO09::09HallowWiz	737,{
	mes "[Assistente de Halloween]";
	mes "...";
	mes "Você quer fazer um truque em alguém?";
	next;
	while(1) {
		switch(select("Que truque?: Claro: não.")) {
			case 1:
				mes "[Assistente de Halloween]";
				mes "Posso convocar monstros em outras partes do mundo com apenas alguns materiais.";
				next;
				mes "[Assistente de Halloween]";
				mes "Parece interessante hein?";
				next;
				mes "[Assistente de Halloween]";
				mes "Se você me trazer tecido, jack o 'abóboras, tecido gasto ou abóboras esmagadas, posso convocar os monstros.";
				next;
				break;
			case 2:
				mes "[Assistente de Halloween]";
				mes "Em qual cidade você quer fazer um truque?";
				next;
				getmapxy(.@mapname$,.@mapx,.@mapy,BL_PC,""+strcharinfo(0)+"");
				if (.@mapname$ == "Prontera") {
					switch(select("Geffen: Payon: Alberta: Aldebaran")) {
						case 1:
							set .@HallowTown,3;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "teto") {
					switch(select("Prontera:Geffen:Alberta:Aldebaran")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "Se encaixa") {
					switch(select("Prontera:Payon:Alberta:Aldebaran")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,2;
							break;
						case 3:
							set .@HallowTown,4;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "Alberta") {
					switch(select("Pron")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,5;
							break;
					}
				}
				else if (.@mapname$ == "aldebaran") {
					switch(select("Pron")) {
						case 1:
							set .@HallowTown,1;
							break;
						case 2:
							set .@HallowTown,3;
							break;
						case 3:
							set .@HallowTown,2;
							break;
						case 4:
							set .@HallowTown,4;
							break;
					}
				}
				setarray .@HallowTowns$[1],"Prontera","teto","Se encaixa","Alberta","aldebaran";
				mes "[Assistente de Halloween]";
				mes "Ok, então vamos para o próximo passo.";
				next;
				mes "[Assistente de Halloween]";
				mes "Quantos tecidos ou abóboras Jack O 'você quer usar? Não ultrapasse 100 porque essa é a quantidade máxima que eu posso usar.";
				next;
				input .@input;
				if (.@input == 0) {
					mes "[Assistente de Halloween]";
					mes "Você não tem ideia definitiva.";
					mes "Não é grande coisa.";
					mes "Avise.";
					next;
					break;
				}
				else if (.@input > 100) {
					mes "[Assistente de Halloween]";
					mes "Eu disse que deve estar entre 1 e 100!";
					mes "Você não prestou atenção!";
					next;
					break;
				}
				else {
					set .@fabric,countitem(1059);
					set .@jack,countitem(1062);
					set .@worn,countitem(6299);
					set .@crushed,countitem(6298);
					set .@whispers,0;
					set .@darklords,0;

					set .@total,.@fabric + .@jack + .@worn + .@crushed;

					if(.@total < .@input) {
						mes "[Assistente de Halloween]";
						mes "Reconte o número de itens que você tem e me diga o total.";
						mes "Huhuhuhuhuhu ...";
						next;
						break;
					}

					if(.@fabric > 0) {
						if(.@fabric >= .@input) {
							delitem 1059,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 1059,.@fabric;
							set .@input,.@input - .@fabric;
							set .@whispers,.@whispers+.@fabric;
						}
					}
					if(.@worn > 0 && .@input != 0) {
						if(.@worn >= .@input) {
							delitem 6299,.@input;
							set .@whispers,.@whispers+.@input;
							set .@input,0;
						}
						else{
							delitem 6299,.@worn;
							set .@input,.@input - .@worn;
							set .@whispers,.@whispers+.@worn;
						}
					}
					if(.@jack > 0 && .@input != 0) {
						if(.@jack >= .@input) {
							delitem 1062,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 1062,.@jack;
							set .@input,.@input - .@jack;
							set .@darklords,.@darklords+.@jack;
						}
					}
					if(.@crushed > 0 && .@input != 0) {
						if(.@crushed >= .@input) {
							delitem 6298,.@input;
							set .@darklords,.@darklords+.@input;
							set .@input,0;
						}
						else{
							delitem 6298,.@crushed;
							set .@input,.@input - .@crushed;
							set .@darklords,.@darklords+.@crushed;
						}
					}
					if (.@input > 0) {
						mes "Há um problema.";
						close;
					}
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Sussurro de Halloween",3014,.@whispers;
					monster ""+.@HallowTowns$[.@HallowTown]+"",0,0,"Halloween Lorde das Trevas",3015,.@darklords;
					mes "[Assistente de Halloween]";
					mes "Aqui está o que você queria.";
					mes "Imagine o que as pessoas devem estar pensando nas outras aldeias?";
					mes "Muhahahaha";
					close;
				}
			case 3:
				mes "[Assistente de Halloween]";
				mes "Se você mudar de idéia, volte aqui ...";
				mes "Vou ficar aqui por um tempo ...";
				mes "Kkkk...";
				close;
		}
	}
}
