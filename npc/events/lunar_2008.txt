payon_in01,193,30,4	script	Rice Mill Grandma#rat	78,{
	mes "[Rice Mill avó]";
	if (!lunar_rat) {
		mes "Aqueles pequenos imundos";
		mes "Criaturas! Correndo";
		mes "ao redor, arrebatando as coisas";
		mes "de mim! Se ao menos eles";
		mes "Não foi tão culpado rápido ...";
		next;
		select("O que está errado?");
		mes "[Rice Mill avó]";
		mes "Oh, eu estava apenas fazendo";
		mes "alguns bolos de arroz e";
		mes "doces para comemorar";
		mes "o ano novo, mas estes";
		mes "Animais têm roubado";
		mes "As bolsas de arroz que eu preparei.";
		next;
		select("Bolsas de arroz?");
		mes "[Rice Mill avó]";
		mes "Sim, meu filho precisa desses";
		mes "Bolsas de arroz para bater o";
		mes "arroz em uma argamassa, mas eu não posso";
		mes "faça bolos se eu não";
		mes "até tem o arroz. Você";
		mes "Acha que você pode me ajudar?";
		next;
		if (select("Claro .: Eu nem te conheço.") == 1) {
			set lunar_rat,1;
			mes "[" + strcharinfo(0)+"]";
			mes "Claro.";
			mes "O que posso fazer?";
			next;
			mes "[Rice Mill avó]";
			mes "Bem, você pensa que você";
			mes "pode pegar os animais que";
			mes "roubou minhas bolsas de arroz?";
			mes "Eles são esses azuis e";
			mes "ratos brancos e estes";
			mes "Pequenas moles desagradáveis.";
			next;
			select("Não é um problema.");
			mes "[" + strcharinfo(0)+"]";
			mes "Não é um problema.";
			mes "Vou pegar esses animais,";
			mes "e traga de volta qualquer arroz";
			mes "Bolsas que eu posso encontrar.";
			next;
			mes "[Rice Mill avó]";
			mes "Oh, obrigado!";
			mes "Boa sorte pegando";
			mes "aquelas pragas para mim ~";
			close;
		}
		mes "[Rice Mill avó]";
		mes "Eu ... bem ...";
		mes "Isso é verdade, mas eu era";
		mes "planejando te dar";
		mes "Algo bom em troca";
		mes "para sua ajuda. Eu sei que você é";
		mes "Não é uma pessoa má, então ...";
		close;
	}
	else if ((lunar_rat >= 1 && lunar_rat <= 2) || lunar_rat == 4) {
		if (countitem(7770) >= 1 && lunar_rat == 2) {
			mes "Oh, é isso";
			mes "Arroz doce que meu filho fez?";
			mes "Você me deixaria ter isso?";
			mes "Por favor, espere aqui um momento,";
			mes "E vou fazer de você um pouco de massa ~";
			delitem 7770,1; //Sweet_rice
			set lunar_rat,3;
			close;
		}
		else if (lunar_rat == 4) {
			if (countitem(7770) >= 1) {
				mes "Oh, é isso";
				mes "Arroz doce que meu filho fez?";
				mes "Você me deixaria ter isso?";
				mes "Por favor, espere aqui um momento,";
				mes "E vou fazer de você um pouco de massa ~";
				delitem 7770,1; //Sweet_rice
				set lunar_rat,5;
				close;
			}
			mes "Oh, onde estão todos os";
			mes "Bolsas de arroz? Nós precisamos";
			mes "eles para fazer mais bolos de arroz ...";
			close;
		}
		else if (countitem(7869) < 1 && lunar_rat <= 2) {
			mes "Ooh, aqueles brancos e";
			mes "mouses azuis podem ter";
			mes "pegou minhas bolsas de arroz";
			mes "Essas toupeiras provavelmente tomaram";
			mes "eles também. Tão desagradável";
			mes "Criaturas, não são?";
			close;
		}
		mes "Oh, isso é um dos meus";
		mes "Bolsas de arroz? Desculpe,";
		mes "mas a dor meus quadris ..";
		mes "Você se importaria de ser";
		mes "um querido e entregando";
		mes "Isso para meu filho para mim?";
		set lunar_rat,2;
		close;
	}
	else if (lunar_rat == 3 || lunar_rat == 5) {
		set .@reward,rand(1,100);
		if (.@reward <= 5) {
			if (lunar_rat == 3) {
				getitem 9038,1; // New_Year_Doll_Egg
				set lunar_rat,4;
			}
			else if (lunar_rat == 5) {
				getitem 668,1; // Red_Envelope
			}
		}
		else if (.@reward >= 6 && .@reward <= 10) getitem 12198,2; // Lucky_Rice_Cake
		else if (.@reward >= 11 && .@reward <= 30) getitem 12195,3; // Plain_Rice_Cake
		else if (.@reward >= 31 && .@reward <= 40) getitem 12196,2; // Hearty_Rice_Cake
		else if (.@reward >= 41 && .@reward <= 55) getitem 12123,2; // Honey_Pastry
		else if (.@reward >= 56 && .@reward <= 70) getitem 12122,2; // Sesame_Pastry
		else if (.@reward >= 71 && .@reward <= 80) getitem 12124,2; // Rainbow_Cake
		else if (.@reward >= 81 && .@reward <= 90) getitem 12198,2; // Korean_Rice_Cake
		else if (.@reward >= 91 && .@reward <= 100) getitem 12197,3; // Salty_Rice_Cake
		// After getting the egg there is a chance to get an envelope with each turn in
		if (rand(1,20) <= 5 && lunar_rat == 4) getitem 668,1; // Red_Envelope
		// Set the quest accordingly for repeat turn-ins.
		if (lunar_rat == 3) set lunar_rat,2;
		else if (lunar_rat == 5) set lunar_rat,4;
		mes "Não é o suficiente";
		mes "para retribuir o que você";
		mes "feito por mim, mas eu gostaria";
		mes "você ter esta massa que";
		mes "Acabei de fazer. Por favor, aproveite ~";
		next;
		mes "[Rice Mill avó]";
		mes "Obrigado pelo";
		mes "Doce arroz! Eu estarei";
		mes "Certamente fazer algo";
		mes "delicioso para você se você";
		mes "Traga -me mais, ok?";
		close;
	}
}
