// Stephen - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	Stephen	58,{
	mes "[Stephen]";
	mes "Adivinha o que eu tenho?";
	mes "Uma delícia saborosa não é facilmente encontrada em Midgard ....";
	next;
	mes "[Stephen]";
	mes "Chocolate!";
	mes "É isso mesmo, você não ama chocolate .... eu amo.";
	mes "E você está com sorte, porque estou vendendo -os por apenas 5.000 Zeny por peça!";
	next;
	if(select("Eu quero um pouco de chocolate!: Não, obrigado.")==2) {
		mes "[Stephen]";
		mes "Você não quer chocolate?";
		mes "Estou te dizendo! Você vai se arrepender!";
		mes "É melhor você conseguir um pouco agora ... você não vai se deparar com chocolate assim novamente!";
		mes "Pense e me visite novamente algum dia.";
		close; }
	mes "[Stephen]";
	mes "Hah!";
	mes "Eu sabia!";
	mes "Mas não posso vender para você mais de 5 de uma só vez ... mas, se você realmente precisa de mais ....";
	mes "Você pode voltar novamente.";
	mes "Então, quantos você quer?";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[Stephen]";
	if (.@i > 5)  {
		mes "Sinto muito, mas não posso te dar tantos.";
		close; }
	if (Zeny < .@i*5000) {
		mes "Sinto muito, mas parece que você não pode comprar isso de mim.";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "Lá vai você!";
	mes "Você pode dar isso a alguém como um presente ou se divertir!";
	mes "Mmm ... chocolate doce ...";
	mes "Visite -me a qualquer hora ...!";
	close;
}
