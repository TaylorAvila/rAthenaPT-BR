mosk_in,21,246,5	script	Pinkamenia	71,{
	set .@n$,"^0000FF [Pinkamenia]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "Olá"+strcharinfo(0)+", do";
		mes "Você quer comprar alguns itens?";
		mes "Você terá que me dar amor";
		mes "Flores para eles, é claro!";
		next;
		if(select("Sim, claro!","Não, não importa.") == 2) close;
		next;
		mes .@n$;
		mes "Selecione um item.";
		mes "A quantidade de flores de amor que você precisará é entre parênteses.";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00 ["+.@SweetsCost[.@i]+"]^000000"+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "Você precisará de mais flores de amor, se quiser isso!";
			close;
		}
		mes "Tem certeza que deseja trocar ^00aa00"+.@SweetsCost[.@choice]+"X FLOR AMOR ^000000 PARA ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000?";
		if(select("Não, eu mudei de idéia.") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "Divirta -se com o seu item!";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "Você tem que me trazer 15 flores de amor!";
			close;
		}
		mes "Muito obrigado por";
		mes "Recuperando nossas flores de volta!";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "Tome isso como um pouco";
		mes "'obrigado'.";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "Se você conseguir mais amor";
		mes "Flores, você pode trocar";
		mes "eles para alguns doces aqui.";
		mes "Vejo você em breve!";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "Por favor, traga um chapéu de bolo +8";
		mes "para o extraordinário padeiro";
		mes "Parado ao meu lado!";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "Olá"+strcharinfo(0)+", isso é";
		mes "Dia dos Namorados e nós";
		mes "Adoro entregar doce";
		mes "chocolates.";
		next;
		mes .@n$;
		mes "Nosso problema é que nós";
		mes "Precisa de uma pessoa forte";
		mes "Como você que poderia ajudar";
		mes "nós, mas primeiro você terá";
		mes "para trazer um chapéu de bolo +8";
		mes "para o extraordinário padeiro, que";
		mes "está parado ao meu lado!";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "Olá"+strcharinfo(0)+"!";
	close;

OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{Bônus bdex, 1; bônus bmaxsp, 80; Bonus3 BaddmonsterdropItem, 7864, RC_DEMIHUMAN, 500; }",0;
	end;
}

mosk_in,21,244,3	script	Baker Extraordinaire	410,{
	set .@n$,"^0000FF [BAKER extraordinário]^000000";
	if(#V_QUE12==3) {
		mes .@n$;
		mes "Trocar suas flores de amor com Pinkamenia!";
		close;
	} else if(#V_QUE12==2) {
		mes .@n$;
		mes "Traga 15 flores de amor para Pinkamenia!";
		close;
	} else if(#V_QUE12==1) { 
		mes .@n$;
		if (getequipid(EQI_HEAD_TOP) == 5024 && getequiprefinerycnt(EQI_HEAD_TOP) >= 8) {
			mes "Ah, com Pinkenami te disse";
			mes "para me trazer o bolo +8";
			mes "Chapéu. Agora estou indo";
			mes "Para trocar seu bolo +8";
			mes "Chapéu para outro chapéu de bolo.";
			next;
			mes .@n$;
			delitem 5024,1;
			getitem 5105,1;
			equip 5105;
			mes "Agora, se você usar o";
			mes "novo chapéu de bolo lá";
			mes "é uma chance de que";
			mes "Monstros demi-humanos irão";
			mes "soltar uma flor de amor!";
			next;
			mes .@n$;
			mes "Os mobs roubaram nossas flores,";
			mes "que precisamos criar";
			mes "Nossos chocolates. Agora pegue";
			mes "15 Adoro flores e traga";
			mes "Eles Pinkamenia!";
			set #V_QUE12,2;
			close;
		} else {
			mes "Desculpe, mas onde está o seu";
			mes "+8 chapéu de bolo? Trazer";
			mes "para mim, e lembre -se";
			mes "para equipá -lo!";
			close;
		}
	} else {
		mes .@n$;
		mes "Olá,"+strcharinfo(0)+"!";
		close;
	}
}
