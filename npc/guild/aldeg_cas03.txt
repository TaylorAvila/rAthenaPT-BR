// Guild Flag (Outside Castle)
//============================================================
-	script	Nuernberg#1::OutsideFlagsA3	722,{

	set .@GID, GetCastleData("ALDEG_CAS03",1);

	if (.@GID == 0) {
		mes "[Edito do reino divino de Rune-Midgarts]";
		mes "";
		mes "1. Siga a ordenança do Divino Rune-Midgarts Kingdom,";
		mes "Nós declaramos isso";
		mes "Não existe um mestre formal deste castelo.";
		mes "";
		mes "2. Para quem pode";
		mes "superar todos os ensaios";
		mes "e destruir o Empeio,";
		mes "o rei vai dotar aquele com";
		mes "Propriedade deste castelo.";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[Voz ecoando]";
		mes "Corajosos ...";
		mes "Deseja voltar ao seu lugar honrado?";
		next;
		if(select("Retorne ao castelo da guilda.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("ALDEG_CAS03",1)) {
				warp "aldeg_cas03",205,186;
				end;
			}
			end;
		}
		close;
	}
	mes "[Edito do reino divino de Rune-Midgarts]";
	mes "";
	mes "1. Siga a ordenança do Divino Rune-Midgarts Kingdom,";
	mes "Aprovamos que este lugar está em";
	mes "a prossessão privada de ^ff0000"+GetGuildName(.@GID)+"^000000 Guild.";
	mes "";
	mes "2. O mestre da guilda de ^ff0000"+GetGuildName(.@GID)+"^000000 Guild é";
	mes "^FF0000"+GetGuildMaster(.@GID)+"^000000";
	mes "Se há alguém que se opõe a isso,";
	mes "Prove sua força e honra com uma lâmina de aço na sua mão.";
	close;

OnRecvCastleA03:
	FlagEmblem GetCastleData("ALDEG_CAS03",1);
	end;
}
