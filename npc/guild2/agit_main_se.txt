		maprespawnguildid strnpcinfo(2),getcastledata(strnpcinfo(2),CD_GUILD_ID),2;

	end;

		set .@str$,substr(strnpcinfo(2),0,1)+substr(strnpcinfo(2),8,9);
		killmonster strnpcinfo(2),"Steward#"+.@str$+"::OnStartArena";
		donpcevent strnpcinfo(0)+"::OnReset";
		donpcevent "Steward#"+.@str$+"::OnStop";

// Link Flags (function)
//============================================================
function	script	LinkFlag	{
	if (!getcharid(2) || getcharid(2) != getcastledata(strnpcinfo(4),CD_GUILD_ID)) end;
	if (getarg(0) == "Instalação de conveniência") {
		mes "^3355ffhis é a fortaleza";
		mes "Serviço de Teleporte. Seria";
		mes "Você gosta de se teletransportar para o";
		mes "Instalação de conveniência para";
		mes "membros da guilda?^000000";
		if(select("Vá para a instalação de conveniência: cancelar") == 1)
			warp strnpcinfo(4),getarg(1),getarg(2);
		close;
	}
	if (getarg(0) == "Centro de Emperorium") {
		mes "^3355ffhis é a fortaleza";
		mes "Serviço de Teleporte. Seria";
		mes "você gosta de se teletransportar para";
		mes "O Centro de Empeio?^000000";
		if(select("Teleport: Cancelar") == 1)
			warp strnpcinfo(4),getarg(1),getarg(2);
		close;
	}
	mes "^3355ffhis é a fortaleza";
	mes "Serviço de Teleporte. Por favor";
	mes "Escolha um destino";
	mes "dentro da fortaleza.^000000";
	for(set .@i,0; .@i<getargcount(); set .@i,.@i+3)
		set .@menu$, .@menu$+getarg(.@i)+":";
	set .@menu$, .@menu$+"Cancelar";
	set .@i, select(.@menu$)-1;
	if (.@i != getargcount()/3)
		warp strnpcinfo(4),getarg(.@i*3+1),getarg(.@i*3+2);
	close;
}

// Return Flags (function)
//============================================================
function	script	ReturnFlag	{
	.@map$ = getarg(0);
	set .@str$, (compare(strnpcinfo(4),"entendimento"))?"Arunafeltz":"Floresta Negra";
	.@GID =  getcastledata(.@map$,CD_GUILD_ID);
	if (!.@GID) {
		mes "["+.@str$+"ECRETO REAL]";
		mes "O Reino Santo de";
		mes .@str$+"declara isso";
		mes "ainda não se reivindicou";
		mes "sobre essa fortaleza. O que";
		mes "que quebra o Empeio vai";
		mes "ser reconhecido como seu novo proprietário.";
		close;
	}
	if (getcharid(2) == .@GID && getarg(1,0)) {
		mes "[Voz tocando]";
		mes "Corajoso,";
		mes "você deseja retornar";
		mes "para sua fortaleza?";
		next;
		if(select("Retorne ao fortalezas: Cancelar") == 1 && getcharid(2) == getcastledata(.@map$,CD_GUILD_ID)) {
			if (compare(.@map$,"Prag")) {
				if (.@map$ == "ARUG_CAS01") setarray .@i[0],67,193;
				else if (.@map$ == "ARUG_CAS02") setarray .@i[0],43,256;
				else setarray .@i[0],121,318;	// Castles 3,4,5 are identical.
			}
			else {
				if (.@map$ == "SCHG_CAS02") setarray .@i[0],136,188;
				else if (.@map$ == "SCHG_CAS03") setarray .@i[0],308,202;
				else setarray .@i[0],120,290;	// Castles 1,4,5 are identical.
			}
			warp .@map$,.@i[0],.@i[1];
		}
		close;
	}
	mes "["+.@str$+"ECRETO REAL]";
	mes "O Reino Santo de";
	mes .@str$+"decreta isso";
	mes "Esta fortaleza é de propriedade";
	mes "pelo ^ff0000"+getguildname(.@GID)+"^000000 Guild.";
	next;
	mes "["+.@str$+"ECRETO REAL]";
	mes "^FF0000"+getguildmaster(.@GID)+"^000000 IS";
	mes "Mestre da guilda de ^ff0000"+getguildname(.@GID)+"^000000.";
	mes "Qualquer objeto deve reivindicar isso";
	mes "fortalenda através da força de";
	mes "aço e magia durante o";
	mes "nomeado Times de cerco da guilda.";
	close;
}
