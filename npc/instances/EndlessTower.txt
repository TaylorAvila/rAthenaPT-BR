e_tower,81,105,0	script	Tower Protection Stone	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Torre interminável";

	set .@etower_timer,checkquest(60200,PLAYTIME); // 1 week
	set .@etower_timer2,checkquest(60201,PLAYTIME); // 4 hours

	if (!instance_check_party(.@party_id,2)) {
		mes "Faça ou participe de uma festa com mais de um membro e tente novamente.";
		close;
	}
	switch(.@etower_timer) {
	case -1:
		if (is_party_leader() == true) {
			mes "Confirmou que a parte foi feita. Gostaria de reservar a entrada da Torre Sem Endlem?";
			next;
			switch(select("Gerar masmorra"+.@md_name$+": Digite a masmorra: retornar a Alberta: Cancelar")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "Nome da festa:"+ getpartyname(.@party_id);
					mes "Líder do partido:"+strcharinfo(0);
					mes "^0000FF"+.@md_name$+"^000000- A reserva falhou!";
					close;
				}
				mes "^0000FF"+.@md_name$+"^000000 - Tente reservar";
				mes "Depois de fazer uma reserva, você deve conversar com o NPC atrás e selecionar o menu 'Digite a masmorra' para entrar na masmorra.";
				close;
			case 2:
				callsub L_Enter,0,1;
			case 3:
				mes "Eu vou te mudar para Alberta.";
				close2;
				warp "alberta",223,36;
				end;
			case 4:
				close;
			}
		}
		switch(select("Digite o"+.@md_name$+": Retorne a Alberta: Cancelar")) {
		case 1:
			callsub L_Enter,1,1;
		case 2:
			mes "Eu vou te mudar para Alberta.";
			close2;
			warp "alberta",223,36;
			end;
		case 3:
			end;
		}
	case 0:
	case 1:
		if (.@etower_timer2 < 2 && getcharid(1) == etower_partyid) {
			mes "Se você já gerou a masmorra, pode inseri -lo.";
			next;
			switch(select("Digite o"+.@md_name$+": Retorne a Alberta: Cancelar")) {
			case 1:
				callsub L_Enter,0,0;
			case 2:
				mes "Eu vou te mudar para Alberta.";
				close2;
				warp "alberta",223,36;
				end;
			case 3:
				close;
			}
		} else {
			set .@dun_lim_time,etower_timer+604800; // 1 week
			// set .@dun_lim_time2,etower_timer+14400; // 4 hours
			set .@dun_cur_time,gettimetick(2);
			set .@dun_ent_t,(.@dun_lim_time - .@dun_cur_time);
			set .@dun_h,(.@dun_ent_t / 3600);
			set .@dun_m,(.@dun_ent_t - (.@dun_h * 3600)) / 60;
			set .@dun_s,.@dun_ent_t - ((.@dun_h * 3600) + (.@dun_m * 60));

			mes "Devido aos efeitos posteriores da torre, você não pode entrar na masmorra agora," + .@dun_h + "horas" + .@dun_m + "minutos" + .@dun_s + "segundos restantes para entrar na próxima masmorra.";
			next;
			mes "É perigoso aqui. Deixe -me movê -lo para Alberta.";
			close2;
			warp "alberta",223,36;
			end;
		}
	case 2:
		set etower_timer,0;
		erasequest 60200;
		erasequest 60201;
		mes "^0000ffthe registros e efeitos posteriores relacionados à torre sem fim foram removidos. Você pode gerar e entrar na torre sem fim.^000000";
		close;
	}

L_Enter:
	switch(instance_enter("Endless Tower")) {
	case IE_OTHER:
		mes "Um erro desconhecido ocorreu.";
		close;
	case IE_NOINSTANCE:
		mes "A torre sem fim de masmorra memorial não existe.";
		mes "O líder do partido ainda não gerou a masmorra.";
		close;
	case IE_NOMEMBER:
		mes "Você pode entrar na masmorra depois de fazer a festa.";
		close;
	case IE_OK:
		mapannounce "e_tower", strcharinfo(0) +"da festa,"+ getpartyname( getcharid(1) ) +", está entrando na masmorra, torre sem fim.",bc_map,"0 buceta 00",FW_NORMAL,12;
		if (getarg(1)) {
			set etower_timer,gettimetick(2);
			set etower_partyid, getcharid(1);
			setquest 60200;
			setquest 60201;
		}
		//warp "1@tower",52,354;
		if (getarg(0) == 0) close;
		else end;
	}
}
