gef_fild10,242,202,0	script	Dimensional Gorge Piece	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Memória de Orc";

	if (!instance_check_party(.@party_id,2,30,80)) {
		mes "Somente os usuários entre os níveis ^FF000030 ~ 80 ^000000 podem entrar nesta masmorra.";
		close;
	}

	set .@orctime,checkquest(12059,PLAYTIME);
	if (.@orctime == -1) {
		if (is_party_leader() == true) {
			mes "O status da parte confirmado. Você gostaria de reservar entrada para o"+.@md_name$+"?";
			next;
			switch(select("Reserve o"+.@md_name$+": Digite a masmorra: Cancelar")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "Nome da festa:"+ getpartyname(.@party_id);
					mes "Líder do partido:"+strcharinfo(0);
					mes "^0000FF"+.@md_name$+"^000000 - A reserva falhou.";
					close;
				}
				mes "^0000FF"+.@md_name$+"^000000- Tentando reservar uma entrada";
				mes "Depois de fazer uma reserva, você deve selecionar 'Digite a masmorra' no menu, se desejar entrar no"+.@md_name$+".";
				close;
			case 2:
				callsub L_Enter,0;
			case 3:
				close;
			}
		}
		if(select(.@md_name$ + "Digite a masmorra memorial: Cancelar") == 2)
			end;
		callsub L_Enter,1;
	} else if (.@orctime == 0 || .@orctime == 1) {
		mes "Você pode entrar na masmorra se tiver sido gerado.";
		next;
		if(select("Entre na masmorra"+.@md_name$+":Cancelar") == 2)
			close;
		callsub L_Enter,0;
	} else if (.@orctime == 2) {
		mes "^0000fffall registros e efeitos posteriores relacionados à masmorra de memória do ORC são excluídos. Agora você pode regenerar ou entrar novamente na masmorra.^000000";
		erasequest 12059;
		close;
	}
	mes "Para gerar uma masmorra, você deve ser o líder do partido e ter pelo menos 2 membros do partido.";
	close;
L_Enter:
	switch(instance_enter("Orc's Memory")) {
	case IE_OTHER:
		mes "Um erro desconhecido ocorreu.";
		close;
	case IE_NOINSTANCE:
		mes "A memória do Memorial Dungeon Orc não existe.";
		mes "A masmorra memorial foi destruída pelo líder do partido ou por causa do prazo. Por favor, tente novamente depois de 2 horas.";
		close;
	case IE_NOMEMBER:
		mes "Somente um membro do partido pode entrar na masmorra memorial.";
		close;
	case IE_OK:
		mapannounce "Gef_fild10", getpartyname( getcharid(1) ) +"membro do partido"+strcharinfo(0)+"entrou na memória do orc.",bc_map,"0 buceta 00";
		if (checkquest(12059) == -1) setquest 12059;
		//warp "1@orcs",179,15;
		if (getarg(0) == 0) close;
		else end;
	}

}
