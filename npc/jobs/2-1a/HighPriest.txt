valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Suto Sacerdote]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Suto Sacerdote]";
		mes "Através do poder";
		mes "de santidade, podemos";
		mes "Encontre paz, força";
		mes "e proteção. Entregar";
		mes "nós das forças do mal ...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[Suto Sacerdote]";
		mes "Nosso mundo está em";
		mes "Precisa de pessoas de";
		mes "talento e convicção.";
		mes "Por favor, continue o seu";
		mes "Boas obras como uniformes";
		mes "Maior herói da santidade ...";
		next;
		mes "[Suto Sacerdote]";
		mes "Você gostaria";
		mes "Para se tornar um sumo sacerdote?";
		next;
		if (select("Não.") == 1) {
			mes "[Suto Sacerdote]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Suto Sacerdote]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[Suto Sacerdote]";
		mes "Parabéns.";
		mes "Como um sumo sacerdote,";
		mes "Espero que você guie";
		mes "outros no caminho";
		mes "para a santidade ...";
		close;
	}
	mes "[Suto Sacerdote]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Suto Sacerdote]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
