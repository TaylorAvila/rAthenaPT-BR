valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[High Wizard]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[High Wizard]";
		mes "Nós altos bruxos temos";
		mes "a responsabilidade de";
		mes "Usando nossa magia destrutiva";
		mes "Para os propósitos certos.";
		next;
		mes "[High Wizard]";
		mes "Uma vida inteira de treinamento";
		mes "é necessário antes de se tornar";
		mes "um mago alto. Você pode imaginar";
		mes "O que aconteceria se nosso poder";
		mes "foi colocado nas mãos erradas?!";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[High Wizard]";
		mes "É hora.";
		mes "E Midgard tem";
		mes "necessidade daqueles que podem";
		mes "empunham o mais forte da magia ...";
		next;
		mes "[High Wizard]";
		mes "Você gostaria de";
		mes "se tornar um mago alto?";
		next;
		if (select("Não.") == 1) {
			mes "[High Wizard]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honras para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[High Wizard]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[High Wizard]";
		mes "Parabéns.";
		mes "Como um mago alto,";
		mes "Espero usar você use";
		mes "seus poderes para trazer";
		mes "paz para os oprimidos.";
		close;
	}
	mes "[High Wizard]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[High Wizard]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
