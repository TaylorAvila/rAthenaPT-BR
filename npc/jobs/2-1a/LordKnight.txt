valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Lord Knight]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Lord Knight]";
		mes "Nós, cavaleiros, temos um";
		mes "Responsabilidade incrível ...";
		mes "Para servir e proteger.";
		next;
		mes "[Lord Knight]";
		mes "Mesmo ao custo";
		mes "de nossas próprias vidas,";
		mes "Devemos proteger o";
		mes "bem -estar de nossos camaradas.";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Lord Knight]";
		mes "Sua hora chegou!";
		mes "O mundo ainda precisa de você.";
		mes "Por favor continue sua vida";
		mes "Como um herói com uma nova aparência.";
		next;
		mes "[Lord Knight]";
		mes "Você gostaria";
		mes "Para se tornar um lorde cavaleiro?";
		next;
		if (select("Não.") == 1) {
			mes "[Lord Knight]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Lord Knight]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[Lord Knight]";
		mes "Parabéns!";
		mes "Como um lorde cavaleiro,";
		mes "Espero que você seja";
		mes "na vanguarda da batalha,";
		mes "E lidere seus aliados à vitória!";
		close;
	}
	mes "[Lord Knight]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Lord Knight]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
