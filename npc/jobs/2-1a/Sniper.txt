valkyrie,44,55,5	script	Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Sniper]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Sniper]";
		mes "Um tiro.";
		mes "Uma morte.";
		mes "Não é tão difícil";
		mes "Depois de desenvolver o";
		mes "visão para esse estilo";
		mes "de lutar.";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Sniper]";
		mes "O mundo está em";
		mes "necessidade de poderosos poderosos";
		mes "como você. Você está pronto para";
		mes "A incrível responsabilidade?";
		next;
		mes "[Sniper]";
		mes "Você está disposto a";
		mes "dê o próximo passo e";
		mes "se tornar um atirador de elite?";
		next;
		if (select("Não.") == 1) {
			mes "[Sniper]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Sniper]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[Sniper]";
		mes "Parabéns!";
		mes "Como atirador, espero";
		mes "que os servos do mal";
		mes "nunca estará seguro então";
		mes "desde que estejam em";
		mes "Sua visão!";
		close;
	}
	mes "[Sniper]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Sniper]";
	mes "Por favor, faça";
	mes "yourself comfortable";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
