valkyrie,44,50,5	script	MasterSmith#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Mastersmith]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Mastersmith]";
		mes "Um ...";
		mes "O auge";
		mes "de artesanato.";
		mes "Esse é o trabalho de";
		mes "Um mestrado.";
		next;
		mes "[Mastersmith]";
		mes "Quando você é o";
		mes "o melhor dos melhores,";
		mes "você começa a forjar com";
		mes "o objetivo de descobrir";
		mes "maneiras novas e melhores de";
		mes "criação ...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[Mastersmith]";
		mes "Chegou a hora!";
		mes "Nosso mundo precisa de corajosos,";
		mes "aventureiros trabalhadores";
		mes "como você...";
		next;
		mes "[Mastersmith]";
		mes "Você gostaria de";
		mes "Tornar -se um mestrado?";
		next;
		if (select("Não.") == 1) {
			mes "[Mastersmith]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Mastersmith]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[Mastersmith]";
		mes "Parabéns!";
		mes "Como um mestrado,";
		mes "Espero que você forje";
		mes "um caminho para um brilhante";
		mes "futuro para Midgard.";
		close;
	}
	mes "[Mastersmith]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Mastersmith]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
