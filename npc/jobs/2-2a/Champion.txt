valkyrie,53,42,3	script	Champion#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Campeão]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Campeão]";
		mes "Habilidade.";
		mes "Velocidade.";
		mes "Força.";
		mes "Agilidade.";
		next;
		mes "[Campeão]";
		mes "Um campeão pode";
		mes "beneficiar de todos";
		mes "essas coisas. Mas";
		mes "Só se pode dominar";
		mes "Tanta coisa na vida ...";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[Campeão]";
		mes "É hora.";
		mes "Hora de grandes heróis";
		mes "para se levantar contra o";
		mes "forças do mal que atormentam";
		mes "O mundo de Midgard!";
		next;
		mes "[Campeão]";
		mes "Você gostaria";
		mes "Para se tornar um campeão?";
		next;
		if (select("Não.") == 1) {
			mes "[Campeão]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Campeão]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[Campeão]";
		mes "Parabéns!";
		mes "Viver como campeão,";
		mes "e trazer luz para";
		mes "o mundo através do";
		mes "força de seus punhos.";
		close;
	}
	mes "[Campeão]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Campeão]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
