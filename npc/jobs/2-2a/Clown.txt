valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Menestrel]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Menestrel]";
		mes "Você quer";
		mes "Cante uma música comigo?";
		mes "Sha o la la ~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Menestrel]";
		mes "O mundo sombrio";
		mes "de mortais está em necessidade";
		mes "de música mais alegre.";
		mes "Você vai trazê -lo para eles";
		mes "e virar a maré no";
		mes "batalha contra o mal?";
		next;
		mes "[Menestrel]";
		mes "Você vai fazer isso";
		mes "Para Midgard ...";
		mes "Como um menestrel?";
		next;
		if (select("Não.") == 1) {
			mes "[Menestrel]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Menestrel]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[Menestrel]";
		mes "Parabéns!";
		mes "Como um menestrel, seu";
		mes "Suas músicas trarão";
		mes "espero para seus aliados e";
		mes "desespero para seus inimigos.";
		close;
	}
	mes "[Menestrel]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Menestrel]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
