valkyrie,53,50,3	script	Biochemist#Valkyrie	122,{
	if (Class == Job_Creator && bioeth == 13 && getskilllv("AM_BIOETHICS") == 0) {
		mes "[Bioquímico]";
		mes "Ah, você veio para";
		mes "recuperar as memórias";
		mes "Perdido para você? Sim, você";
		mes "deve estar aqui para o";
		mes "Segredos da vida que";
		mes "já foram seus ...";
		next;
		if (select("Sim: Não") == 1) {
			mes "[Bioquímico]";
			mes "Feche seus olhos e";
			mes "Coloque sua mente em repouso.";
			mes "Voltaremos ao seu";
			mes "passado para lembrar o";
			mes "Fragmentos do seu Lost";
			mes "memórias.";
			next;
			mes "[Bioquímico]";
			mes "Quando você abre seus olhos,";
			mes "você vai se lembrar claramente";
			mes "o segredo da vida. Você vai";
			mes "Lembre -se também do peso de";
			mes "responsabilidade em usar estes";
			mes "Segredos para as extremidades certas ...";
			next;
			skill "AM_BIOETHICS",1,SKILL_PERM;
			mes "[Bioquímico]";
			mes "Abra seus olhos ...";
			mes "Agora que você tem";
			mes "lembrou -se de como";
			mes "Crie vida artificial, eu só";
			mes "Peça que você trate todos os seus";
			mes "criações com respeito.";
			close;
		}
		mes "[Bioquímico]";
		mes "Se você deseja";
		mes "Recupere você perdido";
		mes "Memórias, por favor";
		mes "volte para mim.";
		mes "O segredo para criar";
		mes "A vida não é uma coisa insignificante ...";
		close;
	}
	else if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Bioquímico]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Bioquímico]";
		mes "É estranho isso";
		mes "Alguém como eu está aqui.";
		mes "Mas até alguém hábil";
		mes "nas maneiras da ciência";
		mes "pode conseguir ser um herói.";
		next;
		mes "[Bioquímico]";
		mes "Neste caso,";
		mes "Não é necessariamente";
		mes "os meios que usei, mas";
		mes "as pontas pelas quais eu";
		mes "lutou que me ganhou";
		mes "Um lugar em Valhalla ...";
		close;
	}
	else if (ADVJOB == Job_Creator && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[Bioquímico]";
		mes "Sim...";
		mes "Está na hora.";
		mes "Precisamos de mais gênios";
		mes "Como você no Midgard.";
		next;
		mes "[Bioquímico]";
		mes "Você gostaria de";
		mes "se tornar um bioquímico?";
		next;
		if (select("Não.") == 1) {
			mes "[Bioquímico]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Bioquímico]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Creator;
		if (bioeth == 13)
			skill "AM_BIOETHICS",1,SKILL_PERM;
		set ADVJOB,0;
		mes "[Bioquímico]";
		mes "Parabéns!";
		mes "Como um bioquímico,";
		mes "Espero que você use o seu";
		mes "vasto conhecimento para o";
		mes "propósitos certos.";
		close;
	}
	else {
		mes "[Bioquímico]";
		mes "Bem-vindo";
		mes "para Valhalla,";
		mes "o Hall de Honra.";
		next;
		mes "[Bioquímico]";
		mes "Por favor, faça";
		mes "você mesmo confortável";
		mes "Enquanto você está aqui.";
		mes "Honra para os Warriors!";
		close;
	}
}
