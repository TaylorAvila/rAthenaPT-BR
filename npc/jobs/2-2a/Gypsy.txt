		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Cigano]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Cigano]";
		mes "Mover para a esquerda,";
		mes "Mova -se para a direita ~!";
		mes "E passo ...!";
		mes "Dançar pode ser";
		mes "Mais do que um hobby.";
		mes "Para mim, é um modo de vida ~";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Cigano]";
		mes "A terra de Midgard";
		mes "precisa de mulheres talentosas";
		mes "Para alterar sutilmente os saldos";
		mes "Na batalha entre o bem";
		mes "e mal.";
		next;
		mes "[Cigano]";
		mes "Você está pronto";
		mes "Para assumir esse papel,";
		mes "e se tornar um cigano?";
		next;
		if (select("Não.") == 1) {
			mes "[Cigano]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Cigano]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[Cigano]";
		mes "Parabéns!";
		mes "Como cigano, eu sei";
		mes "que suas performances";
		mes "influenciar o coração de todos";
		mes "Aqueles que estarão assistindo ...";
		close;
	}
	mes "[Cigano]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Cigano]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
