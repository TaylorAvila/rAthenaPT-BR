valkyrie,53,39,3	script	Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Paladino]";
			mes "Parabéns.";
			mes "Honra para os Warriors!";
			close;
		}
		mes "[Paladino]";
		mes "Não se esqueça";
		mes "Que a Guerra Santa";
		mes "está se aproximando rapidamente!";
		mes "Nós devemos nos preparar!";
		mes "Que a luz da justiça";
		mes "Sempre ilumine nosso caminho!";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Paladino]";
		mes "A Guerra Santa vai";
		mes "esteja sobre nós antes de nós";
		mes "Saiba disso. Mais do que nunca,";
		mes "Precisamos de homens fortes";
		mes "e mulheres para lutar pelo que";
		mes "é bom e certo.";
		next;
		mes "[Paladino]";
		mes "Você vai lutar em";
		mes "o lado da justiça";
		mes "Como um paladino?";
		next;
		if (select("Não.") == 1) {
			mes "[Paladino]";
			mes "Quando você está pronto,";
			mes "Sinta -se à vontade para voltar.";
			mes "Honra para os Warriors!";
			close;
		}
		if (SkillPoint) {
			mes "[Paladino]";
			mes "Ainda é possível aprender mais habilidades. Por favor, use";
			mes "Todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[Paladino]";
		mes "Parabéns.";
		mes "Como paladino, espero";
		mes "você vai proteger aqueles";
		mes "mais fraco que você, e trazer";
		mes "Vitória dos EUA no próximo";
		mes "guerra entre o bem e o mal.";
		close;
	}
	mes "[Paladino]";
	mes "Bem-vindo";
	mes "para Valhalla,";
	mes "o Hall de Honra.";
	next;
	mes "[Paladino]";
	mes "Por favor, faça";
	mes "você mesmo confortável";
	mes "Enquanto você está aqui.";
	mes "Honra para os Warriors!";
	close;
}
