			mes "Honra para os Warriors!";

		mes "Sorrateiro ...";

		mes "Meus amigos sabem que eles";
		mes "pode contar comigo.";
		mes "Eu preciso deles e";
		mes "Eles precisam de mim.";

		mes "Parabéns!";
