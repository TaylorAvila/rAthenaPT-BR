//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[Equipe Cool Event Corp.]";
	mes "Bem -vindo ao Cool Event Corp.";
	mes "Nossa equipe está sempre trabalhando";
	mes "para superar suas expectativas";
	mes "para serviço de qualidade. Então, como";
	mes "Posso ajudá -lo hoje?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[Equipe Cool Event Corp.]";
		mes "Seu ponto de reaparecimento";
		mes "foi salvo aqui";
		mes getarg(1)+".";
		mes "Obrigado por usar o";
		mes "Serviço Cool Event Corp. ~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[Equipe Cool Event Corp.]";
		if(!callfunc("F_CanOpenStorage")){
			mes "Sinto muito, mas você";
			mes "Precisa do novato";
			mes "Habilidade básica Nível 6 para";
			mes "use o serviço de armazenamento.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "Sinto muito, mas você não";
			mes "tem zeny suficiente para usar";
			mes "o serviço de armazenamento. Nosso";
			mes "A taxa de acesso ao armazenamento é 40 Zeny.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "Deixe -me abrir seu pessoal";
		mes "armazenamento para você imediatamente.";
		mes "Obrigado por apoiar legal";
		mes "Event Corp. usando nosso";
		mes "serviços. Tenha um bom dia ~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "Por favor, escolha";
		mes "seu destino.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +"->"+ .@cost +"Z: Cancelar") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[Equipe Cool Event Corp.]";
				mes "Sinto muito, mas você não tem";
				mes "Zeny suficiente para o teleporte";
				mes "Serviço. A taxa para se teletransportar";
				mes "para"+ .@destination$ +"é"+.@cost+"mulheres.";
			}
		}
		close2;
		break;
	case 4:
		mes "[Equipe Cool Event Corp.]";
		if(BaseClass != Job_Merchant){
			mes "Sinto muito, mas o";
			mes "Serviço de aluguel de carrinhos de pressão";
			if (checkre(0))
				mes "está disponível apenas para classes de comerciantes.";
			else {
				mes "está disponível apenas para comerciantes,";
				mes "Blacksmiths, Smiths brancos,";
				mes "Alquimistas e criadores.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "Você só pode alugar um carrinho depois";
			mes "Aprendendo a habilidade do carrinho de pressão.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "Você já tem";
			mes "um carrinho de pressão equipado.";
			mes "Infelizmente, não podemos";
			mes "alugar mais de um para";
			mes "cada cliente de cada vez.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "O aluguel do carrinho";
		mes "A taxa é 800 Zeny. Seria";
		mes "Você gosta de alugar um carrinho de pressão?";
		next;
		if(select("Alugue um carrinho de pressão: Cancelar") == 1){
			if(Zeny < 800){
				mes "[Equipe Cool Event Corp.]";
				mes "Sinto muito, mas você";
				mes "não tem o suficiente";
				mes "Zeny para pagar o carrinho";
				mes "Taxa de aluguel de 800 Zeny.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[Equipe Cool Event Corp.]";
		mes "Cool Event Corp. é sempre";
		mes "esforçando -se para fornecer o melhor";
		mes "Serviços para nossos clientes.";
		mes "Ajude -nos a se tornar o melhor de";
		mes "fornecendo -nos suas opiniões";
		mes "e feedback honesto. Obrigado.";
		close2;
	}
	cutin "", 255;
	end;
}

	callfunc "F_CoolEventCorp",
		"Salvar: Use Storage :: Rent a PushCart: Storage Senha Service: Cancelar",
		"na cidade de Lighthalzen","Lighthalzen",95,240;
	end;
}
