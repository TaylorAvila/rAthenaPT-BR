//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[Então funcionário]";
		mes "Bem -vindo ao";
		mes "Caffra Corporation.";
		mes "Os serviços Kafra";
		mes "estão sempre do seu lado.";
		mes "Como posso ajudá -lo?";
		break;
	case 1:// Niflheim specific message
		mes "[Então funcionário]";
		mes "^666666W-WEEEELC-COME";
		mes "Para Th-The K-Kaafrrrrra";
		mes "C-Ro Thor-Irs-Ride .... 000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[Então funcionário]";
		mes "Bem-vindo. ^FF0000" + getguildname( getcharid(2) ) + "^000000 Membro.";
		mes "A coportação Kafra ficará com você onde quer que você vá.";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[Então funcionário]";
		mes "Então, você veio de uma terra distante para estudar nossa cultura, ou está apenas passeando?";
		mes "Em ambos os casos, por que não ficar um pouco?";
		mes "O ar é eternamente pesado com o";
		mes "perfume de flores silvestres agradáveis.";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[Então funcionário]";
		mes "Com nossos muitos kafra";
		mes "Locais de serviço, você nunca";
		mes "longe de casa.";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "Use armazenamento", "Use o armazenamento da guilda", "Alugue um carrinho de pressão", "Use o serviço de teleporte", "Cancelar";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"Salvar","Use armazenamento","Cancelar"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"Use armazenamento","Cancelar"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"Salvar","Use armazenamento","Alugue um carrinho de pressão","Verifique outras informações","Cancelar"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"Use armazenamento","Alugue um carrinho de pressão","Verifique outras informações","Cancelar"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"Use armazenamento","Verifique outras informações","Cancelar"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"Salvar","Use armazenamento","Alugue um carrinho de pressão","Cancelar"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Salvar","Use armazenamento","Verifique outras informações","Cancelar"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"Use armazenamento","Alugue um carrinho de pressão","Use o serviço de teleporte","Verifique outras informações","Cancelar"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"Use armazenamento","Salvar","Alugue um carrinho de pressão","Cancelar"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"Salvar","Use armazenamento","Use o serviço de teleporte","Alugue um carrinho de pressão","Verifique outras informações","Cancelar"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Salvar")
			return;
		else if (.@K_Menu0$[.@j] == "Use armazenamento") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Use o serviço de teleporte") {
			if (.@menu_num != 4)// Display Einbroch "Sem serviço de teletransporte" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Então funcionário]";
				mes "Por causa do ^ff0000limited";
				mes "Contrato de transporte^000000, o";
				mes "A Kafra Corporation não pode";
				mes "Forneça serviços de teletransporte";
				mes "na República de Schwarzwald.";
				next;
				mes "[Então funcionário]";
				mes "Pedimos que você por favor";
				mes "Use o serviço de dirigível";
				mes "em vez de. Obrigado pelo seu";
				mes "compreensão e cooperação.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Alugue um carrinho de pressão") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Verifique outras informações") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancelar") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Use o armazenamento da guilda") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Então funcionário]";
			mes "Sinto muito, mas outro membro da guilda está usando o armazenamento da guilda";
			mes "agora mesmo.  Por favor, espere até que essa pessoa termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Então funcionário]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^6666666S-S-Ssoooorry,";
			mes "Você é um a-aaaa";
			mes "Novice ... e necessidades";
			mes "B-Basic SSSSKILL L-LEVEL 6 ...^000000";
			return;
		}
		mes "Sinto muito, mas você";
		mes "Precisa do novato";
		mes "Habilidade básica Nível 6 para";
		mes "use o serviço de armazenamento.";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Bilhete grátis para armazenamento Kafra" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Então funcionário]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^6666666Zeeeeny ...";
					mes "M-More Z-Zena ...!";
					mes "N-Need 150 ... Zeny ...";
					mes "Ergh! BL-BLOOD BL ~!^000000";
					return;
				}
				// Standard Message
				mes "Sinto muito, mas você não";
				mes "tem zeny suficiente para usar";
				mes "o serviço de armazenamento. Nosso";
				mes "Taxa de acesso ao armazenamento é" + .@fee + "mulheres.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "Fim" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Então funcionário]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666 que você ... para ... usando ...^000000";
		}
		// Normal message
		else {
			mes "[Então funcionário]";
			mes "Aqui, deixe -me abrir";
			mes "seu armazenamento para você.";
			mes "Obrigado por usar";
			mes "O serviço Kafra.";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[Então funcionário]";
	mes "Por favor, escolha";
	mes "seu destino.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancelar")
		return;
	// Consume "Bilhete grátis para transporte de kafra" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Então funcionário]";
			mes "Sinto muito, mas você não tem";
			mes "Zeny suficiente para o teleporte";
			mes "Serviço. A taxa para se teletransportar";
			mes "para" + @wrpD$[.@j] + "é" + @wrpP[.@j] + "mulheres.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Comodo Pharos Beacon") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Chocado") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mjolnir Dead Pit") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morocc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Orc Dungeon") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[Então funcionário]";
		mes "Sinto muito, mas o";
		mes "Serviço de aluguel de carrinhos de pressão";
		if (checkre(0) == true)
			mes "está disponível apenas para classes de comerciantes.";
		else {
			mes "está disponível apenas para comerciantes,";
			mes "Blacksmiths, Master Smiths,";
			mes "Alquimistas e bioquímicos.";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[Então funcionário]";
		mes "Você já tem";
		mes "um carrinho de pressão equipado.";
		mes "Infelizmente, não podemos";
		mes "alugar mais de um para";
		mes "cada cliente de cada vez.";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Então funcionário]";
		mes "Você só pode alugar um carrinho depois de aprender o \"Push Cart\"habilidade.";
		return 1;
	}
	// Consume "Bilhete grátis para o serviço de carrinho" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Então funcionário]";
		mes "O aluguel do carrinho";
		mes "taxa é" + .@rental_fee + "Zeny. Seria";
		mes "Você gosta de alugar um carrinho de pressão?";
		next;
		if (select("Alugue um carrinho de pressão") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Então funcionário]";
			mes "Sinto muito, mas você";
			mes "não tem o suficiente";
			mes "Zeny para pagar o carrinho";
			mes "Taxa de aluguel de" + .@rental_fee + "mulheres.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "Digite sua senha de armazenamento:";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Senha de armazenamento incorreto.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + "Orgulhosamente apresenta um novo serviço:";
		mes "Proteção de armazenamento adicional com uma senha.";
		next;
		if (select( "Defina nova senha -> 5000Z", "Cancelar" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Seu armazenamento está protegido com uma senha. O que você faria agora?";
	next;
	switch( select( "Alterar senha antiga -> 5000Z", "Remova a senha de armazenamento -> 1000Z", "Cancelar" ) ) {
	case 1:
		mes .@npc_name$;
		mes "No começo, digite sua senha ^0000ffold ^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "Senha errada. Você não pode definir uma nova senha.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Por favor, digite sua senha antes da remoção.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "A senha não foi removida.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Você não tem zeny suficiente.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "Você limpou com sucesso sua senha de armazenamento.";
			mes "Obrigado por usar" + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Senha errada. Não vamos devolver seu 1000Z.";
			mes "Por favor, da próxima vez, digite a senha correta.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Agora insira sua senha ^ff0000New ^000000 para proteger seu armazenamento de ladrões.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "A senha não foi alterada.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Você não tem zeny suficiente.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "Você protegeu seu armazenamento com uma senha secreta.";
	mes "Obrigado por usar" + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "Digite um número 1000 ~ 10000000:";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tentei ajustar a senha de armazenamento.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "Você não pode usar uma senha tão grande.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "Você não deve usar uma senha tão curta.";
		return 0;
	}
	return .@code_;
}
