	cutin "kafra_05",2;

	mes "HM ...?";
	mes "Oh, bem -vindo a";
	mes "A Kafra Corporation";
	mes "Sede. Você fez";
	mes "Precisa de algo?";
	next;
	switch(select("Salvar: Use Armazenamento: Alugue um carrinho de pressão: Cancelar")) {
	case 1:
		mes "[Kafra Leilah]";
		mes "Seu ponto de reaparecimento tem";
		mes "foi salvo aqui, por dentro";
		mes "da corporação Kafra";
		mes "Sede. Obrigado.";
		next;
		savepoint "aldeba_in",96,179,1,1;
		mes "[Kafra Leilah]";
		mes "Por favor, use";
		mes "os serviços de kafra que são";
		mes "disponível em todo";
		mes "Midgard. Obrigado por";
		mes "Visitando a sede da Kafra.";
		close2;
		break;
	case 2:
		if (Zeny < 20) {
			mes "[Kafra Leilah]";
			mes "Com licença, mas isso";
			mes "parece que você não";
			mes "Tenha os 20 Zeny a pagar";
			mes "a taxa de acesso ao armazenamento ...";
			close3;
		}
		mes "[Kafra Leilah]";
		mes "Embora esta instalação seja";
		mes "destinada exclusivamente a";
		mes "O treinamento do funcionário da Kafra";
		mes "e funções administrativas,";
		mes "Vou acessar seu armazenamento para você.";
		next;
		set Zeny, Zeny-20;
		set RESRVPTS, RESRVPTS + (20/5);
		mes "[Kafra Leilah]";
		mes "No futuro, por favor";
		mes "Pergunte ao funcionário da Kafra em";
		mes "dever se você deseja usar";
		mes "Qualquer um dos serviços Kafra.";
		mes "Obrigado pelo seu patrocínio.";
		callfunc("F_CheckKafCode");    //check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "[Kafra Leilah]";
		mes "Minhas desculpas, mas eu sou";
		mes "não de plantão. Eu te ajudaria";
		mes "se eu pudesse, mas na verdade não";
		mes "tem algum carrinho disponível.";
		mes "Por que você não pergunta a outro kafra";
		mes "Funcionário para obter assistência?";
		next;
		mes "[Kafra Leilah]";
		mes "Os funcionários da Kafra são";
		mes "estacionado em todo o";
		mes "Midgard Continent,";
		mes "E você deve ser capaz de encontrar";
		mes "Muito lá fora em Al de Baran.";
		close2;
		break;
	case 4:
		mes "[Kafra Leilah]";
		mes "^666666*Ufa ...*^000000";
		mes "Ótimo, porque eu sou";
		mes "Na verdade, no meu intervalo";
		mes "agora mesmo. Escolhendo";
		mes "'' Cancelar '' foi um bom";
		mes "mova de sua parte.";
		close2;
		break;
	}
	cutin "",255;
	end;
}
