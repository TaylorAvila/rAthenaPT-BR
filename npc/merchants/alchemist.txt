alde_alche,24,188,3	script	Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "- Espere um minuto! -";
		mes "- Atualmente você está carregando -";
		mes "- Muitos itens com você. -";
		mes "- Por favor, volte de novo -";
		mes "- Depois de armazenar alguns itens no armazenamento da Kafra. -";
		close;
	}
	mes "[Gever al Sharp]";
	mes "Bem -vindo ao";
	mes "União Alquimista.";
	mes "Como posso ajudá -lo hoje?";
	next;
	switch(select("Material de compra.")) {
	case 1:
		mes "[Gever al Sharp]";
		mes "O que você gostaria?";
		next;
		if (select("Medicine Bowl - 8 Zeny: Cancelar.") == 1) {
			mes "[Gever al Sharp]";
			mes "Quantos você quer?";
			mes "Digite '0' se você quiser sair.";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[Gever al Sharp]";
					mes "O acordo foi cancelado.";
					mes "Venha de novo na próxima vez.";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[Gever al Sharp]";
					mes "O número deve";
					mes "ser menor que 2000.";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[Gever al Sharp]";
				mes "Você não";
				mes "tem zeny suficiente.";
				mes "Verifique quanto Zeny";
				mes "você tem primeiro.";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[Gever al Sharp]";
				mes "Não parece";
				mes "Você pode carregar tudo.";
				mes "Por favor, verifique o espaço";
				mes "em seu inventário.";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[Gever al Sharp]";
			mes "Obrigado.";
			mes "Venha de novo.";
			close;
		}
		mes "[Gever al Sharp]";
		mes "Bem então,";
		mes "Venha de novo quando";
		mes "você precisa comprar";
		mes "Materiais relacionados a";
		mes "Alquimia, tudo bem?";
		close;
	case 2:
		mes "[Gever al Sharp]";
		mes "O que você precisa?";
		mes "Os manuais geralmente são 100.000 Zeny. Mas existem alguns manuais especiais que custarão mais.";
		next;
		switch(select("Guia da criação de poções: Guia de criação de álcool: Guia de criação de granadas de garrafa: Guia de criação de garrafas ácidas: Guia de criação de garrafas de planta: Guia de criação de garrafas da esfera marinha: Guia de criação de casaco brilhante: Guia de criação de poção condensada: cancelamento de acordo.")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[Gever al Sharp]";
			mes "Bem então ...";
			mes "Volte se você";
			mes "sempre preciso comprar";
			mes "um manual de produção.";
			close;
		}
	case 3:
		mes "[Gever al Sharp]";
		mes "Tudo bem então,";
		mes "Tenha um bom dia.";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[Gever al Sharp]";
		mes "Você não";
		mes "tem zeny suficiente.";
		mes "Verifique quanto Zeny";
		mes "você tem primeiro.";
		close;
	}
	mes "[Gever al Sharp]";
	mes getitemname(.@item_id) +"?";
	mes "Isso será"+ .@zeny_req +"mulheres.";
	next;
	if (select("Compra.:Quit.") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[Gever al Sharp]";
		mes "Obrigado por";
		mes "Seu patrocínio.";
		close;
	}
	mes "[Gever al Sharp]";
	mes "Venha de novo";
	mes "próxima vez.";
	close;
}
