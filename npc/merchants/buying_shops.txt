// Black Marketeer (Buy Licenses - Non-Merchant Classes) :: qskill_buy_store
//============================================================ 
que_job01,68,84,1	script	Black Marketeer#Buying	881,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "[Morrer. Jas]";
		mes "Você é muito ganancioso, mesmo comparado a alguém como eu!";
		mes "Por que você não vai clarear sua bolsa primeiro?";
		close;
	}
	if (getskilllv("All_buying_store") == 1) {
		mes "[Morrer. Jas]";
		mes "Ei, você já fez um contrato com Hugh.";
		mes "Eu não tenho nenhum negócio com você.";
		close;
	}
	mes "[Morrer. Jas]";
	mes "Você deve precisar de algo muito para vir me encontrar.";
	mes "O que você quer?";
	next;
	switch(select("Licença de loja de compradores em massa: Quem é você ?: Nada, nada!")) {
	case 1:
		mes "[Morrer. Jas]";
		mes "Eu sabia!";
		mes "Claro, eu posso fazer isso para você.";
		mes "O meu se parece com o autêntico que os comerciantes recebem daquele bastardo Sr. Hugh!";
		next;
		mes "[Morrer. Jas]";
		mes "E minha licença é melhor 'porque você não precisa' aprender nenhuma habilidade.";
		mes "Quantos você quer?";
		next;
		mes "[Morrer. Jas]";
		mes "Só para você saber, só posso fazê -los em pequenas quantidades, até 10 por vez.";
		mes "Vai custar 500 Zeny para cada um.";
		next;
		while(1) {
			input .@input;
			mes "[Morrer. Jas]";
			if (.@input == 0) {
				mes "Você não precisa dessas licenças?";
				close;
			} else if (.@input > 10) {
				mes "Só posso fazer até 10 de cada vez, você sabe.";
				next;
			} else {
				mes "Vai custar"+(.@input*500)+"mulheres.";
				if (Zeny < (.@input*500)) {
					mes "Mas você não tem dinheiro suficiente.";
					next;
					mes "[Morrer. Jas]";
					mes "Você não conhece o básico dos negócios? Tudo tem um preço.";
					mes "Se você quiser algo, você precisa pagar por isso.";
				} else {
					mes "Ha ... ha ha ha!";
					mes "Sr. Hugh, vou assumir o seu negócio de licença. Você verá!";
					mes "*Risada de risada*";
					getitem 12548,.@input; //Buy_Market_Permit2
					set Zeny,Zeny-(.@input*500);
				}
				close;
			}
		}
	case 2:
		mes "[Morrer. Jas]";
		mes "Deixei minha cidade natal há muito tempo.";
		mes "Não faz sentido perguntar quem eu sou, porque tudo o que me resta agora é meu ódio.";
		next;
		mes "[Morrer. Jas]";
		mes "...";
		mes "Hugh é um comerciante corrupto, sem senso de ética nos negócios.";
		mes "Meu único objetivo na vida é destruir Hugh.";
		next;
		mes "[Morrer. Jas]";
		mes "Ah, eu bebi demais ... (*hic*)";
		mes "Isso é apenas o álcool falando.";
		mes "Por favor, esqueça tudo o que eu disse.";
		close;
	case 3:
		mes "[Morrer. Jas]";
		mes "Tudo bem, tudo bem! Você não precisa gritar.";
		mes "Apenas me deixe em paz se você não tiver negócios comigo.";
		close;
	}
}

alberta_in,58,52,4	script	Purchasing Team#Buying	59,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "- Você não pode conversar ou executar a missão porque está carregando muitos itens. -";
		close;
	}
	mes "[Senhor. Hugh]";
	if (BaseClass == Job_Merchant && getskilllv("Mc_vending") >= 1) {
		if (getskilllv("All_buying_store") == 1) {
			mes "Eu sou Hugh da equipe de compras.";
			mes "Como posso ajudá -lo hoje?";
			next;
			if(select("Compre licença de loja de compras em massa: Saia") == 2) {
				mes "[Senhor. Hugh]";
				mes "Sinta -se à vontade para me perguntar se você precisa de alguma licença de loja de compradores em massa.";
				mes "Venha de novo ~";
				close;
			}
			mes "[Senhor. Hugh]";
			mes "São 200 zeny para cada licença de loja de compradores em massa e você pode comprar até 50 por vez.";
			mes "Quantas licenças você precisa?";
			next;
			while(1) {
				input .@input;
				mes "[Senhor. Hugh]";
				if (.@input == 0) {
					mes "Você cancelou o comércio.";
					mes "Tenha um bom dia.";
					close;
				} else if (.@input > 50) {
					mes "Por favor, insira um valor de 50 ou menos.";
					next;
				} else {
					mes "Vai custar"+(.@input*200)+"Zeny para"+.@input+"licenças.";
					if (Zeny < (.@input*200)) {
						mes "Mas você parece não ter dinheiro suficiente.";
					} else {
						mes "Obrigado pelo seu patrocínio.";
						getitem 6377,.@input; //Buy_Market_Permit
						set Zeny,Zeny-(.@input*200);
					}
					close;
				}
			}
		} else {
			mes "Sou Hugh da equipe de compras da Alberta Merchant Guild.";
			mes "Você é...";

			mes "[Senhor. Hugh]";
			mes "Você sabia? Nossa guilda emitiu uma licença para permitir que os indivíduos comprem mercadorias de outras pessoas.";

			select("Eu nunca tive problemas para comprar itens ...");
			mes "[Senhor. Hugh]";
			mes "Você está certo, mas pense nisso:";
			mes "Você não teve dificuldade em comprar a granel?";
			mes "Você teria que encontrar e conversar com todos que têm um item que você deseja.";
			mes "Muito inconveniente, não é?";
			next;
			mes "[Senhor. Hugh]";
			mes "Como comprar em massa é uma questão importante para os comerciantes dos EUA,";
			mes "Eu propus um plano inovador à nossa guilda, com base nos meus 10 anos de experiência em fazer compras.";
			next;
			mes "[Senhor. Hugh]";
			mes "'Deixe os indivíduos abrirem";
			mes "Uma loja de compradores a granel! '";
			mes "Esse é o título da minha proposta.";
			mes "Você verá, quando lê ...";
			next;
			mes "[Senhor. Hugh]";
			mes "...";
			mes "... (Sr. Hugh Yammers sem parar com todos os detalhes.)";
			next;
			select("Tudo bem, qual é o seu ponto?");
			mes "[Senhor. Hugh]";
			mes "Oh sim. Resumindo,";
			mes "Você pode comprar certos itens a granel através de vendas.";
			next;
			mes "[Senhor. Hugh]";
			mes "Para abrir uma loja de compradores em massa, você precisa de uma licença emitida da Merchant Guild.";
			next;
			mes "[Senhor. Hugh]";
			mes "Você precisa disso toda vez que abrir a loja. Esperamos um aumento significativo nos lucros com esse novo tipo de licenciamento.";
			next;
			mes "[Senhor. Hugh]";
			mes "Esse é o ponto da minha proposta!";
			mes "Nosso presidente ficou tão feliz em saber que vamos ganhar muito dinheiro!";
			next;
			mes "["+strcharinfo(0)+"]";
			mes "Por favor, até o ponto já!";
			next;
			mes "[Senhor. Hugh]";
			mes "Não seja tão impaciente, tudo bem?";
			mes "O que quero dizer é que podemos deixar você abrir a loja de compradores em massa se você aprendeu as vendas.";
			next;
			mes "[Senhor. Hugh]";
			mes "Obviamente, cobramos 10.000 Zeny como uma taxa de inscrição única.";
			mes "Você precisará da licença de loja de compras ^4A4ffbulk ^000000 toda vez que abrir a loja.";
			next;
