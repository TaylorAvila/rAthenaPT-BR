payon,173,238,5	script	Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355ffwait um segundo! No momento, você está carregando muitos itens com você. Por favor, volte depois de colocar algumas de suas coisas no armazenamento Kafra.^000000";
		close;
	}
	mes "[Saída]";
	mes "Traga -me dois";
	mes "Pedras preciosas do";
	mes "a mesma cor, e eu vou";
	mes "Mude -os para pedras preciosas";
	mes "de uma cor diferente.";
	next;
	switch(select("Pedras azuis em vermelhas !: pedras vermelhas vermelhas em amarelo!: Pedras amarelas em azuis!")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[Saída]";
		mes "Hah ...!";
		mes "Você está brincando comigo, certo?";
		mes "Eu não posso te fornecer isso";
		mes "serviço se você não";
		mes "me dê pelo menos";
		mes "2"+ getitemname(.@item_req) +"S!";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[Saída]";
		mes "Eu acredito que posso criar";
		mes "um total de"+ .@gems +""+ getitemname(.@item_id) +"s";
		mes "usando o"+ getitemname(.@item_req) +"s";
		mes "que você tem atualmente.";
		mes "O que você quer fazer?";
		next;
		switch(select("Dê -me o máximo que puder.")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[Saída]";
			mes "Então, quantos";
			mes "você quer?";
			mes "O número máximo";
			mes "que você pode entrar é 100.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Saída]";
					mes "Nenhum?";
					mes "Eu acho que você";
					mes "mudou de idéia ...";
					close;
				}
				else if (.@input > 100) {
					mes "[Saída]";
					mes "Errm ...";
					mes "Eu pedi para você entrar";
					mes "Uma quantidade não maior";
					mes "do que 100, lembra ...?";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[Saída]";
					mes "Errm ...";
					mes "Você não tem isso";
					mes "Muitas jóias para negociar ...";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[Saída]";
			mes "Claro, não há problema.";
			mes "Volte a qualquer momento.";
			close;
		}
		mes "[Saída]";
		mes "Lá vai você.";
		mes "Sinta -se à vontade para vir";
		mes "de volta a qualquer momento.";
		mes "HM, o que é essa procura?";
		mes "Há algo no meu rosto?";
		close;
	}
}
