-	script	Ice Cream Maker::IceCreamer	-1,{
	mes "[Sorveteira]";
	mes "Sorvete fresco feito com neve com Lutie!";
	mes "Aproveite agora, não estará à venda por muito tempo!";
	mes "^3355ff100 Zeny^000000 sorvete,";
	mes "Sorvete!";
	next;
	if(select("Gimme Ice Cream!: Cancelar o comércio") == 2) {
		mes "Tem certeza de que não quer nenhum?";
		mes "Eu não vou vender por muito tempo,";
		mes "E quando eu acabo,";
		mes "Não haverá mais !!!";
		close;
	}
	mes "[Sorveteira]";
	mes "Sorvete fresco feito com neve com Lutie!";
	mes "Todo mundo quer nosso delicioso sorvete,";
	mes "Mas temos uma quantia limitada,";
	mes "Então você só pode comprar 5 por vez !!";
	next;
	while(true) {
		input .@input;
		if(.@input < 1) {
			mes "[Sorveteira]";
			mes "Se você não quiser comprar nenhum,";
			mes "você poderia, por favor, deixar o próximo cliente";
			mes "fazer uma compra?";
			mes "Obrigado.";
			close;
		}
		if(.@input > 5) {
			mes "[Sorveteira]";
			mes "Ai";
			mes "Você espera demais.";
			mes "Querido cliente,";
			mes "Se você comer mais de 5 sorvetes,";
			mes "Se você pode fazer muitas viagens";
			mes "para o banheiro esta noite.";
			next;
			continue;
		}
		break;
	}
	if(Zeny<.@input*100) {
		mes "[Sorveteira]";
		mes "Caro cliente, sua carteira parece ser leve.";
		mes "O preço é ^3355FF100 Zeny ^000000 por sorvete.";
		close;
	}
	if(checkweight(536,.@input) == 0) {
		mes "[Sorveteira]";
		mes "Caro cliente, parece que você está carregando muito.";
		mes "Sorvete está bem,";
		mes "Mas você deve considerar seu peso";
		mes "antes de fazer uma compra.";
		close;
	}
	set Zeny,Zeny-100*.@input;
	getitem 536,.@input;	// Ice_Cream
	close;
}

// Alberta
