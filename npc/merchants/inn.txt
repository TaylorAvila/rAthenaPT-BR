//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "Bem -vindo a";
	mes getarg(1) + ".";
	mes "Como posso ajudá -lo?";
	next;
	switch(select("Salvar: Descanse -> 5000 Zeny: Cancelar")) {
	case 1:
		mes .@npc_name$;
		mes "Seu ponto de reaparecimento";
		mes "foi salvo.";
		mes "Obrigado,";
		mes "Por favor, venha novamente.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "Sinto muito, mas a taxa de serviço é de 5.000 anos. Certifique -se de ter dinheiro suficiente para fazer o check -in na próxima vez, ok?";
			close;
		}
		mes "Obrigado.";
		mes "Eu espero que você";
		mes "Aproveite o seu descanso ~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
