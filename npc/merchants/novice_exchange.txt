function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[Comerciante]";
		mes "Ei, onde estão";
		mes "Todos aqueles"+ getitemname(.@item_req) +"";
		mes "que você prometeu?";
		mes "Me dê"+ getitemname(.@item_req) +"!";
		close;
	}
	else {
		mes "[Comerciante]";
		mes "Ok, deixe -me verificar";
		mes "quantos"+ getitemname(.@item_req) +"você";
		mes "tem em você. Hum...";
		next;
		mes "[Comerciante]";
		mes "Você tem";
		mes "um total de"+ countitem(.@item_req) +""+ getitemname(.@item_req) +".";
		mes "Eu posso te dar um total";
		mes "de"+ (countitem(.@item_req)/.@req_amount) +"Poções vermelhas para isso.";
		next;
		mes "[Comerciante]";
		mes "O que você diz?";
		mes "Temos um acordo?";
		next;
		if (select("Deal.:No deal.") == 1) {
			mes "[Comerciante]";
			mes "Você sabe o exato";
			mes "Número de poções vermelhas";
			mes "você quer receber por";
			mes "aqueles"+ getitemname(.@item_req) +", não é?";
			next;
			mes "[Comerciante]";
			mes "Agora eu posso trocar você";
			mes "um mínimo de 1 poção vermelha";
			mes "e um máximo de 100 poções vermelhas";
			mes "ao mesmo tempo. Se você mudar seu";
			mes "Mente, basta entrar '0' para cancelar.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Comerciante]";
					mes "O que...?";
					mes "Por que diabos você";
					mes "até se incomoda em falar comigo?";
					mes "Muito indeciso, não é?";
					close;
				}
				else if (.@input > 100) {
					mes "[Comerciante]";
					mes "Eu não posso te dar mais";
					mes "de 100 poções vermelhas";
					mes "de uma vez. Vamos tentar";
					mes "isso novamente.";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[Comerciante]";
				mes "Uh oh, o número que você entrou não parece certo. É melhor você verificar o número de poções vermelhas pelas quais você pode negociar novamente.";
				close;
			}
			mes "[Comerciante]";
			mes "Lá vai você!";
			mes "Verifique quantas poções vermelhas";
			mes "Eu te dei, deve ser bom. Obrigado, isso foi um bom negócio ~";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[Comerciante]";
		mes "Huh ~";
		mes "Tudo bem.";
		mes "Embora não seja";
		mes "Poções vermelhas mais úteis";
		mes "Para um aventureiro como você?";
		close;
	}
}
