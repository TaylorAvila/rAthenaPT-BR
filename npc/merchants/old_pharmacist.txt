		mes "- Muitos itens com você. -";
		mes "- Por favor, volte mais tarde -";
		mes "- Depois de colocar alguns itens no armazenamento da Kafra. -";
		close;
	}
	mes "[Antigo farmacêutico]";
	mes "Ummmm ...";
	mes "O que te traz aqui ...?";
	next;
	switch(select("Faça Poção: Fale .: Mistura de Informações: Cancelar")) {
	case 1:
		if (MaxWeight - Weight < 5000) {
			mes "[Antigo farmacêutico]";
			mes "Por que você está carregando esses tantos!";
			mes "Não seja ganancioso, carregue apenas o máximo necessário!";
			next;
			mes "[Antigo farmacêutico]";
			mes "Você é muito pesado para receber poções de mim ...";
			mes "Vá armazenar alguns itens em seu armazenamento primeiro!";
			close;
		} else {
			mes "[Antigo farmacêutico]";
			mes "Você tem todas as coisas prontas, certo? Qual você gostaria?";
			next;
			switch(select("Poção vermelha.")) {
			case 1:
				callsub L_Making,507,3,501;
			case 2:
				mes "[Antigo farmacêutico]";
				if (countitem(507) < countitem(713)) {
					set .@max,countitem(507);
				}
				else if (countitem(508) < countitem(713)) {
					set .@max,countitem(508);
				}
				else {
					set .@max,countitem(713);
				}
				if ((countitem(507) < 1) || (countitem(508) < 1) || (countitem(713) == 0)) {
					mes "Você Rascal! O que você esperava?! Vindo aqui sem nada. TSK!";
					mes "Se perder!";
					close;
				}
				if (Zeny < 3) {
					mes "Você Rascal! O que você esperava?! Vindo aqui sem nada. TSK!";
					mes "Se perder!";
					close;
				}
				mes "Quantos?";
				next;
				switch(select("Faça o máximo que puder.")) {
				case 1:
					if ((countitem(507) < .@max) || (countitem(508) < .@max) || (countitem(713) < .@max) || (Zeny < .@max*3)) {
						mes "[Antigo farmacêutico]";
						mes "Você Rascal! Você nem tem todos os materiais e quer que eu faça suas poções?!";
						close;
					}
					set Zeny,Zeny-(.@max*5);
					delitem 507,.@max; //Red_Herb
					delitem 508,.@max; //Yellow_Herb
					delitem 713,.@max; //Empty_Bottle
					getitem 502,.@max; //Orange_Potion
					break;
				case 2:
					mes "[Antigo farmacêutico]";
					mes "Em seguida, escolha um número abaixo de 100. Se você não quiser, basta digitar '0'. Com os materiais que você tem, você pode fazer sobre"+.@max+"poções.";
					next;
					input .@amount,0,101;
					if (.@amount == 0) {
						mes "[Antigo farmacêutico]";
						mes "Decida -se, sim?!";
						close;
					}
					if (.@amount > 100) {
						mes "[Antigo farmacêutico]";
						mes "Você é surdo? Eu disse menos de 100!";
						close;
					}
					if ((countitem(507) < .@amount) || (countitem(508) < .@amount) || (countitem(713) < .@amount) || (Zeny < .@amount*3)) {
						mes "[Antigo farmacêutico]";
						mes "Você Rascal! Você nem tem todos os materiais e quer que eu faça suas poções?!";
						close;
					}
					next;
					set Zeny,Zeny-(.@amount*5);
					delitem 507,.@amount; //Red_Herb
					delitem 508,.@amount; //Yellow_Herb
					delitem 713,.@amount; //Empty_Bottle
					getitem 502,.@amount; //Orange_Potion
					break;
				case 3:
					mes "[Antigo farmacêutico]";
					mes "O que?!";
					mes "Grrr ...";
					mes "Bleh!";
					mes "Se perder!";
					close;
				}
				mes "[Antigo farmacêutico]";
				mes "Aqui você vai. Tudo está feito para que você possa aceitar. Mas lembre -se! Abusar de remédios não é bom.";
				close;
			case 3:
				callsub L_Making,508,10,503;
			case 4:
				callsub L_Making,509,20,504;
			case 5:
				callsub L_Making,510,30,505;
			case 6:
				callsub L_Making,511,3,506;
			case 7:
				mes "[Antigo farmacêutico]";
				mes "O que?!";
				mes "Grrr ...";
				mes "Bleh!";
				mes "Se perder!";
				close;
			}
		}
	case 2:
		mes "[Antigo farmacêutico]";
		mes "Com a medicina, você pode aumentar a capacidade de uma pessoa de regenerar. Mas, eles são apenas bons até certo ponto. * Suspiro* Estou começando a pensar nos dias em que era jovem. Eu devo estar envelhecendo.";
		next;
		mes "[Antigo farmacêutico]";
		mes "De qualquer forma, uma poção é apenas uma poção. Nada mais e nada menos.";
		close;
	case 3:
		mes "[Antigo farmacêutico]";
		mes "Hrrrmm ...";
		mes "Vocês jovens podem ser bastante irritantes. Mas, desde que você perguntou, eu vou explicar.";
		next;
		mes "[Antigo farmacêutico]";
		mes "As ervas funcionam bem sozinhas, mas se você usar minhas técnicas e habilidades especiais para criar poções, o efeito é muito maior.";
		next;
		mes "[Antigo farmacêutico]";
		mes "Se você perguntar ansiosamente e educadamente, eu os farei para você. Mas, não de graça ... não se preocupe, eu apenas cobro uma pequena taxa, então não é tão caro.";
		next;
		mes "[Antigo farmacêutico]";
		mes "Poção vermelha - ^0098E52 ervas vermelhas, 1 garrafa vazia, 2 Taxa de Zeny. ^000000";
		mes "Poção de laranja - ^0098E51 erva vermelha, 1 erva amarela, 1 garrafa vazia, 5 zeny. ^000000";
		mes "Poção Amarela - ^0098E52 Ervas amarelas, 1 garrafa vazia, 10 Zeny. ^000000";
		next;
		mes "[Antigo farmacêutico]";
		mes "Poção branca - ^0098E52 Ervas brancas, 1 garrafa vazia, 20 Zeny Taxa. ^000000";
		mes "Poção azul - ^0098E52 ervas azuis, 1 garrafa vazia, 30 Zeny Fee. ^000000";
		mes "Poção verde - ^0098E52 ervas verdes, 1 garrafa vazia, 3 Zeny Taxa. ^000000";

		mes "[Antigo farmacêutico]";
		mes "Que pessoa chata. Se você tem algo a dizer, por que você não diz isso?!";
		close;
	}
	
L_Making:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	mes "[Antigo farmacêutico]";
	if ((countitem(.@item_req)/2) < countitem(713)) {
		.@max = countitem(.@item_req) /2;
	} else {
		.@max = countitem(713);
	}
	if (countitem(.@item_req) < 2 || countitem(713) == 0) {
		mes "Você Rascal! O que você esperava?! Vindo aqui sem nada. TSK!";
		mes "Se perder!";
		close;
	}
	if (Zeny < 3) {
		mes "Você Rascal! O que você esperava?! Vindo aqui sem nada. TSK!";
		mes "Se perder!";
		close;
	}
	mes "Quantos?";
	next;
	switch(select("Faça o máximo que puder.")) {
	case 1:
		if ((countitem(.@item_req) < .@max*2) || countitem(713) < .@max || (Zeny < .@max*.@req_amount)) {
			mes "[Antigo farmacêutico]";
			mes "Você Rascal! Você nem tem todos os materiais e quer que eu faça suas poções?!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@max*2;
		delitem 713,.@max; //Empty_Bottle
		getitem getarg(2),.@max; 
		break;
	case 2:
		mes "[Antigo farmacêutico]";
		mes "Em seguida, escolha um número abaixo de 100. Se você não quiser, basta digitar '0'. Com os materiais que você tem, você pode fazer sobre"+.@max+"poções.";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[Antigo farmacêutico]";
			mes "Decida -se, sim?!";
			close;
		}
		if (.@amount > 100) {
			mes "[Antigo farmacêutico]";
			mes "Você é surdo? Eu disse menos de 100!";
			close;
		}
		if ((countitem(.@item_req) < .@amount*2) || countitem(713) < .@amount || (Zeny < .@amount*.@req_amount)) {
			mes "[Antigo farmacêutico]";
			mes "Você Rascal! Você nem tem todos os materiais e quer que eu faça suas poções?!";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@amount*2;
		delitem 713,.@amount; //Empty_Bottle
		getitem getarg(2),.@amount; 
		break;
	case 3:
		mes "[Antigo farmacêutico]";
		mes "O que?!";
		mes "Grrr ...";
		mes "Bleh!";
		mes "Se perder!";
		close;
	}
	mes "[Antigo farmacêutico]";
	mes "Aqui você vai. Tudo está feito para que você possa aceitar. Mas lembre -se! Abusar de remédios não é bom.";
	close;
}
