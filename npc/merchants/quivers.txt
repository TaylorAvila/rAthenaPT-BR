payon_in01,5,134,5	script	Inventor Jaax	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[Inventor Jaax]";
		mes "Ei, você está carregando";
		mes "coisas demais. Por que você não o esconde no armazenamento de kafra? Podemos conversar depois que você faz isso, certo?";
		close;
	}
	mes "[Inventor Jaax]";
	mes "Meu nome é Jaxx.";
	mes "Sem ego, eu posso";
	mes "diga que eu talvez sou o";
	mes "^663300Greatest Inventor do nosso tempo^000000.";
	next;
	mes "[Inventor Jaxx]";
	mes "Desta vez, eu tenho";
	mes "criou algo";
	mes "verdadeiramente extraordinário. Eu os chamo de ... ^663300 Magic Afiadores ^000000 !! Isso será lembrado na história como um";
	mes "Revolução da Arrow!";
	emotion ET_BEST;
	next;
	mes "[Inventor Jaax]";
	mes "Estudei mágica e altividores há anos, trabalhando noite e dia até finalmente imaginei como condensar flechas com magia! Com alertas mágicas, você estará carregando mais flechas, mas com menos peso!";
	next;
	mes "[Inventor Jaax]";
	mes "Gostaria de tentar usar um dos meus altinários de flechas? Não tenho dúvidas de que alguém como você pode apreciar meu gênio!";
	next;
	switch(select("Almonte: Aljúgria de seta de ferro: aljava de seta de aço: OrideCon Arrow Quiver: Fire Arrow Quiver: Silver Arronc")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}
	//This doesn't seem to be accessable...
	//mes "[Inventor Jaax]";
	//mes "Existe";
	//mes "Algo que você quer?";
	//close;

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[Inventor Jaax]";
		mes "Excelente!";
		mes "Você está carregando flechas com você? Vou fornecer uma aljava que pode levar"+ .@amount +"do seu"+ getitemname(.@item_id) +"s apenas ^ff3131"+ .@zeny_req +"Mulheres^000000.";
		next;
		switch(select("Armazene o maior número possível de flechas em afiadores: Compre 1 aljava: Cancelar")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "Número de";
			mes "Setas: ^3131ff"+.@arrows+"^000000";
			mes "Número máximo";
			mes "de compras";
			mes "Afiadores: ^3131ff"+.@quiver+"^000000";
			mes "Zeny necessário: ^3131ff"+.@arrow_zeny01+"Mulheres^000000";
			next;
			mes "[Inventor Jaax]";
			mes "Você gostaria de";
			mes "compre tantos afiadores";
			mes "Como você pode para as flechas";
			mes "Você está carregando atualmente?";
			next;
			if (select("Sim: Cancelar") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[Inventor Jaax]";
					mes "Lá vai você!";
					mes "Lembre -se, ^ff0000 Você não poderá usar a aljava quando o seu peso transportado for 90% do seu limite máximo de peso ^000000.";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[Inventor Jaax]";
					mes "Então...";
					mes "Apenas mantenha o controle";
					mes "de quanto você é";
					mes "carregando do tempo";
					mes "para o tempo e você deveria";
					mes "Esteja bem.";
					close;
				}
				else {
					mes "[Inventor Jaax]";
					mes "Sinto muito, mas você não tem Zeny o suficiente. Eu não posso simplesmente doar isso depois dos anos de trabalho";
					mes "Para desenvolver essa tecnologia revolucionária!";
					close;
				}
			}
			mes "[Inventor Jaax]";
			mes "Você mudou de idéia?";
			mes "Quando a glória de possuir";
			mes "Uma aljava está tão perto?";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[Inventor Jaax]";
				mes "Lá vai você!";
				mes "Lembre -se, ^ff0000 Você não poderá usar a aljava quando o seu peso transportado for 90% do seu limite máximo de peso ^000000.";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[Inventor Jaax]";
				mes "Então...";
				mes "Apenas mantenha o controle";
				mes "de quanto você é";
				mes "carregando do tempo";
				mes "para o tempo e você deveria";
				mes "Esteja bem.";
				close;
			}
			else {
				mes "[Inventor Jaax]";
				mes "Você nem mesmo";
				mes "ter"+ .@zeny_req +"Zeny?";
				mes "Eu sinto muito. Eu não tinha";
				mes "ideia de que você era tão ...";
				mes "^Eu subi^000.";
				close;
			}
		case 3:
			mes "[Inventor Jaax]";
			mes "O que...?";
			mes "Você não vê que esta invenção pode mudar para sempre a maneira como as setas são transportadas?! O futuro é agora!";
			close;
		}
	}
	else {
		mes "[Inventor Jaax]";
		mes "Você pode carregar um máximo de"+ .@amount +"Setas dentro desta aljava. Foi feito usando meu método secreto,";
		mes "Portanto, o peso total das setas e aljava é menor do que carregar as setas sozinhas.";
		next;
		mes "[Inventor Jaax]";
		mes "É um milagre da ciência! Um que você pode experimentar por si mesmo se me traga pelo menos"+ .@amount +"Setas e"+ .@zeny_req +"Zeny para cada aljava.";
		close;
	}
}
