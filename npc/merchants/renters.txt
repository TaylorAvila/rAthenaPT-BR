// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	Peco Peco Breeder#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[PECO PECO Breeder]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "Bem-vindo.";
		mes "Cavaleiro Honorável,";
		mes "você gostaria de alugar";
		mes "um peco pico? O aluguel";
		mes "taxa é"+.@price+"mulheres.";
		next;
		switch(select("Alugue PECO PECO: Cancelar")) {
		case 1:
			if (Zeny < .@price) {
				mes "[PECO PECO Breeder]";
				mes "Você não";
				mes "tem zeny suficiente.";
				mes "Você é...";
				mes "falido?";
				close;
			}
			else if(getskilllv("Kn_riding") == 0) {
				mes "[PECO PECO Breeder]";
				mes "Sinto muito, mas você está";
				mes "não é elegível para isso";
				mes "serviço. Por favor, vá aprender";
				mes "A habilidade PECO PECO Ride primeiro.";
				close;
			}
			else if(checkriding()) {
				mes "[PECO PECO Breeder]";
				mes "Você já é";
				mes "montado em um";
				mes "PECO PECO.";
				close;
			}
			else if(ismounting()) {
				mes "[PECO PECO Breeder]";
				mes "Por favor, remova sua montagem em dinheiro.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[PECO PECO Breeder]";
			mes "Eu vejo.";
			mes "Bem então,";
			mes "Tenha um bom dia.";
			close;
		}
	} else {
		mes "Sinto muito, mas estes";
		mes "PECO PECOs são apenas";
		mes "Disponível para cavaleiros";
		mes "e Lord Knights.";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	Peco Peco Breeder#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[Criador de PECEPECO]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "Bem -vindo, Crusader.";
		else mes "Bem -vindo, Paladin.";
		mes "Temos um especial";
		mes "PECO PECO preparado";
		mes "para você. Para alugar um";
		mes "custará"+.@price+"mulheres.";
		next;
		switch(select("Alugue um PECEPECO: Pare")) {
		case 1:
			if (Zeny < .@price) {
				mes "[PECO PECO Breeder]";
				mes "Você não";
				mes "tem zeny suficiente.";
				mes "Se você gostaria";
				mes "um peco pico por favor";
				mes "trazer"+.@price+"Zeny ...";
				close;
			}
			else if(getskilllv("Kn_riding") == 0) {
				mes "[PECO PECO Breeder]";
				mes "Você deve primeiro aprender";
				mes "Para andar de peco antes";
				mes "Eu posso alugar um para você.";
				close;
			}
			else if(checkriding()) {
				mes "[PECO PECO Breeder]";
				mes "Você já está";
				mes "montado em um peco pico.";
				close;
			}
			else if(ismounting()) {
				mes "[PECO PECO Breeder]";
				mes "Por favor, remova sua montagem em dinheiro.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Criador de PECEPECO]";
			mes "Vejo você por aí.";
			close;
		}
	} else {
		mes "O que posso fazer para você?";
		mes "Por favor esteja ciente de que";
		mes "Este PECO PECO RENTAL";
		mes "O serviço é estritamente para";
		mes "Cruzados e paladinos.";
		close;
	}
}
