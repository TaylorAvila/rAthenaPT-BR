function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Seiyablem]";
	mes "Você quer adicionar um slot a um"+ getitemname(.@item_id) +"?";
	mes "Tudo bem, por favor me traga ^ff0000"+ .@req_amount[0] +""+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000,"+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +""+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000") : "") + "e meu ^ff0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +"000 Zeny^000000 Taxa de serviço.";
	mes "Ah, e não se esqueça de trazer isso"+ getitemname(.@item_id) +"!";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "Posso tentar adicionar um slot agora se você tiver os itens necessários e o zeny.";
		mes "No entanto, você deve saber que há uma chance de eu falhar.";
		mes "Portanto, preciso lhe dar um aviso justo ...";
		next;
		mes "[Seiyablem]";
		mes "Se essa tentativa de adicionar um slot à sua arma falhar, então o ^ff0000weapon ^000000, ^ff0000 e quaisquer cartões compostos será destruída ^000000.";
	}
	else // armor
	{
		mes "Se você possui todos os materiais necessários, minha taxa de serviço de Zeny e a armadura, podemos seguir em frente com a tentativa de adição de slot.";
		mes "Mas antes disso, devo avisá -lo sobre o risco.";
		next;
		mes "[Seiyablem]";
		mes "Se essa tentativa de adicionar um slot à sua armadura falhar, então o ^ff0000Armor ^000000, ^FF0000Ir upgrades ^000000 ^ff0000 e quaisquer cartões compostos serão destruídos ^000000.";
	}
	mes "Você ainda quer tentar adicionar um slot?";
	next;
	switch(select("Tente adição de slot: cancelar"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Seiyablem]";
				mes "Tudo bem, então, deixe o trabalho começar!";
				mes "É melhor você orar por um resultado bem -sucedido.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[Seiyablem]";
					mes "Ótimo, parece ter sucesso.";
					mes "Parece muito bem feito. Parabéns!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Seiyablem]";
					mes "Vejo você de novo, amigo!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Seiyablem]";
					mes "Wah! ... Sinto muito, falhou.";
					mes "No entanto, sou completamente inocente.";
					mes "Esta é a sua sorte, e é destinado a Deus, ok?";
					mes "Não fique tão desapontado e tente na próxima vez.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Seiyablem]";
					mes "Desejo -lhe boa sorte na próxima vez!";
					close;
				}
			}
			else
			{
				mes "[Seiyablem]";
				mes "Eu gostaria de prosseguir com essa tentativa de adição de slot, mas você está perdendo algumas coisas.";
				mes "Você tem certeza de que tem o equipamento, os materiais necessários e o Zeny?";
				close;
			}
		case 2:
			mes "[Seiyablem]";
			mes "Precisa de algum tempo para pensar sobre isso, hein?";
			mes "Tudo bem, eu posso entender.";
			mes "Lembre -se de que a vida não é divertida se você está sempre jogando seguro ~";
			close;
	}
}
