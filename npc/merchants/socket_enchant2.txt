function	script	Func_Socket2	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[SLEAPLE]";
	if (.@item_req[1] != 0 && .@req_amount[1] != 0)
	{
		mes getitemname(.@item_id) +"... ok, então você precisa me trazer";
		mes "^FF0000"+ (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +"000 zeny como cobrança de serviço,";
		mes "^FF0000"+ .@req_amount[0] +"de"+ getitemname(.@item_req[0]) +"e"+ .@req_amount[1] +"de"+ getitemname(.@item_req[1]) +".";
		mes "^Ff0000 de claro, você precisa de um"+ getitemname(.@item_id) +".^000000";
	}
	else
	{
		mes getitemname(.@item_id) +"? Ok, então você precisa me trazer";
		mes "^FF0000"+ (.@zeny_req >= 1000 ? (.@zeny_req/1000 +",000") : .@zeny_req) +"000 zeny como cobrança de serviço e"+ .@req_amount[0] +""+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"como requisito.";
		mes "^Ff0000 de claro, não se esqueça de me trazer um"+ getitemname(.@item_id) +".";
		mes "^Ff0000 Você deve ter todos os itens.^000000";
	}
	next;
	mes "[SLEAPLE]";
	mes "Você já trouxe todos eles?";
	mes "Para suas informações, se você não criar um slot,";
	mes "você perderá todo o requisito do item, bem como o alvo"+ ((getiteminfo(.@item_id, ITEMINFO_LOCATIONS)&EQP_HAND_R) ? "arma" : "armadura") +".";
	mes "Lembre -se também, se o"+ ((getiteminfo(.@item_id, ITEMINFO_LOCATIONS)&EQP_HAND_R) ? "arma" : "armadura") +"foi atualizado e foi inserido com um cartão,";
	mes "Você os perderá mesmo se conseguir criar um slot.";
	next;
	switch(select("Peça a criação de slot.: Tente da próxima vez."))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no .@req_amount[0] and .@req_amount[1] provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[SLEAPLE]";
				mes "Tudo bem, então, deixe o trabalho começar!";
				mes "É melhor você orar por um resultado bem -sucedido.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect EF_LORD;
					mes "[SLEAPLE]";
					mes "Ótimo, parece ter sucesso.";
					mes "Parece muito bem feito. Parabéns!";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[SLEAPLE]";
					mes "Vejo você de novo, amigo!";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[SLEAPLE]";
					mes "Wah! ... Sinto muito, falhou.";
					mes "No entanto, sou completamente inocente.";
					mes "Esta é a sua sorte, e é destinado a Deus, ok?";
					mes "Não fique tão desapontado,";
					mes "e tente na próxima vez.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
