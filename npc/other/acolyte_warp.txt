prontera,98,121,4	script	Acolyte#Prtclear	90,{
	mes "[Q]";
	mes "Olá, aventureiro.";
	mes "Eu tenho estudado magia de todo o Rune-Midgarts para atualizar o que acredito ser uma das maiores habilidades disponíveis para a classe Acolyte.";
	next;

	if (is_function("VIP_IRO_ACOLYTE"))
		callfunc "VIP_iRO_Acolyte","[Keiki]","Juno";

	mes "[Q]";
	mes "Eu sou o único acólito que atingiu a habilidade do portal de Warp Nível 10!";
	emotion ET_THINK, playerattached();
	next;
	mes "[Q]";
	mes "Isso mesmo! E...";
	mes "Eu prometo a você que não esqueço os locais que já memorizei.";
	mes "One day I will level up my skills to warp to wherever I please~";
	next;
	mes "[Q]";
	mes "I am willing to warp you to the many locations that I have memorized for a small fee.";
	mes "Você gostaria de usar este serviço?";
	next;
	switch(select("Sim: Não")) {
	case 1:
		mes "[Q]";
		mes "Para onde você gostaria de ir?";
		mes "Desejo -lhe boa sorte em sua jornada.";
		next;

		setarray .@towns$[0], "Chocado", "Se encaixa", "Teto", "Marrocos", "Alberta", "Baran", "Confortável", "Cor", "Juno";
		setarray .@cost[0],        600,     1200,    1200,     1200,      1800,          2200,     2200,     2200,   1800;

		set .@size, getarraysize(.@towns$);
		for(set .@i,0; .@i<.@size; set .@i,.@i+1)
			set .@menu$, .@menu$+.@towns$[.@i]+"->"+.@cost[.@i]+"Com:";
		set .@i, select(.@menu$+"Cancelar")-1;
		if (.@i == .@size)
			close;
		if (Zeny < .@cost[.@i]) {
			mes "[Q]";
			mes "Sinto muito, mas você não tem";
			mes "Zeny suficiente para o teleporte";
			mes "Serviço. A taxa para se teletransportar";
			mes "para"+.@towns$[.@i]+"é"+.@cost[.@i]+"mulheres.";
			close;
		}
		specialeffect2 EF_READYPORTAL;
		specialeffect2 EF_TELEPORTATION;
		specialeffect2 EF_PORTAL;
		next;
		set Zeny, Zeny - .@cost[.@i];
		switch(.@i) {
			case 0: if (checkre(0)) warp "izlude",128,98; else warp "izlude",91,105; break;
			case 1: warp "geffen",120,39; break;
			case 2: warp "payon",161,58; break;
			case 3: warp "morocc",156,46; break;
			case 4: warp "alberta",117,56; break;
			case 5: warp "aldebaran",168,112; break;
			case 6: warp "comodo",209,143; break;
			case 7: warp "umbala",100,154; break;
			case 8: warp "yuno",158,125; break;
		}
		close;
	case 2:
		close;
	}
}
