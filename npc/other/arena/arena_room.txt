function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[Givu]";
	mes "Você gostaria de trocar seus pontos de arena com"+ getitemname(.@item_id) +"?";
	mes "Você pode trocar ^3131ff"+ .@arena_point +"Arena aponta com"+ .@amount +""+ getitemname(.@item_id) +"^000000.";
	mes "Se você deseja cancelar, digite 0. Se não o fizer, digite quantos ^3131ffTimes ^000000 dos pontos de arena que você deseja gastar.";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[Givu]";
		mes "Você cancelou seu pedido.";
		close;
	} else if (.@reward > 1500) {
		mes "[Givu]";
		mes "Você excedeu a capacidade máxima.";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[Givu]";
		mes "Você não tem pontos de arena suficientes.";
		mes "Por favor, verifique a quantidade total de pontos de arena que você tem.";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[Givu]";
	mes "Obrigado, por favor volte.";
	close;
}
