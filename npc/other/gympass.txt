payon,173,141,4	script	Ripped Cabus#GymPass	899,{
	mes "[Rasgado]";
	mes "Ei. As pessoas não são";
	mes "Tão fisicamente ativo quanto eles";
	mes "costumava ser. Mesmo se você lutar";
	mes "Para viver, seu corpo pode";
	mes "Seja fraco e flácido em alguns";
	mes "áreas. Sabe o que quero dizer?";
	next;
	mes "[Rasgado]";
	mes "Ei, treine comigo, e eu posso";
	mes "Garanta que você será capaz";
	mes "para levantar e carregar mais de seu";
	mes "coisa. Apenas me dê o seu";
	mes "^Ff0000GYM PASS^000000 A cada vez,";
	mes "E estaremos prontos para ir.";
	next;
	mes "[Rasgado]";
	mes "Mas não fique muito animado:";
	mes "Não importa quanto treinamento";
	mes "Eu te levo, você pode";
	mes "exagerou. Você já ouviu";
	mes "Alguém que ficou muito buff?";
	mes "Isso é porque eles estão mortos. Ver?";
	next;
	mes "[Rasgado]";
	mes "Eu diria que seria seguro";
	mes "Para você treinar seriamente";
	mes "comigo e aumente o seu";
	mes "Item de capacidade de transporte ^FF000010 vezes ^000000.";
	mes "Então ... você está pronto para suar?";
	next;
	switch(select("Sim: Não: Hum, meus exercícios se depararam.")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[Rasgado]";
				mes "Oh, incrível, eu vejo você";
				mes "trouxe seu passe de academia.";
				mes "Tudo bem, apenas faça o que eu faço,";
				mes "e tente sentir a queimadura.";
				mes "Preparar? Vamos fazer isso.";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[Rasgado]";
				mes "Lá, você deve ser capaz";
				mes "para carregar mais coisas com você.";
				mes "Vamos ver, podemos aumentar";
				mes "Seu item de capacidade de transporte";
				mes "^Ff00000" + .@remain_carry + "^000000 mais vezes se continuarmos";
				mes "treinando juntos assim.";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "All_inccarry",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[Rasgado]";
				mes "Cara, o que eu disse a você?";
				mes "Você tem que me trazer seu";
				mes "^Ff0000GYM PASS^000000 Se você quiser";
				mes "malhar e construir seu";
				mes "item carregando músculos.";
				close;
			}
		}
		else {
			mes "[Rasgado]";
			mes "Cara, acho que não podemos";
			mes "Construa seu item carregando";
			mes "músculos mais do que isso.";
			mes "É muito perigoso para o seu";
			mes "corpo se até tentarmos! Vamos lá,";
			mes "Eu te contei sobre os limites.";
			close;
		}
	case 2:
		mes "[Rasgado]";
		mes "Ah, isso é muito ruim.";
		mes "Bem, volte se você";
		mes "mude de idéia. Dizer";
		mes "Seus amigos sobre mim:";
		mes "Se eles forem flácidos, eu vou ajudar";
		mes "coloque -os em forma.";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "[Rasgado]";
			mes "O que aconteceu?";
			mes "Você deixa seu item carregar";
			mes "Músculos apenas atrofiaram? Sortudo";
			mes "Para você, existe uma coisa dessas";
			mes "como memória muscular. Não vai demorar";
			mes "Tanto para construí -los de volta ...";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[Rasgado]";
			mes "Que tal isso?";
			mes "Seu item carregando";
			mes "Os músculos voltaram,";
			mes "Assim assim! Tente não";
			mes "Wimp Out de novo, ok?";
			skill "All_inccarry",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[Rasgado]";
			mes "Uhh ...";
			mes "Não deu certo";
			mes "juntos antes.";
			mes "Tenho certeza disso.";
			close;
		}
	}
}
