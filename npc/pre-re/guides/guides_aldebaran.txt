aldebaran,139,63,4	script	Soldier#alde	105,{
	cutin "prt_soldier",2;
	mes "[Para Baran Guard]";
	mes "Eu sou apenas um guarda comum";
	mes "que você pode encontrar em qualquer outra cidade. Acho que nem tenho um nome ...";
	next;
	mes "[Para Baran Guard]";
	mes "Estou encarregado dos guias de serviço do Al de Baran Garrison. Deixe -me guiar você";
	mes "Através da nossa cidade!";
	next;
	if (select("Obtenha o guia de localização.: Conversação.") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^Ff6633+^000000 -> Escritório principal da CAFRA";
		mes "^0000FF+^000000 -> loja de armas";
		mes "^00ffff+^000000 -> Guilda de feiticeiro (fechada)";
		mes "^515151+^000000 -> Pub";
		mes "^3355ff+^000000 -> Loja de itens";
		mes "^Ff5555+^000000 -> Guilda de Alquimista";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[Para Baran Guard]";
	mes "Justamos proteger Al de Baran! Que as forças do mal sempre sejam esmagadas pelo";
	mes "Punho justo de bom!";
	close2;
	cutin "prt_soldier",255;
	end;
}
