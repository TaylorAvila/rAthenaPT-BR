rachel,138,146,5	script	Rachel Guide	934,{
	mes "[Rachel Guide]";
	mes "Bem -vindo à capital de";
	mes "Arunafeltz, Rachel onde um calor";
	mes "Breath of deusa Freya chega.";
	mes "Se esta é a primeira vez para você";
	mes "Para usar os serviços guia, por que";
	mes "Você não verifica o \"Notice\"menu primeiro?";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("Guia da vila: Remova as notas do Mini-mapa: Aviso: Cancelar")) {
		case 1:
			mes "[Rachel Guide]";
			mes "Posso lhe dizer qualquer local de construção, desde que seja em Rachel.";
			mes "Então, para onde você quer ir?";
			if (.@compass_check == 0) {
				mes "Você gostaria de mim";
				mes "Para marcar locais";
				mes "em seu mini-mapa?";
				next;
				if (select("Sim: Não") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("Cheshrumnir: Inn: Armas Shop: Loja de ferramentas: Aeroporto: Cancelar")) {
				case 1:
					mes "[Rachel Guide]";
					mes "Cheshrumnir é um terreno sagrado onde o papa, a encarnação da deusa Freya fica.";
					mes "Pegue a estrada para o norte para encontrar o prédio.";
					if (.@compass_check == 1) viewpoint 1,150,249,1,0xFF0000;
					break;
				case 2:
					mes "[Rachel Guide]";
					mes "Você pode descansar sua fadiga fora da jornada na pousada.";
					mes "O prédio esquerdo ao meu lado é a pousada de Rachel.";
					if (.@compass_check == 1) viewpoint 1,115,149,2,0xFF00FF;
					break;
				case 3:
					mes "[Rachel Guide]";
					mes "Você quer conferir as armas que são vendidas em Rachel?";
					mes "A loja de armas está localizada nas proximidades do portão ocidental.";
					if (.@compass_check == 1) viewpoint 1,42,87,3,0x99FFFF;
					break;
				case 4:
					mes "[Rachel Guide]";
					mes "A Rachel Tool Shop vende as poções da melhor qualidade.";
					mes "Está localizado nas proximidades do portão ocidental.";
					if (.@compass_check == 1) viewpoint 1,83,78,4,0x0000FF;
					break;
				case 5:
					mes "[Rachel Guide]";
					mes "O aeroporto está localizado fora do portão oriental.";
					if (.@compass_check == 1) viewpoint 1,273,125,5,0x00FF00;
					break;
				case 6:
					mes "[Rachel Guide]";
					mes "Se você gosta de se livrar de todas as marcas de localização em seu mini-mapa,";
					mes "Apenas me pergunte novamente e escolha \"Remove Marks from Mini-Map\"menu.";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,150,249,1,0xFF0000;
			viewpoint 2,115,149,2,0xFF00FF;
			viewpoint 2,42,87,3,0x99FFFF;
			viewpoint 2,83,78,4,0x0000FF;
			viewpoint 2,273,125,5,0x00FF00;
			mes "[Rachel Guide]";
			mes "Ok, eles se foram agora. Se você tiver mais locais para perguntar, me avise.";
			break;
		case 3:
			mes "[Rachel Guide]";
			mes "Quando você está usando o menu \"village\",";
			mes "Verifique se os locais de construção serão marcados no seu mini-mapa no lado superior direito da tela.";
			mes "Se você não puder ver seu mini-mapa, use a tecla de atalho \"Ctrl+Tab\" ou pressione o botão nas suas janelas de informações básicas, ok?";
			mes "E você também pode ampliar seu mini-mapa usando o botão \"-\", caso não possa visualizar o mapa inteiro da vila.";
			break;
		case 4:
			mes "[Rachel Guide]";
			mes "Espero que você tenha uma jornada maravilhosa";
			mes "em Arunafeltz.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}
