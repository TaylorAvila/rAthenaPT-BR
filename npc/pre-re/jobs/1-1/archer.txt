payon_in02,64,71,4	script	Archer Guildsman#archer	85,{
	if (Upper == 1) {
		if (Class == Job_Novice_High && (ADVJOB == Job_Sniper || ADVJOB == Job_Clown || ADVJOB == Job_Gypsy)) {
			mes "[Archer Guildsman]";
			mes "Ei, eu te conheço.";
			mes "Você fez este teste";
			mes "Antes, não é?";
			next;
			mes "[Archer Guildsman]";
			mes "Ah, você deve ter sido";
			mes "a Valhalla e foi renascido.";
			mes "Uau, isso é tão impressionante!";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Archer Guildsman]";
				mes "Errar...";
				mes "É melhor você aprender todas as habilidades básicas antes de se tornar um arqueiro.";
				next;
				mes "[Archer Guildsman]";
				mes "Tudo bem, vejo você mais tarde.";
				close;
			}
			mes "[Archer Guildsman]";
			mes "Bem então. Eu não";
			mes "Precisa dizer qualquer outra coisa.";
			mes "Eu sei que você fará um ótimo arqueiro ...";
			next;
			skill "Nv_trickdead",0,SKILL_PERM;
			jobchange Job_Archer_high;
			skill "Ac_makingarrow",1,SKILL_PERM;
			skill "AC_CHARGEARROW",1,SKILL_PERM;
			mes "[Archer Guildsman]";
			mes "Embora não haja especial";
			mes "Recompensa por você desta vez, espero que você entenda. Cuide-se.";
			close;
		}
		else {
			mes "[Archer Guildsman]";
			mes "Oh...?";
			mes "Ei, o que é";
			mes "Você está fazendo aqui ...?";
			next;
			mes "[Archer Guildsman]";
			mes "Posso dizer que você não é cortado para ser um arqueiro. Parece que você deve fazer";
			mes "outra coisa ...";
			close;
		}
	}
	mes "[Archer Guildsman]";
	mes "Bom dia. Como posso ajudá -lo?";
	next;
	switch(select("Eu quero ser um arqueiro .: Eu preciso dos requisitos, por favor.")) {
	case 1:
		if (BaseJob == Job_Archer) {
			mes "[Archer Guildsman]";
			mes "Você já se tornou um arqueiro ...";
			close;
		}
		else if (BaseJob != Job_Archer && BaseJob != Job_Novice) {
			mes "[Archer Guildsman]";
			mes "Hum...";
			mes "Você não parece muito um novato ...";
			next;
			mes "[Archer Guildsman]";
			mes "De qualquer forma, seja o que for, você não pode escolher um trabalho como arqueiro, porque já tem um emprego.";
			close;
		}
		if (job_archer_q == 0) {
			mes "[Archer Guildsman]";
			mes "Você quer ser um arqueiro?";
			mes "Nesse caso, você precisa preencher este formulário de inscrição.";
			next;
			if (select("Aplicar.:Cancel") == 1) {
				set job_archer_q,1;
				setquest 1004;
				mes "[Archer Guildsman]";
				mes "Ok, assine aqui. Tudo bem, hum, eu vou promovê -lo assim que atender aos requisitos.";
				next;
				mes "[Archer Guildsman]";
				mes "Se você acha que já os conheceu, podemos verificar isso agora.";
				mes "Você está pronto?";
				next;
				if (select("Sim, eu sou.") == 1) {
					mes "[Archer Guildsman]";
					mes "Tudo bem, deixe -me verificar.";
					next;
				}
				else {
					mes "[Archer Guildsman]";
					mes "Eu entendo. Seja meu convidado se quiser olhar para os requisitos.";
					close;
				}
			}
			else {
				mes "[Archer Guildsman]";
				mes "Bem, tudo bem.";
				mes "Vejo você na próxima vez.";
				close;
			}
		}
		mes "[Archer Guildsman]";
		mes "Você é..." + strcharinfo(0) + "?";
		next;
		if (!callfunc("F_CanChangeJob")) {
			mes "[Archer Guildsman]";
			mes "Bem, você não está no nível certo. Por favor, verifique os requisitos novamente.";
			next;
			mes "[Archer Guildsman]";
			mes "Seu nível de trabalho deve ter pelo menos 10 e não se esqueça de que você deve aprender todas as habilidades básicas. Depois de fazer isso, volte.";
			close;
		}
		if (job_archer_q == 1) {
			set .@archer_item1,countitem(1066) * 5;
			set .@archer_item2,countitem(1067) * 3;
			set .@archer_item3,countitem(1068) * 2;
			set .@archer_item4,countitem(1019);
			set .@total_archer,.@archer_item1 + .@archer_item2 + .@archer_item3 + .@archer_item4;
			set .@total_archer2,(((.@archer_item2 + .@archer_item3) * 2) + .@archer_item4);
			mes "[Archer Guildsman]";
			mes "Excelente!";
			mes "Agora então,";
			mes "Vamos ver ...";
			next;
			mes "[Archer Guildsman]";
			mes "Avaliarei o valor dos vários tipos de troncos, necessários para produzir um arco, que você trouxe.";
			next;
			if (countitem(1066) == 0 && countitem(1067) == 0 && countitem(1068) == 0 && countitem(1019) == 0) {
				mes "[Archer Guildsman]";
				mes "Um...";
				mes "Infelizmente, você não trouxe nenhum dos itens necessários. Não há nada para eu avaliar.";
				close;
			}
			mes "[Archer Guildsman]";
			if (countitem(1066) != 0) mes "Grade S:" + countitem(1066) + "De, nota:" + .@archer_item1 + ".";
			if (countitem(1067) != 0) mes "Grau A:" + countitem(1067) + "De, nota:" + .@archer_item2 + ".";
			if (countitem(1068) != 0) mes "Grau B:" + countitem(1068) + "De, nota:" + .@archer_item3 + ".";
			if (countitem(1019) != 0) mes "Grau C:" + countitem(1019) + "De, nota:" + .@archer_item4 + ".";
			if (.@total_archer < 25) {
				mes "Notas totais: ^FF0000" + .@total_archer + "^000000 / 40";
				next;
				mes "[Archer Guildsman]";
				mes "Menos de 25!? Você tem que obter uma nota de pelo menos 25! Vamos lá, tente mais!";
				close;
			}
			else {
				mes "Notas totais: ^0000ff" + .@total_archer + "^000000 / 40";
				next;
				mes "[Archer Guildsman]";
				if (.@total_archer > 40) {
					mes "Uau! Mais de 40!";
					mes "Excelente! Parabéns!";
				}
				else if (.@total_archer > 30) {
					mes "Mais de 30! Bom trabalho!";
					mes "Parabéns!";
				}
				else {
					mes "* Suspiro* Bem, você mal passou ... de qualquer maneira, muito bem.";
				}
			}
			next;
			mes "[Archer Guildsman]";
			mes "Vou transferir esses troncos para o nosso departamento de produção de arco. Agora que você atendeu aos requisitos, deixe -me promovê -lo imediatamente!";
			if (countitem(1066) != 0) delitem 1066,countitem(1066); //Tree_Of_Archer_1
			if (countitem(1067) != 0) delitem 1067,countitem(1067); //Tree_Of_Archer_2
			if (countitem(1068) != 0) delitem 1068,countitem(1068); //Tree_Of_Archer_3
			if (countitem(1019) != 0) delitem 1019,countitem(1019); //Wooden_Block
		}
		next;
		callfunc "Job_Change",Job_Archer;
		callfunc "F_ClearJobVar";
		completequest 1004;
		mes "[Archer Guildsman]";
		mes "Parabéns!";
		mes "Você agora é um arqueiro!";
		next;
		mes "[Archer Guildsman]";
		mes "Obviamente, esperamos que você ajude a contribuir para o futuro da guilda de Archer com seus esforços.";
		next;
		mes "[Archer Guildsman]";
		mes "Ah, seu arco chegou do departamento de produção de arco. Aqui, leve! É seu ~";
		getitem 1702,1; //Bow_
		getitem 1750,.@total_archer2; //Arrow
		next;
		mes "[Archer Guildsman]";
		mes "Agora, você vai. Caça de orgulho, sabendo que você foi treinado por um dos melhores!";
		close;
	case 2:
		mes "[Archer Guildsman]";
		mes "Vou explicar os requisitos para ser um arqueiro.";
		if (BaseJob != Job_Novice) {
			if (BaseJob == Job_Archer) {
				next;
				mes "[Archer Guildsman]";
				mes "Mas...";
				mes "Você já é um arqueiro. Você já deve saber isso ...";
			}
			else {
				next;
				mes "[Archer Guildsman]";
				mes "Espere um segundo. Você já escolheu um trabalho diferente. Você não precisa saber disso ~";
			}
			mes "Então ... sim ... sem motivo real para lhe dizer os requisitos ...";
		}
		next;
		mes "[Archer Guildsman]";
		mes "Primeiro de tudo, você tem o nível 9 como um novato e conhece todas as habilidades básicas.";
		next;
		mes "[Archer Guildsman]";
		mes "Um arqueiro precisa de concentrações e reflexos extremamente altos, por isso não aceitamos aqueles que têm pouca paciência.";
		next;
		mes "[Archer Guildsman]";
		mes "Você também precisa reunir ^ff0000Trunks ^000000. Existem 4 tipos diferentes de troncos, cada um com qualidade diferente. Você receberá graus diferentes para seus troncos, dependendo da qualidade deles.";
		next;
		mes "[Archer Guildsman]";
		mes "Para se tornar um arqueiro, você deve receber uma nota de pelo menos ^0000FF25 ^000000 pontos em 40. Você pode obter troncos de 'Willow', a árvore. Tenha cuidado, no entanto. Eles podem ser monstros difíceis.";
		close;
	case 3:
		close;
	}
}
