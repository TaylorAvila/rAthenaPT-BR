que_ng,187,149,3	script	Magazine Dealer Kenny::mdk	4_M_01,{
	mes "[Kenny]";
	if (BaseClass == Job_Gunslinger) {
		mes "Bem -vindo à minha loja de revistas.";
		mes "Como você deve saber, grandes números";
		mes "de balas podem ser transportadas mais";
		mes "facilmente quando eles estão em revistas. Agora, posso lhe interessar em";
		mes "alguma coisa em particular?";
		next;
		switch(select("Pacote de esfera do vento: Shadow Sphere Pack: Poison Sphere Pack: Water Sphere Pack: Fire Sphere Pack: Cartucho: Cartucho de Sangue: Cartucho de Prata: Cancelar")) {
		case 1: callsub S_Casing,13204,12144;// <bullet id>, <pack id>
		case 2: callsub S_Casing,13206,12145;
		case 3: callsub S_Casing,13205,12146;
		case 4: callsub S_Casing,13207,12147;
		case 5: callsub S_Casing,13203,12148;
		case 6: callsub S_Casing,13200,12149;
		case 7: callsub S_Casing,13202,12150;
		case 8: callsub S_Casing,13201,12151;
		case 9:
			mes "[Kenny]";
			mes "Bem, se você encontrar";
			mes "que você tem muitos";
			mes "balas, venha me ver.";
			mes "É uma ideia inteligente para armazenar";
			mes "balas com minhas revistas.";
			close;
		}
	}
	mes "Bem -vindo à minha loja.";
	mes "Aqui, eu forneço revistas";
	mes "e cartuchos para pistoleiros.";
	mes "Desculpe, mas não parece";
	mes "Meus serviços seriam de qualquer";
	mes "Use a você, aventureiro.";
	next;
	mes "[Kenny]";
	mes "Eh, mas se você acontecer";
	mes "Conheça qualquer pistoleiro, envie";
	mes "eles do meu jeito. Você nunca pode";
	mes "tem muitas balas.";
	close;

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);

	mes "[Kenny]";
	mes "Agora, você pode negociar";
	mes callfunc("F_InsertPlural",500,getitemname(.@bullet_id));
	mes "e 500 Zeny para 1"+getitemname(.@pack_id)+", então certifique -se de";
	mes "Você tem balas suficientes";
	mes "e Zeny para esta troca.";
	next;
	mes "[Kenny]";
	mes "You can exchange "+callfunc("F_InsertPlural",500,getitemname(.@bullet_id))+" and 500 zeny";
	mes "com 1"+getitemname(.@pack_id)+".";
	next;
	mes "[Kenny]";
	mes "Lembre -se de que não posso dar";
	mes "Vocês mais de 50 revistas";
	mes "de cada vez. Agora, por favor, digite";
	mes "o número de revistas que você";
	mes "quero receber. Se você deseja cancelar, basta entrar '' 0. ''";
	next;
	input .@amount;
	mes "[Kenny]";
	if (.@amount > 50 || .@amount == 0) {
		mes "Ei, eu não posso te dar";
		mes "tantas revistas.";
		mes "Por favor, tente novamente e";
		mes "Digite um número não";
		mes "maior que 50.";
	}
	else if (countitem(.@bullet_id) >= (500*.@amount)) {
		if (Zeny >= (500*.@amount)) {
			if (checkweight(.@pack_id,.@amount) == 0) {
				mes "Ei, você tem muito";
				mes "de lixo amontoado em seu";
				mes "Inventário. Liberte alguns";
				mes "espaço e depois volte";
				mes "e trocar suas balas por";
				mes "Algumas revistas depois, ok?";
			} else {
				mes "Tudo bem, aqui estão";
				mes "suas revistas. Obrigado";
				mes "para visitar minha loja e";
				mes "Espero que você use tudo";
				mes "da sua munição sabiamente.";
				Zeny = Zeny -500*.@amount;
				delitem .@bullet_id,(500*.@amount);
				getitem .@pack_id,.@amount;
			}
		} else {
			mes "Desculpe, mas você não";
			mes "tem zeny suficiente para";
			mes "Esta troca de revistas.";
			mes "Volte para minha loja";
			mes "Depois que você salvou";
			mes "Mais um pouco de dinheiro.";
		}
	} else {
		mes "Desculpe, mas você não";
		mes "tem balas suficientes para";
		mes "Esta troca de revistas.";
		mes "Talvez você deva dobrar";
		mes "Verifique seu inventário e";
		mes "Então volte para mim mais tarde.";
	}
	close;
}
