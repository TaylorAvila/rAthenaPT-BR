alberta,118,160,3	script	Bullet Dealer Tony#alb::bdt	4_M_04,{
	mes "[Tony]";
	if (BaseClass == Job_Gunslinger) {
		mes "Eu sou Tony, o revendedor de balas.";
		mes "Venha até mim sempre que você estiver";
		mes "curto em munição. Apenas me traga";
		mes "os materiais, e eu vou fazer";
		mes "Vocês as balas que você precisa.";
		next;
		switch(select("Esfera de veneno: Flare Sphere: Iluminação Esfera: Esfera cega: Esfera congelante: Cancelar")) {
		case 1: callsub S_BulletTrade,937,10,13205;// <item required>, <amount req>, <bullet id>
		case 2: callsub S_BulletTrade,7097,2,13203;
		case 3: callsub S_BulletTrade,7053,3,13204;
		case 4: callsub S_BulletTrade,1024,5,13206;
		case 5: callsub S_BulletTrade,7054,2,13207;
		case 6:
			mes "[Tony]";
			mes "Mudou de idéia?";
			mes "Bem, se você precisar";
			mes "Quaisquer balas, eu estarei certo";
			mes "aqui. Volte sempre que";
			mes "Você acha que você precisará de mais";
			mes "munição, pistoleiro.";
			close;
		}
	}
	mes "Ei, eu sou Tony. Estou dentro";
	mes "acusação de distribuição";
	mes "e fazendo balas para";
	mes "Pistoleiros. É apenas";
	mes "Como nossa guilda gosta";
	mes "fazer coisas.";
	next;
	mes "[Tony]";
	mes "Sinto muito se você veio";
	mes "Aqui para comprar algumas balas.";
	mes "Eu só posso fazer negócios com";
	mes "Gunslingers totalmente fugidos.";
	close;

S_BulletTrade:
	.@required_id = getarg(0);
	.@required_amount = getarg(1);
	.@bullet_id = getarg(2);

	mes "[Tony]";
	mes "Para cada conjunto de";
	mes callfunc("F_InsertPlural",30,getitemname(.@bullet_id))+",";
	mes "você deve me dar";
	mes "1 ênfasecon,";
	mes "1 phracon, e";
	mes callfunc("F_InsertPlural",.@required_amount,getitemname(.@required_id))+".";
	next;
	mes "[Tony]";
	mes "Lembre -se de que posso dar";
	mes "um máximo de 500 conjuntos de";
	mes "30 balas por vez. Por favor";
	mes "Digite o número de conjuntos de balas";
	mes "isso que você gostaria. Se você quiser";
	mes "Cancelar e depois digitar '' '' ''";
	next;
	input .@amount;
	mes "[Tony]";
	if (.@amount < 1 || .@amount > 500) {
		mes "Ei, eu não posso te dar";
		mes "tantas balas. Não";
		mes "esqueça de entrar em um número";
		mes "Isso não é superior a 500";
		mes "Se você quiser trocar seu";
		mes "itens para algumas balas.";
	}
	else if (countitem(1010) >= .@amount && countitem(1011) >= .@amount && countitem(.@required_id) >= (.@amount*.@required_amount)) {
		if (checkweight(.@bullet_id,.@amount * 30) == 0) {
			mes "Eh? Seu inventário não";
			mes "tem espaço suficiente para isso";
			mes "muitas balas. Volte mais tarde";
			mes "Depois de fazer mais espaço";
			mes "disponível. Tente colocar um pouco de";
			mes "Suas coisas no armazenamento de kafra.";
		} else {
			mes "Ótimo, tudo parece";
			mes "estar em ordem. Deixe -me levar";
			mes "esses materiais, e aqui estão";
			mes "suas balas. É um prazer";
			mes "Para fazer negócios com você ~";
			delitem 1010,.@amount; //Phracon
			delitem 1011,.@amount; //Emveretarcon
			delitem .@required_id,.@amount * .@required_amount;
			getitem .@bullet_id,.@amount * 30;
		}
	} else {
		mes "Huh. Parece você";
		mes "Não tem materiais suficientes";
		mes "por tantas balas. Bem,";
		mes "Não é problema. Apenas venha";
		mes "de volta depois de reunir tudo";
		mes "que você precisa, ok?";
	}
	close;
}
