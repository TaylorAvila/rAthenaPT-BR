prt_in,243,168,4	script	Jovovich	91,{
	mes "[Cabeleireiro Jovovich]";
	mes "Bem -vindo ~!";
	mes "Como posso ajudá -lo?";
	next;
	mes "[Cabeleireiro Jovovich]";
	if (Sex == SEX_MALE) mes "Oh não! Seu cabelo está danificado. Parece que você pode precisar de tratamento profissional. Venha sentar aqui, por favor. Vir.";
	else {
		if (rand(20) > 11) {
			mes "Eh!? Oh meu Deus! Oh não, não, não, não! Seu cabelo está tão danificado! Não é bom se você deixar seu cabelo assim.";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "Você me deixaria tratar seu cabelo? Por favor?";
		}
		else mes "Uau! Seu cabelo seria perfeito quando for tingido ~ que tal morrer para uma mudança?";
	}
	next;
	while(1) {
		switch(select("Hair de tingimento: Dicas e informações: Cancelar")) {
		case 1:
			mes "[Cabeleireiro Jovovich]";
			mes "Sim sim, boa escolha ~";
			mes "Bem, então, esta é sua chance de uma reforma, sua chance de florescer em beleza! Escolha a cor que você gostaria.";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[Cabeleireiro Jovovich]";
					mes "O que você acha? Você queria uma cor diferente?";
					next;
					if (select("Sim: Não") == 1) {
						mes "[Cabeleireiro Jovovich]";
						mes "OK! Escolha a cor que você gostaria.";
						next;
					}
					else  {
						mes "[Cabeleireiro Jovovich]";
						mes "Hmm, estou meio decepcionado. Eu queria fazer um trabalho melhor. Mas prometo que farei melhor da próxima vez. Por favor, venha de novo ~";
						close;
					}
				}

				while(1) {
					switch(select("Vermelho, por favor.")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[Cabeleireiro Jovovich]";
							mes "Você deve gostar da cor do seu cabelo ~";
							close;
						}
						mes "[Cabeleireiro Jovovich]";
						mes "Eehh ~? Você não vai tingir seu cabelo? Estou um pouco triste ...";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[Cabeleireiro Jovovich]";
						mes "Eh? Mas essa é a cor do cabelo que você já tem. Escolha uma cor diferente.";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"amarelo"; break;
						case 2: callsub S_NoDye,978,"violeta"; break;
						case 3: callsub S_NoDye,980,"laranja"; break;
						case 4: callsub S_NoDye,979,"verde"; break;
						case 5: callsub S_NoDye,981,"azul"; break;
						case 6: callsub S_NoDye,982,"branco"; break;
						case 7: callsub S_NoDye,983,"preto"; break;
						case 8: callsub S_NoDye,975,"vermelho"; break;
						}
						if (Zeny < 1000) {
							mes "[Cabeleireiro Jovovich]";
							mes "A taxa é 1000 Zeny. Você não tem o suficiente ...?";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[Cabeleireiro Jovovich]";
			mes "Quando você está se sentindo baixo, quando você é despejado, quando quiser impressionar alguém, ou mesmo quando você só quer se destacar ...";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "Para aquele lugar e hora especiais, você não quer um penteado próprio? Enquanto você tiver os corantes apropriados, farei seu cabelo parecer maravilhoso.";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "E tente não se preocupar muito com a taxa. A aquisição da beleza é a mesma que adquirir tudo. Hehe ~";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "* Taxas e informações *";
			mes "- 1 Item dos corantes da cor do cabelo que você deseja.";
			mes "- 1000 mulheres Taxa.";
			close;

		case 3:
			mes "[Cabeleireiro Jovovich]";
			mes "Homens ou mulheres ...";
			mes "Todo mundo tem o direito e a obrigação de ser linda.";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[Cabeleireiro Jovovich]";
		mes "Eh?! Mas você precisa do item '"+getitemname(getarg(0))+"'Para tingir seu cabelo"+getarg(1)+"...";
		close;
	}
	return;
}
