moc_para01,36,38,3	script	Mission [11 - 25]#Tuto	4_BOARD3,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Você precisa ter um -";
		mes "- ^4d4dff'eden Grupo Mark ' ^000000 -";
		mes "- Para receber essas missões. -";
		close;
	}
	for ( .@i = 11124; .@i < 11134; .@i++ )
		if (checkquest(.@i,PLAYTIME) == 2) erasequest .@i;
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			mes "- Você deve coletar seu -";
			mes "- Recompensa antes de começar -";
			mes "- Uma nova missão. -";
			mes "";
			mes "- Gerente: Spike -";
			close;
		}
		else if (.@j >= 0) {
			mes "- Você só pode solicitar -";
			mes "- Uma missão de cada vez. -";
			mes "";
			mes "- Gerente: Spike -";
			close;
		}
	}
	if (BaseLevel < 11) {
		mes "- Não há tarefas -";
		mes "- Aberto para você agora. -";
		close;
	}
	if (BaseLevel > 25) {
		mes "- essas missões também são -";
		mes "- fácil para o seu calibre -";
		mes "- Procure por um nível mais";
		mes "- Missão apropriada. -";
		close;
	}
	mes "- Conselho de Boletim da Missão -";
	mes "- Para iniciantes. -";
	mes "";
	mes "- Gerente: Spike -";
	next;
	switch(select("Caça ao Hornet: Caça do Condor: Perna de Grasshopper: Caça a cauda de Worm: Caça de esporos: Controle de pragas: Muka Hunting: Hunting Familiar: Colete Feathers: Colete esporos de veneno: Cancelar")) {
		case 1: callsub L_Quest,11114,"Caça a vizinhos","- Hunt 10 ^4d4dff'hornets ' ^000000 -","- Noroeste de Protera. -";
		case 2: callsub L_Quest,11115,"Caça condor","- Hunt 20 ^4d4dff'condors ' ^000000 -","- ao sul de Marrocos. -";
		case 3: callsub L_Quest,11116,"A perna de Grasshopper","- Hunt 10 ^4d4dff'rockers ' ^000000 -","- a sudoeste de Protera e -","- Traga 10 ^4d4dffgrasshopper as pernas ^000000. -";
		case 4: callsub L_Quest,11117,"Caça à cauda de minhocas","- Hunt 20 ^4d4dff'worm Tails ' ^000000 -","- Sudeste de Payon.-";
		case 5: callsub L_Quest,11118,"Caça de esporos","- Hunt 30 ^4d4dff'Spores ' ^000000 -","- em torno de Payon. -";
		case 6: callsub L_Quest,11119,"Controle de pragas","- Hunt 20 ^4d4dffthief bug ovos ^000000 -","- no bueiro contendo -","- e traga 10 ^4d4dffchrysalis ^000000. -";
		case 7: callsub L_Quest,11120,"Cara de caça","- Hunt 20 ^4d4dff'muka ' ^000000 -","- ao sul de Pronntera. -";
		case 8: callsub L_Quest,11121,"Caça familiar","- Hunt 20 ^4d4dfffamiliar ^000000 -","- nos esgotos do bueiro -","- Oeste de Pronntera. -";
		case 9: callsub L_Quest,11122,"Colete penas","- Hunt 30 ^4d4dff'peco pecos ' ^000000 -","- ao sul de Pronntera. -";
		case 10: callsub L_Quest,11123,"Colete esporos de veneno","- Hunt 30 ^4d4dffpOison esporos ^000000 -","- dentro da torre de Geffen e -","- Traga 5 ^4d4dffpoison esporos ^000000. -";
		case 11:
			mes "- Pare de ler o quadro de avisos. -";
			close;
	}

L_Quest:
	.@quest_status = checkquest(getarg(0)+10,PLAYTIME);
	if (.@quest_status == 0 || .@quest_status == 1) {
		mes "- Missão ^4d4dff"+ getarg(1) +"^000000 -";
		mes "- já está terminado para hoje. -";
		mes "";
		mes "- Gerente: Spike -";
		close;
	}
	mes "- Missão '"+ getarg(1) +"' -";
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	mes "";
	mes "- Gerente: Spike -";
	next;
	if (select("Eu farei esta missão .: Olhe para outras missões.") == 2) {
		mes "- Vamos procurar por -";
		mes "- Outras missões. -";
		close;
	}
	for ( .@i = 2; .@i < getargcount(); .@i++ )
		mes getarg(.@i);
	setquest getarg(0);
	close;
}

moc_para01,32,30,6	script	Spike	4_F_CHILD,{
	for( .@i = 11114; .@i < 11124; .@i++ ) {
		.@j = checkquest(.@i,HUNTING);
		if (.@j == 2) {
			switch(.@i) {
				case 11114: callsub L_Quest,.@i,"Caça a vizinhos",900;
				case 11115: callsub L_Quest,.@i,"Caça condor",1800;
				case 11116: callsub L_Quest,.@i,"A perna de Grasshopper",1400,940,10;
				case 11117: callsub L_Quest,.@i,"Caça à cauda de minhocas",2600;
				case 11118: callsub L_Quest,.@i,"Caça de esporos",3900;
				case 11119: callsub L_Quest,.@i,"Extirpate insetos",2900,915,10;
				case 11120: callsub L_Quest,.@i,"Cara de caça",3200;
				case 11121: callsub L_Quest,.@i,"Caça familiar",3200;
				case 11122: callsub L_Quest,.@i,"Colete penas",5400;
				case 11123: callsub L_Quest,.@i,"Colete esporos de veneno",6000,7033,5;
			}
		}
		else if (.@j >= 0) {
			mes "[Espinho]";
			mes "Você ainda não terminou sua missão.";
			mes "Continue com o bom trabalho. Você está quase pronto.";
			close;
		}
	}
	mes "[Espinho]";
	mes "Você está procurando um emprego?";
	mes "Sempre temos muito trabalho a fazer.";
	next;
	mes "[Espinho]";
	mes "Por que você não olha em volta e encontra algumas tarefas interessantes?";
	close;

L_Quest:
	.@quest_id = getarg(0);
	if (checkquest(.@quest_id,HUNTING) == 2) {
		.@item_req = getarg(3,0);
		.@req_amount = getarg(4,0);
		if (.@item_req) {
			if (countitem(.@item_req) < .@req_amount) {
				mes "[Espinho]";
				mes "Você não tem o suficiente"+ getitemname(.@item_req) +".";
				mes "Você precisa trazer"+ .@req_amount +""+ getitemname(.@item_req) +".";
				close;
			}
		}
		mes "[Espinho]";
		mes "Hmm ... deixe -me ver ~";
		mes "Você completou o '"+ getarg(1) +"missão.";
		next;
		mes "[Espinho]";
		mes "Bom trabalho ~ !!";
		mes "Nosso cliente está tão feliz com seu trabalho. Ele apreciará o que você fez.";
		mes "Por favor nos ajude novamente. Obrigado!";
		if (.@item_req)
			delitem .@item_req,.@req_amount;
		erasequest .@quest_id;
		setquest .@quest_id+10;
		getexp getarg(2),0;
		close;
	}
}
