	if ((BaseLevel < 26) || (BaseLevel > 40)) {
		mes "Seu nível não está qualificado. Você não pode usar o quadro de avisos.";
		close;
	}
	switch(select("Missão de caça: Missão de Entrega: Missão de Reunião: Missão de Escorte")) {
	case 1:
		mes "[Conselho de Missão de Caça]";
		mes "- Detalhes da missão de caça -";

		next;

		mes "- Detalhes da missão de entrega -";
		mes "- estão no quadro. -";

			case 1: callsub L_Quest,8271,0,"Entregar um inseticida","Dieshin Man","Mrocc ruins","Obtenha um 'inseticida' de Dieshin Man e entregue -o a sudeste de Payon";

		mes "- estão no quadro. -";

	//QuestID,ReqLvl,Quest,Client,Loc,Desc
	.@string$ = getarg(2);
	if (BaseLevel < getarg(1,26)) {
		mes "Seu nível não atende aos requisitos. Você não pode tomar o '"+ .@string$ +"missão.";
		close;
	}
	mes "["+ .@string$ +"!!]";
	mes "Cliente:"+getarg(3);
	mes "Localização do cliente:"+getarg(4);
	mes "Missão solicitada:"+getarg(5);
	mes "";
	mes "Você vai aceitar esta missão?";
	next;
	if(select("Não, não vou aceitar.") == 1) {
		mes "Você recusou o"+ .@string$ +"missão.";
		close;
	}
	if (checkquest(getarg(0)) > -1) mes "Você já está fazendo esta missão.";
	else {
		setquest getarg(0);
		mes "Você aceitou o"+ .@string$ +"missão.";
	}
	mes "Você pode receber uma recompensa do cliente depois de terminar a missão.";
	next;
	mes "Para mais detalhes, pressione 'alt + u', para obter mais informações da janela da missão.";
	close;
}
