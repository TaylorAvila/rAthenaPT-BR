moc_para01,44,38,3	script	Mission [71 - 85]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "- Apenas membros do -";
		mes "- Eden Group pode ler -";
		mes "- Este quadro de avisos. -";
		close;
	}

	// Clear the original quest data (no longer used).
	if (slv_quest) set slv_quest,0;
	if (checkquest(10102) > -1 || checkquest(10103) > -1 || checkquest(10104) > -1 || checkquest(10105) > -1 || checkquest(10106) > -1) {
		for(set .@quest,10102; .@quest<=10106; set .@quest,.@quest+1) {
			if (checkquest(.@quest) > -1)
				erasequest .@quest;
		}
		mes "- Desde que o cliente se moveu -";
		mes "- para outro lugar, -";
		mes "- O formulário de inscrição foi removido. -";
		next;
	}

	if (BaseLevel < 71) {
		mes "- Apenas membros do -";
		mes "- O nível certo pode ler -";
		mes "- Este quadro de avisos. -";
		mes "- Tente encontrar um quadro -";
		mes "- Isso se adapta ao seu nível. -";
		close;
	} else if (BaseLevel > 85) {
		// Quest IDs: 10107-10123, 5055-5057
		setarray .@quests[0],
			10107,10108,10109,
			10110,10111,10112,10113,
			10114,10115,10116,10117,
			10118,10119,10120,
			10121,10122,10123,5055,5056;
		for(set .@i,0; .@i<getarraysize(.@quests); set .@i,.@i+1) {
			if (checkquest(.@quests[.@i],HUNTING) == 2)
				set .@complete[getarraysize(.@complete)], .@quests[.@i];
		}
		if (checkquest(5057) > -1 && countitem(7187) >= 30)
			set .@complete[getarraysize(.@complete)],5057;
		if (getarraysize(.@complete)) {
			mes "Você tem missões em andamento.";
			mes "Você quer entregá -los?";
			next;
			if(select("Claro.") == 2)
				close;
			for(set .@i,0; .@i<getarraysize(.@complete); set .@i,.@i+1)
				callsub L_Quest,.@complete[.@i];
			close;
		}
		mes "- Apenas membros do -";
		mes "- O nível certo pode ler -";
		mes "- Este quadro de avisos. -";
		mes "- Tente encontrar um quadro -";
		mes "- Isso se adapta ao seu nível. -";
		close;
	}
	mes "- Escolha onde você -";
	mes "- gostaria de caçar. -";
	next;
	switch(select("Masmorra de Esfinge: Glast Heim: Juno Area: Clock Tower: Localized Islands")) {
	case 1:
		mes "- Existem vários pedidos -";
		mes "- Relacionado à masmorra da Esfinge -";
		mes "- Neste conselho de missão. -";
		next;
		set .@str1$,"Os moradores de Marrocc estão preocupados com o crescente número de monstros na masmorra da Esfinge.";
		set .@str2$,"- Meidi de Marrocos -";
		switch(select("Hunt Requiem: Hunt Marduk: Hunt Pasana")) {
			case 1: callsub L_Quest,10107,.@str1$+"Hunt 10 descansos.",.@str2$;
			case 2: callsub L_Quest,10108,.@str1$+"Hunt 10 Marduks.",.@str2$;
			case 3: callsub L_Quest,10109,.@str1$+"Hunt 10 PASSENUS.",.@str2$;
		}
	case 2:
		mes "- Existem vários pedidos -";
		mes "- Relacionado com Glast Heim -";
		mes "- Neste conselho de missão. -";
		next;
		set .@str1$,"Tenho muito medo de que os monstros do Glast Heim nos atacem.";
		set .@str2$,"- Ancellia de Geffen -";
		switch(select("Hunt Dark Frame: Hunt Evil Druid: Hunt Wraith: Hunt Raydric Archer")) {
			case 1: callsub L_Quest,10110,.@str1$+"Caça 10 quadros escuros.",.@str2$;
			case 2: callsub L_Quest,10111,.@str1$+"Caça 10 druidas malignos.",.@str2$;
			case 3: callsub L_Quest,10112,.@str1$+"Hunt 10 Wraiths.",.@str2$;
			case 4: callsub L_Quest,10113,.@str1$+"Hunt 10 Raydric Archers.",.@str2$;
		}
	case 3:
		mes "- Existem vários pedidos -";
		mes "- relacionado à área -";
		mes "- em torno de Juno. -";
		next;
		set .@str1$,"Aqueles que não podem lutar, como eu, acham muito mais difícil viajar devido ao grande número de monstros.";
		set .@str2$,"- Jeanbai, viajante -";
		switch(select("Hunt Grand Peco: Hunt Sleeper: Hunt Goat: Hunt Harpy")) {
			case 1: callsub L_Quest,10114,.@str1$+"Hunt 20 Grand Pecos.",.@str2$;
			case 2: callsub L_Quest,10115,.@str1$+"Caça 20 dormentes.",.@str2$;
			case 3: callsub L_Quest,10116,.@str1$+"Caça 20 cabras.",.@str2$;
			case 4: callsub L_Quest,10117,.@str1$+"Hunt 20 Harpies.",.@str2$;
		}
	case 4:
		mes "- Existem vários pedidos -";
		mes "- relacionado à torre do relógio -";
		mes "- Neste conselho de missão. -";
		next;
		set .@str1$,"Eu adorava ouvir a doce melodia da torre do relógio Al de Baran, mas não consigo ouvir mais nada por causa de todos os ruídos que os monstros estão fazendo.";
		set .@str2$,"- Rizingsetter, presidente da comunidade do relógio de amor -";
		switch(select("Hunt Clock: Hunt Punk: Hunt Rideword")) {
			case 1: callsub L_Quest,10118,.@str1$+"Caça 15 relógios antes de quebrar.",.@str2$;
			case 2: callsub L_Quest,10119,.@str1$+"Caça 15 punks antes de quebrar.",.@str2$;
			case 3: callsub L_Quest,10120,.@str1$+"Hunt 15 Ridewords antes de quebrar.",.@str2$;
		}
	case 5:
		mes "- Existem vários pedidos -";
		mes "- relacionado às ilhas localizadas -";
		mes "- Neste conselho de missão. -";
		next;
		set .@str1$,"Eu quero fazer uma aventura, mas há tantas coisas com as quais tenho que lidar. Se eu não posso, então outros também não podem!";
		set .@str2$,"- Funfy, quem quer ir a uma aventura -";
		switch(select("Hunt Kikimora: Hunt Miyabi Doll: Hunt Mi Gao: Hunt Headless Mule: Hunt Tamruan: Reúna máscaras do festival para o festival")) {
			case 1: callsub L_Quest,10121,.@str1$+"Hunt 15 Kikimorus em Moscowia!",.@str2$;
			case 2: callsub L_Quest,10122,.@str1$+"Caça 15 bonecas Miyabi em Amatsu!",.@str2$;
			case 3: callsub L_Quest,10123,.@str1$+"CHOQUE 15 MISSAYANG!",.@str2$;
			case 4: callsub L_Quest,5055,"Eu sei que existem muitos monstros aterrorizantes fora da cidade, mas o que me aterroriza são essas mulas sem cabeça à espreita. Espero que eles nunca se aventurem na cidade. Caça 15 mulas sem cabeça!","- Mariana from Brasilis -";
			case 5: callsub L_Quest,5056,"Os tamruans fazem muito barulho. Eu não me importo de manhã, mas eles me impedem de adormecer à noite. Quero dormir. Hunt 15 Tamruans!","- igualmente pum -fars invisíveis -";
			case 6: callsub L_Quest,5057,"As crianças do bolinho roubaram todas as máscaras que eu havia preparado para o festival. Tente conseguir 30 máscaras! Eu te devo ~","- Saymmo Chefe de Kunlun -";
		}
	}
	end;

//callsub L_Quest,<quest ID>{,"<Descrição1>","<Descrição2>"};
L_Quest:
	set .@quest, getarg(0);
	switch(.@quest) {
		case 10107: set .@name$,"Caça descanso"; set .@exp,15000; break;
		case 10108: set .@name$,"Hunt Marduk"; set .@exp,15000; break;
		case 10109: set .@name$,"Assombrar"; set .@exp,15000; break;
		case 10110: set .@name$,"Caça a moldura escura"; set .@exp,20000; break;
		case 10111: set .@name$,"Hunt Druid do mal"; set .@exp,20000; break;
		case 10112: set .@name$,"Hunt Wraith"; set .@exp,20000; break;
		case 10113: set .@name$,"Hunt Raydric Archer"; set .@exp,20000; break;
		case 10114: set .@name$,"Caça Grand Peco"; set .@exp,30000; break;
		case 10115: set .@name$,"Caça dorminhoco"; set .@exp,30000; break;
		case 10116: set .@name$,"Caça cabra"; set .@exp,30000; break;
		case 10117: set .@name$,"Caça harpy"; set .@exp,30000; break;
		case 10118: set .@name$,"Hunt Clock"; set .@exp,25000; break;
		case 10119: set .@name$,"Caça punk"; set .@exp,25000; break;
		case 10120: set .@name$,"Hunt Rideword"; set .@exp,25000; break;
		case 10121: set .@name$,"Hunt Kakoror"; set .@exp,30000; break;
		case 10122: set .@name$,"Hunt Miyabi Doll"; set .@exp,30000; break;
		case 10123: set .@name$,"Hunt Mi Gao"; set .@exp,30000; break;
		case 5055: set .@name$,"Caça mula sem cabeça"; set .@exp,30000; break;
		case 5056: set .@name$,"Hunt Tamruan"; set .@exp,30000; break;
		case 5057: set .@name$,"Reúna máscaras do festival para o festival"; set .@exp,30000; break;
	}
	if (getargcount() == 1) {
		mes "'"+.@name$+"'Concluído. Você quer entregá -lo e receber sua recompensa?";
		next;
		if(select("Vire em busca .: Cancel.") == 1) {
			if (.@quest == 5057) delitem 7187,30; //Festival_Mask
			erasequest .@quest;
			getexp .@exp,0;
			mes "Você recebeu exp por terminar a missão.";
			next;
			return;
		}
		close;
	} else {
		if (.@quest == 5057) {
			// Quest 5057 is the only collection quest, so separate conditions are unnecessary.
			if (checkquest(5057) == -1)
				set .@hunting,-1;
			else if (countitem(7187) >= 30)
				set .@hunting,2;
		} else
			set .@hunting, checkquest(.@quest,HUNTING);
		if (.@hunting == 0 || .@hunting == 1) {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "Volte aqui quando você terminar a missão.";
		} else if (.@hunting == 2) {
			mes "Você completou a missão.";
			next;
			if(select("Vire em busca .: Cancel.") == 1) {
				if (.@quest == 5057) delitem 7187,30; //Festival_Mask
				erasequest .@quest;
				getexp .@exp,0;
				mes "Você recebeu exp por terminar a missão.";
			}
		}/* else if (.@hunting == 3) {
//			recall_completequest .@quest
			erasequest .@quest;
			mes "Você re-aceitou a missão '"+.@name$+"'.";
		}*/ else {
			mes getarg(1);
			mes getarg(2);
			next;
			mes "Você quer aceitar a missão?";
			next;
			if(select("Sim.:no.") == 1) {
				mes "Você recebeu a missão '"+.@name$+"'. Veja suas informações de missão para obter mais detalhes.";
				setquest .@quest;
			}
		}
		close;
	}
}
