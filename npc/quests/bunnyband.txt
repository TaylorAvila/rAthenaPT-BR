alberta,26,229,0	script	Kafra Employee#bunny	83,{
	if (BUNYBND == 1) {
		mes "[Então funcionário]";
		mes "Olá ~!";
		mes "Como você gostaria de";
		mes "Participação e Campra";
		mes "Especial da corporação";
		mes "^529dffbunny Band Event^000000?";
		next;
		switch(select("Claro, eu trouxe os itens.")) {
		case 1:
			mes "[Então funcionário]";
			mes "Tudo bem ~";
			mes "Deixe -me verificar";
			mes "Veja se você trouxe";
			mes "Todos os itens ...";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[Então funcionário]";
					mes "Ooh, me desculpe";
					mes "Mas você precisa";
					mes "traga pelo menos";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+".";
					close;
				}
			mes "[Então funcionário]";
			mes "Ótimo, eu vejo isso";
			mes "você se reuniu";
			mes "tudo que eu preciso";
			mes "Faça a banda do coelho.";
			mes "Por favor, espere um momento";
			mes "Enquanto eu juntei ...";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[Então funcionário]";
					mes "HM? Desculpe,";
					mes "Mas eu realmente não posso";
					mes "Faça isso agora. Você";
					mes "need "+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "para que eu coloque isso";
					mes "Bunny Band juntos ...";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[Então funcionário]";
			mes "Ah, aqui está ~";
			mes "A banda de coelho perfeita!";
			mes "Bem, espero que você goste.";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[Então funcionário]";
			mes "Obrigado por";
			mes "participando disso";
			mes "evento especial e seu";
			mes "patrocínio contínuo";
			mes "dos serviços Kafra ~";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[Então funcionário]";
	mes "Oi aí ~! Você gostaria";
	mes "Para se juntar ao nosso especial ^529dffbunny";
	mes "Evento da banda^000000 hospedado pelo Kafra";
	mes "Corporação e patrocinada por";
	mes "A Alberta Merchant Guild?";
	next;
	switch(select("Participe do evento: Informações do evento: Cancelar")) {
	case 1:
		set BUNYBND,1;
		mes "[Então funcionário]";
		mes "Ótimo! Obrigado por";
		mes "participando! Se você";
		mes "Já não ouvi,";
		mes "você precisa coletar estes";
		mes "itens se você me quiser colocar um";
		mes "Bunny Band juntos para você ...";
		next;
		mes "[Então funcionário]";
		mes "100 penas,";
		mes "1 trevo de quatro folhas,";
		mes "1 banda de gatinhos e";
		mes "1 Pearl. É isso!";
		mes "Boa sorte e eu estarei";
		mes "Esperando por você aqui ~";
		close;
	case 2:
		break;
	case 3:
		close;
	}
L_Info:
	mes "[Então funcionário]";
	mes "Para agradecer ao nosso valor";
	mes "Clientes, Kafra Corporation";
	mes "preparou um evento especial";
	mes "Onde o funcionário da Kafra vai montar";
	mes "Bandas de coelho para aventureiros";
	mes "que trazem os itens necessários.";
	next;
	mes "[Então funcionário]";
	mes "Para este especial,";
	mes "um item único,";
	mes "Traga 100 penas,";
	mes "1 trevo de quatro folhas,";
	mes "1 banda de gatinhos e";
	mes "1 Pearl.";
	next;
	mes "[Então funcionário]";
	mes "Quando você está pronto,";
	mes "Volte e traga";
	mes "Esses itens para mim.";
	mes "(Patrocinado pelo";
	mes "Alberta Merchant Guild).";
	close;
}
