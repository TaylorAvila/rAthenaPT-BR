	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);

		input .@sell;
		if (.@sell == 0) {
			mes "[Madeleine Chu]";
			mes "Você mudou o seu";
			mes "mente? Bem, se você precisar";
			mes "para comprar"+ getitemname(.@item_id);
			mes "Kits depois, basta voltar";
			mes "Para mim a qualquer momento, tudo bem?";
			close;
		}
		else if (.@sell > 100) {
			mes "[Madeleine Chu]";
			mes "Oh, sinto muito, mas";
			mes "Eu não vendo mais do que";
			mes "100"+ getitemname(.@item_id);
			mes "de cada vez, apenas para estar seguro.";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[Madeleine Chu]";
		mes "Oh, me desculpe, mas você";
		mes "Não posso pagar tantos";
		mes getitemname(.@item_id);
		mes "Por favor, verifique seu zeny";
		mes "Antes de comprar meus kits ~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[Madeleine Chu]";
		mes "Sinto muito, mas você não";
		mes "tem espaço suficiente em seu";
		mes "Inventário para tantos";
		mes getitemname(.@item_id) +"...";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[Madeleine Chu]";
	mes "Aqui está ~";
	mes "Boa sorte com";
	mes "Seu treinamento culinário!";
	return;
