payon_in03,188,146,5	script	Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- Espere um momento! -";
		mes "- Atualmente você está carregando -";
		mes "- Muitos itens com você. -";
		mes "- Por favor, volte mais tarde -";
		mes "- Depois de colocar alguns itens no armazenamento da Kafra. -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[Merchant Marx Hansen]";
		mes "Bem-vindo.";
		mes "Você veio para";
		mes "Processar frutas também?";
		next;
		switch(select("Faça suco.: falte e obtenha informações sobre o processamento de frutas .: Cancel.")) {
		case 1:
			mes "[Merchant Marx Hansen]";
			mes "Que tipo de suco de frutas você gostaria de fazer?";
			next;
			switch(select("Suco de maçã: suco de banana: suco de cenoura: suco de uva: cancelar")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[Merchant Marx Hansen]";
				mes "Bem então ...";
				mes "Vejo você na próxima vez.";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[Merchant Marx Hansen]";
				mes "Oh não ...";
				mes "Você não tem todos os materiais necessários. Fazer"+getitemname(.@juice)+", Eu preciso de 1"+getitemname(.@fruit)+"e 1 garrafa vazia. Também precisarei de uma taxa de 3 zeny.";
				next;
				mes "[Merchant Marx Hansen]";
				mes "Quando você preparar tudo, misturarei a fruta para lhe dar suco delicioso.";
				close;
			}

			mes "[Merchant Marx Hansen]";
			mes "Quantos você gostaria?";
			next;
			switch(select("O máximo possível.")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[Merchant Marx Hansen]";
				mes "Escolha um número menor que 100. Se você não quiser, coloque '0'. Você pode fazer as pazes" + countitem(.@fruit) + "garrafas de suco.";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[Merchant Marx Hansen]";
						mes "Bem então ...";
						mes "Venha de novo.";
						close;
					}
					else if (.@input > 100) {
						mes "[Merchant Marx Hansen]";
						mes "Mais de 100 garrafas é impossível. Escolha uma quantidade diferente.";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[Merchant Marx Hansen]";
				mes "Bem então ...";
				mes "Venha de novo.";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[Merchant Marx Hansen]";
				mes "Oh não ...";
				mes "Você não tem todos os materiais necessários. Não posso ajudar uma situação como essa. Eu acho que você coleta o que precisa.";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[Merchant Marx Hansen]";
			mes "Aqui você vai! Suco fresco e delicioso, como prometido. Deve ser muito refrescante e palatável.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Bem então ...";
			mes "Venha de novo.";
			close;
		case 2:
			mes "[Merchant Marx Hansen]";
			mes "Antes que os seres humanos pudessem desenvolver sociedades tão vastas, eles reuniram frutas das árvores para sobreviver. As frutas podem ter sido a bênção da natureza que nos permitiu existir no mundo.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Como a vida se tornou tão próspera, a geração mais jovem parece não comer mais frutas. Então, comecei a pensar em maneiras de tornar as frutas mais fáceis de comer.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Percebi que, quando você faz suco de frutas, é mais conveniente comer e tem um gosto muito melhor.";
			next;
			mes "# Informações sobre suco de frutas #";
			mes "^Cc4e5c- suco de maçã -^000000";
			mes "Apple x 1 de, garrafa vazia x 1 de, 3 Zeny.";
			mes "^E3cf57- suco de banana -^000000";
			mes "Banana x 1 de, garrafa vazia x 1 de, 3 zeny.";
			mes "^Ed9121- Juice de cenoura -^000000";
			mes "Cenoura x 1 ea, garrafa vazia x 1 ea, 3 Zeny.";
			mes "^Cc00ff- suco de uva -^000000";
			mes "Uva x 1 de, garrafa vazia x 1 de, 3 zeny.";
			close;
		case 3:
			mes "[Merchant Marx Hansen]";
			mes "Ei!";
			mes "Se você visitar";
			mes "Alguém, fale com eles!";
			close;
		}
	}
	else {
		mes "[Merchant Marx Hansen]";
		mes "Bem-vindo.";
		mes "Como posso ajudá -lo?";
		next;
		if (select("Conversa: Cancelar") == 1) {
			mes "[Merchant Marx Hansen]";
			mes "Antes que os seres humanos pudessem desenvolver sociedades tão vastas, eles reuniram frutas das árvores para sobreviver. As frutas podem ter sido a bênção da natureza que nos permitiu existir no mundo.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Como a vida se tornou tão próspera, a geração mais jovem parece não comer mais frutas. Então, comecei a pensar em maneiras de tornar as frutas mais fáceis de comer.";
			next;
			mes "[Merchant Marx Hansen]";
			mes "Percebi que, quando você faz suco de frutas, é mais conveniente comer e tem um gosto muito melhor.";
			close;
		}
		mes "[Merchant Marx Hansen]";
		mes "Ei!";
		mes "Se você visitar";
		mes "Alguém, fale com eles!";
		close;
	}
