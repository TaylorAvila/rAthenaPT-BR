// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	Nia#yagu	700,{
	mes "[Nosso]";
	mes "Olá, eu sou Nia, o quinto";
	mes "discípulo de Mawong, o";
	mes "Mastermind. Agora,";
	mes "Como você gostaria de jogar um";
	mes "Number Match Game With Me?";
	next;
	switch(select("Number Match Game ?: Prática Número Match: Play Number Match: Record titular: Cancelar")) {
	case 1:
		mes "[Nosso]";
		mes "Ah, bem, o objetivo de";
		mes "O jogo de partida numérico é para";
		mes "acho que o número que estou pensando";
		mes "de. Claro, existem alguns";
		mes "regras que limitam os números";
		mes "que eu posso escolher.";
		next;
		mes "[Nosso]";
		mes "Todos os números que eu faço";
		mes "Up deve ter 3 dígitos, eu não posso";
		mes "Use 0 para qualquer um dos dígitos,";
		mes "e eu não posso usar o mesmo";
		mes "Número para qualquer um dos dígitos";
		mes "mais de uma vez.";
		next;
		mes "[Nosso]";
		mes "Então, por exemplo, eu não posso";
		mes "Use os números 103, 112,";
		mes "252, ou 701. Esses números";
		mes "tem um 0 ou eles usam";
		mes "o mesmo número nos dígitos";
		mes "mais de uma vez. Entendeu agora?";
		next;
		mes "[Nosso]";
		mes "Agora, você só tem 5 suposições,";
		mes "E depois de cada palpite, eu vou contar";
		mes "você quantos dígitos você adivinhou";
		mes "corretamente, seguido pelo número de dígitos corretamente adivinhados em";
		mes "o local de dígito correto.";
		next;
		mes "[Nosso]";
		mes "Por exemplo, digamos o";
		mes "O número que eu inventei é 168.";
		mes "Se o seu primeiro palpite é 678, você adivinhou dois dos dígitos,";
		mes "6 e 8, corretamente. No entanto,";
		mes "Apenas 8 está no local correto.";
		next;
		mes "[Nosso]";
		mes "Você sabe, seria uma boa ideia";
		mes "Se você tentar o modo de prática";
		mes "primeiro. Então, quando você é mais";
		mes "confortável, jogue o real";
		mes "Number Match Game. Não se esqueça que a velocidade conta para sua pontuação!";
		next;
		mes "[Nosso]";
		mes "Ah, se você acontecer";
		mes "fazer um novo recorde no";
		mes "Number Match Game, você";
		mes "Tenha seu nome gravado,";
		mes "Portanto, não perca esta chance para";
		mes "fama e talvez até fortuna ~";
		close;
	case 2:
		mes "[Nosso]";
		mes "Ótimo, vamos fazer uma prática";
		mes "Number Match Game. Por favor";
		mes "Tente adivinhar meu número, ok?";
		mes "Se você deseja cancelar, digite um número superior a 1.000.";
		set .@practice,1;
		break;
	case 3:
		mes "[Nosso]";
		mes "Ótimo, vamos jogar o número de números!";
		mes "Você terá 5 chances de adivinhar o número em que estou pensando.";
		mes "Se você deseja cancelar, digite um número superior a 1.000.";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(nulo)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[Nosso]";
		mes "^FF0000" + .@name_record$ + "^000000 é o";
		mes "titular do registro para o número";
		mes "Jogo de partida com um tempo de";
		mes "^FF0000" + .@score_min + "minutos," + .@score_sec + "segundos^000000.";
		close;
	case 5:
		mes "[Nosso]";
		mes "Cuide-se,";
		mes "aventureiro. Se você já";
		mes "sinto vontade de jogar um número";
		mes "Jogo de partida, venha falar";
		mes "Para mim, tudo bem? Vejo você ~";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[Nosso]";
				mes "Você cancelou o";
				mes "Number Match Game.";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[Nosso]";
			mes "Você entrou em um dos";
			mes "dígitos como 0, ou insira o";
			mes "mesmo valor numérico em";
			mes "os dígitos mais de uma vez.";
			mes "Por favor, digite outro palpite.";
			next;
		}

		set .@retry, .@retry+1;
		mes "[Nosso]";
		mes "^FF0000"+strcharinfo(0)+"^000000,";
		mes "Seu palpite é ^0000ff"+.@input+"^000000.";
		mes "Me dê um momento para";
		mes "crie seus resultados.";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[Nosso]";
				mes "Sim, o número que";
				mes "Eu imaginei que era ^ff0000"+.@yagu+"^000000!";
				mes "Parabéns, você";
				mes "Acabei de ganhar esta prática";
				mes "jogo! Agora você deveria ser";
				mes "pronto para a coisa real ~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(nulo)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[Nosso]";
					mes "Ótimo! Você adivinhou";
					mes "Meu número, que foi";
					mes "^FF0000" + .@yagu + "^000000. Você terminou";
					mes "^FF0000" + .@total_min + "minutos," + .@total_sec + "segundos^000000,";
					mes "quebrando o recorde anterior";
					mes "de ^ff0000" + .@score_min + "minutos," + .@score_sec + "segundos^000000.";
					next;
					mes "[Nosso]";
					mes "^FF0000" + .@name_record$ + "^000000";
					mes "Defina esse registro antigo,";
					mes "Mas agora você é o";
					mes "nova pessoa para vencer no";
					mes "Jogo de partida. Parabéns";
					mes "Para estabelecer um novo registro!";
					close;
				}
				mes "[Nosso]";
				mes "Bom trabalho, você adivinhou";
				mes "Meu número! A resposta, de";
				mes "Claro, foi ^ff0000" + .@yagu + "^000000. Você conseguiu";
				mes "para adivinhar este número corretamente";
				mes "em ^ff0000" + .@total_min + "minutos," + .@total_sec + "segundos^000000.";
				next;
				mes "[Nosso]";
				mes "^FF0000" + .@name_record$ + "^000000 IS";
				mes "a correspondência de número atual";
				mes "titular do registro com um tempo";
				mes "de ^ff0000" + .@score_min + "minutos," + .@score_sec + "segundos^000000.";
				mes "Boa sorte, e espero que você";
				mes "pode quebrar este recorde algum dia ~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[Nosso]";
				mes "Seu palpite tem ^ff0000" + .@ball + "^000000 do";
				mes "números corretos. ^FF0000" + .@strike + "^000000 do";
				mes "dígitos em seu palpite têm";
				mes "o número correto no";
				mes "posicionamento correto do dígito.";
				mes "Tudo bem, tente de novo ~";
				next;
			} else {
				mes "[Nosso]";
				mes "Seu último palpite teve ^ff0000" + .@ball + "^000000";
				mes "dos números corretos,";
				mes "e ^ff0000" + .@strike + "^000000 dígitos em seu palpite";
				mes "tinha o número correto em";
				mes "o posicionamento correto do dígito.";
				switch(.@retry) {
					case 1: mes "Você tem quatro palpites restantes."; break;
					case 2: mes "Você tem três palpites restantes."; break;
					case 3: mes "Você tem duas suposições restantes."; break;
					case 4: mes "Você só tem um palpite ..."; break;
					case 5: mes "A resposta foi ^ff0000" + .@yagu + "^000000."; close;
				}
				next;
			}
		}
	}
}
