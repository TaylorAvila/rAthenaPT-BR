// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[Edgar]";
		mes "Então você está indo para Alberta novamente? Deixe -me dar o mesmo desconto e cobrar apenas 250 Zeny, assim como da última vez.";
		mes "Como esse som?";
		next;
		if (select("Tudo bem ~!: Por que você está sendo tão legal comigo?!") == 1) {
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Um...";
				mes "Isso não é";
				mes "dinheiro suficiente.";
				mes "Por que você não vai";
				mes "receber mais dinheiro?";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "É assim que eu sou. Isso, e seu sorriso diabólico me lembra meu amado filho de cabelos loiros que saiu de casa anos atrás para se tornar um marinheiro em seu próprio navio. Abençoe sua alma, onde quer que ele esteja.";
		close;
	}
	mes "[Edgar]";
	mes "Minha cidade, Izlude, está conectada a Alberta ao porto no oeste. Há muito tráfego entre nós, quase me tornei um Albertian.  Hehehe ~";
	next;
	mes "[Edgar]";
	mes "Há um cara que eu conheço muito bem, Phelix, que vive em Alberta. Esse cara é realmente mesquinho ... ele cobra por tudo!";
	next;
	mes "[Edgar]";
	mes "Mas ele é realmente um cara legal e gosta de ajudar outras pessoas. Ele tem um bom coração e lhe dará seu apoio se você atingir o preço dele.";
	next;
	mes "[Edgar]";
	mes "Ultimamente, as pessoas em Alberta dizem que ele está realmente tentando ajudar as pessoas e que sua demanda por geleia é apenas um encobrimento.  Bem, você deve dar uma olhada no que ele tem a oferecer.";
	next;
	if (select("Você pode me dizer como chegar a Alberta ?:") == 1) {
		mes "[Edgar]";
		mes "Huh? Bem, você pode usar os pés e apenas caminhar.  Mas se você tiver dinheiro, gostaria de sugerir que você pegue um navio.";
		next;
		if (select("Ok, peguei.: Mas estou cansado de andar e estou sem dinheiro!") == 1) {
			mes "[Edgar]";
			mes "Tudo bem, tome cuidado ~";
			close;
		}
		mes "[Edgar]";
		mes "OK...";
		mes "Você não quer andar e não tem dinheiro, mas ainda quer ir para lá? Oh nossa ...";
		next;
		mes "[Edgar]";
		mes "Fine bem. Eu, sendo o capitão de um navio, pode -se dar ao luxo de trazê -lo para um preço mais baixo. Como o 250 Zeny soa?";
		next;
		if (select("Tudo bem ~!: Bah, que rasgo !!") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Um...";
				mes "Isso não é";
				mes "dinheiro suficiente.";
				mes "Vá e pegue";
				mes "um pouco mais.";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "Garoto, garoto,";
		mes "Se você pensa";
		mes "Isso é um rasgo ...";
		close;
	}
	mes "[Edgar]";
	mes "Sim, tudo bem.";
	mes "Vejo você mais tarde ~";
	close;
}
