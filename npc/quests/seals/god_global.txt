sec_in02,15,170,0	script	Golbal var	817,{
	callfunc "F_GM_NPC";
	mes "[Verificar]";
	mes "Por favor, digite a senha.";
	next;
	if (callfunc("F_GM_NPC",68392411,0) == 1) {
		mes "[Verificar]";
		mes "Escolha um menu.";
		next;
		switch(select("Agora: No.1: No.2: No.3: No.4: Reset: GOD_SL_1: GOD_EREMES: GOD_BRISING: GOD_MJO_0: GOD_MJO_1: GOD_MJO_2: GOD_MJO_3: GOD_MJO_4")) {
		case 1:
			mes "^0000FF $ GOD1 ^000000 = ^FF0000" + $God1 + "^000000.";
			mes "^0000FF $ GOD2 ^000000 = ^FF0000" + $God2 + "^000000.";
			mes "^0000FF $ GOD3 ^000000 = ^FF0000" + $God3 + "^000000.";
			mes "^0000FF $ GOD4 ^000000 = ^FF0000" + $God4 + "^000000.";
			close;
		case 2: callsub L_Number,"$ Yr1"; break;
		case 3: callsub L_Number,"$ God2"; break;
		case 4: callsub L_Number,"$ God3"; break;
		case 5: callsub L_Number,"$ God4"; break;
		case 6:
			mes "[Verificar]";
			mes "Você tem certeza de que deseja redefinir toda a lista de Deus Globalvar?";
			mes "Por favor, digite a senha.";
			next;
			if (callfunc("F_GM_NPC",68392411,0) == 1) {
				mes "[Verificar]";
				mes "Agora, toda a lista de Deus Globalvar está sendo redefinida.";
				next;
				set $God1,0;
				set $God2,0;
				set $God3,0;
				set $God4,0;
				mes "^0000FF $ GOD1 ^000000 = ^FF0000" + $God1 + "^000000.";
				mes "^0000FF $ GOD2 ^000000 = ^FF0000" + $God2 + "^000000.";
				mes "^0000FF $ GOD3 ^000000 = ^FF0000" + $God3 + "^000000.";
				mes "^0000FF $ GOD4 ^000000 = ^FF0000" + $God4 + "^000000.";
				close;
			} else {
				mes "[Verificar]";
				mes "O comando foi cancelado.";
				close;
			}
		case 7: callsub L_Var,"GOD_SL_1"; break;
		case 8: callsub L_Var,"God_eremes"; break;
		case 9: callsub L_Var,"god_bring"; break;
		case 10: callsub L_Var,"GOD_MJO_0"; break;
		case 11: callsub L_Var,"GOD_MJO_1"; break;
		case 12: callsub L_Var,"GOD_MJO_2"; break;
		case 13: callsub L_Var,"GOD_MJO_3"; break;
		case 14: callsub L_Var,"GOD_MJO_4"; break;
		}
	} else {
		mes "[Verificar]";
		mes "Senha incorreta.";
		close;
	}

L_Number:
	input .@input,0,$@god_check2;
	mes "[Verificar]";
	mes "Você gostaria de mudar para"+.@input+"?";
	mes "Por favor, digite a senha.";
	next;
	if (callfunc("F_GM_NPC",68392411,0) == 1) {
		mes "[Verificar]";
		mes "O comando"+.@input+"foi confirmado.";
		next;
		setd getarg(0),.@input;
		mes getarg(0)+""+.@input;
		close;
	} else {
		mes "[Verificar]";
		mes "O comando foi cancelado.";
		close;
	}

L_Var:
	input .@input;
	setd getarg(0),.@input;
	next;
	mes "[Verificar]";
	mes "Feito.";
	close;

OnInit:
	// Seals roll at 25/50 in Renewal and 50/100 in Pre-Renewal.
	if (checkre(0)) {
		set $@god_check1,25;
		set $@god_check2,50;
	} else {
		set $@god_check1,50;
		set $@god_check2,100;
	}
	end;
}
