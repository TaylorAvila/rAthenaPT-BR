prontera,113,294,3	script	Sign Post#god	837,{
	mes "======== God Seal Status ========";
	setarray .@seal$[1], "Escorregadio","Trapaceiro","BRISINGAMEN","Refeição";
	for(set .@i,1; .@i<=4; set .@i,.@i+1) {
		set .@val, getd("$ Deus"+.@i);
		if (.@val == 0)
			set .@status$,"Invisível";
		else if (.@val < $@god_check1)
			set .@status$,"Ativo";
		else if (.@val < $@god_check2)
			set .@status$,"Apareceu";
		else
			set .@status$,"Lançado";
		mes .@seal$[.@i]+"Selo:"+.@status$;
	}
	mes "";
	mes "======= seu status de selo ========";
	if (god_sl_1 < 51)
		mes "^ff0000sleipnir selo: não concluído^000000";
	else
		mes "^00FF00SLEIPNIR SEAL: Concluído^000000";
	if (god_eremes < 28)
		mes "^ff0000MegingJard Seal: não concluído^000000";
	else
		mes "^00FF00MEGINGJARD SEAL: Concluído^000000";
	if (god_brising < 50)
		mes "^ff0000BRISHAMEN SEAL: não concluído^000000";
	else
		mes "^00FF00BRISHAMEN SEAL: Concluído^000000";
	if (god_mjo_0 < 11)
		mes "^ff0000mjolnir selo: não concluído^000000";
	else
		mes "^00ff00mjolnir selo: concluído^000000";
	close;
}
