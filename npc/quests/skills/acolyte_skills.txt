prt_church,173,23,4	script	Cleric	79,{
	mes "[Acolyte Klift]";
	mes "Ahh . Irmãos! Faz a tarefa de";
	mes "Cuidando com o ônus de ovelha perdida";
	mes "e cansar você?";
	mes "Estou aqui para ajudá -lo.";
	next;
	switch (select("^3355ff 'Sobre a habilidade oculta de Acolyte'^000000 ..:^3355ff 'Holy Light'^000000 Treinamento: Conversação final")) {
	case 1:
		mes "[Acolyte Klift]";
		mes "Nossos membros do clero";
		mes "naturalmente aprenda uma habilidade como";
		mes "como eles amadurecem. Quando eles se aproximam";
		mes "seus anos seniores como um acólito";
		mes "esta habilidade especial.";
		next;
		mes "[Acolyte Klift]";
		mes "Diante de grande perigo e";
		mes "uma luta sem fim com o mal,";
		mes "Nossos irmãos precisam muito de assistência.";
		mes "^3355ff 'Santo Luz'^000000 é essa habilidade.";
		mes "Para ganhar essa habilidade por si mesmo,";
		mes "leva algum trabalho.";
		next;
		mes "[Acolyte Klift]";
		mes "Para obter a habilidade, você deve encontrar esses itens.";
		mes "^Ff33551 Opal^000000";
		mes "^FF33551 Blue de cristal^000000";
		mes "^FF33551 Rosário^000000";
		next;
		mes "[Acolyte Klift]";
		mes "Para poder usar essa habilidade,";
		mes "É preciso ter experiência suficiente.";
		mes "^Ff3355 Nível de trabalho 30^000000";
		mes "é necessário para aprender essa habilidade.";
		next;
		mes "[Acolyte Klift]";
		mes "Quanto a um padre, por causa de seus";
		mes "já vasta quantidade de experiência,";
		mes "Eles são capazes de ganhar essa habilidade em qualquer";
		mes "nível de trabalho.";
		close;
	case 2:
		if (getskilllv("Al_HolyLight") == 1) {
			mes "[Acolyte Klift]";
			mes "Irmão, você já possui";
			mes "A habilidade da 'luz sagrada'.";
			mes "Você não pode ganhar uma habilidade que você";
			mes "já possui ..";
			mes "Eu rezo para que você esteja usando";
			mes "Essa habilidade para o trabalho do bem. .";
			close;
		}
		if ((countitem(727) > 0) && (countitem(991) > 0) && (countitem(2608) > 0) && (JobLevel > 29 || (BaseJob == Job_Priest || BaseJob == Job_Monk)) && (BaseClass == Job_Acolyte)) {
			mes "[Acolyte Klift]";
			mes "Sua fé se mostrou digna";
			mes "para você ganhar a habilidade de 'luz sagrada'.";
			mes "Sua habilidade é adequada";
			mes "Para usar essa habilidade.";
			mes "Use -o com sabedoria. . .";
			next;
			delitem 727,1; //White_Jewel
			delitem 991,1; //Crystal_Blue
			delitem 2608,1; //Rosary
			skill "Al_HolyLight",1,SKILL_PERM;
			mes "[Acolyte Klift]";
			mes "Você agora conhece 'Santo Luz'";
			mes "Você pode usar essa habilidade apenas no";
			mes "melhor conduta. . . . .";
			close;
		}
		mes "[Acolyte Klift]";
		mes "Oh, está claro. . .";
		mes "Você ainda não está pronto para";
		mes "Receba a habilidade da 'luz sagrada'.";
		next;
		mes "[Acolyte Klift]";
		mes "Você deve ouvir com cuidado o que";
		mes "é necessário para essa habilidade.";
		mes "Se você ouvir de perto,";
		mes "Você pode aprender o que falta.";
		mes "Para que você possa melhorar sobre si mesmo.";
		close;
	case 3:
		mes "[Acolyte Klift]";
		mes ". . . . .";
		mes "Eu entendo seu zelo.";
		mes "Você ainda tem muito tempo";
		mes "Pratique e ganhe experiência.";
		mes "Bênçãos sobre você. . . . .";
		close;
	}
}
