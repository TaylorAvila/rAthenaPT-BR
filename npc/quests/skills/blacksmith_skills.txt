		mes "De volta, não é?";
		mes "Tudo bem, deixe -me verificar";
		mes "as coisas que você está carregando.";
		mes "Vamos ver, inventário, inventário ...";
		next;
		mes "[Goodman]";
		mes "Bom trabalho: você passa.";
		mes "Você percebe agora que o";
		mes "tempo e experiência gastos em";
		mes "Reunir esses itens vale a pena";
		mes "Mais do que seus preços em Zeny?";
		next;
		mes "[Goodman]";
		if (Upper == 1)
			mes "Eu acredito que mestre smiths";
		else
			mes "Eu acredito que ferreiros";
		mes "em primeiro lugar deve ser";
		mes "artesãos. Sendo rico e famoso";
		mes "é apenas um efeito colateral disso.";
		if (Upper == 1)
			mes "Honorável Master Smiths Know Value está no artesanato ...";
		else
			mes "Blacksmiths Honorable Know Value está no artesanato ...";
		next;
		if (Upper == 1) {
			mes "[Goodman]";
			mes "Agora, se você não coloca também";
			mes "muita importância para o item";
			mes "vale a pena em termos de zeny, você";
			mes "ser capaz de deixá -los muito";
			mes "mais facilmente. Observe -me fazer isso.";
			mes "Ver? Agora você pode fazer isso também.";
			next;
		}
		mes "[Goodman]";
		mes "A habilidade que você aprendeu é";
		mes "chamado ganância, mas espero";
		mes "você o usa de uma maneira que";
		mes "é diferente do seu nome.";
		mes "Não se esqueça disso, e espero";
		mes "Você cria verdadeiras obras -primas.";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[Goodman]";
	mes "Como você tem usado";
	mes "a habilidade de ganância? Esperançosamente,";
	mes "tem sido conveniente em";
	mes "Seus esforços para forjar bens";
	mes "de maior qualidade para todos";
	mes "de seus camaradas.";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
