ecl_in02,164,56,3	script	Receptionist#Laphine	835,{
	mes "[Recepcionista]";
	mes "Bem -vindo à clínica Eclage.";
	next;
	switch (select("Salvar Local: Rest: Termine a conversa")) {
	case 1:
		mes "[Recepcionista]";
		mes "A clínica foi definida como seu ponto de salvamento. Tenha cuidado, embora ~";
		savepoint "ECL_IN02",162,50,1,1;
		close;
	case 2:
		mes "[Recepcionista]";
		mes "Restaure sua mente e corpo em nossa clínica. Está cheio de energia limpa do Eclage.";
		mes "Carregamos 1 moeda esplendida normalmente, mas oferecemos um desconto especial para os viajantes. Seu total é de 5.000 Zeny.";
		next;
		switch (select("Pague com moeda esplendida: pague com zeny: esqueça")) {
		case 1:
			if (!countitem(6081)) {
				mes "[Recepcionista]";
				mes "Parece que você não tem moedas suficientes. Você veria sua conta novamente, por favor?";
				close;
			}
			mes "[Recepcionista]";
			mes "Faça -se em casa ~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[Recepcionista]";
				mes "Parece que você não tem recursos suficientes para prosseguir. Você se importa de verificar sua conta novamente?";
				close;
			}
			mes "[Recepcionista]";
			mes "Faça -se em casa ~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
