// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[Oficial de Warp Channel]";
	mes "Izlude é um lugar muito animado, não é?";
	mes "Estou aqui para ajudar a fazer Izlude se sentir mais suave!";
	mes "Posso enviar -lhe para outra cópia de Izlude.";
	mes "Você quer ir para outra cópia?";
	next;
	set .@i, select(getarg(0)+":Deixa para lá");
	if (.@i < 6) {
		setarray .@maps$[1],"chocado","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
