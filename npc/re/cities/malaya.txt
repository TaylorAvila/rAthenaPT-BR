function	script	F_Malaya_Jeepney	{
	.@mapName$        = getarg(0);
	.@passengers      = getarg(1);

	// set the other messages of varying amount)
	for (.@i = 5; .@i < getargcount(); .@i++) {
		setd(".@msgjeepneyinfo $ [" + (.@i - 5) + "]", getarg(.@i));
	}

	if (malaya_hi < 10) {
		mes "[Driver de Jeepney]";
		mes getarg(2);
		close;
	} else if ((malaya_hi >= 10) && (malaya_hi < 20)) {
		mes "[Driver de Jeepney]";
		mes getarg(3);
		close;
	} else {
		mes "[Driver de Jeepney]";
		mes getarg(4);
		next;
	}
	switch(select("Conselho [Passageiro" + getmapusers(.@mapName$) + "/" + .@passengers +"]: Jeepney ?: Existem outros jeepneys ?: Ah ... sim ...")) {
	case 1:
		if (getmapusers(.@mapName$) >= .@passengers) {
			mes "[Driver de Jeepney]";
			mes "Receio que o jeepney esteja cheio.";
			mes "Sinto muito, mas que tal outro jeepney?";
			close;
		} else {
			mes "[Driver de Jeepney]";
			mes "Tenha um bom dia.";
			close2;
			warp .@mapName$,29,24;
			end;
		}
	case 2:
		// iterate through all the jeepney information for this specific NPC
		for (.@i = 0; .@i < getarraysize(.@msgJeepneyInfo$); .@i++) {
			mes "[Driver de Jeepney]";
			mes getd(".@msgjeepneyinfo $ [" + .@i + "]");

			if (.@i < (getarraysize(.@msgJeepneyInfo$) - 1)) {
				next;
			}
		}
		close;
	case 3:
		mes "[Driver de Jeepney]";
		mes "Oh! Outros jeepneys estão em operação, é claro.";
		next;
		mes "[Driver de Jeepney]";
		mes "Em Port Malaya, existem 12, 30 e 60 jeepneys de passageiros com 3 cada um.";
		next;
		mes "[Driver de Jeepney]";
		mes "Você gostaria de saber a localização de outros carros?";
		next;
		switch (select("Eu gostaria de saber onde fica o carro de 12 passageiros.")) {
		case 1:
			viewpoint 1, 237, 240, 1, 0xF7E009;
			viewpoint 1, 67 , 44 , 2, 0xF7E009;
			viewpoint 1, 282, 129, 3, 0xF7E009;
			set .@zif_in, 12;
			break;
		case 2:
			viewpoint 1, 134, 250, 4, 0xF7E009;
			viewpoint 1, 341, 153, 5, 0xF7E009;
			viewpoint 1, 293, 290, 6, 0xF7E009;
			set .@zif_in, 30;
			break;
		case 3:
			viewpoint 1, 242, 221, 7, 0xF7E009;
			viewpoint 1, 62 , 245, 8, 0xF7E009;
			viewpoint 1, 257, 58 , 9, 0xF7E009;
			set .@zif_in, 60;
			break;
		}
		mes "[Driver de Jeepney]";
		mes .@zif_in+"A localização do Jeepney do Passageiro foi marcada no seu mapa.";
