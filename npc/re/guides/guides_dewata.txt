dewata,202,106,4	script	Dewata Guide#dewata01	4_M_DEWWOMAN,{
	mes "[Guia Dewata]";
	mes "Bem -vindo a ^8B4513Dewata Island ^000000,";
	mes "Um lugar com ondas deslumbrantes e vistas encantadoras, é um ótimo lugar para relaxar.";
	mes "Se você tiver alguma dúvida, pergunte -me.";
	next;
	mes "[Guia Dewata]";
	mes "Onde posso guiá -lo?";
	F_Navi("[Guia Dewata]");
	while (1) {
		next;
		switch(select("[Funcionário da Kafra]: [revendedor de armas]: [Dealer de armadura]: [revendedor de ferramentas]: [RestaurantEur]: [Krakatau Gatekeeper]: [Alberta Sailor]: Remova as marcas do Mini-map: Cancelar")) {
		case 1:
			callsub L_Mark, "^0000ffkafra funcionário^000000";
			viewpoint 1,202,184,0,0x000FFF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Revendedor de armas","Dewata, 218.164","^008080");
			viewpoint 1,218,163,1,0x008080;
			break;
		case 3:
			callsub L_Mark, F_Navi("Revendedor de armaduras","Dewata, 158.182","^008080");
			viewpoint 1,158,182,2,0x008080;
			break;
		case 4:
			callsub L_Mark, F_Navi("Revendedor de ferramentas","Dewata, 182.164","^F -1493");
			viewpoint 1,182,164,3,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("Restaurando","Dewata, 146.109","^F -1493");
			viewpoint 1,146,109,4,0xFF1493;
			break;
		case 6:
			callsub L_Mark, F_Navi("Gatekeeper de Cracóvia","Dewata, 235,56","^Eles são");
			viewpoint 1,235,56,5,0x8A2BE2;
			break;
		case 7:
			callsub L_Mark, F_Navi("Alberta Sailor","Dewata, 229,49","^00bfff");
			viewpoint 1,229,49,6,0x00BFFF;
			break;
		case 8:
			mes "[Guia Dewata]";
			mes "Vou remover todas as marcas do seu mini-mapa.";
			mes "Há mais alguma coisa que eu possa fazer por você?";
			viewpoint 2,202,184,0,0xFFFFFF;
			viewpoint 2,218,163,1,0xFFFFFF;
			viewpoint 2,158,182,2,0xFFFFFF;
			viewpoint 2,182,164,3,0xFFFFFF;
			viewpoint 2,232,59,4,0xFFFFFF;
			viewpoint 2,195,235,5,0xFFFFFF;
			viewpoint 2,230,52,6,0xFFFFFF;
			break;
		case 9:
			mes "[Guia Dewata]";
			mes "Aproveite sua viagem !!";
			close;
		}
	}

L_Mark:
	mes "[Guia Dewata]";
	mes "O"+getarg(0)+"é";
	mes "Marcado em seu mini-mapa.";
	mes "Verifique seu mini-mapa.";
	mes "Há mais alguma coisa que eu possa fazer por você?";
	return;
}
