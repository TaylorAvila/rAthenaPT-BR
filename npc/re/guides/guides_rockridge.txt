harboro1,356,211,5	script	Guide#rockridge01	4_F_ANYA,{
	mes "[Assistente do xerife]";
	mes "Bem -vindo a Rock Ridge, o deserto quente e seco no oeste, onde a aventura está esperando!";
	mes "Sou o assistente do xerife e estou aqui para orientar os visitantes. Existe um lugar que você quer ir?";
	next;
	mes "[Assistente do xerife]";
	mes "Diga -me para onde você quer ir, e eu mostrarei onde está.";
	mes "Além disso, você pode usar o modo ^B9062FNAVigation ^000000 por ^b9062Flicking o nome do local ^000000 Eu mostrei a você.";
	while(true) {
		next;
		.@s = select( "[Então funcionário]", "[Pousada]", "[Loja de armas/armadura]", "[Loja de ferramentas]", "[Gabinete do xerife]", "Mini-mapa claro.", "Cancelar." );
		switch(.@s) {
		case 1:
			mes "[Assistente do xerife]";
			mes "O funcionário ^006400kafra ^000000, entendeu. Clique no ícone ^0000ffbox ^000000 no seu mini-mapa.";
			viewpoint 1,291,207,1,0xFFFF33;
			break;
		case 2:
			mes "[Assistente do xerife]";
			mes "Marquei a localização do <Vi> inn <offo> Harboro1.156,215,0.000,0 </foste> </avi> em seu mini-mapa.";
			mes "Clique no nome para ativar o modo de navegação.";
			viewpoint 1,156,215,2,0xAAFF00;
			break;
		case 3:
			mes "[Assistente do xerife]";
			mes "Marquei a localização da loja de armas/armaduras <Vi> <FOUBS> Harboro1.312,193,0.000,0 </foste> </avi> em seu mini-mapa.";
			mes "Clique no nome para ativar o modo de navegação.";
			viewpoint 1,312,193,3,0x008080;
			break;
		case 4:
			mes "[Assistente do xerife]";
			mes "Marquei a localização da loja de ferramentas <Vi> <FOUS> Harboro1.312,193,0.000,0 </foste> </avi> em seu mini-mapa.";
			mes "Clique no nome para ativar o modo de navegação.";
			viewpoint 1,312,193,4,0xFF1493;
			break;
		case 5:
			mes "[Assistente do xerife]";
			mes "Marquei a localização do escritório do xerife <Vi> <FOFFE> Harboro1.241,218,0.000,0 </fo> </avi> em seu mini-mapa.";
			mes "Clique no nome para ativar o modo de navegação.";
			viewpoint 1,241,218,5,0xFFCCFF;
			break;
		case 6:
			mes "[Assistente do xerife]";
			mes "Deixe-me remover todas as marcas em seu mini-mapa.";
			viewpoint 2,291,207,1,0xFFFF33;
			viewpoint 2,156,215,2,0xAAFF00;
			viewpoint 2,312,193,3,0x008080;
			viewpoint 2,312,193,4,0xFF1493;
			viewpoint 2,241,218,5,0xFFCCFF;
			break;
		case 7:
			mes "[Assistente do xerife]";
			mes "Aproveite a sua estadia em Rock Ridge!";
			close;
		}
		if (.@s == 7)
			mes "Você precisava de mais alguma coisa?";
		else
			mes "Existe algum outro lugar que você deseja verificar?";
	}
}
