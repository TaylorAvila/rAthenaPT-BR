	} else if(.@zone$ == "Gqgefg"){

		.@edition = 1;
	} else if(.@zone$ == "gqpayg"){
		.@zoneid = 1;
		.@edition = 1;
	} else if(.@zone$ == "Gqprtg"){
		.@zoneid = 0;
		.@edition = 1;
	} else if(.@zone$ == "GQTH"){

		.@edition = 2;
	} else if(.@zone$ == "gqtealde"){
		.@zoneid = 6;
		.@edition = 3;

			.@gid = getcharid(2);
			if(.@gid && is_guild_leader() == true){
				.@time_check1 = checkquest(.conquer_delay_questid[.@zoneid], PLAYTIME);
				if(.@time_check1 == 0){
					mes .@npcname$;
					mes "Faz menos de uma hora desde a última vez que verificamos!";
					mes "Volte quando o tempo de espera acabar.";

					mes "Se você tiver sorte do seu lado, terá bons resultados.";
					close;

					erasequest .conquer_delay_questid[.@zoneid];

					mes "Confirmei que o tempo de espera para as missões de conquista terminou.";

				} else {
					.@siege_check = (.@edition == 1 ? agitcheck() : .@edition == 2 ? agitcheck2() : agitcheck3());

					.@after_time = ((.weekday[.@zoneid] == gettime(4)) && ((.@time > .start_time[.@zoneid]) && (.@time < .end_time[.@zoneid])));

					if(.@siege_check && .@after_time){
						mes .@npcname$;
						mes "De acordo com ^4d4dff, o número de fortalezas que sua guilda conquistou neste exato momento, ^000000 Eu te darei uma recompensa.";

								mes .@npcname$;
								mes "Infelizmente, não há não fortaleza que sua guilda tenha conquistado.";
								mes "Tente de novo!";
								close;

						mes "Bem, faça o que desejar.";

	}
	end;

	function calcexp {
		.@zoneid = getarg(0);
		.@index = getarg(1);
		.@maxlevel = .max_level[.@index];
		.@isboss = (.@zoneid < 4 && !((.@index+1)%3));
		
		if(!.@isboss && BaseLevel > .@maxlevel){
			return 1000 * (100 + ((.@maxlevel - 100) * 3));
		} else {
			return 1000 * (100 + ((BaseLevel - 100) * 3)) * (1 + .@isboss);
		}
	}
	
	function calcjobexp {
		.@edition = getarg(0);
		
		if(.@edition == 3){
			if(jobcanentermap("in_prtcas01")){
				if (JobLevel > 50)
					return 1000 * (50 + ((JobLevel - 50) * 3));
				else
					return 5000;
			} else {
				debugmes "Caso inesperado para o trabalho:" + Class + "(" + jobname(Class) + ") Para Gld_mission#Core :: CalcJobExp";
				return 0;
			}
		} else if(.@edition == 2 ) {
			if(JobLevel > 50)
				return 1000 * (50 + ((JobLevel - 50) * 3));
			else
				return 3000;
		} else {
			return 5000 * JobLevel;
		}
		return 0;
	}
