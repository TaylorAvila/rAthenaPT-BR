function	script	F_flag_woe_TE	{
	.@castle$ = getarg(1);
	.@guild_id = getcastledata( .@castle$,CD_GUILD_ID );
	if (.@guild_id == 0) {
		mes "[Edito do reino divino de Rune-Midgarts]";
		mes "";
		mes "1. Siga a ordenança do Divino Rune-Midgarts Kingdom,";
		mes "Nós declaramos isso";
		mes "Não existe um mestre formal deste castelo.";
		mes "";
		mes "2. Para quem pode";
		mes "superar todos os ensaios";
		mes "e destruir o Empeio,";
		mes "o rei vai dotar aquele com";
		mes "Propriedade deste castelo.";
		close;
	}
	if (.@guild_id == getcharid(2) && getarg(0)) {
		mes "[Voz ecoando]";
		mes "Corajosos ...";
		mes "Deseja voltar ao seu lugar honrado?";
		next;
		if (select( "Volte ao castelo da guilda.","Desistir." ) == 1) {
			if (jobcanentermap(.@castle$) == 0) {
				mes "A classe de trabalho terciária e os usuários com níveis expandidos não podem participar de batalhas de cerco de treinamento.";
				close;
			}
			if (getcastledata( getarg(1),CD_GUILD_ID ) == getcharid(2))
				warp getarg(1),getarg(2),getarg(3);
			end;
		}
	}
	.@guildname$ = getguildname(.@guild_id);
	.@guildmaster$ = getguildmaster(.@guild_id);
	mes "[Edito do reino divino de Rune-Midgarts]";
	mes "";
	mes "1. Siga a ordenança do Divino Rune-Midgarts Kingdom,";
	mes "Aprovamos que este lugar está em";
	mes "a prossessão privada de ^ff0000"+.@guildname$+"^000000 Guild.";
	mes "";
	mes "2. O mestre da guilda de ^ff0000"+.@guildname$+"^000000 Guild é";
	mes "^FF0000"+.@guildmaster$+"^000000";
	mes "Se há alguém que se opõe a isso,";
	mes "Prove sua força e honra com uma lâmina de aço na sua mão.";
	close;
}
