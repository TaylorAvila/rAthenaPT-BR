ma_fild02,312,317,5	script	Guard#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[Guarda]";
		mes "Pessoas sob ^ff0000LEVEL 130 ^000000";
		mes "não pode entrar neste lugar.";
		mes "Este lugar é perigoso. Por favor, volte.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Caverna da caverna";

	if (!.@party_id) {
		mes "[Guarda]";
		mes "O crocodilo é perigoso.";
		mes "Por favor, volte depois de formar um";
		mes "equipe com outras pessoas.";
		close;
	}
	if (is_party_leader() == true) {
		mes "[Guarda]";
		mes "Este lugar é perigoso. Por favor, volte.";
		mes "Se você não tem nenhum negócio aqui, volte.";
		next;
		while(1) {
			switch(select("Por que é tão perigoso ?: Estou aqui para caçar Buwaya.: Conversação.")) {
			case 1:
				mes "[Guarda]";
				mes "Recentemente, tem havido um caso";
				mes "onde os moradores desapareceram.";
				mes "Depois de investigar as faixas,";
				mes "Descobrimos o monstro gigante buwaya";
				mes "foi responsável.";
				next;
				mes "[Guarda]";
				mes "Soldados e mercenários foram despachados para";
				mes "caçar Buwaya, mas nunca foi";
				mes "completamente bem -sucedido.";
				next;
				mes "[Guarda]";
				mes "O crocodilo ainda está sequestrando";
				mes "Os moradores até hoje.";
				mes "Por favor, junte algumas pessoas para se livrar de Buwaya.";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[Guarda]";
					mes "O crocodilo ainda está escondido.";
					mes "Mesmo se você entrar agora, Buwaya não sairá.";
					mes "Por favor, volte mais tarde.";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[Guarda]";
					mes "O nome da festa é ..."+getpartyname(.@party_id)+".";
					mes "O líder do partido é ..."+strcharinfo(0);
					mes "^0000FF"+.@md_name$+"^000000 não pode ser aberto agora.";
					mes "Por favor, tente um momento depois.";
					close;
				}
				mes "[Guarda]";
				mes "Vou abrir o túnel para ^0000ff"+.@md_name$+"^000000.";
				mes "Desejo -lhe boa sorte.";
				close;
			case 3:
				mes "[Guarda]";
				mes "Este lugar é perigoso. Você não pode entrar.";
				close;
			}
		}
	} else {
		mes "[Guarda]";
		mes "Deixe -me falar com o líder do partido.";
		mes "Por favor, chame o líder.";
		close;
	}
}

ma_fild02,315,323,0	script	Cave Entrance#buwaya	45,2,2,{
	end;
OnTouch:
	if (BaseLevel < 130) {
		mes "[Guarda]";
		mes "Pessoas sob ^ff0000LEVEL 130 ^000000";
		mes "não pode entrar neste lugar.";
		mes "Este lugar é perigoso. Por favor, volte.";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[Guarda]";
		mes "O crocodilo ainda está escondido.";
		mes "Mesmo se você entrar agora, Buwaya não sairá.";
		mes "Por favor, volte mais tarde.";
		close;
	} else {
		erasequest 4229;
		// fall through
	}
	switch(select("Digite.:Turn de volta.")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[Guarda]";
			mes "Oh, agora não é um bom momento.";
			mes "Por favor, tente novamente um momento depois.";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[Guarda]";
			mes "Este lugar é perigoso. Por favor, não entre.";
			close;
		case IE_OK:
			mapannounce "MA_FILD02",getpartyname(getcharid(1))+"Partido"+strcharinfo(0)+"Membro começou a caçar negócios na caverna.",bc_map,"0 buceta 00"; //FW_NORMAL 12 0 0
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "- Não foi possível entrar devido a uma força não identificada. -";
			close;
		}
	case 2:
		mes "[Guarda]";
		mes "Você deve valorizar sua vida.";
		mes "É melhor pensar com cuidado antes de agir.";
		close;
	}
}
