// callfunc "F_Mora_Mist",<warp to enable>,<success message>,<fail message>{,<end flag>}
function	script	F_Mora_Mist	{
	set .@map$, instance_mapname("1@mist");
	if (mobcount(.@map$,instance_npcname(strnpcinfo(0))+"::OnMyMobDead") == 0) {
		mes "Obviamente, ninguém está cuidando disso.";
		mes "Parece que você pode cortar a árvore do jardim.";
		next;
		if(select("Pique a árvore do jardim.") == 2)
			close;
		mes "Você corta a árvore, que estava bloqueando o caminho do labirinto,"+((getarg(3,0))?"limpando a saída da floresta.":"Então agora você pode continuar.");
		mapannounce .@map$,getarg(1),bc_map,"0xccffcc"; //FW_NORMAL 12 0 0
		enablenpc instance_npcname(getarg(0));
		disablenpc();
		if (getarg(4,0))
			initnpctimer;
		close;
	} else
		mapannounce .@map$,((getarg(3,0))?getarg(2):getarg(2)+"é chorar: hein? Quem está fazendo coisas ruins na minha árvore?!"),bc_map,"0xccffcc"; //FW_NORMAL 12 0 0
	end;
}
