dali,139,118,4	script	Dimensional Device#sara	10007,{
	if (BaseLevel < 99) {
		mes "[Aventureiro Leon]";
		mes "Ei! Você é jovem demais para brincar com isso! Volte quando estiver pelo menos ^0000ff Nível 99 ^000000!";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[Aventureiro Leon]";
		mes "Antes de brincar com isso; Por que você não vem aqui e fala comigo?";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"As memórias de Sara";

		if (!.@party_id) {
			mes "[Aventureiro Leon]";
			mes "Antes de entrar, você precisa organizar uma festa! Você não sabe como? Digite /Organize PartyNamehere.";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Inicialize o dispositivo dimensional .: Use o dispositivo dimensional.:Cancel.";
		else
			set .@menu$, ": Use o dispositivo dimensional .: Cancel.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "Nome da festa:"+.@p_name$;
				mes "Líder do partido:"+strcharinfo(0);
				mes "^0000FF"+.@md_name$+"^000000- A reserva falhou!";
				close;
			}
			mes "^Ff0000 O dispositivo dimensional inicializa de maneira limpa. Use o dispositivo para entrar na crack no espaço e no tempo.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "Um erro desconhecido ocorreu.";
				close;
			case IE_NOINSTANCE:
				mes "A masmorra memorial"+.@md_name$+"não existe.";
				mes "O líder do partido ainda não gerou a masmorra.";
				close;
			case IE_NOMEMBER:
				mes "Somente os membros registrados podem entrar na instância"+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "facilidade",strcharinfo(0)+", membro do partido"+.@p_name$+"entrou na instância"+.@md_name$+".",bc_map,"0 buceta 00";
				setquest 15002;
				end;
			}
		case 3:
			mes "^Ff0000O líder do partido não inicializou o dispositivo dimensional.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[Aventureiro Leon]";
		mes "Penso que em cerca de 20 horas depois que você entrou pela última vez na fenda, será estável o suficiente para você transverter novamente.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF [Memórias de Sara] A recarga foi liberada. Agora é possível usar o dispositivo dimensional novamente.^000000";
		close;
	}
}
