	if (.@day != SATURDAY && .@day != SUNDAY) {

		next;
		mes "[Casar Jae]";
		mes "Qual é a masmorra de fim de semana? Na verdade, é um espaço especialmente projetado para novos aventureiros treinarem.";
		next;

		mes "[Casar Jae]";
		mes "Para aqueles que não fazem uma festa nos fins de semana, este lugar é caçar e experimentar ganhando motivos para aventureiros solitários!";
		next;
		mes "[Casar Jae]";
		mes "Além disso, o ganho de experiência aqui é muito generoso.";
		next;
		select( "Quando é o horário de abertura?" );
		mes "[Casar Jae]";
		mes "A masmorra estará aberta durante:";
		mes "^0000ffsaturday 00:00 ~ domingo 23: 59^000000";
		next;
		mes "[Casar Jae]";
		mes "Espero vê -lo quando o fim de semana chegar ~";
		close;
	}
	.@md_name$ = "Dungeon de fim de semana";

	if (getcharid(1) < 1) {
		mes "[Casar Jae]";
		mes "Você precisa formar uma parte de um membro ou mais.";
		close;
	}
	switch( checkquest(12378,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		switch( checkquest(12377,PLAYTIME) ) {
		case -1:
		case 2:
			mes "[Casar Jae]";
			mes "Desculpe, mas o período de entrada para o" + .@md_name$ + "expirou.";
			mes "Descanse enquanto espera que a masmorra esteja disponível novamente.";
			close;
		case 0:
		case 1:
			break;
		}
		break;
	case 2:
		erasequest 12377;
		erasequest 12378;
		mes "[Casar Jae]";
		mes "A recarga expirou.";
		mes "Você pode entrar novamente na masmorra memorial.";
		close;
	}

	if (is_party_leader() == true)
		.@menu$ = "Prepare a masmorra memorial";

	mes "[Casar Jae]";
	mes "Eu acho que você está quase pronto.";
	mes "Agora, você gostaria de entrar agora?";
	next;
	switch( select( .@menu$, "Entre na masmorra de fim de semana", "Cancelar" ) ) {
	case 1:
		instance_create(.@md_name$);
		mes "[Casar Jae]";
		mes "A masmorra memorial foi criada.";
		mes "Agora você pode entrar.";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			end;
		case IE_NOINSTANCE:
			mes "Não há masmorra memorial registrada.";
			close;
		case IE_OTHER:
			mes "Um erro desconhecido ocorreu.";
			close;
		case IE_OK:
			mes "[Casar Jae]";
			mes "Você vai!";
			mapannounce "pay_arche", "" + strcharinfo(0) + "da festa," + getpartyname( getcharid(1) ) + ", está entrando no" + .@md_name$ + ".", bc_map, 0xFF99;
			if (isbegin_quest(12377) == 0)
				setquest 12377;
			if (isbegin_quest(12378) == 0)
				setquest 12378;
			end;
		}
		end;
	case 3:
		mes "[Casar Jae]";
		mes "Você não se preparou?";
		mes "Não posso realmente evitar, eu acho.";
		close;
	}
}
