prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {

				mes "[Padre Mareusis]";
				mes "Ah, sinto que você suportou";
				mes "uma experiência de vida passada. Você deve ter aprendido muitas coisas antes de entrar em Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Padre Mareusis]";
					mes "Infelizmente, acho que você ainda não está pronto para se tornar um acólito. Por favor, termine de aprender todas as habilidades básicas primeiro.";
					next;
					mes "[Padre Mareusis]";
					mes "Enquanto isso,";
					mes "Vou esperar até";
					mes "você está pronto.";
					mes "Que Deus seja";
					mes "com você.";
					close;
				}
				mes "[Padre Mareusis]";
				mes "Bem, eu te dou te douendo";
				mes "de volta de Valhalla e";
				mes "Desejo -lhe sorte no seu";
				mes "Jornada de Nova Vida.";
				next;
				skill "Nv_trickdead",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "Al_HolyLight",1,SKILL_PERM;
				mes "[Padre Mareusis]";
				mes "Agora, aventure -se e procure aqueles que precisam da sua ajuda. Que Deus acenda seu caminho.";
				close;
			}else{
				mes "[Padre Mareusis]";
				mes "Agora, aventure -se a procurar pessoas que precisam de sua ajuda. Que Deus esclareça o seu caminho.";
				close;
			}
		}else{
			mes "[Padre Mareusis]";
			mes "Sinto que você sofreu uma experiência de vida passada. Você deve ter aprendido muitas coisas antes de entrar em Valhalla.";
			next;
			mes "[Padre Mareusis]";
			mes "No entanto, posso dizer que você não é adequado para ser um acólito. Lembre -se de quem você estava em sua vida passada e encontre seu caminho.";
			close;
		}
	}
	mes "[Padre Mareusis]";
	mes "O que você procura?";
	next;
	switch(select("Mude seu trabalho para Acolyte.:Compre os requisitos para ser um acólito.")) {
	case 1:
		mes "[Padre Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Você está se sentindo bem hoje? Eu posso dizer pelo seu traje que você já é um acólito. Você não está brincando, está?";
			else
				mes "Sinto muito, mas parece que você já tem seu próprio trabalho, não é?";
			close;
		}else{
			mes "Você realmente deseja se tornar um servo de Deus?";
			mes "Vamos ver se você está pronto para isso ou não ... hmm ...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Padre Mareusis]";
				mes "Oh meu Deus?! Você ainda não realizou a prática básica?! Você tem um longo caminho a percorrer! Venha de novo depois de aumentar seu nível de trabalho!";
				close;
			}
			mes "[Padre Mareusis]";
			mes "Hmm ... seu nível de trabalho é suficiente ...";
			mes "Bom. Agora vou lhe dar a qualificação para se tornar um acólito.";
			next;
			mes "[Padre Mareusis]";
			mes "Lembre -se sempre de ser grato a Deus, que cuida de nós o tempo todo. Em caos e tempos de dificuldade, enfrente suas dificuldades com fé inabalável.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Padre Mareusis]";
			mes "Por fim, quero sinceramente parabenizá -lo por perseverar seu julgamento de penitência.";
			close;
		}
	case 2:
		mes "[Padre Mareusis]";
		mes "Você deseja se tornar um acólito?";
		mes "Então, você deve cumprir os seguintes requisitos que pensam que essas são as práticas dadas por Deus.";
		next;
		mes "[Padre Mareusis]";
		mes "Primeiro, você deve alcançar pelo menos o Nível 9 do Trabalho Novice e aprender todas as habilidades básicas.";
		mes "Essa é a coisa mais básica a se fazer, então você precisa considerá -lo como a maneira de treinar a si mesmo.";
		next;
		mes "[Padre Mareusis]";
		mes "Quando você pensa que atendeu a esse requisito, volte para mim novamente. Então você terá um trabalho sagrado no qual poderá espalhar a vontade de Deus.";
		close;
	case 3:
		close;
	}
}
