job3_guil01,91,93,3	script	Rare Herb Collector	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- Espere um minuto !! -";
		mes "- Atualmente você está carregando -";
		mes "- Muitos itens com você. -";
		mes "- Por favor, tente novamente -";
		mes "- Depois de perder algum peso. -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[Colecionador de ervas venenosas raras]";
		mes "Eu passeio pelo mundo e coleco ervas raras de veneno. Recentemente, comecei a lidar com a erva chamada Izidor. Se você estiver interessado, pode comprá -los.";
		next;
		.@i = select("Como posso comprá -los ?: Trocar por sangue animal: trocar por uma erva amarga: trocar por uma erva nociva mortal: troque por uma rosa congelada: trocá -lo por ment: trocá -lo por Hinalle")-2;
		if (.@i == -1) {
			mes "[Colecionador de ervas venenosas raras]";
			mes "Você pode comprá -los por 10.000 Zeny e um desses itens raros, sangue animal, erva amarga, erva nociva mortal, rosa congelada, ment ou uma hinal.";
			next;
			mes "[Colecionador de ervas venenosas raras]";
			mes "Por que eu pergunto dinheiro? Isso é apenas para prestar um pouco de respeito a um colecionador como eu. Ha ha ha ...";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[Colecionador de ervas venenosas raras]";
			mes "Quantos você quer?";
			mes "Você só pode comprar um máximo de '2000' EA.";
			mes "Digite '0' se você deseja cancelar.";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[Colecionador de ervas venenosas raras]";
				mes "Você cancelou o comércio.";
				close;
			}
			if (.@amount > 2000) {
				mes "[Colecionador de ervas venenosas raras]";
				mes "O número deve ser inferior a 2.000!";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[Colecionador de ervas venenosas raras]";
				mes "Ei, volte quando tiver todos os requisitos para a troca.";
				close;
			}
			mes "[Colecionador de ervas venenosas raras]";
			mes "Bom. Recebi o dinheiro e o item especial.";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[Colecionador de ervas venenosas raras]";
	mes "Eu passeio pelo mundo e coleco ervas raras de veneno. Mas não sinto vontade de vender minhas ervas para uma pessoa como você ... ha ha ha ...";
	close;
}
