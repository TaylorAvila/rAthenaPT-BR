//= Refiner that uses Enriched ores to increase upgrade success.
//===== Changelog: ===========================================

//= 1.1 Removed re-roll behavior. [Secret]

// Main NPC :: mal_jerun
//============================================================
malangdo,221,174,6	script	Holink#mal_cash	559,{
	disable_items;
	mes "[Holink]";
	mes "Eu sou o miaw ~ Blacksmith Holink ~";
	mes "Mestre em Refinamento, Holink ~";
	mes "Eu sou o Catholink especial que aprendeu com o Marrocos ~";
	mes "Minha filha está tão orgulhosa de mim, Holink ~";
	mes "O que Holink ~ refinar hoje?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Vazio]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) {
		mes "[Holink]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Meu professor Aragam disse miau ~";
			mes "Não há cura para a estupidez ...";
			break;
		case EQI_ARMOR:
			mes "Não há nada para ver aqui, miaw !!";
			break;
		case EQI_HAND_L:
			mes "Miau? O que você quer que eu faça com esta mão esquerda ...?";
			break;
		case EQI_HAND_R:
			mes "Miau? O que você quer que eu faça com essa mão direita ...?";
			break;
		case EQI_GARMENT:
			mes "Miau? Você não tem nada.";
			break;
		case EQI_SHOES:
			mes "Kyang ~! Não mexa com o meu olfativo sensível, miaw ~.";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Miau? Onde está o acessório?";
			break;
		case EQI_HEAD_MID:
		case EQI_HEAD_LOW:
			mes "Miau? Você está falando sobre as outras partes da cabeça, miau? ~";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Holink]";
		mes "Até Aragam não pode refinar isso, miau.";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[Holink]";
		mes "Miau ~ refino perfeito. Aragam fez isso, miaw? ~";
		close;
	}
	mes "[Holink]";
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
// Holink has different hardcoded prices
//	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "armadura";
		mes "Você escolheu uma armadura, miau ~";

		switch(.@equip_lv) {
			case 1:
				.@price = 15000;
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "arma";

		switch( .@equip_lv ){
			case 1: // Level 1 Weapon
				.@price = 500;
				mes "Uma arma de nível 1 ...?";
				break;
			case 2: // Level 2 Weapon
				.@price = 2000;
				mes "Miau, uma arma de nível 2 ...?";
				break;
			case 3: // Level 3 Weapon
				.@price = 20000;
				mes "Miau miau ~~ uma arma de nível 3 ~~";
				break;
			case 4: // Level 4 Weapon
				.@price = 50000;
				mes "Me-me-mew! ... uma arma de nível 4 ...!";
				mes "Eu só vi isso duas vezes enquanto";
				mes "Aprendendo com Aragam ... Me-Meow !!";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}

	mes "Você precisa ^ff9999"+getitemname(.@material)+"^000000 e ^ff9999"+.@price+"^000000 Zeny para esta refina, miaw ~";
	mes "Quer continuar, miau? ~";
	next;
	if(select("SIM !!: Não !!") == 2) {
		mes "[Holink]";
		mes "Caiaque!!";
		mes "Você não acredita em Refine Mestre Holink, miaw? ~";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[Holink]";
		mes "Miau!!";
		if (.@type$ == "armadura")
			mes "Essa armadura já estava refinada tantas vezes, miau.";
		else {
			mes "Perigo. Perigo ~";
			mes "Esta arma foi muito refinada, miau ~";
			next;
			mes "[Holink]";
		}
		mes "Você pode quebrá -lo se continuar";
		mes "Para tentar refinar ainda mais esse item, miaw.";
		next;
		mes "[Holink]";
		mes "Uma vez o"+.@type$+"está quebrado, você pode";
		mes "Nunca mais use -o, MEOW. Sem mencionar ... tudo atual";
		mes "^FF0000CARDS E ENCHANTES desaparecerão com certeza^000000.";
		mes "Você ainda quer experimentar, miau ~?";
		next;
		if(select("Sim, eu faço !!: Esqueça isso !!") == 2) {
			mes "[Holink]";
			mes "Miau! Escolha sábia, miau.";
			mes "Mas!!";
			mes "Não fico feliz em ver você duvidando do mestre refino Holink, miau ~";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Holink]";
		mes "Você não tem os ingredientes.";
		mes "Você precisa ^ff9999"+getitemname(.@material)+"^000000 e ^ff9999"+.@price+"^000000 Zeny, miaw ~";
		mes "Vá pegar, miau ~";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Holink]";
		emotion ET_FRET;
		mes "Espere um segundo ...";
		mes "Você acha que eu sou estúpido?!";
		mes "Você mudou o item enquanto eu não estava olhando! Saia daqui!";
		close;
	}

	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		mes "[Holink]";
		mes "Eu ~ eu ~ mEow! Refino divertido de diversão ~";
		next;
		emotion ET_CHUP;
		mes "[Holink]";
		mes "Perfeito!! Perfeito, miau !!";
		mes "Eu sou o discípulo do mago de refino Aragam ~";
		mes "Holink !!";
		mes "Outro dia de refino bem -sucedido, meow !!";
		close;
	}
	failedrefitem .@part;
	mes "[Holink]";
	mes "Meo ~ meow ~ kyaaak !!";
	next;
	switch(rand(1,5)) {
		case 1: emotion ET_CRY; break;
		case 2: emotion ET_PROFUSELY_SWEAT; break;
		case 3: emotion ET_KEK; break;
		case 4: emotion ET_SCRATCH; break;
		case 5: emotion ET_BIGTHROB; break;
	}
	mes "[Holink]";
	mes "Miau!! Aaaaakk ~~ !!!!";
	mes "Kyaaak !! Eu falhei, miau !!";
	next;
	mes "[Holink]";
	mes "......";
	mes "......";
	mes "Tudo ~ tudo ~ quebrado, miau ...";
	next;
	mes "[Holink]";
	mes "Miau .... mestre aragam disse uma vez,";
	mes "Aprenda com suas falhas ...";
	mes "Humano, este fracasso será o começo do seu sucesso no futuro.";
	close;
}
