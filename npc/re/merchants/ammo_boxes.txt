	mes "[Kenny]";
	mes "As sacolas de bala ajudarão você a carregar balas com muito mais facilidade.";
	mes "Venha dar uma olhada ~";
	next;
	switch(select("Vários casos de bala: troca de esferas: troca antiga de esfera: troca de slug: cancelar")) {
	case 1:
		switch(select("Caso da bala: Caso de bala sangrento: Caso de bala de prata: munição AP Bullet Caso: Munção Caso da bala de munição: Caso de bala congelando de munição: Caso de bala de raio de munição: Caso de bala de pedra mágica: Caso da bala de raio: Caso de marcadores: Lightning: Lightning Bullet: Caso da bala de gelo: Caso de bala de veneno: Caso de bala cega: Cancelar")) {
			case 1: callsub S_Casing,13200,12149;
			case 2: callsub S_Casing,13222,22737;
			case 3: callsub S_Casing,13221,22738;
			case 4: callsub S_Casing,13215,22744;
			case 5: callsub S_Casing,13216,22745;
			case 6: callsub S_Casing,13217,22746;
			case 7: callsub S_Casing,13218,22747;
			case 8: callsub S_Casing,13219,22748;
			case 9: callsub S_Casing,13220,22749;
			case 10: callsub S_Casing,13228,23123;
			case 11: callsub S_Casing,13229,23124;
			case 12: callsub S_Casing,13230,23125;
			case 13: callsub S_Casing,13231,23126;
			case 14: callsub S_Casing,13232,23127;
			case 15: callsub S_Close;
		}
	case 2:
		switch(select("Bullet de Fogo: Wind Bullet: Bullet congelante: Poison Bullet: Black Bullet: Cancelar")) {
		case 1: callsub S_Sphere_Exchange,13223,13228;
		case 2: callsub S_Sphere_Exchange,13224,13229;
		case 3: callsub S_Sphere_Exchange,13227,13230;
		case 4: callsub S_Sphere_Exchange,13225,13231;
		case 5: callsub S_Sphere_Exchange,13226,13232;
		case 6: callsub S_Close;
		}
	case 3:
		switch(select("Esfera de fogo: Esfera do vento: Esfera congelante: Esfera de veneno: Esfera cega: Cancelar")) {
		case 1: callsub S_Old_Sphere_Exchange,13203,13228;
		case 2: callsub S_Old_Sphere_Exchange,13204,13229;
		case 3: callsub S_Old_Sphere_Exchange,13207,13230;
		case 4: callsub S_Old_Sphere_Exchange,13205,13231;
		case 5: callsub S_Old_Sphere_Exchange,13206,13232;
		case 6: callsub S_Close;
		}
	case 4:
		switch(select("Munição de lesma L: munição de lesma m: munição de lesma H: munição de lesma: munição de lesma XH: Cancelar")) {
		case 1: callsub S_Slug_Exchange,13210,24;
		case 2: callsub S_Slug_Exchange,13211,12;
		case 3: callsub S_Slug_Exchange,13212,8;
		case 4: callsub S_Slug_Exchange,13213,6;
		case 5: callsub S_Slug_Exchange,13214,5;
		case 6: callsub S_Close;
		}
	case 5:
		callsub S_Close;
	}

S_Casing:
	.@bullet_id = getarg(0);
	.@pack_id = getarg(1);
	.@bullet_str$ = getitemname(.@bullet_id);
	.@pack_str$ = getitemname(.@pack_id);
	.@bullet_count = countitem(.@bullet_id);
	.@pack_count = .@bullet_count / 500;

	if (.@bullet_count < 500 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ei ~ você precisa da quantidade exata de itens para trocar. Vá buscar um pouco mais.";
		mes "Você deve ter 500 ou mais de ^ff0000" + .@bullet_str$ + "^000000, ok? Além disso, certifique -se de não ter ^ff0000wear ^000000 se você quiser trocar.";
		close;
	}

	mes "[Kenny]";
	mes "Você escolheu ^ff0000" + .@pack_str$ + "^000000.";
	mes "Você precisa de 500 ^3131ff" + .@bullet_str$ + "^000000 para fazer 1 ^3131ff" + .@bullet_str$ + "^000000 Item e custará ^3131FF500 ^000000 Zeny.";
	mes "Quantos você quer trocar?";
	next;
	switch(select("Trocar tudo: faça apenas 1: Cancelar")) {
		case 1:
			mes "[Kenny]";
			mes "Contagem de balas: ^3131ff" + .@bullet_count;
			mes "Combinação de sacos de bala: ^3131ff" + .@pack_count;
			mes "Zeny necessário: ^3131ff" + .@pack_count * 500;
			next;
			mes "Você quer trocar?";
			next;
			switch(select("Exchange: Cancelar")) {
				case 1:
					if (Zeny < .@pack_count * 500) {
						mes "[Kenny]";
						mes "Ei ~ você precisa da quantidade exata de Zeny para trocar. Vá buscar um pouco mais.";
						mes "Você deve ter ^ff0000" + .@pack_count * 500 + "^000000, ok?";
						close;
					}
					mes "[Kenny]";
					mes "Tudo feito.";
					mes "A bolsa é inútil quando o peso do inventário termina mais de 70%. Por favor, tenha cuidado.";
					Zeny -= .@pack_count * 500;
					delitem .@bullet_id, .@pack_count * 500;
					getitem .@pack_id, .@pack_count;
					close;
				case 2:
					close;
			}
		case 2:
			if (Zeny < 500) {
				mes "[Kenny]";
				mes "Ei ~ você precisa da quantidade exata de Zeny para trocar. Vá buscar um pouco mais.";
				mes "Você deve ter ^ff0000500 ^000000, ok?";
				close;
			}
			mes "[Kenny]";
			mes "Tudo feito.";
			mes "A bolsa é inútil quando o peso do inventário termina mais de 70%. Por favor, tenha cuidado.";
			Zeny -= 500;
			delitem .@bullet_id, 500;
			getitem .@pack_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);
	.@new_count = .@old_count / 10;

	if (.@old_count < 10 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ei ~ você precisa da quantidade exata de itens para trocar. Vá buscar um pouco mais.";
		mes "Você deve ter 10 ou mais de ^ff0000" + .@old_str$ + "^000000, ok? Além disso, certifique -se de não ter ^ff0000wear ^000000 se você quiser trocar.";
		close;
	}

	mes "[Kenny]";
	mes "Você escolheu ^ff0000" + .@old_str$ + "^000000.";
	mes "Você precisa de 10 ^3131ff" + .@old_str$ + "^000000 para fazer 1 ^3131ff" + .@new_str$ + "^000000 Item.";
	mes "Quantos você quer trocar?";
	next;
	switch(select("Trocar tudo: faça apenas 1: Cancelar")) {
		case 1:
			mes "[Kenny]";
			mes .@old_str$ + "CONTA: ^3131FF" + .@old_count;
			mes .@new_str$ + "CONTA: ^3131FF" + .@new_count;
			next;
			mes "Você quer trocar?";
			next;
			switch(select("Exchange: Cancelar")) {
				case 1:
					mes "[Kenny]";
					mes "Tudo feito.";
					delitem .@old_id, .@new_count * 10;
					getitem .@new_id, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tudo feito.";
			delitem .@old_id, 10;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Old_Sphere_Exchange:
	.@old_id = getarg(0);
	.@new_id = getarg(1);
	.@old_str$ = getitemname(.@old_id);
	.@new_str$ = getitemname(.@new_id);
	.@old_count = countitem(.@old_id);

	if (.@old_count < 1 || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ei ~ você precisa da quantidade exata de itens para trocar. Vá buscar um pouco mais.";
		mes "Você deve ter 1 ou mais de ^ff0000" + .@old_str$ + "^000000, ok? Além disso, certifique -se de não ter ^ff0000wear ^000000 se você quiser trocar.";
		close;
	}

	mes "[Kenny]";
	mes "Você escolheu ^ff0000" + .@old_str$ + "^000000.";
	mes "Você precisa de 1 ^3131ff" + .@old_str$ + "^000000 para fazer 1 ^3131ff" + .@new_str$ + "^000000 Item.";
	mes "Quantos você quer trocar?";
	next;
	switch(select("Trocar tudo: faça apenas 1: Cancelar")) {
		case 1:
			mes "[Kenny]";
			mes .@old_str$ + "CONTA: ^3131FF" + .@old_count;
			mes .@new_str$ + "CONTA: ^3131FF" + .@old_count;
			next;
			mes "Você quer trocar?";
			next;
			switch(select("Exchange: Cancelar")) {
				case 1:
					mes "[Kenny]";
					mes "Tudo feito.";
					delitem .@old_id, .@old_count;
					getitem .@new_id, .@old_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tudo feito.";
			delitem .@old_id, 1;
			getitem .@new_id, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Slug_Exchange:
	.@slug_id = getarg(0);
	.@req_count = getarg(1);
	.@slug_str$ = getitemname(.@slug_id);
	.@new_slug_str$ = getitemname(25187);
	.@count = countitem(.@slug_id);
	.@new_count = .@count / .@req_count;

	if (.@count < .@req_count || getequipid(EQI_AMMO) != -1 || BaseClass != Job_Gunslinger) {
		mes "[Kenny]";
		mes "Ei ~ você precisa da quantidade exata de itens para trocar. Vá buscar um pouco mais.";
		mes "Você deve ter" + .@req_count + "ou mais de ^ff0000" + .@slug_str$ + "^000000, ok? Além disso, certifique -se de não ter ^ff0000wear ^000000 se você quiser trocar.";
		close;
	}

	mes "[Kenny]";
	mes "Você escolheu ^ff0000" + .@slug_str$ + "^000000.";
	mes "Você precisa" + .@req_count + "^3131 ss" + .@slug_str$ + "^000000 para fazer 1 ^3131ff" + .@new_slug_str$ + "^000000 Item.";
	mes "Quantos você quer trocar?";
	next;
	switch(select("Trocar tudo: faça apenas 1: Cancelar")) {
		case 1:
			mes "[Kenny]";
			mes .@slug_str$ + "CONTA: ^3131FF" + .@count;
			mes .@new_slug_str$ + "CONTA: ^3131FF" + .@new_count;
			next;
			mes "Você quer trocar?";
			next;
			switch(select("Exchange: Cancelar")) {
				case 1:
					mes "[Kenny]";
					mes "Tudo feito.";
					delitem .@slug_id, .@new_count * .@req_count;
					getitem 25187, .@new_count;
					close;
				case 2:
					close;
			}
		case 2:
			mes "[Kenny]";
			mes "Tudo feito.";
			delitem .@slug_id, .@req_count;
			getitem 25187, 1;
			close;
		case 3:
			callsub S_Close;
	}

S_Close:
	mes "[Kenny]";
	mes "OK.";
	mes "Se você precisar de alguma coisa";
	mes "Por favor, volte.";
	close;
}
