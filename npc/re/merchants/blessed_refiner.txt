//===== rAthena Script ======================================= 
//= Blessed Refiner
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use Blessed ores to refine equipment.
//= Upon failure, the equipment is not destroyed. The success
//= rate is identical to that for Enriched ores.
//= - "Blacksmith Intéricos" only refines from +6~12.
//= NOTE: This NPC is currently disabled on official servers.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Main NPC :: new_smelting612
//============================================================
-	script	::BlacksmithDister	-1,{
	disable_items;
	mes "[Blacksmith Dister]";
	mes "Nesta sociedade altamente competitiva, devemos ser diferentes para sobreviver!";
	mes "Eu só refino o equipamento nos níveis +6 ~ 12.";
	next;
	mes "[Blacksmith Dister]";
	mes "Vou precisar de ^ff9999 OrideCon abençoado ^000000 para armas e o elunium abençoado ^000000 para armaduras.";
	mes "As refinas falhadas ^ff0000 não quebram ou reduzirão o nível de refino ^000000.";
	next;
	mes "[Blacksmith Dister]";
	mes "Que tal? Você quer refinar?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[não equipado]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Blacksmith Dister]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Eu sou um ferreiro, não um cabeleireiro.";
			break;
		case EQI_ARMOR:
			mes "Com meu martelo, farei de você uma estrela do céu.";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fazer mãos artificiais não é minha especialidade.";
			break;
		case EQI_GARMENT:
			mes "Traga o item para que eu possa refiná -lo!";
			break;
		case EQI_SHOES:
			mes "De onde vem esse cheiro estranho?";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Onde está o acessório?";
			break;
		case EQI_HEAD_MID:
			mes "O que você quer que eu refine?";
			break;
		case EQI_HEAD_LOW:
			mes "Huh? O que você quer que eu faça?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Blacksmith Dister]";
		mes "Este item não pode ser refinado.";
		close;
	}
	.@equip_id = getequipid(.@part);
	.@itemtype = getiteminfo( .@equip_id, ITEMINFO_TYPE );
	.@equip_refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (.@equip_refine < 6 || .@equip_refine > 12) {
		mes "[Blacksmith Dister]";
		mes "Este equipamento se refinou para"+.@equip_refine+". Eu apenas lido com itens com níveis de refino de +6 a +12!";
		close;
	}
	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "Armadura";

		switch(.@equip_lv) {
			case 1:
				set .@price,20000;
				set .@material,6439; //Unbreakable_Def
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "Arma";

		switch( .@equip_lv ){
			case 1:
				set .@price,1000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 2:
				set .@price,2000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 3:
				set .@price,20000;
				set .@material,6438; //Unbreakable_Weap
				break;
			case 4:
				set .@price,40000;
				set .@material,6438; //Unbreakable_Weap
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}
	set .@ore$,"^ff9999Benoud"+.@type$+"Ore^000000";
	mes "[Blacksmith Dister]";
	mes "This "+.@type$+" has been refined to "+.@equip_refine+". To refine it, I need "+.@ore$+" and a "+callfunc("F_InsertComma",.@price)+" zeny refining fee.";
	mes "Você quer prosseguir?";
	next;
	if(select("Sim: Não") == 2) {
		mes "[Blacksmith Dister]";
		mes "Estou ocupado, não brinca comigo!";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[Blacksmith Dister]";
		mes "Esse"+.@type$+"Foi refinado muitas vezes. Embora não desapareça após o fracasso, o"+.@ore$+"vai desaparecer!";
		next;
		mes "[Blacksmith Dister]";
		mes "Obviamente, a taxa de refino não será devolvida! Você quer continuar?";
		next;
		if(select("Sim: Não") == 2) {
			mes "[Blacksmith Dister]";
			mes "Bom. Eu ficaria triste se falhasse também.";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Blacksmith Dister]";
		mes "Materiais insuficientes.";
		mes "To refine a level "+.@equip_lv+" "+.@type$+", I need "+.@ore$+" and a "+callfunc("F_InsertComma",.@price)+" zeny refining fee.";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	mes "[Blacksmith Dister]";
	mes "TAC! TAC! TAC!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		specialeffect EF_BLESSING;
		successrefitem .@part;
		next;
		mes "[Blacksmith Dister]";
		mes "Hahahahahahaha !!!";
		next;
		mes "[Blacksmith Dister]";
		mes "Refine foi um enorme sucesso ~ hahaha!";
		close;
	}
	specialeffect EF_CURSEATTACK;
	specialeffect2 EF_SUI_EXPLOSION;
	next;
	emotion (!rand(5))?ET_ANGER:ET_HUK;
	mes "[Blacksmith Dister]";
	mes "Ó meu Deus!!!!";
	next;
	mes "[Blacksmith Dister]";
	mes "Oh! Embora tenha sido abençoado por"+.@ore$+"Para manter o equipamento ...";
	next;
	mes "[Blacksmith Dister]";
	mes "Meu humor é pior quando falhar !!";
	close;
}
