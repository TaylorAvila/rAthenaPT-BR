-	script	::CardSeparation_mal	-1,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Você tem muitos tipos de objetos. Vamos tentar continuar depois de reduzir esses objetos.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Não posso continuar porque você tem muitos objetos pesados. Vamos tentar continuar depois de reduzir o peso.";
		close;
	}
	if (strnpcinfo(1) == "Jeremy") {
		set .@Jeremy,1;
		set .@n$, "[Jeremy]";
		setarray .@equip_name$[0],  "Armadura",  "Sapato",  "Vestuário", "Chapéu superior";
		setarray .@equip_slot[0], EQI_ARMOR,EQI_SHOES,EQI_GARMENT,EQI_HEAD_TOP;

		mes .@n$;
		mes "Muito tempo sem ver ~";
		mes "Aprendi uma nova habilidade que separa cartões de armadura, sapatos, roupas e arnês. Você quer experimentar?";
	} else {
		set .@Jeremy,0;
		set .@n$, "[Richard]";
		setarray .@equip_name$[0], "Mão esquerda", "Mão direita";
		setarray .@equip_slot[0],   EQI_HAND_L,   EQI_HAND_R;

		mes .@n$;
		mes "Bobo Jeremy não quer tocar em armas e escudos por causa de escolher várias cartas que podem ser danificadas. É por isso que preparei uma habilidade de separação de cartas para armas e escudos ...";
	}
	next;
	mes .@n$;
	mes "Geralmente a taxa é de 1.000.000 de Zeny. Durante a separação do cartão, você pode usar ^990000 itens especiais que reduzem a taxa de destruição de equipamentos ou cartões ^000000. Não cobramos mais zeny por isso.";
	next;
	mes .@n$;
	mes "Existe a possibilidade de destruí -los mesmo usando um item especial. Além disso, ^ff0000 O nível de refino pode ser perdido ^000000. Você tem algum equipamento para se separar?";
	next;

	for(set .@i,0; .@i<getarraysize(.@equip_slot); set .@i,.@i+1) {
		if (getequipisequiped(.@equip_slot[.@i]))
			set .@menu$, .@menu$+getequipname(.@equip_slot[.@i])+":";
		else
			set .@menu$, .@menu$+"^999999"+.@equip_name$[.@i]+"(vazio)^000000:";
	}

	set .@i, select("Pare o trabalho:"+((.@Jeremy)?"Como é possível?":"")+":"+.@menu$);
	switch(.@i) {
	case 1:
		mes .@n$;
		mes "Sempre que você precisar do trabalho, visite -me aqui.";
		close;
	case 2:
		mes .@n$;
		mes "Você se pergunta o que é tão especial. Bem, eu odeio dar apenas um guia, então deixe -me contar a história ...";
		next;
		mes "^000099Jeremy está esticando os ombros e as mãos. Ele pode estar esperando alguém conversar com ele.^000000";
		next;
		mes .@n$;
		mes "Você sabia que a especialidade de Malangdo é alimentos enlatados?";
		next;
		select("Eu sabia disso bem. É tão limitado?");
		mes .@n$;
		mes "Hehe ... todo mundo adora. Mas havia alguns problemas antes.";
		next;
		select("Problemas? Existe algum peixe com defeito?");
		mes .@n$;
		mes "Não, o peixe não tem nenhum problema. O problema é muito óleo de peixe produzido após o processamento. Isso é um desperdício industrial.";
		next;
		mes .@n$;
		mes "No entanto, após um processo de refino revitalizante, esse óleo de peixe se tornou valioso para usar para cuidados antigos de equipamentos e lubrificante industrial.";
		next;
		mes .@n$;
		mes "Além disso, este óleo é tão útil para separar relíquias das armas equipadas que não podemos comprar lubrificante premium e lubrificante comum com Zeny.";
		next;
		mes .@n$;
		mes "Bem, não se preocupe com dinheiro. Certamente o lubrificante premium é caro. Se você pagar um pouco de zeny, posso lhe dar um lubrificante mais barato.";
		next;
		mes .@n$;
		mes "Não tenho certeza sobre a qualidade do sucesso. Enfim, isso é tão barato, certo?";
		close;
	default:
		set .@equip_num, .@equip_slot[.@i-3];
		if (!getequipisequiped(.@equip_num)) {
			mes .@n$;
			if (.@Jeremy)
				mes "Nesta parte, não há nada?";
			else
				mes "Não há nada nessa parte?";
			close;
		}
		break;
	}

	setarray .@check_equip_card[0], getequipcardid(.@equip_num,0),getequipcardid(.@equip_num,1),getequipcardid(.@equip_num,2),getequipcardid(.@equip_num,3);
	setarray .@equip_card[0], .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3];
	set .@mvp_list$,
		"|4408|4128|4456|4168|4142"+  //Gloom_Under_Night_Card, Golden_Bug_Card, Nidhogg_Shadow_Card, Dark_Lord_Card, Doppelganger_Card
		"|4134|4137|4386|4407|4357"+  //Dracula_Card, Drake_Card, Detale_Card, Randgris_Card, B_Seyren_Card
		"|4146|4132|4147|4372|4145"+  //Maya_Card, Mistress_Card, Baphomet_Card, Bacsojin_Card, Berzebub_Card
		"|4374|4352|4367|4236|4425"+  //Apocalips_H_Card, B_Ygnizem_Card, B_Shecil_Card, Amon_Ra_Card, Atroce_Card
		"|4359|4123|4144|4135|4143"+  //B_Eremes_Card, Eddga_Card, Osiris_Card, Orc_Load_Card, Orc_Hero_Card
		"|4263|4131|4430|4276|4419"+  //Incant_Samurai_Card, Moonlight_Flower_Card, Ifrit_Card, Lord_Of_Death_Card, Ktullanux_Card
		"|4403|4399|4376|4441|4302"+  //Kiel_Card, Thanatos_Card, Lady_Tanee_Card, Fallen_Bishop_Card, Tao_Gunka_Card
		"|4305|4148|4318|4121|4365"+  //Turtle_General_Card, Pharaoh_Card, Knight_Windstorm_Card, Phreeoni_Card, B_Katrinn_Card
		"|4363|4324|4361|4330|4342|"; //B_Magaleta_Card, Garm_Card, B_Harword_Card, Dark_Snake_Lord_Card, Rsx_0806_Card

	if (.@Jeremy) {
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		if (!getarraysize(.@equip_card)) {
			mes .@n$;
			mes "O cartão não está equipado. Você quer verificar novamente?";
			close;
		}
		if ((.@equip_card[0] && compare(.@mvp_list$,"|"+.@equip_card[0]+"|")) ||
			(.@equip_card[1] && compare(.@mvp_list$,"|"+.@equip_card[1]+"|")) ||
			(.@equip_card[2] && compare(.@mvp_list$,"|"+.@equip_card[2]+"|")) ||
			(.@equip_card[3] && compare(.@mvp_list$,"|"+.@equip_card[3]+"|")))
			set .@boss_chk,1;
	} else {
		// Official "Richard" script uses a hardcoded list including every possible item.
		//if (!getequipisequiped(.@equip_num)) {
		//	mes "[Richard]";
		//	mes "Desculpe. Ainda não fornecemos esse equipamento.";
		//	close;
		//}

		mes "[Richard]";
		mes "Qual é o soquete que você deseja separar o cartão? Do soquete esquerdo, eles são classificados 1,2,3,4.";
		next;
		set .@menu$,"";
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (.@equip_card[.@i] && getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL) // Armor Enchant System
				.@menu$ = .@menu$ + "Soquete" + (.@i+1) + "-" + getitemname(.@equip_card[.@i])+":";
			else {
				.@menu$ = .@menu$ + "^Hahwakkar" + (.@i+1) + "- sem cartão^000000:";
			}
		}
		set .@i, select("Pare o trabalho:"+.@menu$);
		switch(.@i) {
		case 1:
			mes .@n$;
			mes "Sempre que você precisar trabalhar, venha até mim.";
			close;
		default:
			set .@slot, .@i-2;
			if (.@equip_card[.@slot] == 0 || getiteminfo(.@equip_card[.@slot], ITEMINFO_SUBTYPE) == CARD_ENCHANT) {
				mes .@n$;
				mes "Este soquete não está equipado com nenhum cartão. Por que você não verifica novamente?";
				close;
			}
			break;
		}
		if (compare(.@mvp_list$,"|"+.@equip_card[.@slot]+"|"))
			set .@boss_chk,1;
	}
	if (.@boss_chk == 0) {
		mes .@n$;
		if (.@Jeremy)
			mes "Exceto cartões, ^ff0000All ENCHANTED EFEITOS desaparecerão. ^000000 Se você concordar com isso, escolha o tipo de trabalho:";
		else
			mes "Escolha a taxa de trabalho.";
		next;
		set .@menu$,
			"Próxima vez...:"+
			((Zeny >= 1000000)?"Use 1.000.000z (não use o item especial):":"^999999Use 1.000.000z (insuficiente)^000000:")+
			((countitem(6441))?"Use Premium Lubricant:":"^9999999 LUBRICANTE DO PREMO (insuficiente)^000000:")+
			((countitem(6440))?"Use lubrificante comum":"^99999999DINÁRIO LUBRICANTE (insuficiente)^000000");
		switch(select(.@menu$)) {
		case 1:
			mes .@n$;
			mes "Sempre que você precisar do trabalho, visite -me aqui.";
			close;
		case 2:
			if (Zeny < 1000000) {
				mes .@n$;
				mes "Você não tem zeny suficiente. Por favor, volte com taxas suficientes.";
				close;
			}
			mes .@n$;
			mes "Este é um equipamento bastante antigo. Há uma alta taxa de destruição dos cartões ou equipamentos durante o trabalho. Tem certeza de que deseja continuar?";
			next;
			if(select("Da próxima vez ...: Continue") == 1) {
				mes .@n$;
				mes "Sempre que você precisar do trabalho, visite -me aqui.";
				close;
			}
			set .@sf_c_num,150;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			set Zeny, Zeny - 1000000;
			break;
		case 3:
			if (countitem(6441) == 0) {
				mes .@n$;
				mes "Você não tem lubrificante premium.";
				close;
			}
			mes .@n$;
			mes "Se você usar o lubrificante premium, a taxa de destruição diminuirá muito, mas não posso lhe dar uma garantia de 100%. Tem certeza de que deseja continuar?";
			next;
			if(select("Da próxima vez ...: Continue") == 1) {
				mes .@n$;
				mes "Sempre que você precisar do trabalho, visite -me aqui.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,75;
			set .@sf_w_num,75;
			delitem 6441,1; //High_RankLubricant
			break;
		case 4:
			if (countitem(6440) == 0) {
				mes .@n$;
				mes "Você não tem lubrificante comum.";
				close;
			}
			mes .@n$;
			mes "Se você usar o lubrificante comum, a taxa de destruição diminuirá muito, mas não posso lhe dar uma garantia de 100%. Tem certeza de que deseja continuar?";
			next;
			if(select("Da próxima vez ...: Continue") == 1) {
				mes .@n$;
				mes "Sempre que você precisar do trabalho, visite -me aqui.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			delitem 6440,1; //General_Lubricant
			break;
		}
	} else if (.@boss_chk == 1) {
		mes .@n$;
		mes "Este equipamento contém um cartão MVP precioso. Este cartão não pode ser separado com lubrificante. Se você trazer o super surfactante ^0000ffSillit pong ^000000, poderei trabalhar.";
		next;
		switch(select("Da próxima vez ...: Eu tenho um pong de Sillit.")) {
		case 1:
			mes .@n$;
			mes "Sempre que você precisar do trabalho, visite -me aqui.";
			close;
		case 2:
			if (countitem(6443) == 0) {
				mes .@n$;
				mes "Você não tem Sillit Pong.";
				close;
			}
			break;
		}
		mes .@n$;
		if (.@Jeremy) {
			mes "Exceto cartões, ^ff0000All ENCHANTED EFEITOS desaparecerão. ^000000 Se você concordar com isso, escolha o tipo de trabalho:";
			set .@menu$,"Tudo bem, vamos fazer isso!";
		} else {
			mes "Posso continuar?";
			set .@menu$,"Eu entendi. Apenas faça isso rapidamente!";
		}
		next;
		switch(select("Próxima vez...:"+.@menu$)) {
		case 1:
			mes .@n$;
			mes "Sempre que você precisar do trabalho, visite -me aqui.";
			close;
		case 2:
			set .@sf_c_num,60;
			set .@sf_r_num,60;
			set .@sf_w_num,60;
			delitem 6443,1; //Sillit_Pong_Bottle
			break;
		}
	}

	set .@equip_id, getequipid(.@equip_num);
	set .@equip_refine, getequiprefinerycnt(.@equip_num);

	// anti-hack
	if (callfunc("F_IsEquipCardHack", .@equip_num, .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3]))
		close;

	delequip .@equip_num;

	// Chance of retaining refine level.
	if (rand(1,.@sf_r_num) >= 61)
		set .@equip_refine,0;

	if (.@Jeremy) {
		// Chance of retaining equipment.
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,0,0;
		}

		// Chance of retaining cards.
		for(set .@i,0; .@i<4; set .@i,.@i+1) {
			if (.@equip_card[.@i]) {
				if (rand(1,.@sf_c_num) < 61)
					getitem .@equip_card[.@i],1;
				else
					set .@card_break,1;
			}
		}
	} else {
		set .@card, .@equip_card[.@slot];
		set .@check_equip_card[.@slot],0;

		// Chance of retaining equipment (all enchantments are preserved).
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,.@check_equip_card[0],.@check_equip_card[1],.@check_equip_card[2],.@check_equip_card[3];
		}

		// Chance of retaining card.
		if (rand(1,.@sf_c_num) < 61)
			getitem .@card,1;
		else
			set .@card_break,1;
	}

	// Display corresponding effect.
	if (!.@equip_safe && .@card_break)
		specialeffect2 EF_LORD;
	else if (.@equip_safe && .@card_break)
		specialeffect2 EF_SUI_EXPLOSION;
	else if (!.@equip_safe && !.@card_break)
		specialeffect2 EF_FIREPILLAR;
	else
		specialeffect2 EF_MAXPOWER;

	// Output results.
	mes "-resultado da separação de cartões-";
	if (.@equip_safe) {
		mes "O crack não ocorreu durante o processo de separação de cartões.";
		mes "^0000fffEquipment Separation é normal.^000000";
	} else {
		mes "O crack ocorreu durante o processo de separação de cartões.";
		mes "O equipamento foi danificado. ^ff0000UNRECOVABLE.^000000";
	}
	mes "-------------------";
	if (!.@card_break) {
		mes "A erosão da superfície não ocorreu durante o processo de separação de cartões.";
		mes "^0000ffcard A separação foi bem -sucedida.^000000";
	} else {
		mes "A erosão da superfície ocorreu durante o processo de separação de cartões.";
		mes "O cartão foi danificado. ^ff0000UNRECOVABLE.^000000";
	}
	next;
	mes .@n$;
	mes "Isso é tudo para os resultados da separação do cartão. Por favor, venha novamente.";
	close;
}
