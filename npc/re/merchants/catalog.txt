moc_para01,22,16,5	script	Catalog Magician#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "- Você tem muitos itens. Por favor, faça espaço. -";
		close;
	}
	set .@ticket_cost, 200;
	mes "[Magiciano do catálogo]";
	mes "Olha ... a Academia Mágica";
	mes "em Geffen está agora diretamente";
	mes "vendendo o novo";
	mes "^007777 CATÁLOGO UNRIVERSAL PRATA^000000!";
	next;
	mes "Ele está olhando para as palavras";
	mes "escrito em sua mão.";
	mes "Ele parece muito ocupado";
	mes "olhar para você.";
	next;
	mes "[Magiciano do catálogo]";
	mes "Um novo catálogo";
	mes "acabou de sair!";
	mes "Você pode verificar";
	mes "Quais itens você pode negociar";
	mes "e quais itens estão à venda";
	mes "do fornecedor.";
	next;
	mes "[Magiciano do catálogo]";
	mes "O catálogo ^007777";
	mes "Custa apenas 200z!";
	mes "Você pode comprar até 50 peças de uma vez!";
	next;
	switch(select("Compre A ^007777 CATÁLOGO UNIVERSAL PRATA ^000000: Eu não preciso disso!: Você não parece um fornecedor!")) {
	case 1:
		mes "[Magiciano do catálogo]";
		mes "Você ... quer todas as 50 peças?";
		mes "Ou quantos você precisa?";
		next;
		while (1) {
			input .@input;
			mes "[Magiciano do catálogo]";
			if (.@input == 0) {
				mes "O comércio foi interrompido!";
				mes "Eu não sei o que fazer a seguir ...";
				mes "O que devo fazer?";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "Deve ser menos de 50 peças.";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "O número total de catálogos que você está tentando comprar é" + .@input + "pedaços.";
		mes "Custa" + .@sell + "Com.";
		if (Zeny < .@sell) {
			mes "Parece que você não tem dinheiro suficiente.";
			mes "Você pode vender seu equipamento";
			mes "Para ganhar mais dinheiro ... se você quiser.";
			close;
		}
		mes "Ok, aqui está.";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[Magiciano do catálogo]";
		mes "Argh ... você realmente precisa disso";
		mes "item ... você quer que eu";
		mes "explicar de novo? Não sei";
		mes "O que fazer a seguir ...";
		mes "O que devo fazer?";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[Magiciano do catálogo]";
		mes "Meu verdadeiro trabalho é um mágico.";
		mes "Eu ia perguntar ao comerciante";
		mes "guilda para vendê -los, mas eles eram";
		mes "tentando usá -los como um apêndice";
		mes "em vendas de vassouras. É por isso";
		mes "Estou vendendo isso sozinho.";
		next;
		mes "[Magiciano do catálogo]";
		mes "Existem muitos aventureiros";
		mes "aqui quem vem e vai.";
		mes "Tenho certeza de que posso vender para todos eles em breve.";
		mes "Você não acha?";
		mes "Então ... vamos começar !! Compre mais deles, por favor! POR FAVOR...!";
		close;
	}
}
