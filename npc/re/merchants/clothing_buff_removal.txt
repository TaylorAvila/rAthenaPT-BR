prontera,80,106,5	script	Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[Shot de Derek]";
	mes "Você não pode participar da batalha por causa dos fãs de roupas? Tem que ser irritante.";
	next;
	mes "[Shot de Derek]";
	mes "Embora pareça bom para você, também pode ser realmente irritante. Eu posso me livrar disso com um único gesto de mão !! Vou organizar bem. Se você estiver pronto, escolha sua opção!";
	next;
	switch( select( "Mantenha", "Libere -o imediatamente" ) ){
		case 1:
			mes "[Shot de Derek]";
			mes "Volte para mim, se você mudar de idéia.";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[Shot de Derek]";
			mes "A remoção foi concluída. Certifique -se de se mover livremente agora.";
			atcommand "@Changedress";
			close;
	}
}
