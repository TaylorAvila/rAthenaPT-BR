malangdo,220,167,5	script	Coin Exchanger CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "Você tem muitos tipos de objetos. Vamos tentar continuar depois de reduzir esses objetos.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Não posso continuar porque você tem muitos objetos pesados. Vamos tentar continuar depois de reduzir o peso.";
		close;
	}
	mes "[Trocador de moedas CX-1]";
	mes "Bem-vindo ao CX-1. Se você tiver moedas, poderá trocá -las por moedas de alto ou baixo grau.";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "Parar:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+"(Nenhum)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[Trocador de moedas CX-1]";
		mes "Obrigado por vir.";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[Trocador de moedas CX-1]";
		mes "Você escolheu uma moeda que não tem.";
		mes "Obrigado por vir.";
		close;
	}
	mes "[Trocador de moedas CX-1]";
	mes "---- taxa de troca ----";
	mes "Troca mais alta - 30 a 10";
	mes "Troca inferior - 10 a 30";
	mes "------------------------";
	mes "^FF3333 Quando você troca, cobramos uma taxa de acordo com o preço atual de mercado.^000000";
	next;
	setarray .@exchange_name$[0],"mais alto","mais baixo";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Cannot exchange for Silvervine.
	.@menu$ = "Parar:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff33333 ÚNICA EXCPARAÇÃO ATÉ"+.@exchange_name$[.@i]+"moeda^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaaexchange para"+.@exchange_name$[.@i]+"Moeda (NULL)^000000:";
		else
			.@menu$ = .@menu$ + "Trocar para"+.@exchange_name$[.@i]+"moeda -"+getitemname(.@coin)+"("+.@exchange_rate[.@i]+"necessário):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[Trocador de moedas CX-1]";
		mes "Obrigado por vir.";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[Trocador de moedas CX-1]";
		mes "Esta moeda não está disponível para trocar para"+.@exchange_name$[.@i]+"moeda.";
		mes "Obrigado por vir.";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[Trocador de moedas CX-1]";
		mes "Você não tem moedas suficientes para trocar"+.@exchange_name$[.@i]+"moeda.";
		mes "Obrigado por vir.";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[Trocador de moedas CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 Unidade (s) de ^003399"+getitemname(.@coin)+"^000000 Tomado.";
	mes "Eu os trocei por ^003399"+.@exchange_name$[.@i]+"Moedas ^000000, depois de deduzir ^003399"+.@exchange_loss[(!.@i)]+"^000000 unidade (s), ^003399"+.@exchange_total+"^000000 unidades restantes.";
	close;
}
