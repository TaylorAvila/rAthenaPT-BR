-	script	RareDiamondMerchant	58,{

	set .@npc$, "[Rare Diamond Merchant]";

	mes .@npc$;
	mes "Senhoras e senhores! ! !";
	mes "Você já ouviu rumores, mas você nunca viu com seus próprios olhos";
	mes "O diamante mais raro do mundo!";
	mes "O ^ff82ff'17 Carat Diamond ' ^000000 está atualmente à venda por um preço barato !!";
	mes "Se os diamantes valiosos interessam, então ouça!";
	next;
	
	mes .@npc$;
	mes "Você pode dar o diamante como um presente ao seu amante.";
	mes "O receptor do ^ff82ff'17 diamante de quilato ' ^000000";
	mes "e teres em seu coração.";
	emotion ET_BLABLA;
	next;
	
	mes .@npc$;
	mes "O que? Você não tem um amante?";
	mes "Você ainda pode apenas se apegar a si mesmo.";
	mes "A aura encantadora do ^ff82ff'17 diamante de quilate ' ^000000 é provável";
	mes "trazer amor à sua vida.";
	next;
	
	mes .@npc$;
	mes "Conhecido como o cativante diamante de diamantes ...";
	mes "É o ^ff82ff'17 Carat Diamond ' ^000000!";
	next;

	set .@menu$, "^Ff82ff'17 Diamante de quilato '^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ": Trocar meu diamante de 17 quilates por Zeny ...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "Sim!";
		mes "Hehe, o ^ff82ff'17 quilat diamante ' ^000000 você vê agora";
		mes "é um dos diamantes de 17 quilates da mais alta qualidade.";
		mes "Comparando-o com qualquer rocha incolor do tamanho de uma ervilha, você pode ter";
		mes "Definido de Morocc é indizível!";
		emotion ET_HUK;
		next;
		
		mes .@npc$;
		mes "Alguns afirmam que este é o mesmo diamante";
		mes "Isso foi visto por Kachua em Comodo!";
		next;
		
		if (select("Conversação final: Quanto é?") == 1) {
			mes .@npc$;
			mes "É bastante barato!";
			mes "Se você estiver interessado, fale comigo a qualquer momento!";
			close;
		}
		
		mes .@npc$;
		mes "Claro! O mais importante é o preço!";
		mes "Como eu disse anteriormente, o ^ff82ff'17 diamante de quilato ' ^000000 foi";
		mes "avaliado como um diamante de 17 quilates.";
		mes "Não há muitos como isso no mundo.";
		next;
		
		mes .@npc$;
		mes "O preço é de meros 500 milhões de Zeny!";
		mes "Há também uma taxa de serviço de 1.000.000Z.";
		mes "Comparado à qualidade do diamante";
		mes "Este é um preço pequeno a pagar.";
		next;
		
		if (select("^828282 é muito caro.^000000: Vou comprá -lo.") == 1) {
			mes .@npc$;
			mes "Dizer que um diamante como o";
			mes "^Ff82ff'17 Carat Diamond '^000000 é muito caro,";
			mes "Você me deixa sem palavras.";
			close;
		}
		
		mes .@npc$;
		mes "Oooh! Eu sabia desde o momento em que te vi que você era o direito";
		mes "Proprietário deste maravilhoso diamante.";
		mes "Estou feliz por não estar errado!";
		mes "Eu acredito que este diamante se encaixará muito bem em você.";
		mes "Tudo o que preciso de você é concluir este contrato.";
		next;
		
		while(1) {
			if (select("^B9062fi assinará o contrato.") == 2) {
				mes .@npc$;
				mes "Entendo, isso é muito ruim ..";
				close;
			}
			
			mes .@npc$;
			mes "Qual o seu nome?";
			next;
			
			mes "[" + strcharinfo(0) + "]";
			mes "Meu nome é" + strcharinfo(0) + ".";
			next;

			mes .@npc$;
			mes "Tudo bem, então, por favor, assine aqui.";
			next;
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "OK, bom.";
				mes "Tudo o que resta agora é o pagamento.";
				next;
				
				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;
					
					mes .@npc$;
					mes "Calculando o preço e a taxa de serviço ..";
					mes "501.000.000 de mulheres.";
					mes "O valor foi confirmado.";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "Você pode trocar o diamante de volta por Zeny a qualquer momento.";
					mes "No entanto, haverá uma taxa de serviço de 1.000.000Z.";
					mes "Obrigado pelo seu negócio.";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "HM?";
					mes "Sinto muito, mas você não tem fundos suficientes.";
					mes "Incluindo a taxa de serviço, é necessário um total de 501.000.000z.";
					mes "Por favor, verifique seu equilíbrio de Zeny e tente novamente.";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "Esta é realmente a sua assinatura?";
				mes "Vai assinar novamente para confirmar?";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "Aah. Você gostaria de trocar seu ^ff82ff'17 Carat Diamond ' ^000000 por Zeny?";
		emotion ET_SURPRISE;
		next;
		if (select("Sim: Não") == 2) {
			mes .@npc$;
			mes "Então, o que você deseja? ..";
			mes ". . . . . ....";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "Eu vejo. Depois de receber sua assinatura,";
		mes "Você deve oferecer o seu diamante de quilato ^ff82ff'17.";
		mes "Também não se esqueça de que há uma taxa de serviço de 1.000.000Z!";
		next;
		
		mes .@npc$;
		mes "Qual o seu nome?";
		next;
		
		mes "[" + strcharinfo(0) + "]";
		mes "Meu nome é" + strcharinfo(0) + ".";
		next;
		
		mes .@npc$;
		mes "Tudo bem, então, por favor, assine aqui.";
		next;
		
		while (1) {
			if (select("Sign: Não assine") == 2) {
				mes .@npc$;
				mes "O comércio não pode ser concluído sem a sua assinatura.";
				close;
			}
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "Sua assinatura foi recebida.";
				mes "Agora vou levar o seu ^ff82ff'17 diamante de quilato ' ^000000 ..";
				mes "Você receberá ^0000FF499.000.000 ^000000 depois.";
				next;
				
				mes .@npc$;
				mes "Obrigado pelo seu negócio.";
				mes "Sempre que você precisar de um diamante de quilates ' ^000000,";
				mes "Volte a qualquer momento com 500 milhões de zeny.";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "Esta é realmente a sua assinatura?";
				mes "Vai assinar novamente para confirmar?";
				next;
			}
		}
	}
}
