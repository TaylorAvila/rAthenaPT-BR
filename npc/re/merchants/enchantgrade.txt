grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Stratos]";
	mes "Olá, querido cliente que entrou, vamos esperar um milagre hoje.";
	mes "Como posso ajudá-lo?";
	next;
	switch( select( "Aprimore a nota do equipamento.", "Trocar itens ETEL", "O que é aprimoramento de grau?" ) ){
		case 1:
			mes "[Stratos]";
			mes "Nosso cliente. Você deseja aprimorar o grau de equipamento.";
			mes "Não é fácil mover o poder mágico das jóias.";
			next;
			mes "[Stratos]";
			mes "O poder mágico pode explodir no local.";
			mes "Então a arma que herdará o poder mágico também explode!";
			next;
			switch( select( "Eu ainda vou fazer isso!", "Vou pensar nisso." ) ){
				case 1:
					mes "[Stratos]";
					mes "Desejo boa sorte aos nossos clientes corajosos!";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Stratos]";
					mes "Sempre que você tiver a coragem de desafiar, volte.";
					close;
			}
		case 2:
			mes "[Stratos]";
			mes "São jóias usadas para melhorar as notas.";
			mes "Etel Dust, as jóias que serão a base, e se você me der uma pequena taxa, trocarei por Jóias Etel.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Stratos]";
			mes "Ocasionalmente, jóias instáveis com muito poder mágico puro são encontradas.";
			mes "Chamamos essas jóias etéricas, certo?";
			next;
			mes "[Stratos]";
			mes "Você pode transferir o poder mágico da jóia etérica para outras armas, mas é claro que há algum risco.";
			mes "É delicado, então se você não tiver cuidado, pop! Vai explodir.";
			next;
			mes "[Stratos]";
			mes "De qualquer forma, se você usar esta jóia etérica, poderá aprimorar a arma que está usando.";
			mes "O desempenho fica melhor. Chamamos esse aumento de grau.";
			next;
			mes "[Stratos]";
			mes "Você também não seria feliz se as armas que você ama crescem um passo mais longe?";
			mes "Nós somos as pessoas que o ajudam. É um pouco arriscado, no entanto. Ahahaha.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
