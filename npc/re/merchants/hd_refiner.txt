//===== rAthena Script =======================================
//= HD Refiners
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use HD ores to refine equipment. Upon
//= failure, the equipment is not destroyed; rather, its
//= refine level decreases by 1. The success rate is identical
//= to that for Enriched ores.
//= - "Blacksmith Mighty Hammer" only refines from +7~9.
//= - "Basta" only refines from +10 and up.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Blacksmith Mighty Hammer (+7~9) :: cash_smelting79
//============================================================
-	script	::MightyHammer	-1,{
	disable_items;
	mes "[Blacksmith Mighty Hammer]";
	mes "Ao contrário de outros, sou um ferreiro que refina um número muito limitado de itens.";
	mes "Eu refino apenas itens que são ^cc0000 +7 a +9 ^000000.";
	next;
	mes "[Blacksmith Mighty Hammer]";
	mes "Minha especialidade é que, mesmo que meu refinamento falhe, o nível de refino diminui em 1 sem perder a engrenagem. Não é ótimo?";
	next;
	mes "[Blacksmith Mighty Hammer]";
	mes "Então, vamos começar isso em overdrive, o que você diz? Que item você deseja refinar?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[não equipado]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Blacksmith Mighty Hammer]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Eu sou um ferreiro, não um cabeleireiro.";
			break;
		case EQI_ARMOR:
			mes "Com meu martelo, farei de você uma estrela do céu.";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fazer mãos artificiais não é minha especialidade.";
			break;
		case EQI_GARMENT:
			mes "Traga o item para que eu possa refiná -lo!";
			break;
		case EQI_SHOES:
			mes "De onde vem esse odor do pé?";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Onde está o acessório?";
			break;
		case EQI_HEAD_MID:
			mes "O que você quer que eu refine?";
			break;
		case EQI_HEAD_LOW:
			mes "Huh? O que você quer que eu faça?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Blacksmith Mighty Hammer]";
		mes "Este item não pode ser refinado.";
		close;
	}
	switch( getequiprefinerycnt(.@part) ) {
	case 7:
		.@blacksmith_blessing_count = 1;
		break;
	case 8:
		.@blacksmith_blessing_count = 2;
		break;
	case 9:
		.@blacksmith_blessing_count = 4;
		break;
	default:
		mes "[Blacksmith Mighty Hammer]";
		mes "Eu apenas lide com itens com níveis de refino de +7 a +9.";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	mes "[Blacksmith Mighty Hammer]";
	mes "Para refinar o equipamento que você selecionou, você precisa ^ff9999" + getitemname(.@material) + "^000000 e 20.000 Zeny como uma taxa.";
	mes "Você os tem pronto?";
	next;
	if (select("Sim: Não") == 2) {
		mes "[Blacksmith Mighty Hammer]";
		mes "Vou esperar até que você esteja pronto.";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[Blacksmith Mighty Hammer]";
		mes "Parece que esse item provavelmente não será refinado.";
		mes "Bem, mesmo que falhe, diminui apenas 1 nível de refina.";
		mes "Você gostaria de continuar refinando?";
		next;
		if (countitem(6635) < .@blacksmith_blessing_count)
			setarray .@menu$[0], "", "Sim", "Ainda não";
		else {
			mes "[Blacksmith Mighty Hammer]";
			mes "Ah! É a bênção ^0000ffblacksmith ^000000?";
			mes "Com a bênção do ferreiro, o equipamento não desaparecerá se a refina falhar!";
			next;
			mes "[Blacksmith Mighty Hammer]";
			if ( .@itemtype != IT_WEAPON )
				mes "Para +" + getequiprefinerycnt(.@part) + "Equipamento, refine com^316AC5" + .@blacksmith_blessing_count + "Unidades (s) Blening Blessing^000000 podem impedir que o equipamento funcione. Você quer usá -lo para refinar?";
			else
				mes "Para +" + getequiprefinerycnt(.@part) + "Arma, refina -se com^316ac5" + .@blacksmith_blessing_count + "Unidades (s) Blening Blessing^000000 podem impedir que o equipamento funcione. Você quer usá -lo para refinar?";
			next;
			setarray .@menu$[0], "Use -o para refinar", "Refine diretamente sem ele", "Não refine ainda";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[Blacksmith Mighty Hammer]";
			mes "Somente aqueles que superam o medo do fracasso obterão uma obra -prima.";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Blacksmith Mighty Hammer]";
		mes "Você não disse que tinha tudo pronto?";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Blacksmith Mighty Hammer]";
		emotion ET_FRET;
		mes "Espere um segundo ...";
		mes "Você acha que eu sou estúpido?!";
		mes "Você mudou o item enquanto eu não estava olhando! Saia daqui!";
		close;
	}

	mes "[Blacksmith Mighty Hammer]";
	mes "TAC! TAC! TAC!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Blacksmith Mighty Hammer]";
		mes "O som refresca minha mente toda vez que ouço.";
		mes "Aqui, tenha. Refinar conseguiu perfeitamente!";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion ET_HUK;
		mes "[Blacksmith Mighty Hammer]";
		mes "O que?!!";
		next;
		mes "[Blacksmith Mighty Hammer]";
		mes "Aiya! Estou sem rosto agora. Hmhm.";
		close;
	}
	downrefitem .@part;
	next;
	emotion ET_HUK;
	mes "[Blacksmith Mighty Hammer]";
	mes "Opa !!";
	next;
	mes "[Blacksmith Mighty Hammer]";
	mes "Tenho certeza de que uma pessoa como você nunca me culparia por uma diminuição no nível de refina em 1. Hmm.";
	close;
}

// Basta (+10 and up) :: cash_smelting
//============================================================
-	script	::Basta	-1,{
	disable_items;
	mes "[Basta]";
	mes "Sou o melhor ferreiro do mundo inteiro, Basta.";
	mes "Mas não forneço um serviço de refino normal.";
	mes "Eu apenas refino o equipamento ^cc0000Over +10 ^000000.";
	next;
	mes "[Basta]";
	mes "Qual equipamento você deseja refinar?";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[não equipado]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[Basta]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "Sua cabeça é um equipamento?";
			break;
		case EQI_ARMOR:
			mes "O que você quer que eu faça?";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "Fazer mãos artificiais não é minha especialidade.";
			break;
		case EQI_GARMENT:
			mes "Você sabe o que é uma túnica?";
			break;
		case EQI_SHOES:
			mes "Se você quiser refinar os pés, não venha até mim, tente correr uma maratona.";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "Onde está o acessório?";
			break;
		case EQI_HEAD_MID:
			mes "Bem ... eu não vejo nenhum equipamento que valha a pena refinar.";
			break;
		case EQI_HEAD_LOW:
			mes "Não posso te tornar inteligente. Vá ver um professor da escola para isso.";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[Basta]";
		mes "Mesmo eu não posso refinar este item. Não há como.";
		close;
	}
	.@refine_count = getequiprefinerycnt(.@part);
	if (.@refine_count < 10) {
		mes "[Basta]";
		mes "Eu não te disse? Eu apenas refino o equipamento que é +10 e acima.";
		close;
	}
	else if (.@refine_count == 10)
		.@blacksmith_blessing_count = 7;
	else if (.@refine_count == 11)
		.@blacksmith_blessing_count = 11;
	else if (.@refine_count == 20) {
		mes "[Basta]";
		mes "Esta arma é perfeita, não é mais necessária para refiná -la ~";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	if ( .@itemtype != IT_WEAPON )
		.@type$ = "armadura";
	else
		.@type$ = "arma";
	mes "[Basta]";
	mes "Hmm ... é isso que você deseja refinar?";
	mes "To refine this equipment, I need 1 ^ff9999" + getitemname(.@material) + "^000000 and " + callfunc("F_InsertComma",.@price) + " zeny as a fee.";
	mes "Você realmente quer refinar isso?";
	next;
	if (select("Sim: Não") == 2) {
		mes "[Basta]";
		mes "OK. Se é isso que você quer ...";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[Basta]";
		mes "Esse" + .@type$ + "já foi refinado bem alto.";
		mes "Se você tentar refinar mais, o nível de refino poderá diminuir.";
		next;
		mes "[Basta]";
		mes "Sou diferente dos ferreiros em outros lugares.";
		mes "É impossível que o nível de refina apareça, digamos, 3 ou 4 ... isso parece assustador.";
		mes "Aqui ele só pode diminuir em 1 nível.";
		next;
		if (.@blacksmith_blessing_count == 0 || countitem(6635) < .@blacksmith_blessing_count) {
			mes "[Basta]";
			mes "Comparado a outros ferreiros, o risco é menor.";
			mes "Eu dei todas as precauções. Você quer experimentar?";
			next;
			setarray .@menu$[0], "", "Sim", "No";
		}
		else {
			mes "[Basta]";
			mes "Woah ~ é a abençoação de Bênção ^000000? É difícil conseguir ~ Mas, você tem!";
			next;
			mes "[Basta]";
			mes "Para +" + .@refine_count + "" + .@type$ + "Precisa ^316AC5" + .@blacksmith_blessing_count + "Unidades (s) Blening Blessing^000000 podem impedir que o equipamento funcione. Você quer refinar com a Bênção de Blacksmith?";
			next;
			setarray .@menu$[0], "Refine com a Bênção de Blacksmith", "Refine sem a Bênção de Blacksmith", "Não refine ainda";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[Basta]";
			mes "Bem ~";
			mes "Não é um desafio também pode ser um tipo de sabedoria na vida.";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[Basta]";
		mes "Hmm ... você não trouxe todos os materiais necessários.";
		mes "Volte quando você tiver todos eles.";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[Basta]";
		emotion ET_FRET;
		mes "Espere um segundo ...";
		mes "Você acha que eu sou estúpido?!";
		mes "Você mudou o item enquanto eu não estava olhando! Saia daqui!";
		close;
	}

	mes "Prisioneiro de guerra! Prisioneiro de guerra! Prisioneiro de guerra! Prisioneiro de guerra!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[Basta]";
		mes "Ótimo! Bem feito !!";
		mes "Eu realmente sou o melhor ferreiro em todo o mundo!";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[Basta]";
		mes "Aaaaaaaaaaak !!!";
		next;
		mes "[Basta]";
		mes "O refinamento falhou!";
		mes "O melhor ferreiro do mundo como eu ...";
		mes "não garante 100% de sucesso ~";
	}
	else {
		downrefitem .@part;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[Basta]";
		mes "Aaaaaaaaaaak !!!";
		next;
		mes "[Basta]";
		mes "Caramba!";
		mes "O refino falhou e o nível de refino diminuiu!";
		mes "Mesmo o melhor ferreiro do mundo não garante 100% de sucesso!";
	}
	mes "Muito ruim.";
	next;
	mes "[Basta]";
	mes "Farei melhor da próxima vez! Não se preocupe!";
	close;
}
