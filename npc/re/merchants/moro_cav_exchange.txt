moro_cav,32,73,5	script	Hero Token Exchanger#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[Hero Token Exchange]";
	mes "Se você lutou";
	mes "e coletado";
	mes "Tokens de herói,";
	mes "Eu posso trocá -los com um anel especial por você.";
	mes "Você quer dar uma olhada no meu anel?";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[Hero Token Exchange]";
			mes "Sua mochila tem muitos itens,";
			mes "Por favor, volte depois de organizar!";
			close;
		}
		switch( select( "Descrição do anel", "Trocar com o anel.", "END CONVERSA." ) ) {
		case 1:
			mes "[Hero Token Exchange]";
			mes "Este anel de herói";
			mes "não pode";
			mes "olhar";
			mes "impressionante,";
			next;
			mes "[Hero Token Exchange]";
			mes "Mas leve para";
			mes "Especialista Enchanter Byrnes,";
			mes "E ele pode encantar";
			mes "Todos os 4 soquetes nele";
			mes "e trazer para fora";
			mes "suas habilidades latentes.";
			next;
			mes "[Hero Token Exchange]";
			mes "Anel de herói";
			mes "pode ajudá -lo";
			mes "Ao subjugar";
			mes "Marrocos.";
			mes "Mais alguma pergunta?";
			continue;
		case 2:
			mes "- anel de herói -";
			mes "^006400A Ring feito para guerreiros^000000";
			mes "^006400 na busca de derrotar os marrocc ..^000000";
			mes "^006400When Enchanted,^000000";
			mes "^006400it empresta seu proprietário^000000";
			mes "^006400 O poder de enfrentar os marrocc.^000000";
			mes "^006400Clas: acessório/def: 0^000000";
			mes "^006400 Weight: 20/Nível necessário: 160^000000";
			mes "^006400SLOT: 0/Equipe para: todos os trabalhos^000000";
			next;
			mes "[Hero Token Exchange]";
			mes "O anel de herói exige";
			mes "30 tokens de herói";
			mes "para a troca.";
			mes "Você quer trocar?";
			next;
			if (select( "Menu anterior.", "Trocar 30 fichas de herói." ) == 1) {
				mes "[Hero Token Exchange]";
				mes "Pensar";
				mes "de novo.";
				continue;
			}
			if (countitem(6684) < 30) {
				mes "[Hero Token Exchange]";
				mes "^006400 Token^000000";
				mes "é dado a";
				mes "Aqueles que corajosamente lutaram contra Morocc.";
				mes "Você não tem";
				mes "30 ^006400 TOKENS HOURO ^000000,";
				mes "E, portanto, não posso ajudá -lo.";
				continue;
			}
			mes "[Hero Token Exchange]";
			mes "Vou trocar";
			mes "30 ^006400 TOKENS HEROO ^000000";
			mes "com 1 ^006400HERO Ring ^000000.";
			mes "Algo mais?";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
		case 3:
			mes "[Hero Token Exchange]";
			mes "Um dia";
			mes "Você se encontrará";
			mes "Precisando de anel de herói.";
			close;
		}
	}
}

moro_cav,34,65,5	script	Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[Byrnes]";
	mes "Você já ouviu falar do anel de herói? Somente os verdadeiros heróis merecem possuir aquele anel magnífico, e eu estava esperando por um.";
	mes "O que posso fazer para você?";
	next;
	switch( select( "Anel de herói encantado.", "Redefinir encantamentos no anel de herói.", "Cancelar." ) ) {
	case 1:
		mes "[Byrnes]";
		mes "Eu posso encantar apenas anéis equipados. Qual dos anéis equipados você deseja encantar?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Esquerda", "Certo" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Por favor, equipar o anel de herói que você deseja encantar.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Por favor, equipar seu anel de herói. Não posso encantar mais nada.";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[Byrnes]";
			mes "Um anel de herói totalmente encantado não pode se encantar mais.";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "Def", "hospedar";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "ATK", "Caminhada";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "Str", "Int", "Agi", "ANO", "Dex", "ARCO";
			.@slot = 1;
		}
		mes "[Byrnes]";
		mes "Qual habilidade você deseja como encantamento no slot" + .@slot + "?";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Encantamento concluído!";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[Byrnes]";
		mes "Para redefinir encantamentos em seu anel de herói, preciso de três tokens de herói.";
		next;
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Você não tem tokens de herói suficientes. Não posso redefinir seu anel de herói.";
			close;
		}
		mes "[Byrnes]";
		mes "Eu posso redefinir apenas anéis equipados. Qual dos anéis equipados você deseja redefinir?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Esquerda", "Certo" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Você primeiro precisa equipar o item para redefinir seu status de encantamento.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Por favor, equipar seu anel de herói. Não posso redefinir mais nada.";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Você não tem tokens de herói suficientes. Não posso redefinir seu anel de herói.";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[Byrnes]";
			mes "Seu anel de herói está atualmente em um estado sem encantamento.";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Inicialização concluída.";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
