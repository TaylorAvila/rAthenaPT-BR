	}

		showscript "Fique ... aqui comigo ...";
		itemskill "Al_teleport",1;
	}
	else if (.@r == 7) {
		showscript "Você não deveria ter vindo aqui ...";
		sc_start SC_STUN,5000,0;
	}
	else if (.@r == 8) {
		showscript "Olá ... alguém está aí ...?";
		sc_start SC_BLIND,5000,0;
	}
	else if (.@r == 9) {
		showscript "Estou com tanto medo .... assustado ....";
		sc_start SC_CONFUSION,10000,0;
	}
	else if (.@r == 10) {
		showscript "Caro Odin ...";
		sc_start SC_ASSUMPTIO,100000,5;
