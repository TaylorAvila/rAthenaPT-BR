//= arg(1): How many of each 'getarg(0)' arrow per quiver.

// Mora :: ep14_1_tre
//============================================================
mora,106,117,3	script	Quiver Maker#mora	4_M_RAFLE_GR,{
	if (checkweight(1201,1) == 0) {
		mes "[Fabricante de tremores]";
		mes "Você tem muitas coisas com você.";
		mes "Faça algum espaço em seu inventário e volte. Vou te dizer algo interessante.";
		close;
	}
	if (MaxWeight - Weight < 2000) {
		mes "[Fabricante de tremores]";
		mes "Você parece desgastado com todas essas coisas.";
		mes "Faça algum espaço em seu inventário e volte. Vou te dizer algo interessante.";
		close;
	}
	mes "[Fabricante de tremores]";
	mes "Os moradores de Mora perguntam o que são bons afiadores. Eles simplesmente não sabem como o mundo funciona.";
	mes "Não é de admirar que eles não saibam nada sobre afiadores - espalhando geléia sobre as folhas o dia inteiro.";
	next;
	if (getbattleflag("Feeking.Barter")) {
		.@menu$ = "Eu quero flechas";
	}
	switch( select("Por favor, faça -me uma aljava.", .@menu$, "O que é uma aljava?") ) {
	case 1:
		if (getbattleflag("Feeking.Barter")) {
			mes "[Fabricante de tremores]";
			mes "Finalmente, alguém apareceu quem conhece o verdadeiro valor de uma aljava. Farei isso por quem sabe que vale a pena.";
			close2;
			callshop "Quiver_archer", 1;
			end;
		}
		mes "[Fabricante de tremores]";
		mes "Por fim, alguém aprecia uma aljava!";
		mes "Eu posso fazer altalheiros e elíneos e caçar altalhões.";
		mes "Qual você precisa?";
		next;
		switch(select("Uma aljava élfica: uma aljava de caça: não preciso de uma aljava.")) {
		case 1: callsub S_BuyQuiver,1773,500,500,12575; //Arrow_Of_Elf_Cntr
		case 2: callsub S_BuyQuiver,1774,500,500,12576; //Hunting_Arrow_Cntr
		case 3:
			mes "[Fabricante de tremores]";
			mes "Você pode comprar flechas do comerciante da ferramenta ao meu lado.";
			close;
		}
		end;
	case 2:
		if (getbattleflag("Feeking.Barter")) {
			mes "[Fabricante de tremores]";
			mes "Eu não vendo setas comuns. São setas artesanais que só podem ser compradas aqui. Você realmente reconheceria seu valor, certo?";
			close2;
			callshop "Catch_arrows", 1;
		}
		end;
	case 3:
		mes "[Fabricante de tremores]";
		mes "Uma seta pode ser fina e leve, mas carregar centenas e milhares de flechas é como carregar um tronco inteiro.";
		next;
		mes "[Fabricante de tremores]";
		mes "Mas quando você tem altades, pode colocar flechas nelas e economizar peso e espaço.";
		mes "Se você estiver interessado em um, vou costurar um para você.";
		close;
	}
	end;

S_BuyQuiver:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);
	if (countitem(getarg(0)) < .@req_amount) {
		mes "[Fabricante de tremores]";
		mes "Traga mais do que"+ .@req_amount +""+ getitemname(.@item_req) +"E vou fazer de você uma aljava.";
		close;
	}
	mes "[Fabricante de tremores]";
	mes "Oh, vejo que você tem"+ getitemname(.@item_req) +"na sua mão!";
	mes "Você está interessado em usar uma aljava? É realmente conveniente!";
	mes "Se você estiver interessado, eu posso negociar"+ .@req_amount +"Dessas flechas para um desses afiadores para ^ff3131"+getarg(2)+"Zeny^000000.";
	next;
	switch(select("Negocie todas as flechas que você tem: obtenha apenas uma aljava: não negocie")) {
	case 1:
		set .@arrows, countitem(.@item_req);
		set .@quiver, .@arrows / .@req_amount;
		set .@arrows_used, .@quiver * .@req_amount;
		set .@arrow_zeny01, .@quiver * getarg(2);
		mes "O número de setas que você tem: ^3131ff"+.@arrows+"^000000";
		mes "O número de afiadores disponíveis: ^3131ff"+.@quiver+"^000000";
		mes "Zeny necessário para o comércio: ^3131ff"+.@arrow_zeny01+"Mulheres^000000";
		next;
		mes "Troca?";
		next;
		if(select("Comércio: não comércio") == 2) {
			mes "[Fabricante de tremores]";
			mes "Ei, você não duvida das minhas habilidades, não é?";
			close;
		}
		break;
	case 2:
		set .@quiver, 1;
		set .@arrows_used, .@req_amount;
		set .@arrow_zeny01, getarg(2);
		set .@zeny_mes,1;
		break;
	case 3:
		mes "[Fabricante de tremores]";
		mes "Ei, você não duvida das minhas habilidades, não é?";
		close;
	}
	if (Zeny < .@arrow_zeny01) {
		mes "[Fabricante de tremores]";
		if (.@zeny_mes == 1)
			mes "Eu disse que aceitaria moedas humanas só para você, e você ainda não quer gastar o dinheiro?";
		else
			mes "Você realmente não espera colocar suas mãos em uma obra -prima por nada, não é?";
		close;
	}
	mes "[Fabricante de tremores]";
	mes "Ei, aqui está você.";
	mes "Há ^3131ffsoming algo que você precisa conhecer ^000000 - tente se lembrar disso.";
	mes "^Ff0000 Você não pode usar aljava quando seu ônus é superior a 70%.^000000";
	mes "É melhor você ter isso em mente, ou pode estar com problemas mais tarde.";
	set Zeny, Zeny-.@arrow_zeny01;
	delitem .@item_req,.@arrows_used;
	getitem getarg(3),.@quiver;
	close;
}
