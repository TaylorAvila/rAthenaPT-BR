//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "Eu sou o melhor ourivígio de todos os tempos!";
	mes "Não refino itens normais e chatos.";
	mes "Eu só refino itens que são de nível 10 ou superior.";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "De qualquer forma, você pode usar meus serviços se o seu item estiver no nível 10 ou superior.";
	mes "O que você quer que eu refine?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Eu não acho que posso refinar qualquer itens que você tenha ...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Você não está vestindo";
		mes "qualquer coisa aí que";
		mes "Eu posso refinar.";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "Eu não acho que posso";
		mes "Refine este item ...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "Eu disse que não trabalho com itens inferiores ao nível 10.";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Eu não posso refinar isso";
		mes "não mais. Isso é como";
		mes "refinado como ele fica!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "um";
		.@type$ = "arma";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "um";
		.@type$ = "armadura";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "Hum" + .@article$ + "" + .@type$ + ", tudo bem?";
	mes "Se você quiser refinar isso" + .@type$ + ",";
	mes "I will need 1 ^003366" + getitemname(.@material) + "^000000 and " + callfunc("F_InsertComma",.@price) + " zeny.";
	mes "Tem certeza de que deseja continuar?";
	next;
	if (select("Sim: Não") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm ... se você se importa ... não importa ...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Esse"+.@type$+"já foi refinado várias vezes.";
			mes "Pode ser destruído se eu tentar novamente.";
			mes "Não vai quebrar com certeza, mas há uma pequena chance.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Você pode estar ^ff0000Lowinging o nível de atualização ^000000 do"+.@type$+",";
			mes "ou se ele quebrar, você perderá ^FF0000ANDONANDOS ^000000 ou propriedades especiais adicionadas a ele.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Você ainda quer que eu refinasse?";
			mes "Eu acho que já lhe dei um aviso suficiente.";
			next;
			if(select("Sim.:no.") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "Bem, nenhum desafio é um caminho a percorrer ...";
				mes "Sem risco ... isso pode ser sábio.";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "HM. Você não parece ter dinheiro suficiente ou"+getitemname(.@material)+".";
			mes "Por favor, volte quando você os tiver.";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Espere um segundo ...";
			mes "Você acha que eu sou estúpido?!";
			mes "Você mudou o item enquanto eu não estava olhando! Saia daqui!";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "Bom! Sucesso !!!";
			mes "Eu sou o melhor ourivíssimo.";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh !!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh meu Deus!";
				mes "O nível de atualização caiu ...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmmm!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh meu Deus! Eu não consegui refinar coisas ...";
				mes "Eu não quis dizer isso!";
			}
			mes "Eu poderia ter cometido um erro, mesmo que eu seja o melhor ourrívido de todos os tempos.";
			mes "Simplesmente não era para ser.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Farei um trabalho melhor na próxima vez! Não se preocupe!";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "Eu posso refinar isso ao limite seguro ou ao número desejado de vezes. É sua escolha.";
		next;
		.@menu2 = select("Para o limite seguro, por favor.","Vou decidir quantas vezes.","Eu mudei de idéia ...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Quantas vezes você gostaria que eu refinasse seu item?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "Não consigo refinar esse item tantas vezes.";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "Isso tentará refinar o equipamento" + .@refinecheck + "vezes após o limite de segurança. Seu equipamento pode ser destruído ... tudo bem?";
			next;
			if(select("Sim...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Você disse isso ... que seja.";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Você disse isso ... que seja.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "Isso vai te custar" + .@refinecnt + "" + getitemname(.@material) + "e" + .@fullprice + "Zeny. Tudo bem?";
	next;
	if(select("Sim: não ...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "Você disse isso ... que seja.";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "HM. Você não parece ter dinheiro suficiente ou"+getitemname(.@material)+".";
		mes "Por favor, volte quando você os tiver.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "Olhe aqui ... você não tem itens em ...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "Clang ... não, mas você imaginou que eu poderia ser tão estúpido?!";
			mes "Você mudou ...";
			mes "Saia antes de atordoá -lo com meu martelo !!";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh !!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh meu Deus!";
				mes "O nível de atualização caiu ...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmmm!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh meu Deus! Eu não consegui refinar coisas ...";
				mes "Eu não quis dizer isso!";
			}
			mes "Eu poderia ter cometido um erro, mesmo que eu seja o melhor ourrívido de todos os tempos.";
			mes "Simplesmente não era para ser.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Farei um trabalho melhor na próxima vez! Não se preocupe!";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "Bom! Sucesso !!!";
	mes "Eu sou o melhor ferreiro.";
	close;
}
