job3_rune01,88,62,5	script	Dragon Breeder	105,{
	mes "[Criador de dragão]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "Bem-vindo. Você gostaria de alugar um dragão?";
		next;
		switch(select("Sim: Não")) {
		case 1:
			//if (!getskilllv("Rk_dragonTreining")) {
			if (!getskilllv("Kn_riding")) {
				mes "[Criador de dragão]";
				mes "Por favor, aprenda como montar um dragão primeiro.";
				close;
			}
			else if (checkdragon()) {
				mes "[Criador de dragão]";
				mes "Você já tem um dragão.";
				close;
			}
			else if (ismounting()) {
				mes "[Criador de dragão]";
				mes "Por favor, remova sua montagem em dinheiro.";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[Criador de dragão]";
			mes "Eu vejo. Então tenha um ótimo dia.";
			close;
		}
	}
	mes "O que você está fazendo aqui?";
	mes "Apenas os cavaleiros de Rune podem alugar um dragão.";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[Mado Gear Armorer]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "Bem -vindo, mecânico.";
		mes "Você gostaria de alugar um carrinho de pressão ou";
		mes "ride a Mado Gear?";
		next;
		switch(select("Alugue um carrinho de pressão: Ande de um Mado Gear: Compre o Mado Gear: Upgrade Resfrying Dispositivo: Cancelar")) {
		case 1:
			if (checkcart()) {
				mes "[Mado Gear Armorer]";
				mes "Sinto muito, mas você já";
				mes "Tenha um carrinho de pressão.";
				close;
			}
			setcart;
			mes "[Mado Gear Armorer]";
			mes "Lá vai você!";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[Mado Gear Armorer]";
				mes "Sinto muito, mas você já está";
				mes "riding a Mado Gear.";
				close;
			}
			else if (!getskilllv("Nc_madolicence")) {
				mes "[Mado Gear Armorer]";
				mes "Aprenda a habilidade para obter a licença de equipamento Mado primeiro.";
				close;
			}
			else if(ismounting()) {
				mes "[Mado Gear Armorer]";
				mes "Por favor, remova sua montagem em dinheiro.";
				close;
			}
			setmadogear;
			mes "[Mado Gear Armorer]";
			mes "Divirta -se e, por favor, venha de novo!";
			close;
		case 3:
			mes "[Mado Gear Armorer]";
			mes "A Mado Gear de emergência é realmente útil para situações de emergência e é vendida a 1.000.000 de Zeny.";
			next;
			if (select("Compra: Cancelar") == 2) {
				mes "[Mado Gear Armorer]";
				mes "Eu vejo. Por favor, fique à vontade para me perguntar";
				mes "Se você mudar de idéia.";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[Mado Gear Armorer]";
				mes "Sinto muito, mas você já tem um equipamento de emergência.";
				close;
			}
			if (Zeny < 1000000) {
				mes "[Mado Gear Armorer]";
				mes "Sinto muito, mas você não tem Zeny o suficiente para comprar o equipamento de emergência.";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[Mado Gear Armorer]";
			mes "Lá vai você!";
			close;
		case 4:
			mes "[Mado Gear Armorer]";
			mes "Qual dispositivo você deseja atualizar?";
			next;
			if (select("Dispositivo de resfriamento: refrigerador de alta qualidade") == 1) {
				mes "[Mado Gear Armorer]";
				mes "Atualizar o dispositivo de resfriamento para o resfriador de alta qualidade precisa de 1 dispositivo de resfriamento e 2.000.000 de zeny.";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[Mado Gear Armorer]";
				mes "Atualizar o refrigerador de alta qualidade para um refrigerador especial precisa de 1 refrigerador de alta qualidade e 4.000.000 de zeny.";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("Upgrade: Cancelar") == 2) {
				mes "[Mado Gear Armorer]";
				mes "Eu vejo. Por favor, fique à vontade para me perguntar";
				mes "Se você mudar de idéia.";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[Mado Gear Armorer]";
				mes "Sinto muito, mas você não tem o" + getitemname(.@itemid) + ".";
				close;
			}
			if (Zeny < .@cost) {
				mes "[Mado Gear Armorer]";
				mes "Sinto muito, mas você não tem Zeny suficiente para atualizar o dispositivo.";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[Mado Gear Armorer]";
			mes "Olha Você aqui! Seu próprio" + getitemname(.@itemid) + ".";
			close;
		case 5:
			close;
		}
	}
	mes "Como posso ajudá -lo?";
	mes "O Mado Gears está disponível apenas para mecânica.";
	close;
}
