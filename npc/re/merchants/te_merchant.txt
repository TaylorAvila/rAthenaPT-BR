		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
	}
	.@total_item = (getargcount() -1) /2;
	.@s = select( .@menu$ + ": Veja outros itens" ) -1;
	if (.@s <= .@total_item) {
		.@index = .@s *2 +1;
		mes "[Gerente de itens de aluguel]";
		if (rentalcountitem(getarg(.@index)) > isequippedcnt(getarg(.@index)))
			mes "Você já tem"+ getarg(.@index+1) +".";
		else {
			mes "Aqui está"+ getarg(.@index+1) +".";
			mes "^4d4dffH esse item é efetivo para as próximas 24 horas^000000 e depois disso expira.";
			rentitem getarg(.@index),86400;
		}
		mes "Você precisa de mais alguma coisa?";
		next;
	}
