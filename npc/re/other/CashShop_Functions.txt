function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[Voz do nada]";
		mes "O acesso ao conteúdo premium está disponível apenas para aqueles que possuem uma certa quantidade de pedras de redefinição.";
		next;
		if (countitem(6320) < 1) {
			mes "[Voz do nada]";
			mes "Para mudar para a terceira classe, você deve trazer 1 ^0000ccreset pedra ^000000 ou ter uma conta premium.";
			//mes "Visite nosso ishop para comprar a pedra ^0000CCRET ^000000.";
			close;
		} else {
			mes "[Voz do nada]";
			mes "Gostaria de participar dessa busca por 1 ^0000ccret Stone ^000000?";
			next;
			switch(select("Sim: Não")) {
			case 1:
				mes "[Voz do nada]";
				mes "Boa sorte, aventureiro.";
				delitem 6320,1; //Premium_Reset_Stone
				next;
				break;
			case 2:
				mes "[Voz do nada]";
				mes "Muito bem.";
				close;
			}
		}
	}
	return;
}
