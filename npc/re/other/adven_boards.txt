function	script	F_AvenBoard	{
	.@mo_board = select( "Procure monstros por piso",
					"Sistema de recuperação de informações de monstros",
					( countitem(909) == 30000 ? "Solicitar" : "" ),
					"Cancelar" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[Sistema de recuperação de informações de monstros]";
			mes "Este é o sistema de recuperação de informações de monstros. As informações dos monstros que aparecem nesta masmorra podem ser pesquisadas. Digite o nome do monstro, incluindo espaços, ao pesquisar.";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutro","Água","Terra","Fogo","Vento","Tóxico","Sagrado","Sombra","Fantasma","Mortos -vivos";
			setarray .@size$[0],"Médio","Pequeno","Grande";
			setarray .@race$[0],"Sem forma","Mortos -vivos","Animal","Plantar","Inseto","Peixe","Demônio","Demihuman","Anjo","Dragão";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[Sistema de recuperação de informações de monstros]";
			mes "Agora pesquisando";
			next;
			mes "[Sistema de recuperação de informações de monstros]";
			mes "A pesquisa foi concluída.";
			mes "Agora mostrando os resultados da pesquisa.";
			next;
			mes "[Sistema de recuperação de informações de monstros]";
			if (.@i >= .@size) {
				mes "Não há informações disponíveis para esse monstro.";
				mes "Por favor, verifique o nome do seu monstro.";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- Nome :"+ getarg(.@i+1);
				mes "- Elemento:"+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- Tamanho :"+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- Corrida :"+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[Sistema de recuperação de informações de monstros]";
			mes "Você gostaria de procurar outras informações de monstro?";
			next;
			if (select( "Procurar","Cancelar" ) == 1)
				continue;
			mes "[Sistema de recuperação de informações de monstros]";
			mes "O sistema de pesquisa de monstros agora será desligado.";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "Informações solicitadas aparecerão - não há problema se aparecer";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "Erro de operação.";
		close;
	}
}

function	script	F_InfoBoard	{
	next;

	mes "Para mais informações sobre os monstros";
	mes "Sistema de recuperação de informações de monstros";
	mes "Deve ser usado.";
	next;
	return;
}
