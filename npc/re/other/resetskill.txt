izlude,127,175,3	script	Hypnotist#novice	124,{
	mes "[Hipnotizador]";
	mes "Saudações, aventureiro.";
	mes "Eu sou um membro do hipnotizador";
	mes "Academia enviada aqui para Izlude";
	mes "para fornecer serviços de redefinição de habilidade";
	mes "Para certos personagens de primeira classe por um preço muito bom: grátis!";
	next;
	mes "[Hipnotizador]";
	mes "Embora eu ofereça ilimitado";
	mes "Redefinações de estatísticas/habilidades por enquanto, eu tenho";
	mes "duas condições que devem ser";
	mes "cumprido. Primeiro, você deve ser";
	mes "menor que ^ff0000base Nível 50 ^000000.";
	next;
	mes "[Hipnotizador]";
	mes "Segundo, você deve ser um";
	mes "^Ff0000swordman, acólito, mago,";
	mes "Ladrão, arqueiro, comerciante,";
	mes "Taekwon Boy, ou Taekwon Girl,";
	mes "Gunslinger e Ninja^000000";
	mes "Caráter de trabalho para se qualificar.";
	mes "Agora, você tem alguma dúvida?";
	next;
	switch(select("^FF0000STAT/Redefinição de habilidade?: Redefinir estatísticas: Redefinir habilidades^000000: Cancelar")) {
	case 1:
		mes "[Hipnotizador]";
		mes "As redefinições de estatísticas/habilidades permitem aventuras";
		mes "Para redistribuir sua habilidade";
		mes "Pontos se eles estiverem infelizes";
		mes "com suas habilidades atuais.";
		next;
		mes "[Hipnotizador]";
		mes "Antes de prosseguir com";
		mes "uma redefinição de estatística/habilidade, você deve";
		mes "Reduza todo o peso";
		mes "de todos os itens transportados em seu";
		mes "personagem para 0. Você pode colocar";
		mes "Itens extras e armazenamento kafra.";
		next;
		mes "[Hipnotizador]";
		mes "Ah, também é importante";
		mes "Para remover o seu ^ff0000PUSHCART ^000000";
		mes "Se você tem um equipado.";
		mes "Caso contrário, a hipnose não";
		mes "trabalhar, ou vai sair pela culatra ...";
		close;
	case 2:
		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Habilidade";
		mes "[Hipnotizador]";
		mes "Tem certeza que você";
		mes "quero prosseguir com";
		mes "meu ^ff0000"+.@str$+"Redefinir Serviço^000000?";
		next;
		if(select("Sim: Cancelar") == 1)
			break;
	case 4:
		mes "[Hipnotizador]";
		mes "Obrigado e bom";
		mes "sorte em suas aventuras.";
		mes "Por favor, viaje em segurança ~";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[Hipnotizador]";
			mes "Oh! Remova o seu";
			mes "Pushcart antes de prosseguir";
			mes "com o"+.@str$+"Redefinir serviço.";
			mes "Obrigado por cooperar ~";
			close;
		}
		if (Weight != 0) {
			mes "[Hipnotizador]";
			mes "Se você está aqui para o meu"+.@str$;
			mes "Redefinir serviço, por favor";
			mes "Lembre -se de que você não pode";
			mes "reinicie o seu"+.@str$+"s até o";
			mes "^Ff0000weight de seus itens carregados em";
			mes "Seu inventário é reduzido para 0^000000.";
			next;
			mes "[Hipnotizador]";
			mes "Por que você não coloca o seu";
			mes "coisas no armazenamento de kafra";
			mes "por agora? Dessa forma, você pode";
			mes "Mantenha com segurança todos os seus produtos.";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[Hipnotizador]";
			mes "Sinto muito, mas personagens";
			mes "com níveis básicos mais altos";
			mes "de 50 são inelegíveis para o";
			mes .@str$+"Redefinir o serviço que eu forneço.";
			close;
		}
		mes "[Hipnotizador]";
		mes "Obrigado por usar";
		mes "meu"+.@str$+"Redistribuição";
		mes "serviços. Ah, e melhor";
		mes "de sorte para você em seu";
		mes "viaja, aventureiro.";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[Hipnotizador]";
		mes "Sinto muito, mas o seu";
		mes "A aula de trabalho não se qualifica";
		mes "para o"+.@str$+"Redefinir serviço";
		mes "que eu forneço. Eu só posso";
		mes "oferecer"+.@str$+"Redefina para o";
		mes "A seguir, empregos ...";
		next;
		mes "[Hipnotizador]";
		mes "^Ff0000swordman, acólito,";
		mes "Mago, ladrão, arqueiro,";
		mes "Mercante, Taekwon";
		mes "Garoto, Taekwon Girl,";
		mes "Gollinder e Ninja^000000.";
		close;
	}
}
