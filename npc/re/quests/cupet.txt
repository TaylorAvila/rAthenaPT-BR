function	script	cute_pet_manager	{
	// getarg(0) -> .@tame_id Array
	// getarg(1) -> .@tame_amount Array
	// getarg(2) -> .@hunt_id Array
	// getarg(3) -> .@hunt_amount Array
	// getarg(4) -> .@mob_id Array
	// getarg(5) ->  Attempt Variable
	// getarg(6) -> .@tame_gets Variable
	// getarg(7) -> .@hunt_id2 Array
	// getarg(8) -> .@hunt_amount2 Array

	mes "[Gerente de animal de estimação fofo]";
	mes "Olá ~! Eu sou um gerente de animal de estimação fofo";
	mes "quem é responsável por relações públicas";
	mes "Para o novo sistema de animais de estimação fofo atualizado.";
	next;
	
	mes "[Gerente de animal de estimação fofo]";
	mes "Se você tiver alguma preocupação sobre";
	mes "O sistema de estimação fofo que estou aqui para guiá -lo.";
	mes "Deixe -me saber qual domar";
	mes "item que você quer,";
	mes "Então eu vou te contar o monstro";
	mes "que você pode domar com o item.";
	next;
	
	mes "[Gerente de animal de estimação fofo]";
	mes "Também vou explicar quais materiais";
	mes "você precisa trazer para";
	mes "Obtenha o item de domínio.";
	next;
	
	mes "[Gerente de animal de estimação fofo]";
	mes "Se você já trouxe materiais";
	mes "para trocar o item de domesticação,";
	mes "Eu posso trocá -lo por domar itens imediatamente.";
	next;
	
	mes "[Gerente de animal de estimação fofo]";
	mes "No entanto, há um limite";
	mes "para trocar itens de domesticação.";
	mes "Você só pode obter itens domesticados";
	mes getarg(6) + "vezes. Então, por favor, certifique -se de";
	mes "Quantas vezes você trocou.";
	next;
	
	mes "[Gerente de animal de estimação fofo]";
	mes "Então, qual item de domesticação você quer?";
	mes "Escolha um na lista.";
	next;
	
	// Create Menu System
	set .@size, getarraysize(getarg(0));
	for (set .@a, 0; .@a < .@size; set .@a, .@a + 1)
		set .@menu$, .@menu$ + (.@menu$ == "" ? "" : ":") + getitemname(getelementofarray(getarg(0), .@a));
	
	// Query Player Choice
	set .@choice, select(.@menu$) - 1;

	// Store Variables (Less Lookup)
	set .@tame_id, getelementofarray(getarg(0), .@choice);
	set .@tame_amount, getelementofarray(getarg(1), .@choice);
	set .@hunt_id, getelementofarray(getarg(2), .@choice);
	set .@hunt_amount, getelementofarray(getarg(3), .@choice);
	set .@mob_id, getelementofarray(getarg(4), .@choice);
	set .@hunt_id2, getelementofarray(getarg(7), .@choice);
	set .@hunt_amount2, getelementofarray(getarg(8), .@choice);

	// dispbottom "Você Ed:" + .@tame_id;
	// dispbottom "Valor manso:" + .@tame_amount;
	// dispbottom "Hunt ID:" + .@hunt_id;
	// dispbottom "Nome da caça:" + getitemname(.@hunt_id);
	// dispbottom "Quantidade de caça:" + .@hunt_amount;
	// dispbottom "ID mensal:" + .@mob_id;
	if (.@hunt_id2) { 
		// dispbottom "Hunt ID2:" + .@hunt_id2;
		// dispbottom "Hunt Valor2:" + .@hunt_amount2;
		set .@hunt2_count, countitem(.@hunt_id2);
	}
	if (countitem(.@hunt_id) >= .@hunt_amount && countitem(6083) > 0 && .@hunt2_count >= .@hunt_amount2) {
		mes "[Gerente de animal de estimação fofo]";
		mes "Ah, você reuniu todos os itens";
		mes "com seus amigos. Você pode agora";
		mes "troca de um" + getitemname(.@tame_id) + ".";
		mes "Você quer trocar comigo?";
		next;
		if  (select("Exchange: Não troca") == 2) { 
			mes "[Gerente de animal de estimação fofo]";
			mes "Você não? Você voltará novamente.";
			close;
		}
		delitem .@hunt_id, .@hunt_amount;
		delitem 6083, 1;
		if (.@hunt_id2)
			delitem .@hunt_id2, .@hunt_amount2;
		set getarg(5), getarg(5) + 1;
		getitem .@tame_id, .@tame_amount;
		
		mes "[Gerente de animal de estimação fofo]";
		mes "Escolha sábia.";
		mes "Espero que você e seu animal se dêem bem.";
		close;
	} else {
		mes "[Gerente de animal de estimação fofo]";
		mes "Você escolheu" + getitemname(.@tame_id) + ".";
		mes "Você pode usá -lo para domar";
		mes "^FF0000" + strmobinfo(1,.@mob_id) + "^000000 Monstros.";
		next;
		mes "[Gerente de animal de estimação fofo]";
		mes "Traga ^ff0000" + .@hunt_amount + "" + getitemname(.@hunt_id) + "^000000" + (.@hunt_id2 ? ", ^Ff0000" + .@hunt_amount2 + "" + getitemname(.@hunt_id2) : "") + "^000000 e ^FF00001 CAPSULE DOLLY ^000000";
		mes "Então você pode trocá -los";
		mes "para um" + getitemname(.@tame_id) + ".";
		close;
	}
}
