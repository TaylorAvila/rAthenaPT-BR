moc_para01,40,38,3	script	Mission [41 - 55]	857,{
	if (!callfunc("F_HasEdenGroupMark")) {
		mes "Você não é um membro do grupo Eden. Você não está qualificado para acessar o quadro de avisos.";
		close;
	}
	if (checkweight(1201,1) == 0 || (MaxWeight - Weight) < 1000) {
		mes "^008800Wait A Sec !!";
		mes "Você não pode receber mais itens porque está carregando muitos itens. Por favor, tente novamente depois de iluminar.^000000";
		close;
	}
	if (checkquest(12088) == -1) {
		if (BaseLevel < 41 || BaseLevel > 55) {
			mes "Mission Bulletin Board for Lv41 ~ 55 Aventureiros.";
			close;
		}
		mes "[Conselho da Missão]";
		mes "Conselho de avisos da missão.";
		next;
		if(select("Verifique as missões .: Cancel.") == 2) {
			mes "^000077OK. Vamos verificar as missões na próxima vez.^000000.";
			close;
		}
		switch(select("Caça de Sapo THARA: CAÇA DE CRUADOR: KUKRE CAÇA: ORC BABY CAUNTE: ORC WARRIORS CAUTING: VADON CAÇA: MEGALODON CAUTING: Marse Hunting: Cornutus Hunting: Myst Hunting: Caça de mariscos: caça à esfera marinha")) {
			case 1: callsub L_Quest,12072,"Você viu um sapo vermelho? É chamado Thara Frog e é muito raro. Estamos tentando verificar onde ele habita. Então, por favor, caça 20 sapos Thara.";
			case 2: callsub L_Quest,12073,"Ouvi falar de alguns brinquedos atacando pessoas. É ridículo! Vamos mostrar a eles nosso poder! Temos que caçar 15 cruzadores!";
			case 3: callsub L_Quest,12074,"Alguém foi roubado perto da praia! O possível suspeito é um monstro criminoso chamado Kukre. Por favor, caça 30 Kukre.";
			case 4: callsub L_Quest,12075,"Os bebês ORC estão ficando mais violentos nos dias de hoje. Por favor, caça 15 bebês Orc.";
			case 5: callsub L_Quest,12076,"Orc Warriors está aumentando sua agressão. Por favor, caça 20 guerreiros Orc.";
			case 6: callsub L_Quest,12077,"Você já viu caranguejos com conchas grossas? Eles são vadons. Por favor, caça 15 vadons.";
			case 7: callsub L_Quest,12078,"Os peixes mortos estão ao redor da praia e está piorando. Por favor, caça 30 megalodon.";
			case 8: callsub L_Quest,12079,"Você pode caçar Marse? Eu preciso de 15 marse para o meu estudo. Por favor me ajude.";
			case 9: callsub L_Quest,12080,"Um idiota cornutus mordeu o pé do prefeito. O prefeito está chateado, então precisamos caçar 20 cornutus para fazê -lo feliz!";
			case 10: callsub L_Quest,12081,"As pessoas estão enlouquecendo por causa de Myst. Parece um fantasma, então, assusta as pessoas. Por favor, caça 15 myst.";
			case 11: callsub L_Quest,12082,"Não pergunte o porquê, mas por favor, caça 30 frutos do mar. É uma missão secreta!";
			case 12: callsub L_Quest,12083,"Esfera Marinha? Você sabe o que é uma esfera marinha? Eu tenho uma missão para você caçar 15 esfera marítima, então, por favor, faça isso.";
			case 13: callsub L_Quest,12084,"O prefeito quer comer um delicioso prato de peixe. Phen será o ingrediente perfeito para ele. Por favor, caça 20 fen!";
			case 14: callsub L_Quest,12085,"Os monstros de Orc mortos voltaram para a nossa cidade e eles reviveram como esqueletos ORC. Estamos no caos! Por favor, caça o esqueleto ORC 15.";
			case 15: callsub L_Quest,12086,"Alguns monstros estranhos chamados Zenorc estão nos atacando. Por favor, caça 30 Zenorc.";
			case 16: callsub L_Quest,12087,"Esteja preparado! É uma missão cruel! Caça 15 múmias !! Eu tentei uma vez, mas não consegui ter sucesso. É muito assustador!";
		}
		end;
	}
	setarray .@Exp[0],9600,7200,15750,7875,11100,8550,17100,9000,12300,9900,20250,10350,13800,11025,22500,11250;
	for(set .@i,0; .@i<16; set .@i,.@i+1)
		if (checkquest(12072+.@i,HUNTING) == 2) {
			mes "[Conselho da Missão]";
			mes "Você completou a missão.";
			mes "Aqui está a sua recompensa.";
			erasequest 12088;
			erasequest 12072+.@i;
			specialeffect2 EF_STEAL;
			getexp .@Exp[.@i],0;
			close;
		}
	if ((checkquest(12072,HUNTING) == 0) && (checkquest(12073,HUNTING) == 0) && (checkquest(12074,HUNTING) == 0) && (checkquest(12075,HUNTING) == 0) && (checkquest(12076,HUNTING) == 0) && (checkquest(12077,HUNTING) == 0) && (checkquest(12078,HUNTING) == 0) && (checkquest(12079,HUNTING) == 0) && (checkquest(12080,HUNTING) == 0) && (checkquest(12081,HUNTING) == 0) && (checkquest(12082,HUNTING) == 0) && (checkquest(12083,HUNTING) == 0) && (checkquest(12084,HUNTING) == 0) && (checkquest(12085,HUNTING) == 0) && (checkquest(12086,HUNTING) == 0) && (checkquest(12087,HUNTING) == 0)) {
		mes "Agora você pode ver o quadro de avisos.";
		erasequest 12088;
		close;
	}
	mes "Não há nada de especial para fazer.";
	close;
L_Quest:
	mes "[Memorando do Conselho de Missões]";
	mes getarg(1);
	next;
	if(select("Aceite a missão .: Cancel.") == 1) {
		mes "[Conselho da Missão]";
		mes "Você aceitou a missão. Para mais detalhes, pressione as teclas 'Alt + U', você pode obter informações da janela da missão.";
		setquest 12088;
		setquest getarg(0);
	}
	close;
}
