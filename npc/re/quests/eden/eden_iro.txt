function	script	VIP_iRO_Acolyte	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes getarg(0);
		mes "Mas, meus poderes completos estão disponíveis apenas para aqueles que têm VIPs.";
		mes "I can still offer you a warp to "+getarg(1)+" though.";
		next;
		switch(select(getarg(0)+"-> 1800Z: Cancelar")) {
		case 1:
			if (Zeny < 1800) {
				mes getarg(0);
				mes "Sinto muito, mas você não tem";
				mes "Zeny suficiente para o teleporte";
				mes "Serviço. A taxa para se teletransportar";
				mes "para"+getarg(1)+"é 1.800 Zeny.";
				close;
			}
			specialeffect2 EF_READYPORTAL;
			specialeffect2 EF_TELEPORTATION;
			specialeffect2 EF_PORTAL;
			next;
			set Zeny, Zeny - 1800;
			if (getarg(1) == "Juno")
				warp "yuno",158,125;
			else
				warp "prontera",116,72;
			close;
		case 2:
			close;
		}
	}
	return;
}
