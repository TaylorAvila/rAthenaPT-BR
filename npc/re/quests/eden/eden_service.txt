-	script	Goods Cabinet#00::pggc	-1,{
	mes "É um gabinete de armazenamento de mercadorias.";
	mes "Uma mensagem é escrita em um pedaço de papel.";
	next;
	mes "+ Gabinete é exclusivamente +";
	mes "+ Para o Eden Group +";
	mes "Se você quiser usar isso";
	mes "Gabinete, verifique";
	mes "a seguir:";
	next;
	mes "1. Você é um membro";
	mes "do Eden Group?";
	mes "2. Você aprendeu";
	mes "habilidades básicas suficientes?";
	mes "3. Taxa de cabinete é";
	mes "^4d4dff500 zeny^000000!";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "Você precisa inserir Zeny para usar o gabinete.";
		mes "Custo: 500 mulheres";
		mes "Você gostaria de usá -lo?";
		next;
		switch(select("Use o gabinete .: Cancel.")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "O gabinete não está funcionando para mim.";
						mes "Talvez eu ainda não esteja qualificado para usar o gabinete.";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "Eu não tenho Zeny o suficiente.";
				mes "Eu preciso de pelo menos 500 Zeny para usar o gabinete.";
				close;
			case 2:
				mes "Vou usá -lo na próxima vez.";
				close;
		}
	}
	mes "Eu preciso de uma marca do grupo Eden para usar este gabinete.";
	close;
}
