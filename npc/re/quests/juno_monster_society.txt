			// 88, "Rachel North East North Audumla Grassland", "Tur_dun02", 146,256
			87, "Dungeon da Ilha da Tartaruga 2", "tur_dun03", 130,186,
			88, "Dungeon 3 da Ilha da Tartaruga 3", "Tur_dun02", 146,256
		);
		return;
	case 3:
		callsub( S_Menu,
			// 90, "Rachel Southwest Field", "e_fild01", 354,276,
			90, "Veias North North Field", "e_fild01", 354,276,
			90, "Campo norte das veias", "e_fild04", 52,247,
			93, "Lighthalzen North Porto Luna", "RA_FILD06", 287,39,
			95, "Em frente ao vulcão Thor", "e_fild03", 348,226
		);
		return;
	case 4:
		return;
	}
	end;

S_Menu:
	.@count = getargcount();
	.@cancel_index = .@count / 5;

	for ( .@i = 0; .@i < .@count; .@i += 5 )
		.@menu$ += "[" + getarg(.@i) + "]" + getarg(.@i+1) + ":";
	.@menu$ += "Cancelar";

	.@s = select(.@menu$) - 1;
	if (.@s == .@cancel_index)	// cancel
		return;
	.@s = .@s * 5;
	if (BaseLevel < getarg(.@s)) {
		mes "- disponível apenas para níveis mais altos que" + getarg(.@s) + "-";
		close;
	}
	delitem 1000374,1;	// Brilliant_Soda
	warp getarg(.@s+2), getarg(.@s+3),getarg(.@s+4);
	end;
}
