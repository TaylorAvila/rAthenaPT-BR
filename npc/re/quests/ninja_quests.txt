que_ng,21,72,6	script	Master Armor Craftsman	4_M_RASWORD,{
	mes "[REIT]";
	mes "Como você definiria a vitória?";
	mes "Para cair o inimigo ...";
	mes "No entanto, o mais importante é ...";
	mes "Nunca se cair!";
	next;
	mes "[REIT]";
	mes "Para isso, você deve se proteger melhor.";
	mes "Você não pode proteger 'outros' enquanto impede sua proteção 'própria'.";
	next;
	if(select("Combine armadura.: Conversação.") == 2) {
		mes "[REIT]";
		mes "A prudência também é uma virtude importante na batalha.";
		close;
	}
	switch(select("Wolf Armguard [1]: Crescent Armguard [1]: armadura de escala ninja [1]: escuro [1] desiste.")) {
	case 1:
		mes "[REIT]";
		mes "Wolf Armguard ... uma armadura bestial.";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[REIT]";
			mes "Você sente a natureza do lobo?";
			mes "Selvagem e charmosa ...";
			close;
		}
		break;
	case 2:
		mes "[REIT]";
		mes "Crescent Armguard ... uma coisa de beleza e força ... com ninguém menor.";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[REIT]";
			mes "Tanta beleza ...";
			mes "Como se todo o corpo estivesse transbordando de magia.";
			close;
		}
		break;
	case 3:
		mes "[REIT]";
		mes "Armadura de escala ninja ... diz que um ninja lendário uma vez o usava.";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[REIT]";
			mes "Esta é realmente uma armadura lendária ...";
			mes "Mesmo se eu não acreditar em lendas ...";
			close;
		}
		break;
	case 4:
		mes "[REIT]";
		mes "A Armadura do Rei Shadow ... foi feito para proteger os outros.";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[REIT]";
			mes "Como é?";
			mes "A armadura não parece que há dois?";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[REIT]";
	mes "Você não tem ingredientes suficientes.";
	mes "Olhe de novo ...";
	close;

L_Select:
	.@string1$ = "Verifique os ingredientes.";
	.@string2$ = "Verifique as estatísticas.";
	while(1) {
		switch(select(.@string1$,.@string2$,"Combinar.","Desistir.")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[REIT]";
			mes "Parece que você precisa de mais tempo para decidir.";
			close;
		}
	}

L_Ingredients:
	mes "[REIT]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Para fazer uma guarda de braço ^ff0000wolf, você precisa de 1 braço de raposa e 10 sangue de lobo ^000000.";
		mes "Você pode sentir isso? A pulsação do sangue do lobo ...";
		break;
	case 2173:	//Crescent_Armguard
		mes "Para fazer um guarda -braço de gorjeta, você precisa de 1 braço de raposa e 100 fragmentos de cristal ^000000.";
		mes "Um crescente tão bonito e claro quanto um cristal será gravado nele.";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "Para fazer uma armadura de escala ^ff0000ninja, você precisa de 30 escalas de dragão de fogo, 30 escamas de gelo e 30 peças de escala escura ^000000.";
		mes "Você pode sentir seus poderes lendários.";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Para fazer uma armadura do rei, você precisará de um terno ninja especial com um slot e 10 pedaços escuros ^000000.";
		mes "Com a escuridão, ele apaga a própria existência ...";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Chance de ter 5 seg. de ATK+100, fuja -50 no ataque corpo a corpo.";
		mes "Tipo de escudo, defesa 45, slot 1";
		mes "Nível 100 necessário apenas tipo ninja";
		break;
	case 2173:	//Crescent_Armguard
		mes "Diminui o atraso pós -habilidade para 2% para cada aprimoramento.";
		mes "Tipo de escudo, defesa 70, slot 1";
		mes "Nível 100 necessário apenas tipo ninja";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%, MSP-30%";
		mes "Tipo de armadura, defesa 90, slot 1";
		mes "Nível 100 necessário apenas tipo ninja";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Chance de ativar o LVL. 1 Ilusão - Shadow quando atacado por ataques corpo a corpo.";
		mes "Tipo de armadura, defesa 60, slot 1";
		mes "Nível 100 necessário apenas tipo ninja";
		break;
	}
	next;
	return;

L_Combine:
	mes "[REIT]";
	mes "Certifique -se de que ^ff0000 você tenha os ingredientes e equipamentos corretos em mãos ^000000.";
	mes "Não adianta chorar mais tarde.";
	next;
	switch(select("Combine.: Quit.")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- Aguentar!! -";
			mes "- Você não pode receber itens -";
			mes "- Porque você carrega demais. -";
			mes "- Por favor, tente novamente -";
			mes "- depois de iluminar seu fardo. -";
			close;
		}
		return;
	case 2:
		mes "[REIT]";
		mes "Parece que você precisa de mais tempo para decidir.";
		close;
	}
	end;
}
