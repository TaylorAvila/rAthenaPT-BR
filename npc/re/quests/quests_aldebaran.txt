c_tower1,232,222,5	script	Belljamin Button#ct	968,{
	mes "[Botão de Benjamin]";
	mes "Os monstros aqui são capazes de controlar o tempo.";
	if( BaseLevel < 150 ){
		mes "Volte quando chegar ^0000FLEVEL 150 ^000000.";
		close;
	}
	mes "Time, memories, environments... all can become warped here.";
	next;
	mes "[Botão de Benjamin]";
	mes "Estou implorando para não perguntar mais sobre isso. Pode colocar sua vida em perigo.";
	next;
	.@menu = select("Don't ask:Go to 2nd floor of warped tower:Go to 3rd floor of warped tower");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[Botão de Benjamin]";
			mes "Tenha cuidado com sua escolha.";
			next;
			mes "[Botão de Benjamin]";
			mes "Eu preciso de ^0000ff10.000z ^000000 para deixar você entrar. É um lugar bastante perigoso para se estar.";
			next;
			switch( select( "Sair: Pague 10.000z: Por que você precisa de dinheiro?" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[Botão de Benjamin]";
						mes "Realmente? Você não tem dinheiro e me pediu para levá -lo até lá?";
						close;
					}
					Zeny -= 10000;
					mes "[Botão de Benjamin]";
					mes "Não tenho certeza de como você acompanhar o tempo lá dentro, mas tenha o mais cuidado possível.";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[Botão de Benjamin]";
					mes "Estou aqui o dia todo para a segurança dos aventureiros. Eu sempre os alerta sobre o perigo. É por isso que preciso de algum tipo de pagamento.";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[Botão de Benjamin]";
	mes "Ufa ... você fez uma boa escolha. Este lugar é brutal, então você provavelmente deve voltar a uma área segura.";
	close;
}
