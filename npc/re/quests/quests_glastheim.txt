glast_01,195,273,3	script	Hugin's follower#1	2_F_MAGICMASTER,{
	mes "[Seguidor de Hugin]";
	mes "Não posso me mover entre o tempo e as dimensões como o Mestre Hugin, mas posso fingir.";
	next;
	mes "[Seguidor de Hugin]";
	mes "O que está fazendo com que esse glast Heim seja imbuído de uma força sombria tão poderosa?";
	next;
	if( select( "Eu não me importo: isso é interessante, por favor me diga mais" ) == 2 ){
		mes "[Seguidor de Hugin]";
		mes "Graças a Hugin, encontrei o caminho para acessar este velho Glast Heim, que é imbuído por forças sombrias tão poderosas.";
		next;
		mes "[Seguidor de Hugin]";
		mes "O que você diz? Você quer explorar lá?";
		next;
		if( select( "Sem exploração para mim: eu realmente quero explorar!" ) == 2 ){
			mes "[Seguidor de Hugin]";
			mes "Claro, mas este não é um serviço gratuito! Eu tenho que conseguir algo em troca. Hoho ...";
			next;
			mes "[Seguidor de Hugin]";
			mes "Eu posso mandar você para dois lugares. Qual lugar você gostaria que eu lhe enviasse?";
			next;
			switch( select( "Cancelar: Cemitério do Mosteiro (LV100 ~ 130) - 5000Z: corredor do 2º andar (acima de LV140) - 10000Z" ) ){
				case 1:
					mes "[Seguidor de Hugin]";
					mes "Tudo bem, por favor, volte sempre que mudar de idéia. Eu acho que você terá muito o que conversar comigo ...";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "GL_CAS02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[Seguidor de Hugin]";
				mes "O que? Você esperava esse serviço gratuitamente?";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[Seguidor de Hugin]";
	mes "Bem, se você não está interessado nisso, não importa.";
	close;
}
