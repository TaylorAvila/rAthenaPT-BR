			mes "Hmm você não acreditaria, mas essa missão está relacionada a um roubo.";
			mes "Eles me disseram um monte de"+ .@string$[2] +"está faltando na fortaleza em"+ .@fortress$ +".";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "O ladrão provavelmente aproveitou as horas do cerco.";
			mes "É um pedido para encontrar ^4d4dff"+ .@num_items +"ou mais de"+ .@string$[2] +"^000000.";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Como a região tem mais atmosfera casual,";
		mes "Não é divertido fazer apenas batalhas de cerco, certo?";
		mes "Então, eu trouxe ^4d4dffome works que você pode fazer durante as horas de cerco ^000000 ~";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Você pode aceitar missões ^4d4dfffstarting de 1 hora antes do início do cerco e até o final da batalha de cerco, 000000,";
		mes "e toda missão tem ^4d4dffa 4 horas de tempo Limite ^000000 para conclusão.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Simplificando, você aceita uma missão de mim durante as horas de cerco e";
		mes "^4d4dffcome de volta para me dar um relatório dentro do limite de tempo^000000.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Experimente por diversão.";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Eu vejo.";
		mes "Volte a qualquer momento, se você mudar de idéia.";
		mes "Eu posso não estar disponível, no entanto. Hehehe.";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "Parece que você ainda não cumpriu a missão?";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "Você quer cancelar por acaso?";
		next;
		switch( select( "No.","Sim." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Boa sorte!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Como você queria, esta missão é cancelada.";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "Hmm você ainda tem algum tempo, mas ainda não é a hora do cerco";
		mes "Eu não acho que você deveria continuar.";
		mes "Você deve cancelar esta missão. Eu diria que falhou.";
		next;
		switch( select( "Eu gostaria de continuar.","Sim, vou cancelar." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Eu não acho que vale a pena. Bem, boa sorte!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Como você queria, esta missão é cancelada.";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "Aceite.","Não aceite." ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dff O prazo é de 4 horas.^000000";
		mes "Cumprir a missão naquele tempo e me reportar.";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Tente outra missão então.";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "Você não encontrou nada.";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "Você encontrou '"+ getitemname(.@item_id[ .@miss_item01 ]) +"!'";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + "^4d4dffcheck o resultado^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
