	mes "Área superior, onde você pode verificar a força aplicada com base no tamanho do monstro.";
	next;
	mes "[Instrutor de treinamento]";
	mes "A área inferior onde você pode verificar a força aplicada dependendo do atributo e formulário do monstro.";
	next;
	mes "[Instrutor de treinamento]";

	next;
	switch( select( "Área superior", "Área inferior", "Não entre" ) ) {
	case 1:

		mes "Então eu vou te guiar.";
		close2;
		warp "tra_fild",112,167;
		end;
	case 2:
		mes "[Instrutor de treinamento]";
		mes "Então eu vou te guiar.";
		close2;
		warp "tra_fild",111,74;
		end;

		mes "[Instrutor de treinamento]";
		mes "Se você precisar de orientação, venha até mim.";
		close;
	}
}
